  3370→  function toLocalDatetime(iso: string): string {
  3371→    // Strip trailing Z and timezone offset to treat as local time
  3372→    const cleaned = iso.replace(/[Zz]$/, '').replace(/[+-]\d{2}:\d{2}$/, '');
  3373→    const d = new Date(cleaned);

  3375→    const pad = (n: number) => String(n).padStart(2, '0');
  3376→    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
