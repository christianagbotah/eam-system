    71→// ===== Helper: TrendIndicator component =====
    72→function TrendIndicator({ value, label }: { value: number; label?: string }) {
    73→  if (value === 0) return <span className="text-xs text-muted-foreground">{label || 'No change'}</span>;
    74→  const isPositive = value > 0;
    75→  return (
    76→    <span className={`text-xs font-semibold flex items-center gap-0.5 ${isPositive ? 'text-red-600 dark:text-red-400' : 'text-emerald-600 dark:text-emerald-400'}`}>
    77→      {isPositive ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
    78→      {Math.abs(value)}%{label && ` ${label}`}
    79→    </span>
    80→  );
    81→}

    83→// ===== Helper: Clickable KPI Card =====
    84→function KPICard({ label, value, sublabel, color, bgColor, borderColor, iconBg, iconColor, icon, barData, showRing, ringValue, trend, onClick }: {
    85→  label: string;
    86→  value: string | number;
    87→  sublabel?: string;
    88→  color: string;
    89→  bgColor: string;
    90→  borderColor: string;
    91→  iconBg: string;
    92→  iconColor: string;
    93→  icon: React.ElementType;
    94→  barData?: number[];
    95→  showRing?: boolean;
    96→  ringValue?: number;
    97→  trend?: number;
    98→  onClick?: () => void;
    99→}) {
   100→  const Icon = icon;
   101→  const content = (
   102→    <Card className={`border ${borderColor} ${bgColor} hover:shadow-lg transition-all duration-300 group overflow-hidden relative h-full flex flex-col ${onClick ? 'cursor-pointer' : ''}`}>
   103→      <div className="absolute top-0 right-0 w-24 h-24 bg-gradient-to-bl from-white/40 to-transparent dark:from-white/5 dark:to-transparent rounded-bl-full" />
   104→      <CardContent className="p-5 relative flex-1 flex flex-col">
   105→        <div className="flex items-start justify-between mb-4">
   106→          <div className={`h-10 w-10 rounded-xl ${iconBg} flex items-center justify-center group-hover:scale-110 transition-transform duration-300`}>
   107→            <Icon className={`h-5 w-5 ${iconColor}`} />
   108→          </div>
   109→          {showRing ? (
   110→            <ProgressRing value={ringValue || 0} color={color} size={48} strokeWidth={4} />
   111→          ) : barData && barData.some(v => v > 0) ? (
   112→            <MiniBarChart data={barData} color={color} maxVal={Math.max(...barData, 1)} />
   113→          ) : null}
   114→        </div>
   115→        <div className="space-y-1 mt-auto">
   116→          <p className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">{label}</p>
   117→          <p className="text-3xl font-bold tracking-tight" style={{ color: color }}>{value}</p>
   118→          <div className="flex items-center gap-2">
   119→            {sublabel && <p className="text-xs text-muted-foreground font-medium">{sublabel}</p>}
   120→            {trend !== undefined && <TrendIndicator value={trend} />}
   121→          </div>
   122→        </div>
   123→      </CardContent>
   124→    </Card>
   125→  );
   126→  if (onClick) {
   127→    return <div onClick={onClick}>{content}</div>;
   128→  }
   129→  return content;
   130→}
