     1→import { NextRequest, NextResponse } from 'next/server';
     2→import { db } from '@/lib/db';
     3→import { getSession, isAdmin, hasRole } from '@/lib/auth';
     4→import { generateReportPDF, type ReportPDFParams } from '@/lib/generate-report-pdf';
     5→
     6→type ReportType = 'lifecycle' | 'execution' | 'materials' | 'tools' | 'downtime' | 'technician_performance';
     7→
     8→// GET /api/repairs/reports?type=lifecycle&plantId=&from=&to=&priority=&department=
     9→export async function GET(request: NextRequest) {
    10→  try {
    11→    const session = getSession(request);
    12→    if (!session) return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
    13→
    14→    const canViewReports = isAdmin(session) || hasRole(session, 'maintenance_manager') || hasRole(session, 'maintenance_planner') || hasRole(session, 'plant_manager') || hasRole(session, 'maintenance_supervisor');
    15→    if (!canViewReports) {
    16→      return NextResponse.json({ success: false, error: 'Insufficient permissions to view reports' }, { status: 403 });
    17→    }
    18→
    19→    const { searchParams } = new URL(request.url);
    20→    const type = searchParams.get('type') as ReportType | null;
    21→
    22→    if (!type || !['lifecycle', 'execution', 'materials', 'tools', 'downtime', 'technician_performance'].includes(type)) {
    23→      return NextResponse.json(
    24→        { success: false, error: 'Invalid report type. Must be one of: lifecycle, execution, materials, tools, downtime, technician_performance' },
    25→        { status: 400 },
    26→      );
    27→    }
    28→
    29→    const plantId = searchParams.get('plantId') || undefined;
    30→    const from = searchParams.get('from') ? new Date(searchParams.get('from')!) : undefined;
    31→    const to = searchParams.get('to') ? new Date(searchParams.get('to')!) : undefined;
    32→    const priority = searchParams.get('priority') || undefined;
    33→    const department = searchParams.get('department') || undefined;
    34→    const assignee = searchParams.get('assignee') || undefined;
    35→    const format = searchParams.get('format');
    36→
    37→    const dateFilter: Record<string, unknown> = {};
    38→    if (from) dateFilter.gte = from;
    39→    if (to) dateFilter.lte = to;
    40→
    41→    let result: NextResponse;
    42→    switch (type) {
    43→      case 'lifecycle': result = await handleLifecycleReport(plantId, from, to, priority, department, dateFilter); break;
    44→      case 'execution': result = await handleExecutionReport(plantId, from, to, priority, assignee, dateFilter); break;
    45→      case 'technician_performance': result = await handleTechnicianPerformanceReport(plantId, from, to, department, dateFilter); break;
    46→      case 'materials': result = await handleMaterialsReport(plantId, from, to, dateFilter); break;
    47→      case 'downtime': result = await handleDowntimeReport(plantId, from, to, dateFilter); break;
    48→      case 'tools': result = await handleToolsReport(plantId, from, to, dateFilter); break;
    49→      default:
    50→        return NextResponse.json({ success: false, error: 'Unknown report type' }, { status: 400 });
    51→    }
    52→
    53→    // ========== PDF FORMAT ==========
    54→    if (format === 'pdf') {
    55→      const jsonBody = await result.json();
    56→      if (jsonBody.success && jsonBody.data) {
    57→        const pdfBuffer = await generateReportPDF(buildRepairPdfParams(type, jsonBody.data, session.fullName || session.userId, from, to, plantId, priority, department));
    58→        const filename = `repair-${type}-report.pdf`;
    59→        return new NextResponse(pdfBuffer, {
    60→          headers: {
    61→            'Content-Type': 'application/pdf',
    62→            'Content-Disposition': `attachment; filename="${filename}"`,
    63→          },
    64→        });
    65→      }
    66→    }
    67→
    68→    return result;
    69→  } catch (error: unknown) {
    70→    const message = error instanceof Error ? error.message : 'Failed to generate report';
    71→    return NextResponse.json({ success: false, error: message }, { status: 500 });
    72→  }
    73→}

    75→// ── LIFECYCLE REPORT ──────────────────────────────────────────────────────────
    76→// Full lifecycle from MR → WO closure including time at each stage
    77→async function handleLifecycleReport(
    78→  plantId: string | undefined,
    79→  from: Date | undefined,
    80→  to: Date | undefined,
    81→  priority: string | undefined,
    82→  _department: string | undefined,
    83→  dateFilter: Record<string, unknown>,
    84→) {
    85→  const mrWhere: Record<string, unknown> = { plantId: plantId || undefined };
    86→  if (priority) mrWhere.priority = priority;
    87→  if (Object.keys(dateFilter).length > 0) mrWhere.createdAt = dateFilter;
    88→
    89→  const mrs = await db.maintenanceRequest.findMany({
    90→    where: mrWhere,
    91→    include: {
    92→      workOrder: {
    93→        include: {
    94→          statusHistory: { orderBy: { createdAt: 'asc' } },
    95→          timeLogs: true,
    96→        },
    97→      },
    98→      requester: { select: { id: true, fullName: true } },
    99→      supervisor: { select: { id: true, fullName: true } },
   100→      approver: { select: { id: true, fullName: true } },
   101→      assignedPlanner: { select: { id: true, fullName: true } },
   102→    },
   103→    orderBy: { createdAt: 'desc' },
   104→    take: 200,
   105→  });
   106→
   107→  const lifecycleEntries = mrs.map((mr) => {
   108→    const stages: Record<string, { timestamp: Date | null; durationHours: number | null }> = {};
   109→    let previousTimestamp: Date | null = mr.createdAt;
   110→
   111→    stages.mr_created = { timestamp: mr.createdAt, durationHours: null };
   112→
   113→    // Check for supervisor review
   114→    if (mr.workflowStatus === 'supervisor_review' || mr.supervisorId) {
   115→      const nextStage: Date | null = mr.updatedAt;
   116→      stages.mr_supervisor_review = { timestamp: nextStage, durationHours: previousTimestamp && nextStage ? (nextStage.getTime() - previousTimestamp.getTime()) / (1000 * 3600) : null };
   117→      previousTimestamp = nextStage;
   118→    }
   119→
   120→    // Check for approval
   121→    if (mr.status === 'approved' || mr.status === 'converted') {
   122→      const approvalTime = mr.updatedAt;
   123→      stages.mr_approved = { timestamp: approvalTime, durationHours: previousTimestamp && approvalTime ? (approvalTime.getTime() - previousTimestamp.getTime()) / (1000 * 3600) : null };
   124→      previousTimestamp = approvalTime;
   125→    }
   126→
   127→    // Check for planner assignment
   128→    if (mr.assignedPlannerId) {
   129→      stages.mr_planner_assigned = { timestamp: mr.updatedAt, durationHours: previousTimestamp ? (mr.updatedAt.getTime() - previousTimestamp.getTime()) / (1000 * 3600) : null };
   130→      previousTimestamp = mr.updatedAt;
   131→    }
   132→
   133→    // WO lifecycle stages
   134→    const wo = mr.workOrder;
   135→    if (wo) {
   136→      stages.wo_created = { timestamp: wo.createdAt, durationHours: previousTimestamp ? (wo.createdAt.getTime() - previousTimestamp.getTime()) / (1000 * 3600) : null };
   137→      previousTimestamp = wo.createdAt;
   138→
   139→      for (const sh of wo.statusHistory) {
   140→        const duration = previousTimestamp ? (sh.createdAt.getTime() - previousTimestamp.getTime()) / (1000 * 3600) : null;
   141→        stages[`wo_${sh.toStatus}`] = { timestamp: sh.createdAt, durationHours: duration };
   142→        previousTimestamp = sh.createdAt;
   143→      }
   144→    }
   145→
   146→    const totalTurnaround = mr.createdAt && wo?.actualEnd
   147→      ? (wo.actualEnd.getTime() - mr.createdAt.getTime()) / (1000 * 3600)
   148→      : null;
   149→
   150→    return {
   151→      mrId: mr.id,
   152→      mrNumber: mr.requestNumber,
   153→      mrTitle: mr.title,
   154→      priority: mr.priority,
   155→      status: mr.status,
   156→      workflowStatus: mr.workflowStatus,
   157→      hasWorkOrder: !!wo,
   158→      woNumber: wo?.woNumber || null,
   159→      woStatus: wo?.status || null,
   160→      stages,
   161→      totalTurnaroundHours: totalTurnaround,
   162→      requester: mr.requester,
   163→      supervisor: mr.supervisor,
   164→      approver: mr.approver,
   165→      planner: mr.assignedPlanner,
   166→    };
   167→  });
   168→
   169→  // Calculate averages
   170→  const completedWithWo = lifecycleEntries.filter((e) => e.woStatus === 'closed' && e.totalTurnaroundHours !== null);
   171→  const avgTurnaround = completedWithWo.length > 0
   172→    ? completedWithWo.reduce((sum, e) => sum + (e.totalTurnaroundHours || 0), 0) / completedWithWo.length
   173→    : null;
   174→
   175→  const avgByStage: Record<string, number> = {};
   176→  for (const entry of lifecycleEntries) {
   177→    for (const [stageName, stageData] of Object.entries(entry.stages)) {
   178→      if (stageData.durationHours !== null) {
   179→        if (!avgByStage[stageName]) avgByStage[stageName] = [];
   180→        (avgByStage[stageName] as unknown as number[]).push(stageData.durationHours);
   181→      }
   182→    }
   183→  }
   184→  const avgStageDurations: Record<string, number> = {};
   185→  for (const [k, v] of Object.entries(avgByStage)) {
   186→    const arr = v as unknown as number[];
   187→    avgStageDurations[k] = arr.length > 0 ? arr.reduce((s, n) => s + n, 0) / arr.length : 0;
   188→  }
   189→
   190→  return NextResponse.json({
   191→    success: true,
   192→    data: {
   193→      totalRequests: mrs.length,
   194→      convertedToWo: mrs.filter((m) => m.workOrder).length,
   195→      closed: completedWithWo.length,
   196→      avgTurnaroundHours: Math.round((avgTurnaround || 0) * 100) / 100,
   197→      avgStageDurations,
   198→      entries: lifecycleEntries,
   199→    },
   200→  });
   201→}

   203→// ── EXECUTION REPORT ─────────────────────────────────────────────────────────
   204→// WO completion rates, actual vs estimated, rework, team performance
   205→async function handleExecutionReport(
   206→  plantId: string | undefined,
   207→  from: Date | undefined,
   208→  to: Date | undefined,
   209→  priority: string | undefined,
   210→  assignee: string | undefined,
   211→  dateFilter: Record<string, unknown>,
   212→) {
   213→  const woWhere: Record<string, unknown> = { plantId: plantId || undefined };
   214→  if (priority) woWhere.priority = priority;
   215→  if (assignee) woWhere.assignedTo = assignee;
   216→  if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
   217→
   218→  const workOrders = await db.workOrder.findMany({
   219→    where: woWhere,
   220→    include: {
   221→      assignee: { select: { id: true, fullName: true, department: true } },
   222→      teamLeader: { select: { id: true, fullName: true } },
   223→      repairCompletion: true,
   224→      timeLogs: true,
   225→      materials: true,
   226→      statusHistory: true,
   227→    },
   228→    orderBy: { createdAt: 'desc' },
   229→    take: 300,
   230→  });
   231→
   232→  const total = workOrders.length;
   233→  const closed = workOrders.filter((wo) => wo.status === 'closed');
   234→  const completed = workOrders.filter((wo) => ['completed', 'verified', 'closed'].includes(wo.status));
   235→
   236→  // Completion by type
   237→  const byType: Record<string, { total: number; closed: number; rate: number; totalHours: number }> = {};
   238→  for (const wo of workOrders) {
   239→    if (!byType[wo.type]) byType[wo.type] = { total: 0, closed: 0, rate: 0, totalHours: 0 };
   240→    byType[wo.type].total++;
   241→    byType[wo.type].totalHours += wo.actualHours || 0;
   242→    if (wo.status === 'closed') byType[wo.type].closed++;
   243→  }
   244→  for (const t of Object.keys(byType)) {
   245→    byType[t].rate = byType[t].total > 0 ? byType[t].closed / byType[t].total : 0;
   246→  }
   247→
   248→  // Completion by priority
   249→  const byPriority: Record<string, { total: number; closed: number; rate: number }> = {};
   250→  for (const wo of workOrders) {
   251→    if (!byPriority[wo.priority]) byPriority[wo.priority] = { total: 0, closed: 0, rate: 0 };
   252→    byPriority[wo.priority].total++;
   253→    if (wo.status === 'closed') byPriority[wo.priority].closed++;
   254→  }
   255→  for (const p of Object.keys(byPriority)) {
   256→    byPriority[p].rate = byPriority[p].total > 0 ? byPriority[p].closed / byPriority[p].total : 0;
   257→  }
   258→
   259→  // Actual vs estimated hours
   260→  let totalEstimated = 0;
   261→  let totalActual = 0;
   262→  let withEstimate = 0;
   263→  for (const wo of workOrders) {
   264→    if (wo.estimatedHours) {
   265→      totalEstimated += wo.estimatedHours;
   266→      withEstimate++;
   267→    }
   268→    if (wo.actualHours) totalActual += wo.actualHours;
   269→  }
   270→  const avgEstimated = withEstimate > 0 ? totalEstimated / withEstimate : 0;
   271→  const avgActual = completed.length > 0 ? totalActual / completed.length : 0;
   272→  const avgVariance = withEstimate > 0 ? avgActual - avgEstimated : 0;
   273→
   274→  // Rework analysis
   275→  const reworkCount = workOrders.filter((wo) => {
   276→    const completion = wo.repairCompletion;
   277→    return completion && completion.reworkCount > 0;
   278→  }).length;
   279→  const reworkRate = total > 0 ? reworkCount / total : 0;
   280→  const totalReworkInstances = workOrders.reduce((sum, wo) => sum + (wo.repairCompletion?.reworkCount || 0), 0);
   281→
   282→  // Team performance metrics
   283→  const teamMetrics: Record<string, { fullName: string; total: number; closed: number; avgActualHours: number; rework: number }> = {};
   284→  for (const wo of workOrders) {
   285→    const techName = wo.assignee?.fullName || 'Unassigned';
   286→    if (!teamMetrics[wo.assignedTo || 'unassigned']) {
   287→      teamMetrics[wo.assignedTo || 'unassigned'] = { fullName: techName, total: 0, closed: 0, avgActualHours: 0, rework: 0 };
   288→    }
   289→    teamMetrics[wo.assignedTo || 'unassigned'].total++;
   290→    if (wo.status === 'closed') teamMetrics[wo.assignedTo || 'unassigned'].closed++;
   291→    teamMetrics[wo.assignedTo || 'unassigned'].rework += (wo.repairCompletion?.reworkCount || 0);
   292→  }
   293→  for (const tm of Object.values(teamMetrics)) {
   294→    tm.avgActualHours = tm.closed > 0 ? workOrders.filter((wo) => wo.assignedTo && wo.status === 'closed').reduce((sum) => sum + (workOrders.find((w) => w.assignedTo)?.actualHours || 0), 0) / tm.closed : 0;
   295→  }
   296→
   297→  // Convert byType Record to array format for frontend
   298→  const byTypeArray = Object.entries(byType).map(([type, data]) => ({
   299→    type,
   300→    count: data.total,
   301→    closed: data.closed,
   302→    rate: Math.round(data.rate * 10000) / 100,
   303→    avgHours: data.total > 0 ? Math.round((data.totalHours / data.total) * 100) / 100 : 0,
   304→  }));
   305→
   306→  // Convert byPriority Record to array format
   307→  const byPriorityArray = Object.entries(byPriority).map(([priority, data]) => ({
   308→    priority,
   309→    total: data.total,
   310→    closed: data.closed,
   311→    rate: Math.round(data.rate * 10000) / 100,
   312→  }));
   313→
   314→  return NextResponse.json({
   315→    success: true,
   316→    data: {
   317→      totalWOs: total,
   318→      completionRate: Math.round((total > 0 ? completed.length / total : 0) * 10000) / 100,
   319→      avgActualHours: Math.round(avgActual * 100) / 100,
   320→      reworkRate: Math.round(reworkRate * 10000) / 100,
   321→      byType: byTypeArray,
   322→      byPriority: byPriorityArray,
   323→      summary: {
   324→        total,
   325→        completed: completed.length,
   326→        closed: closed.length,
   327→        completionRate: total > 0 ? completed.length / total : 0,
   328→        closureRate: total > 0 ? closed.length / total : 0,
   329→      },
   330→      byPriorityObj: byPriority,
   331→      labor: {
   332→        avgEstimatedHours: Math.round(avgEstimated * 100) / 100,
   333→        avgActualHours: Math.round(avgActual * 100) / 100,
   334→        avgVarianceHours: Math.round(avgVariance * 100) / 100,
   335→        variancePercent: avgEstimated > 0 ? Math.round((avgVariance / avgEstimated) * 10000) / 100 : 0,
   336→      },
   337→      rework: {
   338→        reworkWos: reworkCount,
   339→        reworkRate: Math.round(reworkRate * 10000) / 100,
   340→        totalReworkInstances,
   341→      },
   342→      teamMetrics: Object.values(teamMetrics).sort((a, b) => b.closed - a.closed),
   343→    },
   344→  });
   345→}

   347→// ── TECHNICIAN PERFORMANCE REPORT ────────────────────────────────────────────
   348→async function handleTechnicianPerformanceReport(
   349→  plantId: string | undefined,
   350→  from: Date | undefined,
   351→  to: Date | undefined,
   352→  department: string | undefined,
   353→  dateFilter: Record<string, unknown>,
   354→) {
   355→  // Get all work orders in date range
   356→  const woWhere: Record<string, unknown> = { plantId: plantId || undefined };
   357→  if (department) woWhere.departmentId = department;
   358→  if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
   359→
   360→  const workOrders = await db.workOrder.findMany({
   361→    where: woWhere,
   362→    include: {
   363→      assignee: { select: { id: true, fullName: true, username: true, department: true, primaryTrade: true } },
   364→      repairCompletion: true,
   365→      timeLogs: true,
   366→    },
   367→    orderBy: { createdAt: 'desc' },
   368→    take: 500,
   369→  });
   370→
   371→  // Group by technician
   372→  const techMap: Record<string, {
   373→    user: { id: string; fullName: string; username: string; department: string | null; primaryTrade: string | null };
   374→    woCount: number;
   375→    closedCount: number;
   376→    totalTimeLogged: number;
   377→    totalEstimated: number;
   378→    reworkCount: number;
   379→    avgTimePerWo: number;
   380→    timeAccuracy: number;
   381→  }> = {};
   382→
   383→  for (const wo of workOrders) {
   384→    const uid = wo.assignedTo;
   385→    if (!uid) continue;
   386→
   387→    if (!techMap[uid]) {
   388→      techMap[uid] = {
   389→        user: wo.assignee!,
   390→        woCount: 0,
   391→        closedCount: 0,
   392→        totalTimeLogged: 0,
   393→        totalEstimated: 0,
   394→        reworkCount: 0,
   395→        avgTimePerWo: 0,
   396→        timeAccuracy: 0,
   397→      };
   398→    }
   399→
   400→    techMap[uid].woCount++;
   401→    if (wo.status === 'closed') techMap[uid].closedCount++;
   402→    techMap[uid].totalTimeLogged += wo.actualHours || 0;
   403→    techMap[uid].totalEstimated += wo.estimatedHours || 0;
   404→    techMap[uid].reworkCount += wo.repairCompletion?.reworkCount || 0;
   405→  }
   406→
   407→  // Calculate per-technician metrics
   408→  const technicians = Object.values(techMap).map((tech) => {
   409→    const avgTimePerWo = tech.woCount > 0 ? tech.totalTimeLogged / tech.woCount : 0;
   410→    const timeAccuracy = tech.totalEstimated > 0
   411→      ? 100 - Math.abs(tech.totalTimeLogged - tech.totalEstimated) / tech.totalEstimated * 100
   412→      : 0;
   413→
   414→    return {
   415→      ...tech,
   416→      avgTimePerWo: Math.round(avgTimePerWo * 100) / 100,
   417→      timeAccuracy: Math.max(0, Math.round(timeAccuracy * 100) / 100),
   418→      reworkRate: tech.woCount > 0 ? Math.round((tech.reworkCount / tech.woCount) * 10000) / 100 : 0,
   419→      completionRate: tech.woCount > 0 ? Math.round((tech.closedCount / tech.woCount) * 10000) / 100 : 0,
   420→    };
   421→  }).sort((a, b) => b.closedCount - a.closedCount);
   422→
   423→  return NextResponse.json({
   424→    success: true,
   425→    data: {
   426→      totalTechnicians: technicians.length,
   427→      totalWorkOrders: workOrders.length,
   428→      technicians,
   429→    },
   430→  });
   431→}

   433→// ── MATERIALS REPORT ─────────────────────────────────────────────────────────
   434→async function handleMaterialsReport(
   435→  plantId: string | undefined,
   436→  from: Date | undefined,
   437→  to: Date | undefined,
   438→  dateFilter: Record<string, unknown>,
   439→) {
   440→  const woWhere: Record<string, unknown> = { plantId: plantId || undefined };
   441→  if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
   442→
   443→  const workOrders = await db.workOrder.findMany({
   444→    where: woWhere,
   445→    include: {
   446→      materials: true,
   447→      repairCompletion: true,
   448→    },
   449→    orderBy: { createdAt: 'desc' },
   450→    take: 300,
   451→  });
   452→
   453→  // Material cost by WO
   454→  const materialCostByWo = workOrders.map((wo) => {
   455→    const totalMaterialCost = wo.materials.reduce((sum, m) => sum + (m.totalCost || 0), 0);
   456→    const totalIssued = wo.materials.filter((m) => ['issued', 'returned'].includes(m.status)).length;
   457→    const totalReturned = wo.materials.filter((m) => m.status === 'returned').length;
   458→
   459→    return {
   460→      workOrderId: wo.id,
   461→      woNumber: wo.woNumber,
   462→      title: wo.title,
   463→      materialCount: wo.materials.length,
   464→      issuedCount: totalIssued,
   465→      returnedCount: totalReturned,
   466→      totalMaterialCost: Math.round(totalMaterialCost * 100) / 100,
   467→      completionCost: wo.repairCompletion?.totalMaterialCost || 0,
   468→    };
   469→  }).filter((m) => m.materialCount > 0);
   470→
   471→  // Aggregate cost by category (via items)
   472→  const totalMaterialCost = materialCostByWo.reduce((sum, m) => sum + m.totalMaterialCost, 0);
   473→
   474→  // Spare part return analysis
   475→  const sprWhere: Record<string, unknown> = { plantId: plantId || undefined };
   476→  if (Object.keys(dateFilter).length > 0) sprWhere.createdAt = dateFilter;
   477→
   478→  const sparePartReturns = await db.sparePartReturn.findMany({ where: sprWhere });
   479→  const totalReturns = sparePartReturns.length;
   480→  const returnedToStore = sparePartReturns.filter((r) => r.status === 'returned_to_store').length;
   481→  const disposed = sparePartReturns.filter((r) => r.status === 'disposed').length;
   482→  const totalRefurbCost = sparePartReturns.reduce((sum, r) => sum + (r.actualRefurbCost || 0), 0);
   483→  const returnRate = totalReturns > 0 ? Math.round((returnedToStore / totalReturns) * 10000) / 100 : 0;
   484→
   485→  // Top materials by cost
   486→  const itemIds = [...new Set(workOrders.flatMap((wo) => wo.materials.map((m) => m.itemId).filter(Boolean) as string[]))];
   487→  const topItems = itemIds.length > 0 ? await db.inventoryItem.findMany({
   488→    where: { id: { in: itemIds } },
   489→    select: { id: true, itemCode: true, name: true, category: true, unitCost: true },
   490→    take: 20,
   491→  }) : [];
   492→
   493→  return NextResponse.json({
   494→    success: true,
   495→    data: {

   514→// ── DOWNTIME REPORT ──────────────────────────────────────────────────────────
   515→async function handleDowntimeReport(
   516→  plantId: string | undefined,
   517→  from: Date | undefined,
   518→  to: Date | undefined,
   519→  dateFilter: Record<string, unknown>,
   520→) {
   521→  const dtWhere: Record<string, unknown> = { plantId: plantId || undefined };
   522→  if (Object.keys(dateFilter).length > 0) dtWhere.createdAt = dateFilter;
   523→
   524→  const downtimes = await db.workOrderDowntime.findMany({
   525→    where: dtWhere,
   526→    include: {
   527→      workOrder: { select: { id: true, woNumber: true, title: true, status: true } },
   528→      createdBy: { select: { id: true, fullName: true } },
   529→    },
   530→    orderBy: { createdAt: 'desc' },
   531→    take: 500,
   532→  });
   533→
   534→  const total = downtimes.length;
   535→  const totalDowntimeMinutes = downtimes.reduce((sum, d) => sum + (d.durationMinutes || 0), 0);
   536→  const totalDowntimeHours = Math.round((totalDowntimeMinutes / 60) * 100) / 100;
   537→  const avgDurationHours = total > 0 ? Math.round((totalDowntimeMinutes / 60 / total) * 100) / 100 : 0;
   538→  const totalProductionLoss = downtimes.reduce((sum, d) => sum + (d.productionLoss || 0), 0);
   539→
   540→  // By asset
   541→  const byAsset: Record<string, { assetName: string; count: number; totalHours: number; totalLoss: number }> = {};
   542→  for (const dt of downtimes) {
   543→    const key = dt.assetId || 'unknown';
   544→    if (!byAsset[key]) byAsset[key] = { assetName: dt.assetName, count: 0, totalHours: 0, totalLoss: 0 };
   545→    byAsset[key].count++;
   546→    byAsset[key].totalHours += (dt.durationMinutes || 0) / 60;
   547→    byAsset[key].totalLoss += dt.productionLoss || 0;
   548→  }
   549→
   550→  // By category
   551→  const byCategory: Record<string, { count: number; totalHours: number }> = {};
   552→  for (const dt of downtimes) {
   553→    if (!byCategory[dt.category]) byCategory[dt.category] = { count: 0, totalHours: 0 };
   554→    byCategory[dt.category].count++;
   555→    byCategory[dt.category].totalHours += (dt.durationMinutes || 0) / 60;
   556→  }
   557→
   558→  // By impact level
   559→  const byImpactLevel: Record<string, { count: number; totalHours: number }> = {};
   560→  for (const dt of downtimes) {
   561→    if (!byImpactLevel[dt.impactLevel]) byImpactLevel[dt.impactLevel] = { count: 0, totalHours: 0 };
   562→    byImpactLevel[dt.impactLevel].count++;
   563→    byImpactLevel[dt.impactLevel].totalHours += (dt.durationMinutes || 0) / 60;
   564→  }
   565→
   566→  const topAssetsByDowntime = Object.entries(byAsset)
   567→    .map(([assetId, data]) => ({ assetId, ...data, totalHours: Math.round(data.totalHours * 100) / 100 }))
   568→    .sort((a, b) => b.totalHours - a.totalHours)
   569→    .slice(0, 10);
   570→
   571→  return NextResponse.json({
   572→    success: true,
   573→    data: {
   574→      summary: {
   575→        totalEvents: total,
   576→        totalDowntimeHours,
   577→        avgDurationHours,
   578→        totalProductionLoss: Math.round(totalProductionLoss * 100) / 100,
   579→        avgLossPerHour: totalDowntimeHours > 0 ? Math.round((totalProductionLoss / totalDowntimeHours) * 100) / 100 : 0,
   580→      },
   581→      byAsset: topAssetsByDowntime,
   582→      byCategory: Object.fromEntries(Object.entries(byCategory).map(([k, v]) => [k, { ...v, totalHours: Math.round(v.totalHours * 100) / 100 }])),
   583→      byImpactLevel: Object.fromEntries(Object.entries(byImpactLevel).map(([k, v]) => [k, { ...v, totalHours: Math.round(v.totalHours * 100) / 100 }])),
   584→    },
   585→  });
   586→}

   588→// ── TOOLS REPORT ─────────────────────────────────────────────────────────────
   589→async function handleToolsReport(
   590→  plantId: string | undefined,
   591→  from: Date | undefined,
   592→  to: Date | undefined,
   593→  dateFilter: Record<string, unknown>,
   594→) {
   595→  // Tool damage analysis
   596→  const dtrWhere: Record<string, unknown> = { plantId: plantId || undefined };
   597→  if (Object.keys(dateFilter).length > 0) dtrWhere.createdAt = dateFilter;
   598→
   599→  const damagedReports = await db.damagedToolReport.findMany({
   600→    where: dtrWhere,
   601→    include: {
   602→      tool: { select: { id: true, toolCode: true, name: true, category: true, purchaseCost: true } },
   603→    },
   604→    orderBy: { createdAt: 'desc' },
   605→    take: 300,
   606→  });
   607→
   608→  const totalDamageReports = damagedReports.length;
   609→  const totalRepairCost = damagedReports.reduce((sum, d) => sum + (d.actualRepairCost || 0), 0);
   610→  const writtenOff = damagedReports.filter((d) => d.status === 'written_off').length;
   611→  const repaired = damagedReports.filter((d) => d.status === 'repaired').length;
   612→  const inProgress = damagedReports.filter((d) => d.status === 'repair_in_progress').length;
   613→
   614→  // By damage type
   615→  const byDamageType: Record<string, { count: number; cost: number }> = {};
   616→  for (const dr of damagedReports) {
   617→    if (!byDamageType[dr.damageType]) byDamageType[dr.damageType] = { count: 0, cost: 0 };
   618→    byDamageType[dr.damageType].count++;
   619→    byDamageType[dr.damageType].cost += dr.actualRepairCost || 0;
   620→  }
   621→
   622→  // By severity
   623→  const bySeverity: Record<string, { count: number; cost: number }> = {};
   624→  for (const dr of damagedReports) {
   625→    if (!bySeverity[dr.damageSeverity]) bySeverity[dr.damageSeverity] = { count: 0, cost: 0 };
   626→    bySeverity[dr.damageSeverity].count++;
   627→    bySeverity[dr.damageSeverity].cost += dr.actualRepairCost || 0;
   628→  }
   629→
   630→  // By tool category
   631→  const byCategory: Record<string, { count: number; cost: number }> = {};
   632→  for (const dr of damagedReports) {
   633→    const cat = dr.tool?.category || 'Unknown';
   634→    if (!byCategory[cat]) byCategory[cat] = { count: 0, cost: 0 };
   635→    byCategory[cat].count++;
   636→    byCategory[cat].cost += dr.actualRepairCost || 0;
   637→  }
   638→
   639→  // Tool transfer frequency
   640→  const txWhere: Record<string, unknown> = {};
   641→  if (from || to) txWhere.createdAt = dateFilter;
   642→
   643→  const transfers = await db.toolTransferRequest.findMany({
   644→    where: { ...txWhere, plantId: plantId || undefined },
   645→    orderBy: { createdAt: 'desc' },
   646→    take: 200,
   647→  });
   648→
   649→  const totalTransfers = transfers.length;
   650→  const completedTransfers = transfers.filter((t) => t.status === 'transferred').length;
   651→
   652→  // Most damaged tools
   653→  const damagedToolsMap: Record<string, { toolName: string; toolCode: string; category: string; damageCount: number; totalCost: number }> = {};
   654→  for (const dr of damagedReports) {
   655→    if (!dr.tool) continue;
   656→    const key = dr.toolId;
   657→    if (!damagedToolsMap[key]) {
   658→      damagedToolsMap[key] = { toolName: dr.tool.name, toolCode: dr.tool.toolCode, category: dr.tool.category, damageCount: 0, totalCost: 0 };
   659→    }
   660→    damagedToolsMap[key].damageCount++;
   661→    damagedToolsMap[key].totalCost += dr.actualRepairCost || 0;
   662→  }
   663→
   664→  const mostDamagedTools = Object.values(damagedToolsMap)
   665→    .sort((a, b) => b.damageCount - a.damageCount)
   666→    .slice(0, 10);
   667→
   668→  return NextResponse.json({
   669→    success: true,
   670→    data: {
   671→      summary: {
   672→        totalDamageReports,
   673→        totalRepairCost: Math.round(totalRepairCost * 100) / 100,
   674→        repaired,
   675→        writtenOff,
   676→        inProgress,
   677→        avgRepairCost: totalDamageReports > 0 ? Math.round((totalRepairCost / totalDamageReports) * 100) / 100 : 0,
   678→        totalTransfers,
   679→        completedTransfers,
   680→      },
   681→      byDamageType: Object.fromEntries(Object.entries(byDamageType).map(([k, v]) => [k, { ...v, cost: Math.round(v.cost * 100) / 100 }])),
   682→      bySeverity: Object.fromEntries(Object.entries(bySeverity).map(([k, v]) => [k, { ...v, cost: Math.round(v.cost * 100) / 100 }])),
   683→      byCategory: Object.fromEntries(Object.entries(byCategory).map(([k, v]) => [k, { ...v, cost: Math.round(v.cost * 100) / 100 }])),
   684→      mostDamagedTools,
   685→    },
   686→  });
   687→}
