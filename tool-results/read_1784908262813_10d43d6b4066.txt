   163→function formatTimeAgo(dateStr: string | null): string {
   164→  if (!dateStr) return 'Never';
   165→  const diff = Date.now() - new Date(dateStr).getTime();
   166→  const mins = Math.floor(diff / 60000);
   167→  if (mins < 1) return 'Just now';
   168→  if (mins < 60) return `${mins}m ago`;
   169→  const hrs = Math.floor(mins / 60);
   170→  if (hrs < 24) return `${hrs}h ${mins % 60}m ago`;
   171→  const days = Math.floor(hrs / 24);
   172→  return `${days}d ago`;
   173→}
   174→
   175→function formatUptime(seconds: number): string {
   176→  if (!seconds || seconds <= 0) return 'N/A';
   177→  const hrs = Math.floor(seconds / 3600);
   178→  const mins = Math.floor((seconds % 3600) / 60);
   179→  return `${hrs}h ${mins}m`;
   180→}

   318→function ProtocolBreakdown({ sources }: { sources: DataSource[] }) {
   319→  const breakdown: Record<string, { total: number; connected: number; error: number }> = {};
   320→  for (const s of sources) {
   321→    if (!breakdown[s.sourceType]) breakdown[s.sourceType] = { total: 0, connected: 0, error: 0 };
   322→    breakdown[s.sourceType].total++;
   323→    if (s.status === 'connected') breakdown[s.sourceType].connected++;
   324→    if (s.status === 'error') breakdown[s.sourceType].error++;
   325→  }
   326→
   327→  return (
   328→    <Card>
   329→      <CardHeader className="pb-3">
   330→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   331→          <Radio className="h-4 w-4 text-emerald-600" />
   332→          Protocol Breakdown
   333→        </CardTitle>
   334→      </CardHeader>
   335→      <CardContent>
   336→        <div className="space-y-2">
   337→          {Object.entries(breakdown).length === 0 && (
   338→            <p className="text-xs text-muted-foreground py-2">No sources configured</p>
   339→          )}
   340→          {Object.entries(breakdown).map(([type, counts]) => (
   341→            <div key={type} className="flex items-center justify-between p-2 rounded-lg bg-muted/50">
   342→              <div className="flex items-center gap-2">
   343→                <span className={`inline-flex items-center px-2 py-0.5 rounded text-[10px] font-semibold ${protocolColors[type] || 'bg-gray-100 text-gray-800'}`}>
   344→                  {protocolLabels[type] || type}
   345→                </span>
   346→                <span className="text-xs text-muted-foreground">{counts.total} source{counts.total !== 1 ? 's' : ''}</span>
   347→              </div>
   348→              <div className="flex items-center gap-1.5">
   349→                <span className="flex items-center gap-0.5 text-[10px] text-emerald-600 font-medium">
   350→                  <CheckCircle2 className="h-3 w-3" />{counts.connected}
   351→                </span>
   352→                {counts.error > 0 && (
   353→                  <span className="flex items-center gap-0.5 text-[10px] text-red-500 font-medium">
   354→                    <XCircle className="h-3 w-3" />{counts.error}
   355→                  </span>
   356→                )}
   357→              </div>
   358→            </div>
   359→          ))}
   360→        </div>
   361→      </CardContent>
   362→    </Card>
   363→  );
   364→}
