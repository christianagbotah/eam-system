   167→function autoInterval(range: TimeRange): string {
   168→  switch (range) {
   169→    case '1h': return '1m';
   170→    case '6h': return '5m';
   171→    case '24h': return '1h';
   172→    case '7d': return '1h';
   173→    case '30d': return '1d';
   174→    case '90d': return '1d';
   175→    case '1y': return '1w';
   176→    default: return '1h';
   177→  }
   178→}
   179→
   180→function formatNumber(n: number): string {
   181→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`;
   182→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
   183→  return n.toLocaleString();
   184→}
   185→
   186→function timeAgo(dateStr: string | null): string {
   187→  if (!dateStr) return 'Never';
   188→  const diff = Date.now() - new Date(dateStr).getTime();
   189→  const mins = Math.floor(diff / 60000);
   190→  if (mins < 1) return 'Just now';
   191→  if (mins < 60) return `${mins}m ago`;
   192→  const hrs = Math.floor(mins / 60);
   193→  if (hrs < 24) return `${hrs}h ago`;
   194→  const days = Math.floor(hrs / 24);
   195→  return `${days}d ago`;
   196→}
   197→
   198→function formatTimestamp(ts: string): string {
   199→  const d = new Date(ts);
   200→  return d.toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   201→}

   781→function ComparisonView({ tagMonitor }: { tagMonitor: TagMonitor[] }) {
   782→  const [selectedSources, setSelectedSources] = useState<string[]>([]);
   783→  const [timeRange, setTimeRange] = useState<TimeRange>('24h');
   784→  const [chartData, setChartData] = useState<Record<string, number | null>[]>([]);
   785→  const [chartLoading, setChartLoading] = useState(false);
   786→
   787→  const toggleSource = (srcId: string) => {
   788→    setSelectedSources(prev => {
   789→      if (prev.includes(srcId)) return prev.filter(s => s !== srcId);
   790→      if (prev.length >= 3) return prev;
   791→      return [...prev, srcId];
   792→    });
   793→  };
   794→
   795→  const fetchComparison = useCallback(async () => {
   796→    if (selectedSources.length === 0) { setChartData([]); return; }
   797→    setChartLoading(true);
   798→    try {
   799→      const range = TIME_RANGES.find(r => r.value === timeRange)!;
   800→      const from = new Date(Date.now() - range.ms);
   801→      const to = new Date();
   802→      const interval = autoInterval(timeRange);
   803→
   804→      const res = await api.post('/api/historian/aggregate', {
   805→        operation: 'multi-source',
   806→        sourceIds: selectedSources,
   807→        from: from.toISOString(),
   808→        to: to.toISOString(),
   809→        interval,
   810→      });
   811→      if (res.success) {
   812→        const raw = res.data as Array<{ timestamp: string; sourceId: string; avg: number | null }>;
   813→        // Pivot: one row per timestamp with columns per source
   814→        const timeMap = new Map<string, Record<string, number | null>>();
   815→        for (const d of raw) {
   816→          const key = d.timestamp;
   817→          if (!timeMap.has(key)) timeMap.set(key, { time: new Date(key).getTime() });
   818→          timeMap.get(key)![d.sourceId] = d.avg;
   819→        }
   820→        const pivoted = [...timeMap.values()]
   821→          .sort((a, b) => (a.time! as number) - (b.time! as number))
   822→          .map(d => ({
   823→            time: new Date(d.time! as number).toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' }),
   824→            ...Object.fromEntries(selectedSources.map(s => [s, d[s] ?? null])),
   825→          }));
   826→        setChartData(pivoted);
   827→      }
   828→    } catch (e) {
   829→      console.error('Failed to fetch comparison data', e);
   830→    } finally {
   831→      setChartLoading(false);
   832→    }
   833→  }, [selectedSources, timeRange]);
   834→
   835→  useEffect(() => { fetchComparison(); }, [fetchComparison]);
   836→
   837→  const chartConfig = useMemo(() => {
   838→    const cfg: Record<string, { label: string; color: string }> = {};
   839→    selectedSources.forEach((s, i) => { cfg[s] = { label: s.slice(0, 20), color: COMPARISON_COLORS[i] }; });
   840→    return cfg;
   841→  }, [selectedSources]);
   842→
   843→  return (
   844→    <Card className="border-0 shadow-sm">
   845→      <CardHeader className="pb-3 px-5 pt-5">
   846→        <div className="flex flex-col gap-3">
   847→          <CardTitle className="text-sm font-semibold flex items-center gap-2">
   848→            <ArrowUpDown className="h-4 w-4 text-blue-500" />
   849→            Comparison View
   850→            <Badge variant="outline" className="text-[10px]">{selectedSources.length}/3 selected</Badge>
   851→          </CardTitle>
   852→          <div className="flex flex-wrap gap-1.5">
   853→            {tagMonitor.slice(0, 20).map((t, i) => {
   854→              const isSelected = selectedSources.includes(t.sourceId);
   855→              const colorIdx = selectedSources.indexOf(t.sourceId);
   856→              return (
   857→                <Button key={t.sourceId} variant={isSelected ? 'default' : 'outline'} size="sm"
   858→                  className={`h-7 text-[11px] ${isSelected ? '' : ''}`}
   859→                  style={isSelected ? { backgroundColor: COMPARISON_COLORS[colorIdx], borderColor: COMPARISON_COLORS[colorIdx] } : {}}
   860→                  onClick={() => toggleSource(t.sourceId)}>
   861→                  {t.sourceId.slice(0, 18)}
   862→                </Button>
   863→              );
   864→            })}
   865→          </div>
   866→          <div className="flex gap-1">
   867→            {TIME_RANGES.map(r => (
   868→              <Button key={r.value} variant={timeRange === r.value ? 'default' : 'outline'} size="sm"
   869→                className={`h-7 px-2 text-[11px] ${timeRange === r.value ? 'bg-blue-600 hover:bg-blue-700' : ''}`}
   870→                onClick={() => setTimeRange(r.value)}>
   871→                {r.label}
   872→              </Button>
   873→            ))}
   874→          </div>
   875→        </div>
   876→      </CardHeader>
   877→      <CardContent className="px-5 pb-5">
   878→        {chartLoading && chartData.length === 0 ? (
   879→          <Skeleton className="h-72 rounded-lg" />
   880→        ) : chartData.length === 0 ? (
   881→          <div className="flex items-center justify-center h-72 text-muted-foreground text-sm">
   882→            {selectedSources.length === 0 ? 'Select 2-3 tags to compare' : 'No data available'}
   883→          </div>
   884→        ) : (
   885→          <ChartContainer config={chartConfig} className="h-72">
   886→            <LineChart data={chartData}>
   887→              <CartesianGrid strokeDasharray="3 3" />
   888→              <XAxis dataKey="time" tick={{ fontSize: 10 }} interval="preserveStartEnd" />
   889→              <YAxis tick={{ fontSize: 10 }} />
   890→              <ChartTooltip content={<ChartTooltipContent />} />
   891→              <ChartLegend content={<ChartLegendContent />} />
   892→              {selectedSources.map((s, i) => (
   893→                <Line key={s} type="monotone" dataKey={s} stroke={COMPARISON_COLORS[i]} strokeWidth={2} dot={false} connectNulls />
   894→              ))}
   895→            </LineChart>
   896→          </ChartContainer>
   897→        )}
   898→      </CardContent>
   899→    </Card>
   900→  );
