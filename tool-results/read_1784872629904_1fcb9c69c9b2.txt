     1→     1→import { NextRequest, NextResponse } from 'next/server';
     2→     2→import { db } from '@/lib/db';
     3→     3→import { getSession, hasPermission, isAdmin, hasAnyPermission } from '@/lib/auth';
     4→     4→import { getPlantScope, getPlantFilterWhere } from '@/lib/plant-scope';
     5→     5→import { generateReportPDF } from '@/lib/generate-report-pdf';
     6→     6→
     7→     7→// GET /api/work-orders/reports
     8→     8→// Comprehensive WO reports: downtime, response time, breakdowns, man hours, materials, failure rate, stoppages
     9→     9→export async function GET(request: NextRequest) {
    10→    10→  try {
    11→    11→    const session = getSession(request);
    12→    12→    if (!session) {
    13→    13→      return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
    14→    14→    }
    15→    15→    // Permission gate: require reports.view, work_orders.dashboard, or admin
    16→    16→    if (!hasAnyPermission(session, ['reports.view', 'reports.export', 'work_orders.dashboard', 'analytics.view']) && !isAdmin(session)) {
    17→    17→      return NextResponse.json({ success: false, error: 'Insufficient permissions: reports.view required' }, { status: 403 });
    18→    18→    }
    19→    19→
    20→    20→    const { searchParams } = new URL(request.url);
    21→    21→    const from = searchParams.get('from');
    22→    22→    const to = searchParams.get('to');
    23→    23→    const department = searchParams.get('department');
    24→    24→    const trade = searchParams.get('trade');
    25→    25→    const priority = searchParams.get('priority');
    26→    26→    const format = searchParams.get('format');
    27→    27→    const moduleFilter = searchParams.get('moduleFilter') || 'all';
    28→    28→
    29→    29→    const plantScope = await getPlantScope(request, session);
    30→    30→    const plantFilter = getPlantFilterWhere(plantScope);
    31→    31→
    32→    32→    // Date filter
    33→    33→    const dateFilter: Record<string, unknown> = {};
    34→    34→    if (from) dateFilter.gte = new Date(from + 'T00:00:00');
    35→    35→    if (to) dateFilter.lte = new Date(to + 'T23:59:59');
    36→    36→
    37→    37→    const woWhere: Record<string, unknown> = {
    38→    38→      ...plantFilter,
    39→    39→      status: { notIn: ['draft', 'cancelled'] },
    40→    40→    };
    41→    41→    if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
    42→    42→    if (department) woWhere.departmentId = department;
    43→    43→    if (trade) woWhere.tradeActivity = trade;
    44→    44→    if (priority) woWhere.priority = priority;
    45→    45→    if (moduleFilter === 'repairs') woWhere.type = { in: ['corrective', 'emergency'] };
    46→    46→    if (moduleFilter === 'pm') woWhere.type = { in: ['preventive'] };
    47→    47→
    48→    48→    // Fetch all relevant WOs with relations
    49→    49→    const workOrders = await db.workOrder.findMany({
    50→    50→      where: Object.keys(woWhere).length > 0 ? woWhere : undefined,
    51→    51→      include: {
    52→    52→        assignee: { select: { id: true, fullName: true, department: true } },
    53→    53→        teamLeader: { select: { id: true, fullName: true } },
    54→    54→        materials: true,
    55→    55→        timeLogs: true,
    56→    56→        workOrderDowntimes: true,
    57→    57→        repairCompletion: true,
    58→    58→        statusHistory: { orderBy: { createdAt: 'asc' } },
    59→    59→        taskExecutions: true,
    60→    60→        failureRecords: true,
    61→    61→      },
    62→    62→      orderBy: { createdAt: 'desc' },
    63→    63→    });
    64→    64→
    65→    65→    // Fetch enriched asset data for all referenced assets
    66→    66→    const assetIds = [...new Set(workOrders.map(wo => wo.assetId).filter((id): id is string => !!id))];
    67→    67→    const assets = assetIds.length > 0 ? await db.asset.findMany({
    68→    68→      where: { id: { in: assetIds } },
    69→    69→      include: { category: { select: { name: true } } },
    70→    70→    }) : [];
    71→    71→    const assetMap = new Map(assets.map(a => [a.id, a]));
    72→    72→
    73→    73→    // Fetch enriched inventory item data for all referenced materials
    74→    74→    const itemIds = [...new Set(workOrders.flatMap(wo =>
    75→    75→      (wo.materials || []).map(m => m.itemId).filter((id): id is string => !!id)
    76→    76→    ))];
    77→    77→    const inventoryItems = itemIds.length > 0 ? await db.inventoryItem.findMany({
    78→    78→      where: { id: { in: itemIds } },
    79→    79→      select: { id: true, itemCode: true, name: true, unitOfMeasure: true, supplier: true, supplierPartNumber: true, binLocation: true, shelfLocation: true, specification: true, currentStock: true },
    80→    80→    }) : [];
    81→    81→    const itemMap = new Map(inventoryItems.map(i => [i.id, i]));
    82→    82→
    83→    83→    // Helper: get enriched asset details from a work order
    84→    84→    function getAssetDetails(wo: { assetId?: string | null; assetName?: string | null }) {
    85→    85→      const asset = wo.assetId ? assetMap.get(wo.assetId) : null;
    86→    86→      return {
    87→    87→        assetId: wo.assetId || null,
    88→    88→        assetName: wo.assetName || asset?.name || 'Unassigned',
    89→    89→        assetTag: asset?.assetTag || null,
    90→    90→        manufacturer: asset?.manufacturer || null,
    91→    91→        model: asset?.model || null,
    92→    92→        serialNumber: asset?.serialNumber || null,
    93→    93→        category: asset?.category?.name || null,
    94→    94→        criticality: asset?.criticality || null,
    95→    95→        condition: asset?.condition || null,
    96→    96→        location: asset?.location || null,
    97→    97→        building: asset?.building || null,
    98→    98→        floor: asset?.floor || null,
    99→    99→        area: asset?.area || null,
   100→   100→        purchaseCost: asset?.purchaseCost || null,
   101→   101→        currentValue: asset?.currentValue || null,
   102→   102→      };
   103→   103→    }
   104→   104→
   105→   105→    // Build itemName -> first itemId mapping for material enrichment
   106→   106→    const itemNameToItemId: Record<string, string> = {};
   107→   107→    for (const wo of workOrders) {
   108→   108→      for (const mat of (wo.materials || [])) {
   109→   109→        if (mat.itemId && mat.itemName && !itemNameToItemId[mat.itemName]) {
   110→   110→          itemNameToItemId[mat.itemName] = mat.itemId;
   111→   111→        }
   112→   112→      }
   113→   113→    }
   114→   114→
   115→   115→    // Helper: get enriched inventory item details
   116→   116→    function getItemDetails(itemName: string) {
   117→   117→      const itemId = itemNameToItemId[itemName];
   118→   118→      const item = itemId ? itemMap.get(itemId) : null;
   119→   119→      return {
   120→   120→        itemCode: item?.itemCode || null,
   121→   121→        unitOfMeasure: item?.unitOfMeasure || null,
   122→   122→        supplier: item?.supplier || null,
   123→   123→        supplierPartNumber: item?.supplierPartNumber || null,
   124→   124→        binLocation: item?.binLocation || null,
   125→   125→        shelfLocation: item?.shelfLocation || null,
   126→   126→        specification: item?.specification || null,
   127→   127→        currentStock: item?.currentStock || null,
   128→   128→      };
   129→   129→    }
   130→   130→
   131→   131→    const total = workOrders.length;
   132→   132→    const closedWOs = workOrders.filter(wo => wo.status === 'closed');
   133→   133→    const completedWOs = workOrders.filter(wo => ['completed', 'verified', 'closed'].includes(wo.status));
   134→   134→    const breakdownWOs = workOrders.filter(wo => ['emergency', 'corrective'].includes(wo.type));
   135→   135→
   136→   136→    // ========== 1. SUMMARY KPIs ==========
   137→   137→    const summary = {
   138→   138→      totalWOs: total,
   139→   139→      completedWOs: completedWOs.length,
   140→   140→      closedWOs: closedWOs.length,
   141→   141→      completionRate: total > 0 ? Math.round((completedWOs.length / total) * 10000) / 100 : 0,
   142→   142→      breakdownCount: breakdownWOs.length,
   143→   143→      breakdownRate: total > 0 ? Math.round((breakdownWOs.length / total) * 10000) / 100 : 0,
   144→   144→    };
   145→   145→
   146→   146→    // ========== 2. DOWNTIME (per trade) ==========
   147→   147→    const downtimeByTrade: Record<string, { totalMinutes: number; events: number; productionLoss: number }> = {};
   148→   148→    for (const wo of workOrders) {
   149→   149→      const dtEvents = wo.workOrderDowntimes || [];
   150→   150→      const trade = wo.tradeActivity || 'Unassigned';
   151→   151→      for (const dt of dtEvents) {
   152→   152→        if (!downtimeByTrade[trade]) downtimeByTrade[trade] = { totalMinutes: 0, events: 0, productionLoss: 0 };
   153→   153→        downtimeByTrade[trade].totalMinutes += dt.durationMinutes;
   154→   154→        downtimeByTrade[trade].events++;
   155→   155→        downtimeByTrade[trade].productionLoss += dt.productionLoss || 0;
   156→   156→      }
   157→   157→    }
   158→   158→    const totalDowntimeMinutes = Object.values(downtimeByTrade).reduce((s, d) => s + d.totalMinutes, 0);
   159→   159→    const totalDowntimeEvents = Object.values(downtimeByTrade).reduce((s, d) => s + d.events, 0);
   160→   160→    const totalProductionLoss = Object.values(downtimeByTrade).reduce((s, d) => s + d.productionLoss, 0);
   161→   161→
   162→   162→    const downtimeByTradeArray = Object.entries(downtimeByTrade)
   163→   163→      .map(([tradeName, data]) => ({
   164→   164→        trade: tradeName,
   165→   165→        totalHours: Math.round((data.totalMinutes / 60) * 100) / 100,
   166→   166→        totalMinutes: data.totalMinutes,
   167→   167→        events: data.events,
   168→   168→        productionLoss: data.productionLoss,
   169→   169→      }))
   170→   170→      .sort((a, b) => b.totalHours - a.totalHours);
   171→   171→
   172→   172→    // Downtime by month
   173→   173→    const downtimeByMonth: Record<string, { minutes: number; events: number }> = {};
   174→   174→    for (const wo of workOrders) {
   175→   175→      for (const dt of (wo.workOrderDowntimes || [])) {
   176→   176→        const month = dt.downtimeStart?.toISOString().slice(0, 7) || 'unknown';
   177→   177→        if (!downtimeByMonth[month]) downtimeByMonth[month] = { minutes: 0, events: 0 };
   178→   178→        downtimeByMonth[month].minutes += dt.durationMinutes;
   179→   179→        downtimeByMonth[month].events++;
   180→   180→      }
   181→   181→    }
   182→   182→    const downtimeByMonthArray = Object.entries(downtimeByMonth)
   183→   183→      .map(([month, data]) => ({
   184→   184→        month,
   185→   185→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   186→   186→        events: data.events,
   187→   187→      }))
   188→   188→      .sort((a, b) => a.month.localeCompare(b.month));
   189→   189→
   190→   190→    // Downtime by category
   191→   191→    const downtimeByCategory: Record<string, { minutes: number; events: number }> = {};
   192→   192→    for (const wo of workOrders) {
   193→   193→      for (const dt of (wo.workOrderDowntimes || [])) {
   194→   194→        const cat = dt.category || 'unplanned';
   195→   195→        if (!downtimeByCategory[cat]) downtimeByCategory[cat] = { minutes: 0, events: 0 };
   196→   196→        downtimeByCategory[cat].minutes += dt.durationMinutes;
   197→   197→        downtimeByCategory[cat].events++;
   198→   198→      }
   199→   199→    }
   200→   200→    const downtimeByCategoryArray = Object.entries(downtimeByCategory)
   201→   201→      .map(([category, data]) => ({
   202→   202→        category,
   203→   203→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   204→   204→        events: data.events,
   205→   205→      }))
   206→   206→      .sort((a, b) => b.totalHours - a.totalHours);
   207→   207→
   208→   208→    // ========== 3. RESPONSE TIME ==========
   209→   209→    // Response time = time from WO creation to first status that indicates work started (in_progress)
   210→   210→    const responseTimes: { hours: number; priority: string; type: string; trade: string; woId: string }[] = [];
   211→   211→    for (const wo of workOrders) {
   212→   212→      const history = wo.statusHistory || [];
   213→   213→      if (history.length === 0) continue;
   214→   214→
   215→   215→      // Find the transition to 'in_progress' (or 'assigned' as fallback)
   216→   216→      const startEntry = history.find(h => h.toStatus === 'in_progress')
   217→   217→        || history.find(h => h.toStatus === 'assigned');
   218→   218→
   219→   219→      if (startEntry && wo.createdAt) {
   220→   220→        const hours = (startEntry.createdAt.getTime() - wo.createdAt.getTime()) / (1000 * 60 * 60);
   221→   221→        if (hours >= 0 && hours < 10000) { // sanity check
   222→   222→          responseTimes.push({
   223→   223→            hours: Math.round(hours * 100) / 100,
   224→   224→            priority: wo.priority,
   225→   225→            type: wo.type,
   226→   226→            trade: wo.tradeActivity || 'Unassigned',
   227→   227→            woId: wo.id,
   228→   228→          });
   229→   229→        }
   230→   230→      }
   231→   231→    }
   232→   232→
   233→   233→    const avgResponseTime = responseTimes.length > 0
   234→   234→      ? Math.round(responseTimes.reduce((s, r) => s + r.hours, 0) / responseTimes.length * 100) / 100
   235→   235→      : 0;
   236→   236→
   237→   237→    // Response time by priority
   238→   238→    const responseByPriority: Record<string, number[]> = {};
   239→   239→    for (const r of responseTimes) {
   240→   240→      if (!responseByPriority[r.priority]) responseByPriority[r.priority] = [];
   241→   241→      responseByPriority[r.priority].push(r.hours);
   242→   242→    }
   243→   243→    const responseByPriorityArray = Object.entries(responseByPriority)
   244→   244→      .map(([p, hours]) => ({
   245→   245→        priority: p,
   246→   246→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   247→   247→        minHours: Math.round(Math.min(...hours) * 100) / 100,
   248→   248→        maxHours: Math.round(Math.max(...hours) * 100) / 100,
   249→   249→        count: hours.length,
   250→   250→      }))
   251→   251→      .sort((a, b) => a.avgHours - b.avgHours);
   252→   252→
   253→   253→    // Response time by trade
   254→   254→    const responseByTrade: Record<string, number[]> = {};
   255→   255→    for (const r of responseTimes) {
   256→   256→      if (!responseByTrade[r.trade]) responseByTrade[r.trade] = [];
   257→   257→      responseByTrade[r.trade].push(r.hours);
   258→   258→    }
   259→   259→    const responseByTradeArray = Object.entries(responseByTrade)
   260→   260→      .map(([t, hours]) => ({
   261→   261→        trade: t,
   262→   262→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   263→   263→        count: hours.length,
   264→   264→      }))
   265→   265→      .sort((a, b) => a.avgHours - b.avgHours);
   266→   266→
   267→   267→    // ========== 4. BREAKDOWNS ==========
   268→   268→    // Breakdown by type
   269→   269→    const breakdownByType: Record<string, number> = {};
   270→   270→    for (const wo of breakdownWOs) {
   271→   271→      const t = wo.type || 'unknown';
   272→   272→      breakdownByType[t] = (breakdownByType[t] || 0) + 1;
   273→   273→    }
   274→   274→    const breakdownByTypeArray = Object.entries(breakdownByType)
   275→   275→      .map(([type, count]) => ({ type, count }))
   276→   276→      .sort((a, b) => b.count - a.count);
   277→   277→
   278→   278→    // Breakdown by trade
   279→   279→    const breakdownByTrade: Record<string, number> = {};
   280→   280→    for (const wo of breakdownWOs) {
   281→   281→      const t = wo.tradeActivity || 'Unassigned';
   282→   282→      breakdownByTrade[t] = (breakdownByTrade[t] || 0) + 1;
   283→   283→    }
   284→   284→    const breakdownByTradeArray = Object.entries(breakdownByTrade)
   285→   285→      .map(([trade, count]) => ({ trade, count }))
   286→   286→      .sort((a, b) => b.count - a.count);
   287→   287→
   288→   288→    // Breakdown by month
   289→   289→    const breakdownByMonth: Record<string, number> = {};
   290→   290→    for (const wo of breakdownWOs) {
   291→   291→      const month = wo.createdAt.toISOString().slice(0, 7);
   292→   292→      breakdownByMonth[month] = (breakdownByMonth[month] || 0) + 1;
   293→   293→    }
   294→   294→    const breakdownByMonthArray = Object.entries(breakdownByMonth)
   295→   295→      .map(([month, count]) => ({ month, count }))
   296→   296→      .sort((a, b) => a.month.localeCompare(b.month));
   297→   297→
   298→   298→    // Breakdown by priority
   299→   299→    const breakdownByPriority: Record<string, number> = {};
   300→   300→    for (const wo of breakdownWOs) {
   301→   301→      const p = wo.priority || 'medium';
   302→   302→      breakdownByPriority[p] = (breakdownByPriority[p] || 0) + 1;
   303→   303→    }
   304→   304→    const breakdownByPriorityArray = Object.entries(breakdownByPriority)
   305→   305→      .map(([priority, count]) => ({ priority, count }))
   306→   306→      .sort((a, b) => b.count - a.count);
   307→   307→
   308→   308→    // ========== 5. MAN HOURS ==========
   309→   309→    // From timeLogs and repairCompletion
   310→   310→    const manHoursByTrade: Record<string, { totalHours: number; woCount: number }> = {};
   311→   311→    const manHoursByTechnician: Record<string, { name: string; totalHours: number; woCount: number }> = {};
   312→   312→    let grandTotalManHours = 0;
   313→   313→
   314→   314→    for (const wo of workOrders) {
   315→   315→      const trade = wo.tradeActivity || 'Unassigned';
   316→   316→
   317→   317→      // Get hours from timeLogs (primary), actualHours (fallback), repairCompletion (override)
   318→   318→      let woHours = 0;
   319→   319→      for (const tl of (wo.timeLogs || [])) {
   320→   320→        if (tl.duration && tl.duration > 0) {
   321→   321→          woHours += tl.duration;
   322→   322→        }
   323→   323→      }
   324→   324→      // Fallback to WO's actualHours when no time logs exist
   325→   325→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   326→   326→        woHours = wo.actualHours;
   327→   327→      }
   328→   328→      // Override with repairCompletion if higher (supervisor-verified hours)
   329→   329→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   330→   330→        woHours = wo.repairCompletion.totalLaborHours;
   331→   331→      }
   332→   332→
   333→   333→      grandTotalManHours += woHours;
   334→   334→
   335→   335→      // By trade
   336→   336→      if (!manHoursByTrade[trade]) manHoursByTrade[trade] = { totalHours: 0, woCount: 0 };
   337→   337→      manHoursByTrade[trade].totalHours += woHours;
   338→   338→      manHoursByTrade[trade].woCount++;
   339→   339→
   340→   340→      // By technician
   341→   341→      const techId = wo.assignedTo || 'unassigned';
   342→   342→      const techName = wo.assignee?.fullName || 'Unassigned';
   343→   343→      if (!manHoursByTechnician[techId]) {
   344→   344→        manHoursByTechnician[techId] = { name: techName, totalHours: 0, woCount: 0 };
   345→   345→      }
   346→   346→      manHoursByTechnician[techId].totalHours += woHours;
   347→   347→      manHoursByTechnician[techId].woCount++;
   348→   348→    }
   349→   349→
   350→   350→    const manHoursByTradeArray = Object.entries(manHoursByTrade)
   351→   351→      .map(([trade, data]) => ({
   352→   352→        trade,
   353→   353→        totalHours: Math.round(data.totalHours * 100) / 100,
   354→   354→        woCount: data.woCount,
   355→   355→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   356→   356→      }))
   357→   357→      .sort((a, b) => b.totalHours - a.totalHours);
   358→   358→
   359→   359→    const manHoursByTechnicianArray = Object.values(manHoursByTechnician)
   360→   360→      .map(data => ({
   361→   361→        ...data,
   362→   362→        totalHours: Math.round(data.totalHours * 100) / 100,
   363→   363→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   364→   364→      }))
   365→   365→      .sort((a, b) => b.totalHours - a.totalHours);
   366→   366→
   367→   367→    // Man hours by month
   368→   368→    const manHoursByMonth: Record<string, number> = {};
   369→   369→    for (const wo of workOrders) {
   370→   370→      const month = wo.createdAt.toISOString().slice(0, 7);
   371→   371→      let woHours = 0;
   372→   372→      for (const tl of (wo.timeLogs || [])) {
   373→   373→        if (tl.duration && tl.duration > 0) woHours += tl.duration;
   374→   374→      }
   375→   375→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   376→   376→        woHours = wo.actualHours;
   377→   377→      }
   378→   378→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   379→   379→        woHours = wo.repairCompletion.totalLaborHours;
   380→   380→      }
   381→   381→      manHoursByMonth[month] = (manHoursByMonth[month] || 0) + woHours;
   382→   382→    }
   383→   383→    const manHoursByMonthArray = Object.entries(manHoursByMonth)
   384→   384→      .map(([month, totalHours]) => ({
   385→   385→        month,
   386→   386→        totalHours: Math.round(totalHours * 100) / 100,
   387→   387→      }))
   388→   388→      .sort((a, b) => a.month.localeCompare(b.month));
   389→   389→
   390→   390→    // Man hours by activity type
   391→   391→    const manHoursByActivity: Record<string, number> = {};
   392→   392→    for (const wo of workOrders) {
   393→   393→      for (const tl of (wo.timeLogs || [])) {
   394→   394→        const act = tl.activityType || 'maintenance';
   395→   395→        manHoursByActivity[act] = (manHoursByActivity[act] || 0) + (tl.duration || 0);
   396→   396→      }
   397→   397→    }
   398→   398→    const manHoursByActivityArray = Object.entries(manHoursByActivity)
   399→   399→      .map(([activity, totalHours]) => ({
   400→   400→        activity,
