    78→function getWeekDates(referenceDate: Date): Date[] {
    79→  const monday = startOfWeek(referenceDate, { weekStartsOn: 1 });
    80→  return Array.from({ length: 7 }, (_, i) => addDays(monday, i));
    81→}

    83→function cellColorClass(hours: number): string {
    84→  if (hours === 0) return 'bg-gray-50 text-gray-300';
    85→  if (hours <= 4) return 'bg-emerald-50 text-emerald-700';
    86→  if (hours <= 8) return 'bg-green-100 text-green-800';
    87→  if (hours <= 12) return 'bg-amber-100 text-amber-800';
    88→  return 'bg-red-100 text-red-800';
    89→}
    90→
    91→function cellBgClass(hours: number): string {
    92→  if (hours === 0) return '';
    93→  if (hours <= 4) return 'bg-emerald-50/50';
    94→  if (hours <= 8) return 'bg-green-50/50';
    95→  if (hours <= 12) return 'bg-amber-50/50';
    96→  return 'bg-red-50/50';
    97→}

   123→function isSupervisorOrAbove(): boolean {
   124→  const slugs = JSON.parse(

   230→  isTotal,
   231→}: {
   232→  hours: number;
   233→  breakdown?: { activity: string; hours: number }[];
   234→  isTotal?: boolean;
   235→}) {

   237→

   240→      <Tooltip>
   241→        <TooltipTrigger asChild>
   242→          <td
   243→            className={`text-center px-2 py-2 text-xs font-medium border-r border-b border-gray-100 transition-colors cursor-default
   244→              ${hasData ? cellBgClass(hours) + ' hover:opacity-80' : 'hover:bg-gray-50'}

   246→            `}
   247→          >
   248→            {formatDuration(hours)}
   249→          </td>
   250→        </TooltipTrigger>
   251→        {hasData && breakdown && breakdown.length > 0 && (
   252→          <TooltipContent side="top" className="max-w-[220px]">
   253→            <div className="space-y-1">
   254→              <p className="text-xs font-semibold mb-1">Activity Breakdown</p>
   255→              {breakdown.map((b) => (
   256→                <div key={b.activity} className="flex items-center justify-between gap-4">
   257→                  <span className="text-[11px] text-muted-foreground capitalize">{b.activity.replace(/_/g, ' ')}</span>
   258→                  <span className="text-[11px] font-medium">{formatDuration(b.hours)}</span>
   259→                </div>
   260→              ))}
   261→            </div>
   262→          </TooltipContent>
   263→        )}
   264→      </Tooltip>
   265→    </TooltipProvider>
   266→  );

   303→export function TimesheetPage() {
   304→  const { user, hasPermission, isAdmin } = useAuthStore();
   305→  const abortRef = useAbortRef();
   306→
   307→  // ── State ──
   308→  const [referenceDate, setReferenceDate] = useState(() => new Date());
   309→  const [viewMode, setViewMode] = useState<ViewMode>('mine');
   310→  const [selectedUserId, setSelectedUserId] = useState<string>('');
   311→  const [filterStatus, setFilterStatus] = useState<string>('all');
   312→  const [filterActivity, setFilterActivity] = useState<string>('all');
   313→
   314→  const [timeLogs, setTimeLogs] = useState<TimeLogEntry[]>([]);
   315→  const [teamMembers, setTeamMembers] = useState<TeamMember[]>([]);
   316→  const [loading, setLoading] = useState(true);
   317→  const [exporting, setExporting] = useState(false);
   318→
   319→  // ── Derived: can see team view? ──
   320→  const canViewTeam = useMemo(() => {
   321→    return canViewAllTimeLogs() || (user?.roles || []).some(
   322→      (r: any) => ['maintenance_supervisor', 'maintenance_manager', 'plant_manager', 'hr_manager', 'maintenance_planner', 'admin'].includes(r.slug)
   323→    );
   324→  }, [user]);
   325→
   326→  // ── Week dates ──
   327→  const weekDates = useMemo(() => getWeekDates(referenceDate), [referenceDate]);
   328→  const weekStartStr = format(weekDates[0], 'yyyy-MM-dd');
   329→  const weekEndStr = format(weekDates[6], 'yyyy-MM-dd');
   330→
   331→  // ── Fetch team members ──
   332→  useEffect(() => {
   333→    if (!canViewTeam) return;
   334→    api
   335→      .get<TeamMember[]>('/api/users?status=active&role=maintenance_technician', {
   336→        signal: abortRef.current.signal,
   337→      })
   338→      .then((res) => {
   339→        if (res.success && res.data) {
   340→          const members: TeamMember[] = Array.isArray(res.data)
   341→            ? res.data.map((u: any) => ({
   342→                id: u.id,
   343→                fullName: u.fullName || u.username,
   344→                department: u.department || '',
   345→                status: u.status || 'active',
   346→              }))
   347→            : [];
   348→          setTeamMembers(members);
   349→        }
   350→      })
   351→      .catch(() => {});
   352→  }, [canViewTeam, abortRef]);
   353→
   354→  // ── Fetch time logs ──
   355→  const fetchTimeLogs = useCallback(async () => {
   356→    setLoading(true);
   357→    try {
   358→      const params = new URLSearchParams();
   359→      params.set('from', weekStartStr);
   360→      params.set('to', weekEndStr);
   361→      params.set('limit', '1000');
   362→
   363→      // For mine view or when a user is selected
   364→      if (viewMode === 'mine' || selectedUserId) {
   365→        params.set('userId', selectedUserId || user?.id || '');
   366→      }
   367→
   368→      const res = await api.get<TimeLogEntry[]>(`/api/time-logs?${params}`, {
   369→        signal: abortRef.current.signal,
   370→      });
   371→
   372→      if (res.success && res.data) {
   373→        const logs: TimeLogEntry[] = Array.isArray(res.data) ? res.data : [];
   374→        setTimeLogs(logs);
   375→      } else {
   376→        setTimeLogs([]);
   377→        if (res.error) toast.error(res.error);
   378→      }
   379→    } catch {
   380→      setTimeLogs([]);
   381→    } finally {
   382→      setLoading(false);
   383→    }
   384→  }, [weekStartStr, weekEndStr, viewMode, selectedUserId, user, abortRef]);
   385→
   386→  useEffect(() => {
   387→    fetchTimeLogs();
   388→  }, [fetchTimeLogs]);
   389→
   390→  // ── Build grid data ──
   391→  const gridData = useMemo(() => {
   392→    // Apply client-side filters
   393→    let filtered = timeLogs;
   394→
   395→    if (filterStatus !== 'all') {
   396→      filtered = filtered.filter((log) => log.workOrder?.status === filterStatus);
   397→    }
   398→    if (filterActivity !== 'all') {
   399→      filtered = filtered.filter((log) => log.activityType === filterActivity);
   400→    }
   401→
   402→    // Group by work order
   403→    const woMap = new Map<string, GridRow>();
   404→
   405→    for (const log of filtered) {
   406→      if (!log.workOrderId || !log.duration) continue;
   407→
   408→      const wo = log.workOrder;
   409→      if (!wo) continue;
   410→
   411→      let row = woMap.get(log.workOrderId);
   412→      if (!row) {
   413→        row = {
   414→          workOrderId: log.workOrderId,
   415→          woNumber: wo.woNumber || '—',
   416→          title: wo.title || 'Untitled',
   417→          status: wo.status || 'unknown',
   418→          type: wo.type || 'corrective',
   419→          dailyHours: [0, 0, 0, 0, 0, 0, 0],
   420→          totalHours: 0,
   421→          dailyBreakdown: {},
   422→        };
   423→        woMap.set(log.workOrderId, row);
   424→      }
   425→
   426→      // Determine which day of the week
   427→      const logDate = parseISO(log.timestamp);
   428→      let dayIndex = -1;
   429→      for (let d = 0; d < 7; d++) {
   430→        if (isSameDay(logDate, weekDates[d])) {
   431→          dayIndex = d;
   432→          break;
   433→        }
   434→      }
   435→      if (dayIndex < 0) continue;
   436→
   437→      const dur = log.duration || 0;
   438→      row.dailyHours[dayIndex] += dur;
   439→      row.totalHours += dur;
   440→
   441→      // Track breakdown per day
   442→      if (!row.dailyBreakdown[dayIndex]) {
   443→        row.dailyBreakdown[dayIndex] = [];
   444→      }
   445→
   446→      const act = log.activityType || 'maintenance';
   447→      const existing = row.dailyBreakdown[dayIndex].find((b) => b.activity === act);
   448→      if (existing) {
   449→        existing.hours += dur;
   450→      } else {
   451→        row.dailyBreakdown[dayIndex].push({ activity: act, hours: dur });
   452→      }
   453→    }
   454→
   455→    // Round all hours to 1 decimal
   456→    const rows = Array.from(woMap.values()).map((row) => ({
   457→      ...row,
   458→      dailyHours: row.dailyHours.map((h) => Math.round(h * 10) / 10),
   459→      totalHours: Math.round(row.totalHours * 10) / 10,
   460→      dailyBreakdown: Object.fromEntries(
   461→        Object.entries(row.dailyBreakdown).map(([day, breakdown]) => [
   462→          day,
   463→          breakdown.map((b) => ({ ...b, hours: Math.round(b.hours * 10) / 10 })),
   464→        ])
   465→      ),
   466→    }));
   467→
   468→    return rows;
   469→  }, [timeLogs, filterStatus, filterActivity, weekDates]);
   470→
   471→  // ── Compute summary ──
   472→  const summary = useMemo(() => {
   473→    const dailyTotals = [0, 0, 0, 0, 0, 0, 0];
   474→    let grandTotal = 0;
   475→    let uniqueWOs = new Set<string>();
   476→    let uniqueDays = new Set<number>();
   477→
   478→    for (const row of gridData) {
   479→      uniqueWOs.add(row.workOrderId);
   480→      for (let d = 0; d < 7; d++) {
   481→        dailyTotals[d] += row.dailyHours[d];
   482→        grandTotal += row.dailyHours[d];
   483→        if (row.dailyHours[d] > 0) uniqueDays.add(d);
   484→      }
   485→    }
   486→
   487→    // Round
   488→    const roundedDaily = dailyTotals.map((h) => Math.round(h * 10) / 10);
   489→    grandTotal = Math.round(grandTotal * 10) / 10;
   490→    const overtime = Math.max(0, grandTotal - WEEK_OVERHOUR_THRESHOLD);
   491→    const avgPerWO = gridData.length > 0 ? Math.round((grandTotal / gridData.length) * 10) / 10 : 0;
   492→
   493→    return {
   494→      dailyTotals: roundedDaily,
   495→      grandTotal,
   496→      overtime: Math.round(overtime * 10) / 10,
   497→      woCount: uniqueWOs.size,
   498→      daysWorked: uniqueDays.size,
   499→      avgPerWO,
   500→    };
   501→  }, [gridData]);
   502→
   503→  // ── Handlers ──
   504→  const handlePrevWeek = () => setReferenceDate((d) => subWeeks(d, 1));
   505→  const handleNextWeek = () => setReferenceDate((d) => addWeeks(d, 1));
   506→  const handleCurrentWeek = () => setReferenceDate(new Date());
   507→
   508→  const handleExportCSV = async () => {
   509→    setExporting(true);
   510→    try {
   511→      const params = new URLSearchParams();
   512→      params.set('from', weekStartStr);
   513→      params.set('to', weekEndStr);
   514→      params.set('export', 'csv');
   515→      if (viewMode === 'mine' || selectedUserId) {
   516→        params.set('userId', selectedUserId || user?.id || '');
   517→      }
   518→
   519→      const response = await fetch(`/api/time-logs?${params}`, {
   520→        headers: { Authorization: `Bearer ${localStorage.getItem('eam_token')}` },
   521→      });
   522→
   523→      if (!response.ok) {
   524→        toast.error('Export failed');
   525→        setExporting(false);
   526→        return;
   527→      }
   528→
   529→      const blob = await response.blob();
   530→      const url = window.URL.createObjectURL(blob);
   531→      const a = document.createElement('a');
   532→      a.href = url;
   533→      a.download = `timesheet-${weekStartStr}-to-${weekEndStr}.csv`;
   534→      document.body.appendChild(a);
   535→      a.click();
   536→      document.body.removeChild(a);
   537→      window.URL.revokeObjectURL(url);
   538→      toast.success('CSV exported successfully');
   539→    } catch {
   540→      toast.error('Export failed');
   541→    } finally {
   542→      setExporting(false);
   543→    }
   544→  };
   545→
   546→  // ── Active filters count ──
   547→  const activeFilterCount = useMemo(() => {
   548→    let count = 0;
   549→    if (filterStatus !== 'all') count++;
   550→    if (filterActivity !== 'all') count++;
   551→    return count;
   552→  }, [filterStatus, filterActivity]);
   553→
   554→  const clearFilters = () => {
   555→    setFilterStatus('all');
   556→    setFilterActivity('all');
   557→  };
   558→
   559→  // ── Render ──
   560→  return (
   561→    <div className="page-content">
   562→      {/* ── Header ── */}
   563→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   564→        <div className="flex items-center gap-3">
   565→          <div className="p-2 bg-teal-100 rounded-xl">
   566→            <Timer className="h-6 w-6 text-teal-700" />
   567→          </div>
   568→          <div>
   569→            <div className="flex items-center gap-2">
   570→              <h2 className="text-2xl font-bold tracking-tight">Timesheet</h2>
   571→              <Badge variant="secondary" className="font-mono">

   601→          bgColor="bg-teal-50"
   602→        />

   604→          icon={AlertTriangle}
   605→          label="Overtime Hours"
   606→          value={formatDuration(summary.overtime)}
   607→          color={summary.overtime > 0 ? 'text-red-600' : 'text-gray-500'}

   609→          subtext={`> ${WEEK_OVERHOUR_THRESHOLD}h`}
   610→        />

   612→          icon={Layers}
   613→          label="Work Orders"
   614→          value={summary.woCount}
   615→          color="text-emerald-600"
   616→          bgColor="bg-emerald-50"
   617→        />
   618→        <SummaryCard
   619→          icon={TrendingUp}
   620→          label="Avg Hours/WO"
   621→          value={formatDuration(summary.avgPerWO)}
   622→          color="text-amber-600"
   623→          bgColor="bg-amber-50"

   626→          icon={CalendarDays}

   629→          color="text-slate-600"
   630→          bgColor="bg-slate-50"
   631→        />

   633→
   634→      {/* ── Week Navigation + View Mode + Filters ── */}
   635→      <div className="space-y-4">

   637→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">

   640→            onPrev={handlePrevWeek}

   669→
   670→        {/* Filters row */}

   672→          {/* Technician selector (team view) */}

   674→            <Select

   712→              <SelectItem value="cancelled">Cancelled</SelectItem>
   713→            </SelectContent>
   714→          </Select>
   715→
   716→          {/* Activity type filter */}
   717→          <Select value={filterActivity} onValueChange={setFilterActivity}>
   718→            <SelectTrigger className="w-44">
   719→              <SelectValue placeholder="Activity Type" />
   720→            </SelectTrigger>
   721→            <SelectContent>
   722→              <SelectItem value="all">All Activities</SelectItem>
   723→              <SelectItem value="maintenance">Maintenance</SelectItem>
   724→              <SelectItem value="travel">Travel</SelectItem>
   725→              <SelectItem value="inspection">Inspection</SelectItem>
   726→              <SelectItem value="testing">Testing</SelectItem>
   727→              <SelectItem value="standby">Standby</SelectItem>
   728→              <SelectItem value="other">Other</SelectItem>
   729→            </SelectContent>
   730→          </Select>
   731→
   732→          {/* Clear filters */}
   733→          {activeFilterCount > 0 && (
   734→            <Button variant="ghost" size="sm" onClick={clearFilters} className="h-9 text-muted-foreground hover:text-foreground gap-1.5">

   736→              <Badge variant="secondary" className="h-5 min-w-5 px-1 text-[10px]">{activeFilterCount}</Badge>

   738→          )}
   739→        </div>
   740→      </div>
   741→
   742→      {/* ── Timesheet Grid ── */}
   743→      {loading ? (
   744→        <GridLoadingSkeleton />

   749→              icon={FileSpreadsheet}
   750→              title="No time entries this week"

   757→          <CardContent className="p-0">
   758→            <div className="overflow-x-auto">
   759→              <Table>
   760→                <TableHeader>
   761→                  <TableRow className="bg-muted/50">
   762→                    <TableHead className="sticky left-0 z-10 bg-muted/50 min-w-[220px] w-[220px]">
   763→                      <div className="flex items-center gap-1.5">
   764→                        <Layers className="h-3.5 w-3.5 text-muted-foreground" />
   765→                        Work Order
   766→                      </div>
   767→                    </TableHead>
   768→                    {weekDates.map((date, i) => (
   769→                      <TableHead key={i} className="min-w-[100px] text-center">

   781→                  </TableRow>
   782→                </TableHeader>

   787→                      key={row.workOrderId}

   791→                      <TableCell className="sticky left-0 z-10 bg-background">

   803→                          </div>

   819→                      >
   820→                        {formatDuration(row.totalHours)}

   824→
   825→                  {/* Summary row */}
   826→                  <TableRow className="bg-muted/40 font-semibold">
   827→                    <TableCell className="sticky left-0 z-10 bg-muted/40">
   828→                      <div className="flex items-center gap-1.5">
   829→                        <BarChart3 className="h-3.5 w-3.5 text-muted-foreground" />
   830→                        <span className="text-xs font-semibold">Daily Totals</span>
   831→                      </div>
   832→                    </TableCell>
   833→                    {summary.dailyTotals.map((hours, i) => (
   834→                      <GridCell key={i} hours={hours} isTotal />
   835→                    ))}
   836→                    <TableCell
   837→                      className={`text-center px-2 py-2 text-sm font-bold
   838→                        ${summary.grandTotal > WEEK_OVERHOUR_THRESHOLD ? 'bg-red-100 text-red-800' : summary.grandTotal > 0 ? 'bg-emerald-100 text-emerald-800' : 'text-gray-300'}
   839→                      `}
   840→                    >

   857→
   858→      {/* ── Legend ── */}
   859→      <div className="flex flex-wrap items-center gap-4 text-xs text-muted-foreground">
   860→        <span className="font-medium text-foreground">Hours Legend:</span>

   865→        <div className="flex items-center gap-1.5">
   866→          <span className="h-3 w-3 rounded-sm bg-emerald-200" />

   868→        </div>
   869→        <div className="flex items-center gap-1.5">
   870→          <span className="h-3 w-3 rounded-sm bg-green-200" />

   891→        <CardContent className="p-4 pt-0">
   892→          <div className="space-y-3">
   893→            {weekDates.map((date, i) => {
   894→              const total = summary.dailyTotals[i];
   895→              const maxDaily = Math.max(...summary.dailyTotals, 1);

   897→
   898→              return (

   902→                  </div>
   903→                  <div className="flex-1 h-6 bg-muted/50 rounded-full overflow-hidden relative">
   904→                    <div
   905→                      className={`h-full rounded-full transition-all duration-500 ${
   906→                        total === 0
   907→                          ? ''
   908→                          : total <= 8
   909→                          ? 'bg-emerald-500'
   910→                          : total <= 12
   911→                          ? 'bg-amber-500'
   912→                          : 'bg-red-500'
   913→                      }`}
   914→                      style={{ width: `${pct}%` }}
   915→                    />
   916→                    <span className="absolute inset-0 flex items-center justify-center text-[11px] font-medium">
   917→                      {formatDuration(total)}
   918→                    </span>
   919→                  </div>
   920→                </div>
   921→              );
   922→            })}
   923→          </div>
   924→
   925→          <Separator className="my-4" />
   926→
   927→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 text-center">
   928→            <div className="bg-muted/50 rounded-lg p-3">
   929→              <p className="text-lg font-bold text-teal-600">{formatDuration(summary.grandTotal)}</p>
   930→              <p className="text-[10px] text-muted-foreground">Total Hours</p>
   931→            </div>
   932→            <div className="bg-muted/50 rounded-lg p-3">
   933→              <p className={`text-lg font-bold ${summary.overtime > 0 ? 'text-red-600' : 'text-gray-500'}`}>
   934→                {formatDuration(summary.overtime)}

   938→            <div className="bg-muted/50 rounded-lg p-3">

   946→          </div>
   947→        </CardContent>
   948→      </Card>
   949→    </div>
   950→  );
   951→}
