     1→import { NextRequest, NextResponse } from 'next/server';
     2→import { db } from '@/lib/db';
     3→import { getSession, isAdmin, hasRole } from '@/lib/auth';
     4→import { notifyUser } from '@/lib/notifications';
     5→
     6→// 24-hour threshold for overdue detection
     7→const OVERDUE_THRESHOLD_MS = 24 * 60 * 60 * 1000;
     8→
     9→// GET /api/repairs/material-requests/[id]
    10→export async function GET(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
    11→  try {
    12→    const { id } = await params;
    13→    const matReq = await db.repairMaterialRequest.findUnique({
    14→      where: { id },
    15→      include: {
    16→        requestedBy: { select: { id: true, fullName: true, username: true, department: true } },
    17→        supervisorApprovedBy: { select: { id: true, fullName: true } },
    18→        storekeeperApprovedBy: { select: { id: true, fullName: true } },
    19→        issuedByUser: { select: { id: true, fullName: true } },
    20→        returnedByUser: { select: { id: true, fullName: true } },
    21→        workOrder: {
    22→          select: {
    23→            id: true, woNumber: true, title: true, status: true,
    24→            assignedSupervisor: { select: { id: true, fullName: true } },
    25→            planner: { select: { id: true, fullName: true } },
    26→          },
    27→        },
    28→        item: { select: { id: true, itemCode: true, name: true, currentStock: true, unitOfMeasure: true } },
    29→      },
    30→    });
    31→
    32→    if (!matReq) return NextResponse.json({ success: false, error: 'Material request not found' }, { status: 404 });
    33→
    34→    // Compute overdue flag: pending requests older than 24 hours
    35→    const enriched = {
    36→      ...matReq,
    37→      isOverdue:
    38→        matReq.status === 'pending' &&
    39→        Date.now() - new Date(matReq.createdAt).getTime() > OVERDUE_THRESHOLD_MS,
    40→    };
    41→
    42→    return NextResponse.json({ success: true, data: enriched });
    43→  } catch (error: unknown) {
    44→    const message = error instanceof Error ? error.message : 'Failed to load material request';
    45→    return NextResponse.json({ success: false, error: message }, { status: 500 });
    46→  }
    47→}

    49→// PUT /api/repairs/material-requests/[id]
    50→export async function PUT(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
    51→  try {
    52→    const session = getSession(request);
    53→    if (!session) return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
    54→
    55→    const { id } = await params;
    56→    const existing = await db.repairMaterialRequest.findUnique({ where: { id } });
    57→    if (!existing) return NextResponse.json({ success: false, error: 'Not found' }, { status: 404 });
    58→
    59→    // Ownership check: only the requester (or admin/supervisor/manager) can edit a pending request
    60→    if (!isAdmin(session) && !hasRole(session, 'maintenance_supervisor') && !hasRole(session, 'maintenance_manager') && !hasRole(session, 'plant_manager')) {
    61→      if (existing.requestedById !== session.userId) {
    62→        return NextResponse.json({ success: false, error: 'You can only edit your own requests' }, { status: 403 });
    63→      }
    64→      if (existing.status !== 'pending') {
    65→        return NextResponse.json({ success: false, error: 'Only pending requests can be edited' }, { status: 400 });
    66→      }
    67→    }
    68→
    69→    const body = await request.json();
    70→    const allowedFields: Record<string, unknown> = {};
    71→    if (body.quantityRequested !== undefined) allowedFields.quantityRequested = body.quantityRequested;
    72→    if (body.unit !== undefined) allowedFields.unit = body.unit;
    73→    if (body.unitCost !== undefined) allowedFields.unitCost = body.unitCost;
    74→    if (body.reason !== undefined) allowedFields.reason = body.reason;
    75→    if (body.notes !== undefined) allowedFields.notes = body.notes;
    76→    if (body.urgency !== undefined && ['low', 'normal', 'high', 'critical'].includes(body.urgency)) {
    77→      allowedFields.urgency = body.urgency;
    78→    }
    79→
    80→    const updated = await db.repairMaterialRequest.update({
    81→      where: { id },
    82→      data: allowedFields,
    83→    });
    84→
    85→    await db.auditLog.create({
    86→      data: {
    87→        userId: session.userId,
    88→        action: 'update',
    89→        entityType: 'repair_material_request',
    90→        entityId: id,
    91→        oldValues: JSON.stringify(existing),
    92→        newValues: JSON.stringify(allowedFields),
    93→      },
    94→    });
    95→
    96→    return NextResponse.json({ success: true, data: updated });
    97→  } catch (error: unknown) {
    98→    const message = error instanceof Error ? error.message : 'Failed to update material request';
    99→    return NextResponse.json({ success: false, error: message }, { status: 500 });
   100→  }
   101→}

   103→// DELETE /api/repairs/material-requests/[id] — cancel (only if pending)
   104→export async function DELETE(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
   105→  try {
   106→    const session = getSession(request);
   107→    if (!session) return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
   108→
   109→    const { id } = await params;
   110→    const existing = await db.repairMaterialRequest.findUnique({ where: { id } });
   111→    if (!existing) return NextResponse.json({ success: false, error: 'Not found' }, { status: 404 });
   112→
   113→    if (existing.status !== 'pending') {
   114→      return NextResponse.json({ success: false, error: 'Only pending requests can be cancelled' }, { status: 400 });
   115→    }
   116→
   117→    // Ownership check: only requester or admin/supervisor/manager can cancel
   118→    if (!isAdmin(session) && !hasRole(session, 'maintenance_supervisor') && !hasRole(session, 'maintenance_manager') && !hasRole(session, 'plant_manager')) {
   119→      if (existing.requestedById !== session.userId) {
   120→        return NextResponse.json({ success: false, error: 'You can only cancel your own requests' }, { status: 403 });
   121→      }
   122→    }
   123→
   124→    await db.repairMaterialRequest.delete({ where: { id } });
   125→
   126→    await db.auditLog.create({
   127→      data: {
   128→        userId: session.userId,
   129→        action: 'delete',
   130→        entityType: 'repair_material_request',
   131→        entityId: id,
   132→        oldValues: JSON.stringify(existing),
   133→      },
   134→    });
   135→
   136→    return NextResponse.json({ success: true, message: 'Material request cancelled' });
   137→  } catch (error: unknown) {
   138→    const message = error instanceof Error ? error.message : 'Failed to cancel material request';
   139→    return NextResponse.json({ success: false, error: message }, { status: 500 });
   140→  }
   141→}

   143→// POST /api/repairs/material-requests/[id] — workflow actions
   144→export async function POST(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
   145→  try {
   146→    const session = getSession(request);
   147→    if (!session) return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
   148→
   149→    const { id } = await params;
   150→    const body = await request.json();
   151→    const { action, approvedQuantity, quantityApproved, quantityReturned, notes } = body;
   152→
   153→    const matReq = await db.repairMaterialRequest.findUnique({
   154→      where: { id },
   155→      include: {
   156→        workOrder: {
   157→          select: {
   158→            id: true, woNumber: true, title: true,
   159→            assignedSupervisorId: true, plannerId: true, assignedTo: true,
   160→          },
   161→        },
   162→        requestedBy: { select: { id: true, fullName: true } },
   163→      },
   164→    });
   165→
   166→    if (!matReq) return NextResponse.json({ success: false, error: 'Material request not found' }, { status: 404 });
   167→
   168→    // ── Role-based access control for workflow actions ──
   169→    if (action === 'supervisor_approve' || action === 'supervisor_reject') {
   170→      if (!isAdmin(session) &&
   171→          !hasRole(session, 'maintenance_supervisor') &&
   172→          !hasRole(session, 'maintenance_manager') &&
   173→          !hasRole(session, 'plant_manager')) {
   174→        return NextResponse.json({ success: false, error: 'Only admin, maintenance supervisor, maintenance manager, or plant manager can supervisor-approve material requests' }, { status: 403 });
   175→      }
   176→    }
   177→    if (action === 'storekeeper_approve' || action === 'storekeeper_reject') {
   178→      if (!isAdmin(session) &&
   179→          !hasRole(session, 'store_keeper') &&
   180→          !hasRole(session, 'inventory_manager') &&
   181→          !hasRole(session, 'tools_shop_attendant')) {
   182→        return NextResponse.json({ success: false, error: 'Only admin, store keeper, store manager, or tools shop attendant can store-approve material requests' }, { status: 403 });
   183→      }
   184→    }
   185→
   186→    const now = new Date();
   187→    let updated: any;
   188→
   189→    switch (action) {
   190→      // ──────────────────────────────────────────────
   191→      // SUPERVISOR APPROVE — with optional quantity override
   192→      // ──────────────────────────────────────────────
   193→      case 'supervisor_approve': {
   194→        if (matReq.status !== 'pending') {
   195→          return NextResponse.json({ success: false, error: `Cannot approve: current status is ${matReq.status}` }, { status: 400 });
   196→        }
   197→        const qty = approvedQuantity ?? quantityApproved ?? matReq.quantityRequested;
   198→        updated = await db.repairMaterialRequest.update({
   199→          where: { id },
   200→          data: {
   201→            status: 'supervisor_approved',
   202→            supervisorApprovedById: session.userId,
   203→            supervisorApprovedAt: now,
   204→            supervisorApprovedQuantity: qty !== matReq.quantityRequested ? qty : null,
   205→            quantityApproved: qty,
   206→          },
   207→        });
   208→
   209→        // Audit trail for supervisor approval
   210→        await db.auditLog.create({
   211→          data: {
   212→            userId: session.userId,
   213→            action: 'material_request_supervisor_approve',
   214→            entityType: 'repair_material_request',
   215→            entityId: id,
   216→            newValues: JSON.stringify({
   217→              action: 'supervisor_approve',
   218→              status: 'supervisor_approved',
   219→              approvedQuantity: qty,
   220→              requestedQuantity: matReq.quantityRequested,
   221→              quantityChanged: qty !== matReq.quantityRequested,
   222→            }),
   223→          },
   224→        });
   225→
   226→        // Notify store keepers
   227→        const storeKeepers = await db.user.findMany({
   228→          where: { userRoles: { some: { role: { slug: 'store_keeper' } } }, status: 'active' },
   229→          select: { id: true },
   230→        });
   231→        for (const sk of storeKeepers) {
   232→          await notifyUser(
   233→            sk.id,
   234→            'repair_material_request',
   235→            'Material Request Awaiting Store Approval',
   236→            `${qty} ${matReq.unit} of ${matReq.itemName} approved by supervisor for WO ${matReq.workOrder.woNumber}`,
   237→            'repair_material_request',
   238→            id,
   239→            'maintenance-work-orders',
   240→          );
   241→        }
   242→
   243→        // Notify requester
   244→        await notifyUser(
   245→          matReq.requestedById,
   246→          'repair_material_request',
   247→          'Material Request Supervisor Approved',
   248→          qty !== matReq.quantityRequested
   249→            ? `Your request for ${matReq.itemName} was approved (quantity adjusted from ${matReq.quantityRequested} to ${qty})`
   250→            : `Your request for ${matReq.itemName} was approved by supervisor`,
   251→          'repair_material_request',
   252→          id,
   253→          `material-requests?id=${id}`,
   254→        );
   255→        break;
   256→      }
   257→
   258→      // ──────────────────────────────────────────────
   259→      // SUPERVISOR REJECT — with rejection reason in notes
   260→      // ──────────────────────────────────────────────
   261→      case 'supervisor_reject': {
   262→        if (matReq.status !== 'pending') {
   263→          return NextResponse.json({ success: false, error: `Cannot reject: current status is ${matReq.status}` }, { status: 400 });
   264→        }
   265→        const rejectionNotes = notes
   266→          ? `[${now.toISOString()}] REJECTED by ${session.userId}: ${notes}`
   267→          : `[${now.toISOString()}] REJECTED by ${session.userId}`;
   268→        const updatedNotes = matReq.notes
   269→          ? `${matReq.notes}\n${rejectionNotes}`
   270→          : rejectionNotes;
   271→
   272→        updated = await db.repairMaterialRequest.update({
   273→          where: { id },
   274→          data: {
   275→            status: 'rejected',
   276→            supervisorApprovedById: session.userId,
   277→            supervisorApprovedAt: now,
   278→            notes: updatedNotes,
   279→          },
   280→        });
   281→
   282→        // Audit trail for rejection
   283→        await db.auditLog.create({
   284→          data: {
   285→            userId: session.userId,
   286→            action: 'material_request_supervisor_reject',
   287→            entityType: 'repair_material_request',
   288→            entityId: id,
   289→            newValues: JSON.stringify({
   290→              action: 'supervisor_reject',
   291→              status: 'rejected',
   292→              reason: notes || null,
   293→            }),
   294→          },
   295→        });
   296→
   297→        await notifyUser(
   298→          matReq.requestedById,
   299→          'repair_material_request',
   300→          'Material Request Rejected',
   301→          `Your request for ${matReq.itemName} was rejected by supervisor${notes ? `: ${notes}` : ''}`,
   302→          'repair_material_request',
   303→          id,
   304→          `material-requests?id=${id}`,
   305→        );
   306→        break;
   307→      }
   308→
   309→      // ──────────────────────────────────────────────
   310→      // STOREKEEPER APPROVE — with optional quantity override + stock reservation
   311→      // ──────────────────────────────────────────────
   312→      case 'storekeeper_approve': {
   313→        if (matReq.status !== 'supervisor_approved') {
   314→          return NextResponse.json({ success: false, error: `Cannot approve: current status is ${matReq.status}` }, { status: 400 });
   315→        }
   316→
   317→        const qty = approvedQuantity ?? quantityApproved ?? matReq.quantityApproved;
   318→        let stockReserved = false;
   319→
   320→        // Reserve stock: deduct from inventory via an 'adjustment' movement
   321→        if (matReq.itemId) {
   322→          const invItem = await db.inventoryItem.findUnique({ where: { id: matReq.itemId } });
   323→          if (invItem) {
   324→            if (invItem.currentStock < qty) {
   325→              return NextResponse.json({
   326→                success: false,
   327→                error: `Insufficient stock to reserve. Available: ${invItem.currentStock}, Required: ${qty}`,
   328→              }, { status: 400 });
   329→            }
   330→            // Deduct stock as reservation
   331→            await db.inventoryItem.update({
   332→              where: { id: matReq.itemId },
   333→              data: { currentStock: { decrement: qty } },
   334→            });
   335→            // Create reservation stock movement of type 'adjustment'
   336→            await db.stockMovement.create({
   337→              data: {
   338→                itemId: matReq.itemId,
   339→                type: 'adjustment',
   340→                quantity: qty,
   341→                previousStock: invItem.currentStock,
   342→                newStock: invItem.currentStock - qty,
   343→                reason: `Stock reserved for WO ${matReq.workOrder.woNumber} — ${matReq.itemName}`,
   344→                referenceType: 'work_order',
   345→                referenceId: matReq.workOrderId,
   346→                performedById: session.userId,
   347→                notes: `Reservation: ${qty} ${matReq.unit} reserved for material request ${id.substring(0, 8)}`,
   348→              },
   349→            });
   350→            stockReserved = true;
   351→          }
   352→        }
   353→
   354→        updated = await db.repairMaterialRequest.update({
   355→          where: { id },
   356→          data: {
   357→            status: 'storekeeper_approved',
   358→            storekeeperApprovedById: session.userId,
   359→            storekeeperApprovedAt: now,
   360→            storekeeperApprovedQuantity: qty !== matReq.quantityApproved ? qty : null,
   361→            quantityApproved: qty,
   362→            stockReserved,
   363→          },
   364→        });
   365→
   366→        // Audit trail for storekeeper approval with reservation details
   367→        await db.auditLog.create({
   368→          data: {
   369→            userId: session.userId,
   370→            action: 'material_request_storekeeper_approve',
   371→            entityType: 'repair_material_request',
   372→            entityId: id,
   373→            newValues: JSON.stringify({
   374→              action: 'storekeeper_approve',
   375→              status: 'storekeeper_approved',
   376→              approvedQuantity: qty,
   377→              previousApprovedQuantity: matReq.quantityApproved,
   378→              quantityChanged: qty !== matReq.quantityApproved,
   379→              stockReserved,
   380→              itemId: matReq.itemId || null,
   381→            }),
   382→          },
   383→        });
   384→
   385→        await notifyUser(
   386→          matReq.requestedById,
   387→          'repair_material_request',
   388→          'Material Request Ready for Issuance',
   389→          `${qty} ${matReq.unit} of ${matReq.itemName} approved by store keeper. Ready for pickup.`,
   390→          'repair_material_request',
   391→          id,
   392→          `material-requests?id=${id}`,
   393→        );
   394→        break;
   395→      }
   396→
   397→      // ──────────────────────────────────────────────
   398→      // STOREKEEPER REJECT — with rejection reason in notes
   399→      // ──────────────────────────────────────────────
   400→      case 'storekeeper_reject': {
   401→        if (matReq.status !== 'supervisor_approved') {
   402→          return NextResponse.json({ success: false, error: `Cannot reject: current status is ${matReq.status}` }, { status: 400 });
   403→        }
   404→        const rejectionNotes = notes
   405→          ? `[${now.toISOString()}] REJECTED by store: ${notes}`
   406→          : `[${now.toISOString()}] REJECTED by store keeper ${session.userId}`;
   407→        const updatedNotes = matReq.notes
   408→          ? `${matReq.notes}\n${rejectionNotes}`
   409→          : rejectionNotes;
   410→
   411→        updated = await db.repairMaterialRequest.update({
   412→          where: { id },
   413→          data: {
   414→            status: 'rejected',
   415→            storekeeperApprovedById: session.userId,
   416→            storekeeperApprovedAt: now,
   417→            notes: updatedNotes,
   418→          },
   419→        });
   420→
   421→        // Audit trail for store rejection
   422→        await db.auditLog.create({
   423→          data: {
   424→            userId: session.userId,
   425→            action: 'material_request_storekeeper_reject',
   426→            entityType: 'repair_material_request',
   427→            entityId: id,
   428→            newValues: JSON.stringify({
   429→              action: 'storekeeper_reject',
   430→              status: 'rejected',
   431→              reason: notes || null,
   432→            }),
   433→          },
   434→        });
   435→
   436→        await notifyUser(
   437→          matReq.requestedById,
   438→          'repair_material_request',
   439→          'Material Request Rejected by Store',
   440→          `Your request for ${matReq.itemName} was rejected by store keeper${notes ? `: ${notes}` : ''}`,
   441→          'repair_material_request',
   442→          id,
   443→          `material-requests?id=${id}`,
   444→        );
   445→        break;
   446→      }
   447→
   448→      // ──────────────────────────────────────────────
   449→      // ISSUE — handle both reserved and non-reserved stock
   450→      // ──────────────────────────────────────────────
   451→      case 'issue': {
   452→        if (matReq.status !== 'storekeeper_approved' && matReq.status !== 'picking') {
   453→          return NextResponse.json({ success: false, error: `Cannot issue: current status is ${matReq.status}` }, { status: 400 });
   454→        }
   455→        const qtyToIssue = approvedQuantity ?? quantityApproved ?? matReq.quantityApproved;
   456→
   457→        // If stock was already reserved at storekeeper approval, just create the issue record.
   458→        // If not reserved (no itemId or no reservation happened), deduct stock now.
   459→        if (matReq.itemId) {
   460→          const invItem = await db.inventoryItem.findUnique({ where: { id: matReq.itemId } });
   461→
   462→          if (matReq.stockReserved) {
   463→            // Stock was already deducted during reservation — just record the issuance
   464→            if (invItem) {
   465→              await db.stockMovement.create({
   466→                data: {
   467→                  itemId: matReq.itemId,
   468→                  type: 'out',
   469→                  quantity: qtyToIssue,
   470→                  previousStock: invItem.currentStock,
   471→                  newStock: invItem.currentStock, // already deducted during reservation
   472→                  reason: `Issued for WO ${matReq.workOrder.woNumber} (from reserved stock)`,
   473→                  referenceType: 'work_order',
   474→                  referenceId: matReq.workOrderId,
   475→                  performedById: session.userId,
   476→                  notes: `Issuance from reserved stock for material request ${id.substring(0, 8)}` + (notes ? ` — ${notes}` : ''),
   477→                },
   478→              });
   479→            }
   480→          } else {
   481→            // Stock was NOT reserved — deduct now
   482→            if (invItem) {
   483→              if (invItem.currentStock < qtyToIssue) {
   484→                return NextResponse.json({
   485→                  success: false,
   486→                  error: `Insufficient stock. Available: ${invItem.currentStock}, Requested: ${qtyToIssue}`,
   487→                }, { status: 400 });
   488→              }
   489→              await db.inventoryItem.update({
   490→                where: { id: matReq.itemId },
   491→                data: { currentStock: { decrement: qtyToIssue } },
   492→              });
   493→              await db.stockMovement.create({
   494→                data: {
   495→                  itemId: matReq.itemId,
   496→                  type: 'out',
   497→                  quantity: qtyToIssue,
   498→                  previousStock: invItem.currentStock,
   499→                  newStock: invItem.currentStock - qtyToIssue,
   500→                  reason: `Issued for WO ${matReq.workOrder.woNumber}`,
   501→                  referenceType: 'work_order',
   502→                  referenceId: matReq.workOrderId,
   503→                  performedById: session.userId,
   504→                  notes: notes || null,
   505→                },
   506→              });
   507→            }
   508→          }
   509→        }
   510→
   511→        updated = await db.repairMaterialRequest.update({
   512→          where: { id },
   513→          data: {
   514→            status: 'issued',
   515→            quantityIssued: qtyToIssue,
   516→            issuedById: session.userId,
   517→            issuedAt: now,
   518→          },
   519→        });
   520→
   521→        // Audit trail for issuance
   522→        await db.auditLog.create({
   523→          data: {
   524→            userId: session.userId,
   525→            action: 'material_request_issue',
   526→            entityType: 'repair_material_request',
   527→            entityId: id,
   528→            newValues: JSON.stringify({
   529→              action: 'issue',
   530→              status: 'issued',
   531→              quantityIssued: qtyToIssue,
   532→              wasReserved: !!matReq.stockReserved,
   533→              itemId: matReq.itemId || null,
   534→            }),
   535→          },
   536→        });
   537→
   538→        // Notify requester
   539→        await notifyUser(
   540→          matReq.requestedById,
   541→          'repair_material_request',
   542→          'Materials Issued',
   543→          `${qtyToIssue} ${matReq.unit} of ${matReq.itemName} issued for WO ${matReq.workOrder.woNumber}`,
   544→          'repair_material_request',
   545→          id,
   546→          `material-requests?id=${id}`,
   547→        );
   548→
   549→        // Notify work order's planner when material is issued
   550→        if (matReq.workOrder.plannerId && matReq.workOrder.plannerId !== matReq.requestedById) {
   551→          await notifyUser(
   552→            matReq.workOrder.plannerId,
   553→            'repair_material_request',
   554→            'Material Issued for Planned Work Order',
   555→            `${qtyToIssue} ${matReq.unit} of ${matReq.itemName} issued for WO ${matReq.workOrder.woNumber}`,
   556→            'repair_material_request',
   557→            id,
   558→            'maintenance-work-orders',
   559→          );
   560→        }
   561→
   562→        // Notify assigned supervisor if different from planner
   563→        if (matReq.workOrder.assignedSupervisorId
   564→            && matReq.workOrder.assignedSupervisorId !== matReq.requestedById
   565→            && matReq.workOrder.assignedSupervisorId !== matReq.workOrder.plannerId) {
   566→          await notifyUser(
   567→            matReq.workOrder.assignedSupervisorId,
   568→            'repair_material_request',
   569→            'Material Issued for WO Under Your Supervision',
   570→            `${qtyToIssue} ${matReq.unit} of ${matReq.itemName} issued for WO ${matReq.workOrder.woNumber}`,
   571→            'repair_material_request',
   572→            id,
   573→            'maintenance-work-orders',
   574→          );
   575→        }
   576→        break;
   577→      }
   578→
   579→      // ──────────────────────────────────────────────
   580→      // RECORD RETURN — with cumulative return tracking
   581→      // ──────────────────────────────────────────────
   582→      case 'record_return': {
   583→        if (matReq.status !== 'issued' && matReq.status !== 'partially_returned') {
   584→          return NextResponse.json({ success: false, error: `Cannot record return: current status is ${matReq.status}` }, { status: 400 });
   585→        }
   586→        const qtyToReturn = approvedQuantity ?? quantityApproved ?? quantityReturned ?? 0;
   587→
   588→        if (qtyToReturn <= 0) {
   589→          return NextResponse.json({ success: false, error: 'Return quantity must be greater than 0' }, { status: 400 });
   590→        }
   591→
   592→        // Validate cumulative returns don't exceed issued quantity
   593→        const previousReturned = matReq.quantityReturned || 0;
   594→        const cumulativeReturn = previousReturned + qtyToReturn;
   595→        if (cumulativeReturn > matReq.quantityIssued) {
   596→          return NextResponse.json({
   597→            success: false,
   598→            error: `Cumulative returns (${cumulativeReturn}) would exceed issued quantity (${matReq.quantityIssued}). Already returned: ${previousReturned}. Max additional return: ${matReq.quantityIssued - previousReturned}`,
   599→          }, { status: 400 });
   600→        }
   601→
   602→        // Add back to inventory
   603→        if (matReq.itemId && qtyToReturn > 0) {
   604→          const invItem = await db.inventoryItem.findUnique({ where: { id: matReq.itemId } });
   605→          if (invItem) {
   606→            await db.inventoryItem.update({
   607→              where: { id: matReq.itemId },
   608→              data: { currentStock: { increment: qtyToReturn } },
   609→            });
   610→            await db.stockMovement.create({
   611→              data: {
   612→                itemId: matReq.itemId,
   613→                type: 'in',
   614→                quantity: qtyToReturn,
   615→                previousStock: invItem.currentStock,
   616→                newStock: invItem.currentStock + qtyToReturn,
   617→                reason: `Returned from WO ${matReq.workOrder.woNumber}`,
   618→                referenceType: 'work_order',
   619→                referenceId: matReq.workOrderId,
   620→                performedById: session.userId,
   621→                notes: `Return #${Math.floor(previousReturned) + 1}: ${qtyToReturn} ${matReq.unit}` + (notes ? ` — ${notes}` : ''),
   622→              },
   623→            });
   624→          }
   625→        }
   626→
   627→        const newStatus = cumulativeReturn >= matReq.quantityIssued ? 'fully_returned' : 'partially_returned';
   628→        updated = await db.repairMaterialRequest.update({
   629→          where: { id },
   630→          data: {
   631→            status: newStatus,
   632→            quantityReturned: cumulativeReturn,
   633→            returnedById: session.userId,
   634→            returnedAt: now,
   635→          },
   636→        });
   637→
   638→        // Audit trail for return with cumulative tracking
   639→        await db.auditLog.create({
   640→          data: {
   641→            userId: session.userId,
   642→            action: 'material_request_record_return',
   643→            entityType: 'repair_material_request',
   644→            entityId: id,
   645→            newValues: JSON.stringify({
   646→              action: 'record_return',
   647→              status: newStatus,
   648→              returnQuantity: qtyToReturn,
   649→              previousReturned,
   650→              cumulativeReturned: cumulativeReturn,
   651→              quantityIssued: matReq.quantityIssued,
   652→              itemId: matReq.itemId || null,
   653→            }),
   654→          },
   655→        });
   656→
   657→        // Notify requester about the return
   658→        await notifyUser(
   659→          matReq.requestedById,
   660→          'repair_material_request',
   661→          newStatus === 'fully_returned'
   662→            ? 'All Materials Returned'
   663→            : 'Partial Material Return Recorded',
   664→          newStatus === 'fully_returned'
   665→            ? `All ${matReq.quantityIssued} ${matReq.unit} of ${matReq.itemName} returned for WO ${matReq.workOrder.woNumber}`
   666→            : `${qtyToReturn} ${matReq.unit} of ${matReq.itemName} returned for WO ${matReq.workOrder.woNumber}. Total returned: ${cumulativeReturn}/${matReq.quantityIssued}`,
   667→          'repair_material_request',
   668→          id,
   669→          `material-requests?id=${id}`,
   670→        );
   671→        break;
   672→      }
   673→
   674→      default:
   675→        return NextResponse.json({ success: false, error: `Unknown action: ${action}` }, { status: 400 });
   676→    }
   677→
   678→    // Note: per-action audit logs are written inside each case above.
   679→    // The catch-all audit log below is kept as a safety net but will be a duplicate.
   680→    // Since each action now writes its own detailed audit log, we skip the generic one.
   681→    // await db.auditLog.create({ ... });
   682→
   683→    return NextResponse.json({ success: true, data: updated });
   684→  } catch (error: unknown) {
   685→    const message = error instanceof Error ? error.message : 'Failed to process action';
   686→    return NextResponse.json({ success: false, error: message }, { status: 500 });
   687→  }
   688→}
