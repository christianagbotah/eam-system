   144→export function ProductionResourcePlanningPage() {
   145→  const [search, setSearch] = useState('');
   146→  const [filterType, setFilterType] = useState('all');
   147→  const [resourceData, setResourceData] = useState<any[]>([]);
   148→  const [loading, setLoading] = useState(true);
   149→  const [kpisData, setKpisData] = useState({ total: 0, overAllocated: 0, underUtilized: 0, utilization: 0 });
   150→  useEffect(() => {
   151→    (async () => {
   152→      const [wcRes, poRes] = await Promise.all([
   153→        api.get('/api/work-centers'),
   154→        api.get('/api/production-orders'),
   155→      ]);
   156→      if (wcRes.success && poRes.success) {
   157→        const wcs = (wcRes.data || []) as any[];
   158→        const pos = (poRes.data || []) as any[];
   159→        const resources: any[] = [];
   160→        wcs.forEach((wc: any, i: number) => {
   161→          const wcOrders = pos.filter((po: any) => po.workCenter?.id === wc.id);
   162→          const alloc = wcOrders.length > 0 ? Math.min(Math.round((wcOrders.length / Math.max(wcs.length, 1)) * 100 * 1.2), 115) : 0;
   163→          const status = alloc > 100 ? 'over-allocated' : alloc < 50 ? 'under-utilized' : 'allocated';
   164→          resources.push({ id: `RES-${String(i + 1).padStart(3, '0')}`, name: wc.name, type: 'machine', assignedTo: wcOrders.length > 0 ? wcOrders[0].orderNumber : 'Unassigned', allocation: alloc, available: wc.capacity || 40, status, shift: i % 3 === 0 ? 'Night' : i % 3 === 1 ? 'Day' : 'All' });
   165→        });
   166→        setResourceData(resources);
   167→        const overAllocated = resources.filter(r => r.status === 'over-allocated').length;
   168→        const underUtilized = resources.filter(r => r.status === 'under-utilized').length;
   169→        const avgUtil = resources.length > 0 ? Math.round(resources.reduce((s, r) => s + r.allocation, 0) / resources.length) : 0;
   170→        setKpisData({ total: resources.length, overAllocated, underUtilized, utilization: avgUtil });
   171→      }
   172→      setLoading(false);
   173→    })();
   174→  }, []);
   175→  const statusColors: Record<string, string> = { allocated: 'bg-emerald-50 text-emerald-700 border-emerald-200', available: 'bg-sky-50 text-sky-700 border-sky-200', 'over-allocated': 'bg-red-50 text-red-700 border-red-200', 'under-utilized': 'bg-amber-50 text-amber-700 border-amber-200' };

   178→    if (filterType !== 'all' && r.type !== filterType) return false;
   179→    if (search && !r.name.toLowerCase().includes(search.toLowerCase())) return false;
   180→    return true;
   181→  });
   182→  const kpis = [
   183→    { label: 'Resources Planned', value: kpisData.total, icon: Layers, color: 'text-emerald-600 bg-emerald-50' },
   184→    { label: 'Over-Allocated', value: kpisData.overAllocated, icon: AlertTriangle, color: 'text-red-600 bg-red-50' },
   185→    { label: 'Under-Utilized', value: kpisData.underUtilized, icon: TrendingDown, color: 'text-amber-600 bg-amber-50' },
   186→  Minus,
   187→    { label: 'Utilization', value: `${kpisData.utilization}%`, icon: Gauge, color: 'text-sky-600 bg-sky-50' },
   188→  ];
   189→  return (
   190→    <div className="page-content">
   191→      <div><h1 className="text-2xl font-bold tracking-tight">Resource Planning</h1><p className="text-muted-foreground text-sm mt-1">Plan and allocate resources for production orders</p></div>
   192→      <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
   193→        {kpis.map(k => { const I = k.icon; return (<Card key={k.label} className="bg-card text-card-foreground border border-border/60 rounded-xl shadow-sm"><CardContent className="p-5"><div className="flex items-center gap-4"><div className={`h-11 w-11 rounded-xl ${k.color} flex items-center justify-center`}><I className="h-5 w-5" /></div><div><p className="text-2xl font-bold">{k.value}</p><p className="text-xs text-muted-foreground">{k.label}</p></div></div></CardContent></Card>); })}

   197→        <Select value={filterType} onValueChange={setFilterType}><SelectTrigger className="w-[150px]"><SelectValue placeholder="Type" /></SelectTrigger><SelectContent><SelectItem value="all">All Types</SelectItem><SelectItem value="labor">Labor</SelectItem><SelectItem value="machine">Machine</SelectItem><SelectItem value="material">Material</SelectItem></SelectContent></Select>
   198→      </div>

   491→          const completed = wcOrders.filter(o => o.status === 'completed').length;

   503→        const getMonthKey = (d: Date) => {
   504→          return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
