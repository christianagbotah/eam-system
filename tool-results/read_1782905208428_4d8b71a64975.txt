   487→  console.log('=== Targeted Seed — Creating Missing Data ===\n');
   488→
   489→  // ── 1. Permissions ──
   490→  console.log('[1/7] Seeding permissions...');
   491→  let permCount = 0;
   492→  for (const [moduleName, actions] of Object.entries(modulePermissions)) {
   493→    for (const action of actions) {
   494→      const slug = `${moduleName}.${action}`;
   495→      await db.permission.upsert({
   496→        where: { slug },
   497→        update: {},
   498→        create: { slug, module: moduleName, action, name: `${moduleName} ${action}` },
   499→      });
   500→      permCount++;
   501→    }
   502→  }
   503→  console.log(`  ✅ ${permCount} permissions ensured\n`);
   504→
   505→  // ── 2. Roles ──
   506→  console.log('[2/7] Seeding roles...');
   507→  const createdRoles: Record<string, string> = {};
   508→  
   509→  // Also ensure admin role exists
   510→  const adminRole = await db.role.upsert({
   511→    where: { slug: 'admin' },
   512→    update: {},
   513→    create: { name: 'Administrator', slug: 'admin', description: 'Full system access', level: 100, isSystem: true },
   514→  });
   515→  createdRoles['admin'] = adminRole.id;
   516→
   517→  for (const roleDef of roleDefinitions) {
   518→    const role = await db.role.upsert({
   519→      where: { slug: roleDef.slug },
   520→      update: {},
   521→      create: { name: roleDef.name, slug: roleDef.slug, description: roleDef.description, level: roleDef.level, isSystem: roleDef.isSystem },
   522→    });
   523→    createdRoles[roleDef.slug] = role.id;
   524→  }
   525→  console.log(`  ✅ ${Object.keys(createdRoles).length} roles ensured\n`);
   526→
   527→  // ── 3. Role-Permission Assignments ──
   528→  console.log('[3/7] Assigning role permissions...');
   529→  let rpCount = 0;
   530→  for (const [roleSlug, permSlugs] of Object.entries(rolePermissionBundles)) {
   531→    const roleId = createdRoles[roleSlug];
   532→    if (!roleId) continue;
   533→
   534→    for (const permSlug of permSlugs) {
   535→      const perm = await db.permission.findUnique({ where: { slug: permSlug } });
   536→      if (!perm) continue;
   537→
   538→      await db.rolePermission.upsert({
   539→        where: { roleId_permissionId: { roleId, permissionId: perm.id } },
   540→        update: {},
   541→        create: { roleId, permissionId: perm.id },
   542→      });
   543→      rpCount++;
   544→    }
   545→  }
   546→  console.log(`  ✅ ${rpCount} role-permission assignments ensured\n`);
   547→
   548→  // ── 4. System Modules + Company Modules ──
   549→  console.log('[4/7] Seeding system modules...');
   550→  let modCount = 0;
   551→  for (const mod of systemModules) {
   552→    const sysMod = await db.systemModule.upsert({
   553→      where: { code: mod.code },
   554→      update: {},
   555→      create: {
   556→        code: mod.code,
   557→        name: mod.name,
   558→        description: mod.description,
   559→        version: mod.version,
   560→        isCore: mod.isCore,
   561→        isSystemLicensed: mod.isCore,
   562→      },
   563→    });
   564→
   565→    // Create CompanyModule to enable it
   566→    await db.companyModule.upsert({
   567→      where: {
   568→        systemModuleId_companyId: {
   569→          systemModuleId: sysMod.id,
   570→          companyId: '__default__',
   571→        },
   572→      },
   573→      update: {
   574→        isActive: mod.isCore,
   575→        isEnabled: mod.isCore,
   576→        licensedAt: mod.isCore ? new Date('2024-01-01') : null,
   577→        activatedAt: mod.isCore ? new Date('2024-01-01') : null,
   578→      },
   579→      create: {
   580→        systemModuleId: sysMod.id,
   581→        companyId: '__default__',
   582→        isActive: mod.isCore,
   583→        isEnabled: mod.isCore,
   584→        licensedAt: mod.isCore ? new Date('2024-01-01') : null,
   585→        activatedAt: mod.isCore ? new Date('2024-01-01') : null,
   586→      },
   587→    });
   588→    modCount++;
   589→  }
   590→  console.log(`  ✅ ${modCount} system modules + company modules ensured\n`);

   603→        name: dept.name,
   604→        code: dept.code,

   666→
   667→    // Find department
   668→    const dept = await db.department.findFirst({ where: { name: u.department } });
   669→    const deptId = dept?.id;
   670→

   685→    // Assign role
