   544→function TagBrowser({ tags, qualityScores }: { tags: TagMonitor[]; qualityScores: QualityScore[] }) {
   545→  const [search, setSearch] = useState('');
   546→  const [statusFilter, setStatusFilter] = useState<string>('all');
   547→
   548→  const qualityMap = useMemo(() => {
   549→    const m = new Map<string, QualityScore>();
   550→    for (const q of qualityScores) m.set(q.sourceId, q);
   551→    return m;
   552→  }, [qualityScores]);
   553→
   554→  const filtered = useMemo(() => {
   555→    return tags.filter((t) => {
   556→      if (search && !t.sourceId.toLowerCase().includes(search.toLowerCase())) return false;
   557→      if (statusFilter !== 'all' && t.status !== statusFilter) return false;
   558→      return true;
   559→    });

   563→    <Card className="border-0 shadow-sm">

   786→function ComparisonView({ tagMonitor }: { tagMonitor: TagMonitor[] }) {
   787→  const [selectedSources, setSelectedSources] = useState<string[]>([]);
   788→  const [timeRange, setTimeRange] = useState<TimeRange>('24h');
   789→  const [chartData, setChartData] = useState<Record<string, number | null>[]>([]);
   790→  const [chartLoading, setChartLoading] = useState(false);
   791→
   792→  const toggleSource = (srcId: string) => {
   793→    setSelectedSources(prev => {
   794→      if (prev.includes(srcId)) return prev.filter(s => s !== srcId);
   795→      if (prev.length >= 3) return prev;
   796→      return [...prev, srcId];
   797→    });
   798→  };
   799→
   800→  const fetchComparison = useCallback(async () => {
   801→    if (selectedSources.length === 0) { setChartData([]); return; }
   802→    setChartLoading(true);
   803→    try {
   804→      const range = TIME_RANGES.find(r => r.value === timeRange)!;
   805→      const from = new Date(Date.now() - range.ms);
   806→      const to = new Date();
   807→      const interval = autoInterval(timeRange);
   808→
   809→      const res = await fetch('/api/historian/aggregate', {
   810→        method: 'POST',
   811→        headers: { 'Content-Type': 'application/json' },
   812→        body: JSON.stringify({
   813→          operation: 'multi-source',
   814→          sourceIds: selectedSources,
   815→          from: from.toISOString(),
   816→          to: to.toISOString(),
   817→          interval,
   818→        }),
   819→      });
   820→      const json = await res.json();
   821→      if (json.success) {
   822→        const raw = json.data as Array<{ timestamp: string; sourceId: string; avg: number | null }>;
   823→        // Pivot: one row per timestamp with columns per source
   824→        const timeMap = new Map<string, Record<string, number | null>>();
   825→        for (const d of raw) {
   826→          const key = d.timestamp;
   827→          if (!timeMap.has(key)) timeMap.set(key, { time: new Date(key).getTime() });
   828→          timeMap.get(key)![d.sourceId] = d.avg;
   829→        }
   830→        const pivoted = [...timeMap.values()]
   831→          .sort((a, b) => (a.time! as number) - (b.time! as number))
   832→          .map(d => ({
   833→            time: new Date(d.time! as number).toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' }),
   834→            ...Object.fromEntries(selectedSources.map(s => [s, d[s] ?? null])),
   835→          }));
   836→        setChartData(pivoted);
   837→      }
   838→    } catch (e) {
   839→      console.error('Failed to fetch comparison data', e);
   840→    } finally {
   841→      setChartLoading(false);
   842→    }
   843→  }, [selectedSources, timeRange]);
   844→
   845→  useEffect(() => { fetchComparison(); }, [fetchComparison]);
   846→
   847→  const chartConfig = useMemo(() => {
   848→    const cfg: Record<string, { label: string; color: string }> = {};
   849→    selectedSources.forEach((s, i) => { cfg[s] = { label: s.slice(0, 20), color: COMPARISON_COLORS[i] }; });
   850→    return cfg;
   851→  }, [selectedSources]);
   852→
   853→  return (
   854→    <Card className="border-0 shadow-sm">
   855→      <CardHeader className="pb-3 px-5 pt-5">

   858→            <ArrowUpDown className="h-4 w-4 text-blue-500" />

   862→          <div className="flex flex-wrap gap-1.5">
   863→            {tagMonitor.slice(0, 20).map((t, i) => {
   864→              const isSelected = selectedSources.includes(t.sourceId);
   865→              const colorIdx = selectedSources.indexOf(t.sourceId);
   866→              return (
   867→                <Button key={t.sourceId} variant={isSelected ? 'default' : 'outline'} size="sm"
   868→                  className={`h-7 text-[11px] ${isSelected ? '' : ''}`}
   869→                  style={isSelected ? { backgroundColor: COMPARISON_COLORS[colorIdx], borderColor: COMPARISON_COLORS[colorIdx] } : {}}
   870→                  onClick={() => toggleSource(t.sourceId)}>
   871→                  {t.sourceId.slice(0, 18)}
   872→                </Button>
   873→              );
   874→            })}
   875→          </div>
   876→          <div className="flex gap-1">
   877→            {TIME_RANGES.map(r => (
   878→              <Button key={r.value} variant={timeRange === r.value ? 'default' : 'outline'} size="sm"
   879→                className={`h-7 px-2 text-[11px] ${timeRange === r.value ? 'bg-blue-600 hover:bg-blue-700' : ''}`}
   880→                onClick={() => setTimeRange(r.value)}>
   881→                {r.label}
   882→              </Button>

   884→          </div>
