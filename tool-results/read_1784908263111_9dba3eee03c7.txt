   100→
   101→  // Track autoOpen to avoid race condition between filter effect and fetch effect
   102→  const autoOpenRef = useRef<string | null>(null);
   103→  const hasAutoOpenedRef = useRef(false);
   104→

   106→  useEffect(() => {
   107→    if (pageParams?.status) {
   108→      setFilterStatus(pageParams.status);

   110→    if (pageParams?.id) {

   120→    const q = searchText.toLowerCase();
   121→    return requests.filter(r =>

   123→      r.requestNumber.toLowerCase().includes(q) ||

   145→        setLoading(false);
   146→        // Auto-open first request if navigated from bell/dashboard with autoOpen param

   166→      setDeleteId(null);

   200→                <Pencil className="h-4 w-4 mr-2" />View / Edit

   351→        variant="destructive"
   352→        loading={deleteLoading}
   353→        onConfirm={handleDeleteFromList}
   354→      />
   355→    </div>
   356→  );
   357→}

   363→export function CreateMRForm({ onSuccess }: { onSuccess: () => void }) {

   365→  const [title, setTitle] = useState('');
   366→  const [description, setDescription] = useState('');

   375→  const [manualAssetId, setManualAssetId] = useState('');
   376→  const [manualMode, setManualMode] = useState(false);
   377→  const [location, setLocation] = useState('');
   378→  const [loading, setLoading] = useState(false);
   379→
   380→  const fetchManualAssetOptions = useCallback(async () => {
   381→    const res = await api.get('/api/assets?limit=500');
   382→    if (res.success && res.data) {
   383→      const assets = (Array.isArray(res.data) ? res.data : []).map((a: any) => ({
   384→        value: a.id,
   385→        label: (a.name || a.assetTag) + (a.serialNumber ? ` — ${a.serialNumber}` : ''),
   386→      }));
   387→      return [...assets, { value: '__create_new__', label: '+ Create new asset', group: '' }];
   388→    }
   389→    return [{ value: '__create_new__', label: '+ Create new asset', group: '' }];
   390→  }, []);
   391→
   392→  // Auto-populate department from user's profile (read-only for non-admins)
   393→  useEffect(() => {
   394→    if (!user) return;
   395→    if (user.department) {
   396→      setDepartmentLabel(user.department);
   397→      // Look up department by name to get the ID
   398→      api.get('/api/departments').then(res => {
   399→        if (res.success && Array.isArray(res.data)) {
   400→          const dept = res.data.find((d: any) => d.name === user.department);
   401→          if (dept) setDepartmentId(dept.id);
   402→        }
   403→      });
   404→    }
   405→  }, [user?.department]);
   406→
   407→  const isDepartmentLocked = !isAdmin() && !!user?.department;
   408→
   409→  const handleSubmit = async (e: React.FormEvent) => {
   410→    e.preventDefault();
   411→    setLoading(true);
   412→    const payload: any = { title, description, priority, departmentId, category, machineDownStatus: machineDown, itemType, location };
   413→    if (itemType === 'machine' && assetId) payload.assetId = assetId;
   414→    if (itemType === 'manual') {
   415→      if (manualAssetId) payload.assetId = manualAssetId;
   416→      else if (manualAssetName) payload.assetName = manualAssetName;
   417→      else { toast.error('Please select or enter an asset name'); setLoading(false); return; }
   418→    }
   419→    const res = await api.post('/api/maintenance-requests', payload);
   420→    if (res.success) {
   421→      toast.success('Maintenance request created');
   422→      onSuccess();
   423→    } else {
   424→      toast.error(res.error || 'Failed to create request');
   425→    }
   426→    setLoading(false);
   427→  };
   428→
   429→  return (
   430→    <form id="create-mr-form" onSubmit={handleSubmit} className="space-y-4">
   431→      <div className="space-y-2">
   432→        <Label>Title *</Label>
   433→        <Input value={title} onChange={e => setTitle(e.target.value)} placeholder="Brief description of the issue" required />
   434→      </div>
   435→      <div className="space-y-2">
   436→        <Label>Description</Label>
   437→        <Textarea value={description} onChange={e => setDescription(e.target.value)} placeholder="Detailed description of the issue, including any relevant observations" rows={3} />
   438→      </div>
   439→

   611→// --- SLA Timer Sub-component ---
   612→function SLATimerDisplay({ slaHours, slaStartedAt, status }: { slaHours?: number; slaStartedAt?: string; status: string }) {
   613→  const [remaining, setRemaining] = useState<{ hours: number; minutes: number; seconds: number; breached: boolean } | null>(null);
   614→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   615→
   616→  useEffect(() => {
   617→    if (!slaHours || !slaStartedAt || status === 'converted' || status === 'rejected') {
   618→      setRemaining(null);
   619→      if (timerRef.current) clearInterval(timerRef.current);
   620→      return;
   621→    }
   622→    const calc = () => {
   623→      const deadline = new Date(slaStartedAt).getTime() + slaHours * 3600 * 1000;
   624→      const now = Date.now();
   625→      const diff = deadline - now;
   626→      if (diff <= 0) {
   627→        setRemaining({ hours: 0, minutes: 0, seconds: 0, breached: true });
   628→      } else {
   629→        setRemaining({ hours: Math.floor(diff / 3600000), minutes: Math.floor((diff % 3600000) / 60000), seconds: Math.floor((diff % 60000) / 1000), breached: false });
   630→      }
   631→    };
   632→    calc();
   633→    timerRef.current = setInterval(calc, 1000);
   634→    return () => { if (timerRef.current) clearInterval(timerRef.current); };
   635→  }, [slaHours, slaStartedAt, status]);
   636→
   637→  if (!remaining) return null;
   638→  const pad = (n: number) => String(n).padStart(2, '0');
   639→  return (
   640→    <Card className={`border-0 shadow-sm ${remaining.breached ? 'border-l-4 border-l-red-500 bg-red-50/50' : 'border-l-4 border-l-amber-500 bg-amber-50/50'}`}>
   641→      <CardContent className="p-4 flex items-center gap-3">
   642→        <div className={`h-10 w-10 rounded-lg flex items-center justify-center ${remaining.breached ? 'bg-red-100 text-red-600' : 'bg-amber-100 text-amber-600'}`}>
   643→          <Hourglass className="h-5 w-5" />
   644→        </div>
   645→        <div className="flex-1">
   646→          <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">SLA Timer</p>
   647→          <p className={`text-lg font-bold font-mono ${remaining.breached ? 'text-red-600' : 'text-amber-700'}`}>
   648→            {remaining.breached ? 'BREACHED' : `${pad(remaining.hours)}:${pad(remaining.minutes)}:${pad(remaining.seconds)}`}
   649→          </p>
   650→        </div>
   651→        <div className="text-right">
   652→          <p className="text-xs text-muted-foreground">{remaining.breached ? 'Time exceeded' : 'Time remaining'}</p>
   653→          <p className="text-xs text-muted-foreground">{slaHours}h SLA window</p>
   654→        </div>
   655→      </CardContent>
   656→    </Card>
   657→  );
   658→}
