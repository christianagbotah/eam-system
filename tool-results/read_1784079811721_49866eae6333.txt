   330→// --- AnalyticsKpiPage separator ---
   331→export function AnalyticsKpiPage() {
   332→  const [stats, setStats] = useState<DashboardStats | null>(null);
   333→  const [workOrders, setWorkOrders] = useState<WorkOrder[]>([]);
   334→  const [requests, setRequests] = useState<MaintenanceRequest[]>([]);
   335→  const [loading, setLoading] = useState(true);
   336→
   337→  useEffect(() => {
   338→    let active = true;
   339→    Promise.all([
   340→      api.get<DashboardStats>('/api/dashboard/stats', { timeout: 30_000 }),
   341→      api.get<WorkOrder[]>('/api/work-orders'),
   342→      api.get<MaintenanceRequest[]>('/api/maintenance-requests'),
   343→    ]).then(([statsRes, woRes, mrRes]) => {
   344→      if (!active) return;
   345→      if (statsRes.success && statsRes.data) setStats(statsRes.data);
   346→      if (woRes.success && woRes.data) setWorkOrders(woRes.data);
   347→      if (mrRes.success && mrRes.data) setRequests(mrRes.data);
   348→    }).catch(() => {
   349→      // Silently handle error
   350→    }).finally(() => {
   351→      if (active) setLoading(false);
   352→    });
   353→    return () => { active = false; };
   354→  }, []);
   355→
   356→  const totalWOs = stats?.totalWorkOrders || 0;
   357→  const completedWOs = stats?.completedWorkOrders || 0;
   358→  const completionRate = totalWOs > 0 ? Math.round((completedWOs / totalWOs) * 100) : 0;
   359→  const completedList = workOrders.filter(wo => wo.status === 'completed' || wo.status === 'verified' || wo.status === 'closed');
   360→  const mttr = completedList.length > 0 ? (completedList.reduce((sum, wo) => sum + (wo.actualHours || 0), 0) / completedList.length).toFixed(1) : '0.0';
   361→  const preventiveWOs = stats?.preventiveWO || 0;
   362→  const pmCompliance = totalWOs > 0 ? Math.round((preventiveWOs / totalWOs) * 100) : 0;
   363→
   364→  const kpiCards = [
   365→    { label: 'Total Assets', value: '-', icon: Building2, color: 'text-emerald-600 bg-emerald-50 dark:bg-emerald-900/30 dark:text-emerald-400' },
   366→    { label: 'Active Work Orders', value: stats?.activeWorkOrders || 0, icon: Wrench, color: 'text-amber-600 bg-amber-50 dark:bg-amber-900/30 dark:text-amber-400' },
   367→    { label: 'Completion Rate', value: `${completionRate}%`, icon: CheckCircle2, color: 'text-teal-600 bg-teal-50 dark:bg-teal-900/30 dark:text-teal-400' },
   368→    { label: 'MTTR (Hours)', value: mttr, icon: Clock, color: 'text-sky-600 bg-sky-50 dark:bg-sky-900/30 dark:text-sky-400' },
   369→    { label: 'Overdue WOs', value: stats?.overdueWorkOrders || 0, icon: AlertTriangle, color: 'text-red-600 bg-red-50 dark:bg-red-900/30 dark:text-red-400' },
   370→    { label: 'PM Compliance', value: `${pmCompliance}%`, icon: ClipboardCheck, color: 'text-violet-600 bg-violet-50 dark:bg-violet-900/30 dark:text-violet-400' },
   371→  ];
   372→
   373→  const recentWOs = [...workOrders].sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, 5);
   374→  const recentMRs = [...requests].sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, 5);
   375→
   376→  return (
   377→    <div className="page-content">
   378→      <div><h1 className="text-2xl font-bold tracking-tight">KPI Dashboard</h1><p className="text-muted-foreground mt-1">Organization-wide key performance indicators for maintenance, reliability, and operations</p></div>
   379→      {loading ? <LoadingSkeleton /> : (<>
   380→        <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-3 gap-4">
   381→          {kpiCards.map(k => { const I = k.icon; return (
   382→            <Card key={k.label}><CardContent className="p-5"><div className="flex items-center gap-4"><div className={`h-11 w-11 rounded-xl ${k.color} flex items-center justify-center`}><I className="h-5 w-5" /></div><div><p className="text-2xl font-bold">{k.value}</p><p className="text-xs text-muted-foreground">{k.label}</p></div></div></CardContent></Card>
   383→          ); })}
   384→        </div>
   385→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   386→          <Card className="border"><CardHeader><CardTitle className="text-base">Recent Work Orders</CardTitle><CardDescription className="text-xs">Latest work order activity</CardDescription></CardHeader><CardContent>
   387→            <div className="space-y-3">
   388→              {recentWOs.map(wo => (
   389→                <div key={wo.id} className="flex items-center justify-between p-2.5 rounded-lg bg-muted/30 hover:bg-muted/50 transition-colors">
   390→                  <div className="flex items-center gap-3 min-w-0">
   391→                    <span className="font-mono text-xs text-muted-foreground">{wo.woNumber}</span>
   392→                    <span className="font-medium text-sm truncate">{wo.title}</span>
   393→                  </div>
   394→                  <div className="flex items-center gap-2 shrink-0"><StatusBadge status={wo.status} /></div>
   395→                </div>
   396→              ))}
   397→              {recentWOs.length === 0 && <EmptyState icon={Wrench} title="No work orders" description="Work orders will appear here." />}
   398→            </div>
   399→          </CardContent></Card>
   400→          <Card className="border"><CardHeader><CardTitle className="text-base">Recent Maintenance Requests</CardTitle><CardDescription className="text-xs">Latest request activity</CardDescription></CardHeader><CardContent>

Math.min(95, Math.max(40, 100 - Math.round((readings.filter((r: any) => r.consumption && r.consumption > 0 && r.previousValue && r.previousValue > 0 && ((r.value - r.previousValue) / r.previousValue) > 0.15).length / Math.max(1, readings.filter((r: any) => r.consumption && r.consumption > 0).length)) * 30))) : 0;
   756→
   757→  const summaryCards = [
   758→    { label: 'Total Consumption', value: totalConsumption > 0 ? `${(totalConsumption / 1000).toFixed(1)} MWh` : '0 MWh', icon: Zap, color: 'text-amber-600 bg-amber-50 dark:bg-amber-900/30 dark:text-amber-400' },
   759→    { label: 'Total Cost', value: totalCost > 0 ? `${formatCurrency(totalCost / 1000)}k` : '₵0', icon: TrendingUp, color: 'text-emerald-600 bg-emerald-50 dark:bg-emerald-900/30 dark:text-emerald-400' },
   760→    { label: 'Avg Daily', value: `${avgDailyConsumption} kWh`, icon: BarChart3, color: 'text-sky-600 bg-sky-50 dark:bg-sky-900/30 dark:text-sky-400' },
   761→    { label: 'Efficiency Score', value: `${efficiencyScore}%`, icon: Target, color: 'text-violet-600 bg-violet-50 dark:bg-violet-900/30 dark:text-violet-400' },
   762→  ];
   763→
   764→  // Monthly consumption trend — group real readings by month
   765→  const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   766→  const monthlyMap: Record<string, number> = {};
   767→  readings.forEach((r: any) => {
   768→    if (!r.readingDate) return;
   769→    const d = new Date(r.readingDate);
   770→    const key = `${d.getFullYear()}-${String(d.getMonth()).padStart(2, '0')}`;
   771→    monthlyMap[key] = (monthlyMap[key] || 0) + (r.consumption || 0);
   772→  });
   773→  const monthlyData = Object.entries(monthlyMap)
   774→    .sort(([a], [b]) => a.localeCompare(b))
   775→    .slice(-12)
   776→    .map(([key, kwh]) => {
   777→      const parts = key.split('-');
   778→      return { month: monthNames[parseInt(parts[1], 10)] || key, kwh: Math.round(kwh), cost: Math.round(kwh * UNIT_COST_RATE) };
   779→    });
   780→  const maxKwh = monthlyData.length > 0 ? Math.max(...monthlyData.map(m => m.kwh)) : 1;
   781→
   782→  // Real meter readings for the table (show readings with consumption)
   783→  const meterReadings = readings.filter((r: any) => r.consumption && r.consumption > 0).slice(0, 20).map((r: any) => ({
   784→    id: r.id,
   785→    meter: r.meterName || r.readingNumber || '-',
   786→    reading: String(r.value),
   787→    date: r.readingDate ? new Date(r.readingDate).toISOString().split('T')[0] : '-',
   788→    consumption: Math.round(r.consumption),
   789→    cost: Math.round(r.consumption * UNIT_COST_RATE),
   790→  }));
   791→
   792→  // Top consumers — group consumption by meter name
   793→  const meterConsumption: Record<string, number> = {};
   794→  readings.forEach((r: any) => {
   795→    if (r.meterName && r.consumption && r.consumption > 0) {
   796→      meterConsumption[r.meterName] = (meterConsumption[r.meterName] || 0) + r.consumption;
   797→    }
   798→  });
   799→  const topConsumers = Object.entries(meterConsumption)
   800→    .map(([name, consumption]) => ({ name, consumption: Math.round(consumption) }))
   801→    .sort((a, b) => b.consumption - a.consumption)
   802→    .slice(0, 8);
   803→  const maxConsumption = topConsumers.length > 0 ? topConsumers[0].consumption : 1;
   804→
   805→  if (loading) return <div className="p-6 lg:p-8"><LoadingSkeleton /></div>;
   806→
   807→  return (
   808→    <div className="page-content">
   809→      <div>
   810→        <h1 className="text-2xl font-bold tracking-tight">Energy Analytics</h1>
   811→        <p className="text-muted-foreground mt-1">Monitor energy consumption patterns and optimize usage across assets</p>
   812→      </div>
   813→
   814→      {/* Summary cards */}
   815→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   816→        {summaryCards.map(k => { const I = k.icon; return (
   817→          <Card key={k.label}><CardContent className="p-4"><div className="flex items-center gap-3"><div className={`h-10 w-10 rounded-lg ${k.color} flex items-center justify-center`}><I className="h-5 w-5" /></div><div><p className="text-xl font-bold">{k.value}</p><p className="text-xs text-muted-foreground">{k.label}</p></div></div></CardContent></Card>
   818→        ); })}
   819→      </div>
   820→
   821→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   822→        {/* Monthly consumption trend */}
   823→        <Card>
   824→          <CardHeader className="pb-3"><CardTitle className="text-base">Monthly Consumption Trend</CardTitle><CardDescription>kWh per month from meter readings</CardDescription></CardHeader>
   825→          <CardContent className="space-y-2.5">
   826→            {monthlyData.map(m => (
   827→              <div key={m.month} className="flex items-center gap-3">
   828→                <span className="text-xs font-medium w-8 text-muted-foreground">{m.month}</span>
   829→                <div className="flex-1 bg-muted rounded-full h-5 overflow-hidden">
   830→                  <div className={`h-full rounded-full transition-all ${m.kwh >= 5000 ? 'bg-red-400' : m.kwh >= 4000 ? 'bg-amber-400' : 'bg-emerald-400'}`} style={{ width: `${(m.kwh / maxKwh) * 100}%` }} />
   831→                </div>
   832→                <span className="text-xs font-semibold w-16 text-right">{m.kwh.toLocaleString()}</span>
   833→              </div>
   834→            ))}
   835→            <div className="flex items-center gap-4 pt-2 border-t">
   836→              <div className="flex items-center gap-1.5"><div className="h-2 w-2 rounded-full bg-emerald-400" /><span className="text-[10px] text-muted-foreground">&lt; 4,000 kWh</span></div>
   837→              <div className="flex items-center gap-1.5"><div className="h-2 w-2 rounded-full bg-amber-400" /><span className="text-[10px] text-muted-foreground">4,000-5,000</span></div>
   838→              <div className="flex items-center gap-1.5"><div className="h-2 w-2 rounded-full bg-red-400" /><span className="text-[10px] text-muted-foreground">&gt; 5,000 kWh</span></div>
   839→            </div>
   840→          </CardContent>
   841→        </Card>
   842→
   843→        {/* Top energy consumers */}
   844→        <Card>
   845→          <CardHeader className="pb-3"><CardTitle className="text-base">Top Energy Consumers</CardTitle><CardDescription>Assets ranked by estimated consumption</CardDescription></CardHeader>
   846→          <CardContent className="space-y-3 max-h-80 overflow-y-auto">
   847→            {topConsumers.map((c: any, i: number) => (
   848→              <div key={c.name} className="flex items-center gap-3">
   849→                <span className="text-xs font-bold text-muted-foreground w-5">#{i + 1}</span>
   850→                <span className="text-sm flex-1 truncate">{c.name}</span>
   851→                <div className="w-24 bg-muted rounded-full h-2"><div className="bg-emerald-500 h-full rounded-full" style={{ width: `${(c.consumption / maxConsumption) * 100}%` }} /></div>
   852→                <span className="text-xs font-semibold w-16 text-right">{c.consumption.toLocaleString()} kWh</span>
   853→              </div>
   854→            ))}
   855→          </CardContent>
   856→        </Card>
   857→      </div>
   858→
   859→      {/* Meter readings table */}
   860→      <Card>
   861→        <CardHeader className="pb-3"><CardTitle className="text-base">Meter Readings</CardTitle><CardDescription>Latest meter readings and consumption data</CardDescription></CardHeader>
   862→        <CardContent>
   863→          <div className="overflow-x-auto">
   864→            <Table>
   865→              <TableHeader>
   866→                <TableRow>
   867→                  <TableHead>Meter</TableHead>
   868→                  <TableHead className="hidden md:table-cell">Reading</TableHead>
   869→                  <TableHead className="hidden sm:table-cell">Date</TableHead>
   870→                  <TableHead>Consumption (kWh)</TableHead>
   871→                  <TableHead className="hidden md:table-cell">Cost</TableHead>
   872→                </TableRow>
   873→              </TableHeader>
   874→              <TableBody>
   875→                {meterReadings.map(m => (
   876→                  <TableRow key={m.id}>
   877→                    <TableCell className="font-medium text-sm font-mono">{m.meter}</TableCell>
   878→                    <TableCell className="text-sm text-muted-foreground hidden md:table-cell">{m.reading}</TableCell>
   879→                    <TableCell className="text-sm hidden sm:table-cell">{m.date}</TableCell>
   880→                    <TableCell>
   881→                      <div className="flex items-center gap-2">
   882→                        <div className="w-12 bg-muted rounded-full h-1.5"><div className="bg-amber-400 h-full rounded-full" style={{ width: `${Math.min(100, (m.consumption / 3500) * 100)}%` }} /></div>
   883→                        <span className="text-sm font-semibold">{m.consumption.toLocaleString()}</span>
   884→                      </div>
   885→                    </TableCell>
   886→                    <TableCell className="text-sm text-muted-foreground hidden md:table-cell">{formatCurrency(m.cost)}</TableCell>
   887→                  </TableRow>
   888→                ))}
   889→              </TableBody>
   890→            </Table>
   891→          </div>
   892→        </CardContent>
   893→      </Card>
   894→    </div>
   895→  );
   896→}
