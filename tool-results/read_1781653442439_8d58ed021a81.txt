    31→function buildDbConfig() {
    32→  const dbUrl = process.env.DATABASE_URL || '';
    33→  if (dbUrl.includes('mysql://')) {
    34→    try {
    35→      const url = new URL(dbUrl);
    36→      return {
    37→        host: url.hostname,
    38→        port: parseInt(url.port || '3306', 10),
    39→        user: decodeURIComponent(url.username),
    40→        password: decodeURIComponent(url.password),
    41→        database: url.pathname.slice(1),
    42→      };
    43→    } catch { /* fall through */ }
    44→  }
    45→  const host = process.env.DB_HOST || 'localhost';
    46→  const port = parseInt(process.env.DB_PORT || '3306', 10);
    47→  const user = process.env.DB_USER || 'root';
    48→  const password = process.env.DB_PASSWORD || '';
    49→  const database = process.env.DB_NAME || 'ifleetpro_eam_system';
    50→  return { host, port, user, password, database };
    51→}
    52→
    53→const cfg = buildDbConfig();
    54→console.log(`  📡 Connecting to ${cfg.host}:${cfg.port}/${cfg.database}`);
    55→
    56→const adapter = new PrismaMariaDb({
    57→  host: cfg.host,
    58→  port: cfg.port,
    59→  user: cfg.user,
    60→  password: cfg.password,
    61→  database: cfg.database,
    62→  connectionLimit: 5,
    63→});
    64→
    65→const db = new PrismaClient({ adapter, log: ['warn', 'error'] });

    71→function formatModuleName(mod: string): string {

    75→function formatActionName(action: string): string {

    79→function makeSlug(text: string): string {
    80→  return text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '').slice(0, 80);
    81→}

    83→function logStep(emoji: string, msg: string) {

    87→function logErr(msg: string, err: unknown) {
    88→  const e = err as Error;
    89→  console.error(`  ❌ ${msg}`);
    90→  console.error(`     ${e.message?.slice(0, 200)}`);
    91→  if (process.env.DEBUG) console.error(e.stack);
    92→}
