    54→// Safe trade name extraction — API may return Trade object {id, name, code} or string
    55→function tradeName(trade: any): string {
    56→  if (!trade) return 'Unassigned';
    57→  if (typeof trade === 'string') return trade;
    58→  if (typeof trade === 'object' && trade.name) return trade.name;
    59→  return String(trade);
    60→}

    93→const exportCSV = (filename: string, headers: string[], rows: string[][]) => {
    94→  const csv = [headers.join(','), ...rows.map(r => r.map(c => `"${(c ?? '').toString().replace(/"/g, '""')}"`).join(','))].join('\n');
    95→  const blob = new Blob([csv], { type: 'text/csv' });
    96→  const url = URL.createObjectURL(blob);
    97→  const a = document.createElement('a');
    98→  a.href = url; a.download = `${filename}.csv`; a.click();
    99→  URL.revokeObjectURL(url);
   100→  toast.success(`Exported ${filename}.csv`);
   101→};

   301→export function WOReportsPage() {
   302→  const { startDate, setStartDate, endDate, setEndDate } = useDateRange();
   303→  const [reportData, setReportData] = useState<ReportData | null>(null);
   304→  const [loading, setLoading] = useState(false);
   305→  const [activeTab, setActiveTab] = useState('overview');
   306→  const [pdfLoading, setPdfLoading] = useState(false);
   307→  const [generatedAt, setGeneratedAt] = useState<string | null>(null);
   308→
   309→  // Filters
   310→  const [tradeFilter, setTradeFilter] = useState('');
   311→  const [priorityFilter, setPriorityFilter] = useState('');
   312→  const [deptFilter, setDeptFilter] = useState('');
   313→
   314→  // Fetch report data
   315→  const fetchReport = useCallback(() => {
   316→    setLoading(true);
   317→    api
   318→      .get<ReportData>(
   319→        `/api/work-orders/reports?from=${startDate}&to=${endDate}&trade=${tradeFilter || ''}&priority=${priorityFilter || ''}&department=${deptFilter || ''}`,
   320→        { timeout: 60_000 }
   321→      )
   322→      .then((res) => {
   323→        if (res.success && res.data) {
   324→          setReportData(res.data);
   325→          setGeneratedAt(formatTimestamp());
   326→        } else {
   327→          setReportData(null);
   328→        }
   329→        setLoading(false);
   330→      })
   331→      .catch(() => {
   332→        setReportData(null);
   333→        setLoading(false);
   334→        toast.error('Failed to load report data');
   335→      });
   336→  }, [startDate, endDate, tradeFilter, priorityFilter, deptFilter]);
   337→
   338→  // Auto-fetch on mount
   339→  React.useEffect(() => {
   340→    fetchReport();
   341→  }, [fetchReport]);
   342→
   343→  const s = reportData?.summary;
   344→
   345→  // ============================================================================
   346→  // CSV EXPORT (Enhanced per tab)
   347→  // ============================================================================
   348→
   349→  const handleExportCSV = useCallback(() => {
   350→    if (!reportData) return;
   351→    const tab = activeTab;
   352→    let filename = `wo-report-${startDate}-to-${endDate}`;
   353→    let headers: string[] = [];
   354→    let rows: string[][] = [];
   355→
   356→    switch (tab) {
   357→      case 'overview': {
   358→        filename += '-overview';
   359→        headers = ['Category', 'Sub-Category', 'Count', 'Percentage'];
   360→        const total = s?.totalWOs || 1;
   361→        (reportData.distribution?.byType || []).forEach((d: any) => rows.push(['Type', d.type, String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   362→        (reportData.distribution?.byStatus || []).forEach((d: any) => rows.push(['Status', d.status, String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   363→        (reportData.distribution?.byTrade || []).forEach((d: any) => rows.push(['Trade', tradeName(d.trade), String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   364→        (reportData.distribution?.byPriority || []).forEach((d: any) => rows.push(['Priority', d.priority, String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   365→        break;
   366→      }
   367→      case 'downtime': {
   368→        filename += '-downtime';
   369→        const dtData = reportData.downtime?.byTrade || [];
   370→        const totalHrs = dtData.reduce((a: number, b: any) => a + b.totalHours, 0);
   371→        const totalEvts = dtData.reduce((a: number, b: any) => a + b.events, 0);
   372→        headers = ['Trade', 'Events', 'Hours', 'Avg Hours/Event', '% of Total', 'Production Loss'];
   373→        rows = dtData.map((d: any) => [
   374→          tradeName(d.trade), String(d.events), String(d.totalHours),
   375→          d.events > 0 ? (d.totalHours / d.events).toFixed(1) : '0',
   376→          totalHrs > 0 ? `${((d.totalHours / totalHrs) * 100).toFixed(1)}%` : '0%',
   377→          String(d.productionLoss || 0),
   378→        ]);
   379→        rows.push(['TOTAL', String(totalEvts), String(totalHrs), '-', '100%', String(dtData.reduce((a: number, b: any) => a + (b.productionLoss || 0), 0))]);
   380→        break;
   381→      }
   382→      case 'response-time': {
   383→        filename += '-response-time';
   384→        headers = ['Group', 'Type', 'Count', 'Avg Hours', 'Min Hours', 'Max Hours', 'SLA Target (h)', 'SLA Compliance'];
   385→        (reportData.responseTime?.byPriority || []).forEach((d: any) => {
   386→          const target = SLA_TARGETS[d.priority] || 8;
   387→          const compliance = d.avgHours <= target ? 100 : Math.round((target / d.avgHours) * 100);
   388→          rows.push([d.priority, 'Priority', String(d.count), String(d.avgHours), String(d.minHours), String(d.maxHours), String(target), `${compliance}%`]);
   389→        });
   390→        (reportData.responseTime?.byTrade || []).forEach((d: any) =>
   391→          rows.push([tradeName(d.trade), 'Trade', String(d.count), String(d.avgHours), '-', '-', '-', '-'])
   392→        );
   393→        break;
   394→      }
   395→      case 'breakdowns': {
   396→        filename += '-breakdowns';
   397→        const bdTotal = reportData.breakdowns?.total || 1;
   398→        headers = ['Group', 'Type', 'Count', '% of Total'];
   399→        (reportData.breakdowns?.byTrade || []).forEach((d: any) => rows.push([tradeName(d.trade), 'Trade', String(d.count), `${((d.count/bdTotal)*100).toFixed(1)}%`]));
   400→        (reportData.breakdowns?.byType || []).forEach((d: any) => rows.push([d.type, 'Type', String(d.count), `${((d.count/bdTotal)*100).toFixed(1)}%`]));
   401→        (reportData.breakdowns?.byPriority || []).forEach((d: any) => rows.push([d.priority, 'Priority', String(d.count), `${((d.count/bdTotal)*100).toFixed(1)}%`]));
   402→        break;
   403→      }
   404→      case 'man-hours': {
   405→        filename += '-man-hours';
   406→        const mhTotal = reportData.manHours?.grandTotal || 1;
   407→        const techs = reportData.manHours?.byTechnician || [];
   408→        headers = ['Technician', 'Total Hours', 'WO Count', 'Avg Hrs/WO', '% of Grand Total'];
   409→        rows = techs.map((d: any) => [
   410→          d.name, String(d.totalHours), String(d.woCount),
   411→          String(d.avgHoursPerWO), `${((d.totalHours / mhTotal) * 100).toFixed(1)}%`,
   412→        ]);
   413→        rows.push(['TOTAL', String(reportData.manHours?.grandTotal || 0), '-', '-', '100%']);
   414→        break;
   415→      }
   416→      case 'materials': {
   417→        filename += '-materials';
   418→        const matTotal = reportData.materials?.totalCost || 1;
   419→        const items = reportData.materials?.topItems || [];
   420→        headers = ['Item Name', 'Total Qty', 'Total Cost', 'WOs Used', '% of Total Cost'];
   421→        rows = items.map((d: any) => [
   422→          d.name, String(d.totalQty), String(d.totalCost),
   423→          String(d.woCount), `${((d.totalCost / matTotal) * 100).toFixed(1)}%`,
   424→        ]);
   425→        rows.push(['TOTAL', String(reportData.materials?.totalQty || 0), String(reportData.materials?.totalCost || 0), '-', '100%']);
   426→        break;
   427→      }
   428→      case 'failure-rate': {
   429→        filename += '-failure-rate';
   430→        headers = ['Asset', 'Total WOs', 'Failures', 'Failure Rate', 'MTBF (Days)', 'Risk Level'];
   431→        const assets = reportData.failureRate?.byAsset || [];
   432→        rows = assets.map((d: any) => {
   433→          const mtbf = reportData.failureRate?.mtbf?.find((m: any) => m.assetId === d.assetId);
   434→          const risk = d.failureRate >= 40 ? 'High' : d.failureRate >= 20 ? 'Medium' : 'Low';
   435→          return [d.assetName || d.assetId, String(d.totalWOs), String(d.failures), `${d.failureRate}%`, String(mtbf?.mtbfDays || '-'), risk];
   436→        });
   437→        break;
   438→      }
   439→      case 'stoppages': {
   440→        filename += '-stoppages';
   441→        const spData = reportData.stoppages?.byTrade || [];
   442→        const totalSp = spData.reduce((a: number, b: any) => a + b.count, 0);
   443→        const totalSpHrs = spData.reduce((a: number, b: any) => a + b.totalHours, 0);
   444→        headers = ['Trade', 'Stoppage Count', 'Total Hours', 'Avg Hours', '% of Total'];
   445→        rows = spData.map((d: any) => [
   446→          tradeName(d.trade), String(d.count), String(d.totalHours),
   447→          d.count > 0 ? (d.totalHours / d.count).toFixed(1) : '0',
   448→          totalSp > 0 ? `${((d.count / totalSp) * 100).toFixed(1)}%` : '0%',
   449→        ]);
   450→        rows.push(['TOTAL', String(totalSp), String(totalSpHrs), '-', '100%']);
   451→        break;
   452→      }
   453→      default: {
   454→        headers = ['Metric', 'Value'];
   455→        rows = [
   456→          ['Total WOs', String(s?.totalWOs ?? 0)],
   457→          ['Completed', String(s?.completedWOs ?? 0)],
   458→          ['Completion Rate', String(s?.completionRate ?? 0)],
   459→        ];
   460→      }
   461→    }
   462→
   463→    exportCSV(filename, headers, rows);
   464→  }, [reportData, activeTab, startDate, endDate, s]);
   465→
   466→  // ============================================================================
   467→  // PDF EXPORT
   468→  // ============================================================================
   469→
   470→  const downloadPDF = useCallback(async () => {
   471→    try {
   472→      setPdfLoading(true);
   473→      const params = new URLSearchParams();
   474→      if (startDate) params.set('from', startDate);
   475→      if (endDate) params.set('to', endDate);
   476→      if (tradeFilter) params.set('trade', tradeFilter);
   477→      if (priorityFilter) params.set('priority', priorityFilter);
   478→      if (deptFilter) params.set('department', deptFilter);
   479→      params.set('format', 'pdf');
   480→
   481→      const response = await fetch(`/api/work-orders/reports?${params.toString()}`, {
   482→        headers: getAuthHeaders(),
   483→      });
   484→      if (!response.ok) throw new Error('Failed to generate PDF');
   485→
