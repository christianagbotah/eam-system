   326→function canApproveAsSupervisor(user: any): boolean {
   327→  if (!user) return false;
   328→  const { hasPermission, isAdmin, user: authUser } = useAuthStore.getState();
   329→  // Only supervisors, managers, and planners can approve tool/material requests
   330→  const supervisorRoles = ['admin', 'maintenance_manager', 'maintenance_supervisor', 'maintenance_planner', 'plant_manager'];
   331→  const userRoles = (authUser?.roles || []).map((r: any) => r.slug).filter(Boolean);
   332→  return isAdmin() || userRoles.some((slug: string) => supervisorRoles.includes(slug)) || (authUser?.role?.slug && supervisorRoles.includes(authUser.role.slug)) || hasPermission('repair_material_requests.update');
   333→}
   334→
   335→function canApproveAsStore(user: any): boolean {
   336→  if (!user) return false;
   337→  const { hasPermission, isAdmin, user: authUser } = useAuthStore.getState();
   338→  // Only store keepers, inventory managers, and admins can do store approval
   339→  const storeRoles = ['admin', 'inventory_manager', 'store_keeper', 'tools_shop_attendant'];
   340→  const userRoles = (authUser?.roles || []).map((r: any) => r.slug).filter(Boolean);
   341→  return isAdmin() || userRoles.some((slug: string) => storeRoles.includes(slug)) || (authUser?.role?.slug && storeRoles.includes(authUser.role.slug)) || hasPermission('repair_material_requests.update');
   342→}
