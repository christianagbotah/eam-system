   140→// ---------------------------------------------------------------------------
   141→// Helpers
   142→// ---------------------------------------------------------------------------
   143→
   144→function effColor(v: number): string {
   145→  if (v >= 97) return 'text-emerald-600 font-semibold';
   146→  if (v >= 90) return 'text-amber-600 font-semibold';
   147→  return 'text-red-600 font-semibold';
   148→}
   149→
   150→function effBg(v: number): string {
   151→  if (v >= 97) return 'bg-emerald-50';
   152→  if (v >= 90) return 'bg-amber-50';
   153→  return 'bg-red-50';
   154→}
   155→
   156→function dtColor(v: number): string {
   157→  if (v > 5) return 'text-red-600 font-semibold';
   158→  if (v >= 2) return 'text-amber-600 font-semibold';
   159→  return 'text-emerald-600 font-semibold';
   160→}
   161→
   162→function dtBg(v: number): string {
   163→  if (v > 5) return 'bg-red-50';
   164→  if (v >= 2) return 'bg-amber-50';
   165→  return 'bg-emerald-50';
   166→}
   167→
   168→function fmt(n: number, dec = 1): string {
   169→  if (n == null || Number.isNaN(n)) return '—';
   170→  return n.toFixed(dec);
   171→}
   172→
   173→function fmtInt(n: number): string {
   174→  if (n == null || Number.isNaN(n)) return '—';
   175→  return Math.round(n).toLocaleString();
   176→}
   177→
   178→function statusBadge(breakdowns: number): { label: string; variant: 'destructive' | 'outline' | 'secondary' } {
   179→  if (breakdowns > 5) return { label: 'Critical', variant: 'destructive' };
   180→  if (breakdowns >= 2) return { label: 'Warning', variant: 'outline' };
   181→  return { label: 'Normal', variant: 'secondary' };
   182→}

   195→export function MachineAvailabilityPage() {
   196→  const assetsEnabled = useModuleEnabled(MODULE_CODES.ASSETS);
   197→  const repairsEnabled = useModuleEnabled(MODULE_CODES.REPAIRS);

   207→  useEffect(() => {

   211→        if (res.success && res.data) {
   212→          setData(res.data);
   213→          // default to latest week
   214→          const weeks = res.data.weeklyKPIs.map((k: { week: number }) => k.week);
   215→          setSelectedWeek(weeks.length > 0 ? weeks[weeks.length - 1] : null);

   218→          toast.error(`Failed to load machine availability data: ${res.error || 'Unknown error'}`);

   231→    () => (data && selectedWeek ? data.weeklyKPIs.find((k) => k.week === selectedWeek) ?? null : null),
   232→    [data, selectedWeek],

   241→      const w = m.weekly.find(x => x.week === selectedWeek);
   242→      if (w) {
   243→        totalWE += w.weightedEfficiency;
   244→        totalPM += w.plannedMins;
   245→      }
   246→    }
   247→    return totalPM > 0 ? totalWE / totalPM : 100;
   248→  }, [data, selectedWeek]);

   250→  // --- CSV Export (Tab 1) ---
   251→  const exportCSV = useCallback(() => {
   252→    if (!data || !selectedWeek) return;
   253→    const header = [
   254→      'Machine Name', 'Code', 'Mfg Year', 'Install Year', 'Life',
   255→      'Planned (min)', 'Stoppages (min)', 'Repair DT (min)', 'Breakdowns',
   256→      'Actual Avail (min)', '% DT', 'Efficiency', 'Weighted Eff',
   257→    ];

   259→    for (const m of data.machines) {
   260→      const w = m.weekly.find((x) => x.week === selectedWeek);
   261→      rows.push([
   262→        m.assetName,
   263→        m.assetTag,
   264→        String(m.mfgYear ?? ''),
   265→        String(m.installYear ?? ''),
   266→        String(m.machineLife ?? ''),
   267→        w ? String(w.plannedMins) : '',
   268→        w ? String(w.stoppagesMins) : '',
   269→        w ? String(w.repairDowntimeMins) : '',
   270→        w ? String(w.breakdowns) : '',
   271→        w ? String(w.actualAvailability) : '',
   272→        w ? fmt(w.pctDowntime) : '',
   273→        w ? fmt(w.efficiency) : '',
   274→        w ? fmt(w.efficiency) : '',
   275→      ]);
   276→    }
   277→    const csv = rows.map((r) => r.join(',')).join('\n');
   278→    const blob = new Blob([csv], { type: 'text/csv' });
   279→    const url = URL.createObjectURL(blob);
   280→    const a = document.createElement('a');
   281→    a.href = url;
   282→    a.download = `machine-availability-wk${selectedWeek}.csv`;
   283→    a.click();
   284→    URL.revokeObjectURL(url);
   285→    toast.success('CSV exported');
   286→  }, [data, selectedWeek]);
   287→
   288→  // Module disabled check — AFTER all hooks
   289→  if (!assetsEnabled && !repairsEnabled) {
   290→    return (
   291→      <div className="flex items-center justify-center h-96">
   292→        <div className="text-center">
   293→          <p className="text-muted-foreground">Modules Not Active</p>
   294→          <p className="text-sm text-muted-foreground mt-1">Enable the Assets or Repairs module in Settings → Modules to view machine availability.</p>
   295→        </div>
   296→      </div>
   297→    );
   298→  }
   299→
   300→  // ===================== LOADING =====================
   301→  if (loading) {
   302→    return (
   303→      <div className="space-y-6 p-4 md:p-6">
   304→        <div className="flex items-center justify-between">
   305→          <Skeleton className="h-8 w-72" />
   306→          <Skeleton className="h-10 w-32" />
   307→        </div>
   308→        <div className="grid grid-cols-2 gap-4 md:grid-cols-4">
   309→          {Array.from({ length: 4 }).map((_, i) => (
   310→            <Skeleton key={i} className="h-24 w-full" />
   311→          ))}
   312→        </div>
   313→        <Skeleton className="h-[500px] w-full" />
   314→      </div>
   315→    );
   316→  }
   317→
   318→  if (!data) {
   319→    return (
   320→      <div className="flex flex-col items-center justify-center gap-4 p-12 text-center">
   321→        <BarChart3 className="h-12 w-12 text-muted-foreground" />
   322→        <p className="text-lg font-medium text-muted-foreground">No machine availability data found.</p>
   323→        <p className="text-sm text-muted-foreground">Try selecting a different year.</p>
   324→      </div>
   325→    );
   326→  }
   327→
   328→  // Derived aggregates for Tab 6
   329→  const latestWeek = weeks.length > 0 ? weeks[weeks.length - 1] : null;
   330→  const latestKPI = latestWeek ? data.weeklyKPIs.find((k) => k.week === latestWeek) : null;
   331→  const prevWeek = weeks.length > 1 ? weeks[weeks.length - 2] : null;
   332→  const prevKPI = prevWeek ? data.weeklyKPIs.find((k) => k.week === prevWeek) : null;
   333→
   334→  function trendVal(cur: number | undefined, prev: number | undefined): 'up' | 'down' | 'flat' {
   335→    if (cur == null || prev == null) return 'flat';
   336→    if (cur > prev) return 'up';
   337→    if (cur < prev) return 'down';
   338→    return 'flat';
   339→  }
   340→
   341→  // ===================== RENDER =====================
   342→  return (
   343→    <div className="space-y-4 p-4 md:p-6">
   344→      {/* Header */}
   345→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   346→        <div>
   347→          <h1 className="text-xl font-bold tracking-tight md:text-2xl">Machine Availability &amp; Efficiency</h1>
   348→          <p className="text-sm text-muted-foreground">
   349→            GTP-style weekly machine register &mdash; Year {data.year}
   350→          </p>
   351→        </div>
   352→        <div className="flex items-center gap-2">
   353→          <Select value={String(year)} onValueChange={(v) => setYear(Number(v))}>
   354→            <SelectTrigger className="w-28">
   355→              <SelectValue />
   356→            </SelectTrigger>
   357→            <SelectContent>
   358→              {[2023, 2024, 2025, 2026].map((y) => (
   359→                <SelectItem key={y} value={String(y)}>
   360→                  {y}
   361→                </SelectItem>
   362→              ))}
   363→            </SelectContent>
   364→          </Select>
   365→        </div>
   366→      </div>
   367→
   368→      {/* Tabs */}
   369→      <Tabs value={tab} onValueChange={setTab} className="space-y-4">
   370→        <TabsList className="flex w-full flex-wrap gap-1">
   371→          <TabsTrigger value="machine-details" className="text-xs sm:text-sm">Machine Details</TabsTrigger>
   372→          <TabsTrigger value="efficiency" className="text-xs sm:text-sm">Efficiency Analysis</TabsTrigger>
   373→          <TabsTrigger value="pareto" className="text-xs sm:text-sm">Pareto Analysis</TabsTrigger>
   374→          <TabsTrigger value="trends" className="text-xs sm:text-sm">Weekly Trends</TabsTrigger>
   375→          <TabsTrigger value="breakdown" className="text-xs sm:text-sm">Breakdown Summary</TabsTrigger>
   376→          <TabsTrigger value="kpis" className="text-xs sm:text-sm">Targets &amp; KPIs</TabsTrigger>
   377→        </TabsList>
   378→
   379→        {/* ============================================================= */}
   380→        {/* TAB 1 — Machine Details (GTP Weekly Register)                 */}

   382→        <TabsContent value="machine-details" className="space-y-4">
   383→          {/* KPI Strip */}
   384→          {selectedKPI && (
   385→            <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-5">
   386→              {[
   387→                { label: 'Total Machines', value: String(selectedKPI.totalMachines), icon: BarChart3, color: 'text-slate-600', targetTab: 'machine-details' as const },
   388→                { label: 'Available', value: String(selectedKPI.availableMachines), icon: Activity, color: 'text-emerald-600', targetTab: 'machine-details' as const },
   389→                { label: 'Efficient (≥97%)', value: String(selectedKPI.efficientMachines), icon: Gauge, color: 'text-emerald-600', targetTab: 'efficiency' as const },
   390→                { label: 'Weighted Avg Eff.', value: fmt(selectedKPI.weightedAvgEfficiency) + '%', icon: Target, color: selectedKPI.weightedAvgEfficiency >= 97 ? 'text-emerald-600' : selectedKPI.weightedAvgEfficiency >= 90 ? 'text-amber-600' : 'text-red-600', targetTab: 'efficiency' as const },
   391→                repairsEnabled ? { label: 'Total Breakdowns', value: String(selectedKPI.totalBreakdowns), icon: AlertTriangle, color: selectedKPI.totalBreakdowns <= (data.targets.breakdownsWeekly ?? 2) ? 'text-emerald-600' : 'text-red-600', targetTab: 'breakdown' as const } : null,
   392→              ].filter(Boolean).map((item) => (
   393→                <Card key={item.label} className="border border-border/60 shadow-sm cursor-pointer hover:shadow-md transition-shadow" onClick={() => setTab(item.targetTab)}>
   394→                  <CardContent className="flex items-center gap-3 p-4">
   395→                    <item.icon className={`h-5 w-5 shrink-0 ${item.color}`} />
   396→                    <div className="min-w-0">
   397→                      <p className="truncate text-xs text-muted-foreground">{item.label}</p>
   398→                      <p className="text-lg font-bold leading-tight">{item.value}</p>
   399→                    </div>
   400→                  </CardContent>
   401→                </Card>
   402→              ))}
   403→            </div>
   404→          )}
   405→
   406→          {/* Week Selector + Export */}
   407→          <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   408→            <div className="flex items-center gap-2">
   409→              <span className="text-sm font-medium">Week:</span>
   410→              <Select
   411→                value={selectedWeek ? String(selectedWeek) : ''}
   412→                onValueChange={(v) => setSelectedWeek(Number(v))}
   413→              >
   414→                <SelectTrigger className="w-52">
   415→                  <SelectValue placeholder="Select week" />
   416→                </SelectTrigger>
   417→                <SelectContent>
   418→                  {weeks.map((w) => {
   419→                    const label = data.weeklyKPIs.find((k) => k.week === w)?.weekLabel ?? `Week ${w}`;
   420→                    return (
   421→                      <SelectItem key={w} value={String(w)}>
   422→                        {label}
   423→                      </SelectItem>
   424→                    );
   425→                  })}
   426→                </SelectContent>
   427→              </Select>
   428→            </div>
   429→            <Button variant="outline" size="sm" onClick={exportCSV} disabled={!selectedWeek}>
   430→              <Download className="mr-1.5 h-4 w-4" />
   431→              Export CSV
   432→            </Button>
   433→          </div>
   434→
   435→          {/* Machine Details Table */}
   436→          <Card className="border border-border/60 shadow-sm">
   437→            <CardHeader className="px-4 py-3">
   438→              <CardTitle className="text-sm font-semibold">
   439→                Week {selectedWeek} — Machine Register
   440→              </CardTitle>
   441→            </CardHeader>
   442→            <CardContent className="p-0">
   443→              <div className="max-h-[480px] overflow-auto">
   444→                <table className="w-full min-w-[800px] text-xs">
   445→                  <thead>
   446→                    <tr className="sticky top-0 z-10 bg-background border-b">
   447→                      <th className="px-3 py-2 text-left font-semibold text-muted-foreground">Machine Name</th>
   448→                      <th className="px-3 py-2 text-left font-semibold text-muted-foreground">Code</th>
   449→                      <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Mfg Year</th>
   450→                      <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Install Year</th>
   451→                      <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Life (yrs)</th>

   454→                      {repairsEnabled && dtEnabled && <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Repair DT (min)</th>}
   455→                      {repairsEnabled && <th className="px-3 py-2 text-right font-semibold text-muted-foreground"># BD</th>}
   456→                      <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Actual Avail (min)</th>
   457→                      {repairsEnabled && dtEnabled && <th className="px-3 py-2 text-right font-semibold text-muted-foreground">% Downtime</th>}
   458→                      <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Efficiency</th>
   459→                      <th className="px-3 py-2 text-right font-semibold text-muted-foreground">Weighted Eff</th>
   460→                    </tr>
   461→                  </thead>
   462→                  <tbody>
   463→                    {data.machines.map((m, idx) => {
   464→                      const w = m.weekly.find((x) => x.week === selectedWeek);

   466→                      const pct = w?.pctDowntime ?? 0;
   467→                      return (

   470→                          className={idx % 2 === 0 ? 'bg-background' : 'bg-muted/30'}

   478→                          {repairsEnabled && dtEnabled && <td className="px-3 py-2 text-right">{w ? fmtInt(w.stoppagesMins) : '—'}</td>}
   479→                          {repairsEnabled && dtEnabled && <td className="px-3 py-2 text-right">{w ? fmtInt(w.repairDowntimeMins) : '—'}</td>}
   480→                          {repairsEnabled && <td className="px-3 py-2 text-right">{w ? String(w.breakdowns) : '—'}</td>}
