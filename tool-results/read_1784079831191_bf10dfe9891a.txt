     1→import { NextRequest, NextResponse } from 'next/server';
     2→import { db } from '@/lib/db';
     3→import { getSession, hasPermission, isAdmin, hasAnyPermission } from '@/lib/auth';
     4→import { getPlantScope, getPlantFilterWhere } from '@/lib/plant-scope';
     5→import { generateReportPDF } from '@/lib/generate-report-pdf';
     6→
     7→// GET /api/work-orders/reports
     8→// Comprehensive WO reports: downtime, response time, breakdowns, man hours, materials, failure rate, stoppages
     9→export async function GET(request: NextRequest) {
    10→  try {
    11→    const session = getSession(request);
    12→    if (!session) {
    13→      return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
    14→    }
    15→    // Permission gate: require reports.view, work_orders.dashboard, or admin
    16→    if (!hasAnyPermission(session, ['reports.view', 'reports.export', 'work_orders.dashboard', 'analytics.view']) && !isAdmin(session)) {
    17→      return NextResponse.json({ success: false, error: 'Insufficient permissions: reports.view required' }, { status: 403 });
    18→    }
    19→
    20→    const { searchParams } = new URL(request.url);
    21→    const from = searchParams.get('from');
    22→    const to = searchParams.get('to');
    23→    const department = searchParams.get('department');
    24→    const trade = searchParams.get('trade');
    25→    const priority = searchParams.get('priority');
    26→    const format = searchParams.get('format');
    27→
    28→    const plantScope = await getPlantScope(request, session);
    29→    const plantFilter = getPlantFilterWhere(plantScope);
    30→
    31→    // Date filter
    32→    const dateFilter: Record<string, unknown> = {};
    33→    if (from) dateFilter.gte = new Date(from + 'T00:00:00');
    34→    if (to) dateFilter.lte = new Date(to + 'T23:59:59');
    35→
    36→    const woWhere: Record<string, unknown> = {
    37→      ...plantFilter,
    38→      status: { notIn: ['draft', 'cancelled'] },
    39→    };
    40→    if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
    41→    if (department) woWhere.departmentId = department;
    42→    if (trade) woWhere.tradeActivity = trade;
    43→    if (priority) woWhere.priority = priority;
    44→
    45→    // Fetch all relevant WOs with relations
    46→    const workOrders = await db.workOrder.findMany({
    47→      where: Object.keys(woWhere).length > 0 ? woWhere : undefined,
    48→      include: {
    49→        assignee: { select: { id: true, fullName: true, department: true } },
    50→        teamLeader: { select: { id: true, fullName: true } },
    51→        materials: true,
    52→        timeLogs: true,
    53→        workOrderDowntimes: true,
    54→        repairCompletion: true,
    55→        statusHistory: { orderBy: { createdAt: 'asc' } },
    56→        taskExecutions: true,
    57→        failureRecords: true,
    58→      },
    59→      orderBy: { createdAt: 'desc' },
    60→    });
    61→
    62→    const total = workOrders.length;
    63→    const closedWOs = workOrders.filter(wo => wo.status === 'closed');
    64→    const completedWOs = workOrders.filter(wo => ['completed', 'verified', 'closed'].includes(wo.status));
    65→    const breakdownWOs = workOrders.filter(wo => ['emergency', 'corrective'].includes(wo.type));
    66→
    67→    // ========== 1. SUMMARY KPIs ==========
    68→    const summary = {
    69→      totalWOs: total,
    70→      completedWOs: completedWOs.length,
    71→      closedWOs: closedWOs.length,
    72→      completionRate: total > 0 ? Math.round((completedWOs.length / total) * 10000) / 100 : 0,
    73→      breakdownCount: breakdownWOs.length,
    74→      breakdownRate: total > 0 ? Math.round((breakdownWOs.length / total) * 10000) / 100 : 0,
    75→    };
    76→
    77→    // ========== 2. DOWNTIME (per trade) ==========
    78→    const downtimeByTrade: Record<string, { totalMinutes: number; events: number; productionLoss: number }> = {};
    79→    for (const wo of workOrders) {
    80→      const dtEvents = wo.workOrderDowntimes || [];
    81→      const trade = wo.tradeActivity || 'Unassigned';
    82→      for (const dt of dtEvents) {
    83→        if (!downtimeByTrade[trade]) downtimeByTrade[trade] = { totalMinutes: 0, events: 0, productionLoss: 0 };
    84→        downtimeByTrade[trade].totalMinutes += dt.durationMinutes;
    85→        downtimeByTrade[trade].events++;
    86→        downtimeByTrade[trade].productionLoss += dt.productionLoss || 0;
    87→      }
    88→    }
    89→    const totalDowntimeMinutes = Object.values(downtimeByTrade).reduce((s, d) => s + d.totalMinutes, 0);
    90→    const totalDowntimeEvents = Object.values(downtimeByTrade).reduce((s, d) => s + d.events, 0);
    91→    const totalProductionLoss = Object.values(downtimeByTrade).reduce((s, d) => s + d.productionLoss, 0);
    92→
    93→    const downtimeByTradeArray = Object.entries(downtimeByTrade)
    94→      .map(([tradeName, data]) => ({
    95→        trade: tradeName,
    96→        totalHours: Math.round((data.totalMinutes / 60) * 100) / 100,
    97→        totalMinutes: data.totalMinutes,
    98→        events: data.events,
    99→        productionLoss: data.productionLoss,
   100→      }))
   101→      .sort((a, b) => b.totalHours - a.totalHours);
   102→
   103→    // Downtime by month
   104→    const downtimeByMonth: Record<string, { minutes: number; events: number }> = {};
   105→    for (const wo of workOrders) {
   106→      for (const dt of (wo.workOrderDowntimes || [])) {
   107→        const month = dt.downtimeStart?.toISOString().slice(0, 7) || 'unknown';
   108→        if (!downtimeByMonth[month]) downtimeByMonth[month] = { minutes: 0, events: 0 };
   109→        downtimeByMonth[month].minutes += dt.durationMinutes;
   110→        downtimeByMonth[month].events++;
   111→      }
   112→    }
   113→    const downtimeByMonthArray = Object.entries(downtimeByMonth)
   114→      .map(([month, data]) => ({
   115→        month,
   116→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   117→        events: data.events,
   118→      }))
   119→      .sort((a, b) => a.month.localeCompare(b.month));
   120→
   121→    // Downtime by category
   122→    const downtimeByCategory: Record<string, { minutes: number; events: number }> = {};
   123→    for (const wo of workOrders) {
   124→      for (const dt of (wo.workOrderDowntimes || [])) {
   125→        const cat = dt.category || 'unplanned';
   126→        if (!downtimeByCategory[cat]) downtimeByCategory[cat] = { minutes: 0, events: 0 };
   127→        downtimeByCategory[cat].minutes += dt.durationMinutes;
   128→        downtimeByCategory[cat].events++;
   129→      }
   130→    }
   131→    const downtimeByCategoryArray = Object.entries(downtimeByCategory)
   132→      .map(([category, data]) => ({
   133→        category,
   134→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   135→        events: data.events,
   136→      }))
   137→      .sort((a, b) => b.totalHours - a.totalHours);
   138→
   139→    // ========== 3. RESPONSE TIME ==========
   140→    // Response time = time from WO creation to first status that indicates work started (in_progress)
   141→    const responseTimes: { hours: number; priority: string; type: string; trade: string; woId: string }[] = [];
   142→    for (const wo of workOrders) {
   143→      const history = wo.statusHistory || [];
   144→      if (history.length === 0) continue;
   145→
   146→      // Find the transition to 'in_progress' (or 'assigned' as fallback)
   147→      const startEntry = history.find(h => h.toStatus === 'in_progress')
   148→        || history.find(h => h.toStatus === 'assigned');
   149→
   150→      if (startEntry && wo.createdAt) {
   151→        const hours = (startEntry.createdAt.getTime() - wo.createdAt.getTime()) / (1000 * 60 * 60);
   152→        if (hours >= 0 && hours < 10000) { // sanity check
   153→          responseTimes.push({
   154→            hours: Math.round(hours * 100) / 100,
   155→            priority: wo.priority,
   156→            type: wo.type,
   157→            trade: wo.tradeActivity || 'Unassigned',
   158→            woId: wo.id,
   159→          });
   160→        }
   161→      }
   162→    }
   163→
   164→    const avgResponseTime = responseTimes.length > 0
   165→      ? Math.round(responseTimes.reduce((s, r) => s + r.hours, 0) / responseTimes.length * 100) / 100
   166→      : 0;
   167→
   168→    // Response time by priority
   169→    const responseByPriority: Record<string, number[]> = {};
   170→    for (const r of responseTimes) {
   171→      if (!responseByPriority[r.priority]) responseByPriority[r.priority] = [];
   172→      responseByPriority[r.priority].push(r.hours);
   173→    }
   174→    const responseByPriorityArray = Object.entries(responseByPriority)
   175→      .map(([p, hours]) => ({
   176→        priority: p,
   177→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   178→        minHours: Math.round(Math.min(...hours) * 100) / 100,
   179→        maxHours: Math.round(Math.max(...hours) * 100) / 100,
   180→        count: hours.length,
   181→      }))
   182→      .sort((a, b) => a.avgHours - b.avgHours);
   183→
   184→    // Response time by trade
   185→    const responseByTrade: Record<string, number[]> = {};
   186→    for (const r of responseTimes) {
   187→      if (!responseByTrade[r.trade]) responseByTrade[r.trade] = [];
   188→      responseByTrade[r.trade].push(r.hours);
   189→    }
   190→    const responseByTradeArray = Object.entries(responseByTrade)
   191→      .map(([t, hours]) => ({
   192→        trade: t,
   193→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   194→        count: hours.length,
   195→      }))
   196→      .sort((a, b) => a.avgHours - b.avgHours);
   197→
   198→    // ========== 4. BREAKDOWNS ==========
   199→    // Breakdown by type
   200→    const breakdownByType: Record<string, number> = {};
   201→    for (const wo of breakdownWOs) {
   202→      const t = wo.type || 'unknown';
   203→      breakdownByType[t] = (breakdownByType[t] || 0) + 1;
   204→    }
   205→    const breakdownByTypeArray = Object.entries(breakdownByType)
   206→      .map(([type, count]) => ({ type, count }))
   207→      .sort((a, b) => b.count - a.count);
   208→
   209→    // Breakdown by trade
   210→    const breakdownByTrade: Record<string, number> = {};
   211→    for (const wo of breakdownWOs) {
   212→      const t = wo.tradeActivity || 'Unassigned';
   213→      breakdownByTrade[t] = (breakdownByTrade[t] || 0) + 1;
   214→    }
   215→    const breakdownByTradeArray = Object.entries(breakdownByTrade)
   216→      .map(([trade, count]) => ({ trade, count }))
   217→      .sort((a, b) => b.count - a.count);
   218→
   219→    // Breakdown by month
   220→    const breakdownByMonth: Record<string, number> = {};
   221→    for (const wo of breakdownWOs) {
   222→      const month = wo.createdAt.toISOString().slice(0, 7);
   223→      breakdownByMonth[month] = (breakdownByMonth[month] || 0) + 1;
   224→    }
   225→    const breakdownByMonthArray = Object.entries(breakdownByMonth)
   226→      .map(([month, count]) => ({ month, count }))
   227→      .sort((a, b) => a.month.localeCompare(b.month));
   228→
   229→    // Breakdown by priority
   230→    const breakdownByPriority: Record<string, number> = {};
   231→    for (const wo of breakdownWOs) {
   232→      const p = wo.priority || 'medium';
   233→      breakdownByPriority[p] = (breakdownByPriority[p] || 0) + 1;
   234→    }
   235→    const breakdownByPriorityArray = Object.entries(breakdownByPriority)
   236→      .map(([priority, count]) => ({ priority, count }))
   237→      .sort((a, b) => b.count - a.count);
   238→
   239→    // ========== 5. MAN HOURS ==========
   240→    // From timeLogs and repairCompletion
   241→    const manHoursByTrade: Record<string, { totalHours: number; woCount: number }> = {};
   242→    const manHoursByTechnician: Record<string, { name: string; totalHours: number; woCount: number }> = {};
   243→    let grandTotalManHours = 0;
   244→
   245→    for (const wo of workOrders) {
   246→      const trade = wo.tradeActivity || 'Unassigned';
   247→
   248→      // Get hours from timeLogs (primary), actualHours (fallback), repairCompletion (override)
   249→      let woHours = 0;
   250→      for (const tl of (wo.timeLogs || [])) {
   251→        if (tl.duration && tl.duration > 0) {
   252→          woHours += tl.duration;
   253→        }
   254→      }
   255→      // Fallback to WO's actualHours when no time logs exist
   256→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   257→        woHours = wo.actualHours;
   258→      }
   259→      // Override with repairCompletion if higher (supervisor-verified hours)
   260→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   261→        woHours = wo.repairCompletion.totalLaborHours;
   262→      }
   263→
   264→      grandTotalManHours += woHours;
   265→
   266→      // By trade
   267→      if (!manHoursByTrade[trade]) manHoursByTrade[trade] = { totalHours: 0, woCount: 0 };
   268→      manHoursByTrade[trade].totalHours += woHours;
   269→      manHoursByTrade[trade].woCount++;
   270→
   271→      // By technician
   272→      const techId = wo.assignedTo || 'unassigned';
   273→      const techName = wo.assignee?.fullName || 'Unassigned';
   274→      if (!manHoursByTechnician[techId]) {
   275→        manHoursByTechnician[techId] = { name: techName, totalHours: 0, woCount: 0 };
   276→      }
   277→      manHoursByTechnician[techId].totalHours += woHours;
   278→      manHoursByTechnician[techId].woCount++;
   279→    }
   280→
   281→    const manHoursByTradeArray = Object.entries(manHoursByTrade)
   282→      .map(([trade, data]) => ({
   283→        trade,
   284→        totalHours: Math.round(data.totalHours * 100) / 100,
   285→        woCount: data.woCount,
   286→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   287→      }))
   288→      .sort((a, b) => b.totalHours - a.totalHours);
   289→
   290→    const manHoursByTechnicianArray = Object.values(manHoursByTechnician)
   291→      .map(data => ({
   292→        ...data,
   293→        totalHours: Math.round(data.totalHours * 100) / 100,
   294→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   295→      }))
   296→      .sort((a, b) => b.totalHours - a.totalHours);
   297→
   298→    // Man hours by month
   299→    const manHoursByMonth: Record<string, number> = {};
   300→    for (const wo of workOrders) {
   301→      const month = wo.createdAt.toISOString().slice(0, 7);
   302→      let woHours = 0;
   303→      for (const tl of (wo.timeLogs || [])) {
   304→        if (tl.duration && tl.duration > 0) woHours += tl.duration;
   305→      }
   306→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   307→        woHours = wo.actualHours;
   308→      }
   309→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   310→        woHours = wo.repairCompletion.totalLaborHours;
   311→      }
   312→      manHoursByMonth[month] = (manHoursByMonth[month] || 0) + woHours;
   313→    }
   314→    const manHoursByMonthArray = Object.entries(manHoursByMonth)
   315→      .map(([month, totalHours]) => ({
   316→        month,
   317→        totalHours: Math.round(totalHours * 100) / 100,
   318→      }))
   319→      .sort((a, b) => a.month.localeCompare(b.month));
   320→
   321→    // Man hours by activity type
   322→    const manHoursByActivity: Record<string, number> = {};
   323→    for (const wo of workOrders) {
   324→      for (const tl of (wo.timeLogs || [])) {
   325→        const act = tl.activityType || 'maintenance';
   326→        manHoursByActivity[act] = (manHoursByActivity[act] || 0) + (tl.duration || 0);
   327→      }
   328→    }
   329→    const manHoursByActivityArray = Object.entries(manHoursByActivity)
   330→      .map(([activity, totalHours]) => ({
   331→        activity,
   332→        totalHours: Math.round(totalHours * 100) / 100,
   333→      }))
   334→      .sort((a, b) => b.totalHours - a.totalHours);
   335→
   336→    // ========== 6. MATERIAL QUANTITY AND COST ==========
   337→    let totalMaterialCost = 0;
   338→    let totalMaterialQty = 0;
   339→    const materialsByCategory: Record<string, { qty: number; cost: number; woCount: number }> = {};
   340→    const topMaterials: Record<string, { name: string; qty: number; cost: number; woCount: Set<string> }> = {};
   341→
   342→    for (const wo of workOrders) {
   343→      for (const mat of (wo.materials || [])) {
   344→        const qty = mat.quantity || 0;
   345→        const cost = mat.totalCost || (mat.unitCost && mat.quantity ? mat.unitCost * mat.quantity : 0);
   346→        totalMaterialQty += qty;
   347→        totalMaterialCost += cost;
   348→
   349→        // By WO type
   350→        const woType = wo.type || 'other';
   351→        if (!materialsByCategory[woType]) materialsByCategory[woType] = { qty: 0, cost: 0, woCount: 0 };
   352→        materialsByCategory[woType].qty += qty;
   353→        materialsByCategory[woType].cost += cost;
   354→        materialsByCategory[woType].woCount++;
   355→
   356→        // Top items
   357→        const itemName = mat.itemName || mat.itemId || 'Unknown Item';
   358→        if (!topMaterials[itemName]) topMaterials[itemName] = { name: itemName, qty: 0, cost: 0, woCount: new Set() };
   359→        topMaterials[itemName].qty += qty;
   360→        topMaterials[itemName].cost += cost;
   361→        topMaterials[itemName].woCount.add(wo.id);
   362→      }
   363→    }
   364→
   365→    const materialsByCategoryArray = Object.entries(materialsByCategory)
   366→      .map(([type, data]) => ({
   367→        type,
   368→        totalQty: Math.round(data.qty * 100) / 100,
   369→        totalCost: Math.round(data.cost * 100) / 100,
   370→        woCount: data.woCount,
   371→      }))
   372→      .sort((a, b) => b.totalCost - a.totalCost);
   373→
   374→    const topMaterialsArray = Object.values(topMaterials)
   375→      .map(data => ({
   376→        name: data.name,
   377→        totalQty: Math.round(data.qty * 100) / 100,
   378→        totalCost: Math.round(data.cost * 100) / 100,
   379→        woCount: data.woCount.size,
   380→      }))
   381→      .sort((a, b) => b.totalCost - a.totalCost)
   382→      .slice(0, 20);
   383→
   384→    // Material cost by month
   385→    const materialCostByMonth: Record<string, number> = {};
   386→    for (const wo of workOrders) {
   387→      const month = wo.createdAt.toISOString().slice(0, 7);
   388→      let woMatCost = 0;
   389→      for (const mat of (wo.materials || [])) {
   390→        woMatCost += mat.totalCost || (mat.unitCost && mat.quantity ? mat.unitCost * mat.quantity : 0);
   391→      }
   392→      materialCostByMonth[month] = (materialCostByMonth[month] || 0) + woMatCost;
   393→    }
   394→    const materialCostByMonthArray = Object.entries(materialCostByMonth)
   395→      .map(([month, totalCost]) => ({
   396→        month,
   397→        totalCost: Math.round(totalCost * 100) / 100,
   398→      }))
   399→      .sort((a, b) => a.month.localeCompare(b.month));
   400→
   401→    // ========== 7. FAILURE RATE ==========
   402→    // Failure rate per asset: (failure WOs / total WOs for that asset) * 100
   403→    const assetWOs: Record<string, { name: string; total: number; failures: number }> = {};
   404→    for (const wo of workOrders) {
   405→      const assetId = wo.assetId || 'unassigned';
   406→      const assetName = wo.assetName || 'Unknown Asset';
   407→      if (!assetWOs[assetId]) assetWOs[assetId] = { name: assetName, total: 0, failures: 0 };
   408→      assetWOs[assetId].total++;
   409→      if (['emergency', 'corrective'].includes(wo.type)) {
   410→        assetWOs[assetId].failures++;
   411→      }
   412→    }
   413→
   414→    const failureRateByAsset = Object.entries(assetWOs)
   415→      .filter(([, data]) => data.total >= 2) // Only assets with 2+ WOs
   416→      .map(([assetId, data]) => ({
   417→        assetId,
   418→        assetName: data.name,
   419→        totalWOs: data.total,
   420→        failures: data.failures,
   421→        failureRate: Math.round((data.failures / data.total) * 10000) / 100,
   422→      }))
   423→      .sort((a, b) => b.failureRate - a.failureRate)
   424→      .slice(0, 20);
   425→
   426→    // Failure rate by type
   427→    const failureRateByType: Record<string, { total: number; failures: number }> = {};
   428→    for (const wo of workOrders) {
   429→      const t = wo.type || 'other';
   430→      if (!failureRateByType[t]) failureRateByType[t] = { total: 0, failures: 0 };
   431→      failureRateByType[t].total++;
   432→      if (['emergency', 'corrective'].includes(wo.type)) {
   433→        failureRateByType[t].failures++;
   434→      }
   435→    }
   436→    const failureRateByTypeArray = Object.entries(failureRateByType)
   437→      .map(([type, data]) => ({
   438→        type,
   439→        total: data.total,
   440→        failures: data.failures,
   441→        failureRate: data.total > 0 ? Math.round((data.failures / data.total) * 10000) / 100 : 0,
   442→      }))
   443→      .sort((a, b) => b.failureRate - a.failureRate);
   444→
   445→    // MTBF (Mean Time Between Failures) per asset
   446→    const mtbfByAsset = Object.entries(assetWOs)
   447→      .filter(([, data]) => data.failures >= 2)
   448→      .map(([assetId, data]) => {
   449→        // Calculate operating period from first to last WO
   450→        const assetWOList = workOrders.filter(wo => wo.assetId === assetId);
   451→        if (assetWOList.length < 2) return null;
   452→        const firstDate = assetWOList[assetWOList.length - 1].createdAt;
   453→        const lastDate = assetWOList[0].createdAt;
   454→        const operatingDays = (lastDate.getTime() - firstDate.getTime()) / (1000 * 60 * 60 * 24);
   455→        const mtbfDays = data.failures > 0 ? operatingDays / data.failures : 0;
   456→        return {
   457→          assetId,
   458→          assetName: data.name,
   459→          mtbfDays: Math.round(mtbfDays * 10) / 10,
   460→          mtbfHours: Math.round(mtbfDays * 24 * 10) / 10,
   461→          failureCount: data.failures,
   462→          operatingDays: Math.round(operatingDays),
   463→        };
   464→      })
   465→      .filter(Boolean)
   466→      .sort((a, b) => (a as any).mtbfDays - (b as any).mtbfDays) as any[];
   467→
   468→    // Rework rate (repeat failures)
   469→    const reworkWOs = workOrders.filter(wo => wo.repairCompletion && wo.repairCompletion.reworkCount > 0);
   470→    const reworkRate = total > 0 ? Math.round((reworkWOs.length / total) * 10000) / 100 : 0;
   471→
   472→    // ========== 8. STOPPAGES ==========
   473→    const totalStoppages = totalDowntimeEvents;
   474→    const stoppagesByTrade = downtimeByTradeArray.map(d => ({
   475→      trade: d.trade,
   476→      count: d.events,
   477→      totalHours: d.totalHours,
   478→    }));
   479→
   480→    // Stoppages by impact level
   481→    const stoppagesByImpact: Record<string, { count: number; minutes: number }> = {};
   482→    for (const wo of workOrders) {
   483→      for (const dt of (wo.workOrderDowntimes || [])) {
   484→        const impact = dt.impactLevel || 'medium';
   485→        if (!stoppagesByImpact[impact]) stoppagesByImpact[impact] = { count: 0, minutes: 0 };
   486→        stoppagesByImpact[impact].count++;
   487→        stoppagesByImpact[impact].minutes += dt.durationMinutes;
   488→      }
   489→    }
   490→    const stoppagesByImpactArray = Object.entries(stoppagesByImpact)
   491→      .map(([impact, data]) => ({
   492→        impact,
   493→        count: data.count,
   494→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   495→      }))
   496→      .sort((a, b) => b.count - a.count);
   497→
   498→    // Stoppages by reason
   499→    const stoppagesByReason: Record<string, number> = {};
   500→    for (const wo of workOrders) {
   501→      for (const dt of (wo.workOrderDowntimes || [])) {
   502→        const reason = dt.reason || 'Unassigned';
   503→        stoppagesByReason[reason] = (stoppagesByReason[reason] || 0) + 1;
   504→      }
   505→    }
   506→    const stoppagesByReasonArray = Object.entries(stoppagesByReason)
   507→      .map(([reason, count]) => ({ reason, count }))
   508→      .sort((a, b) => b.count - a.count)
   509→      .slice(0, 10);
   510→
   511→    // ========== 9. COST ANALYSIS ==========
   512→    let grandTotalCost = 0;
   513→    let grandLaborCost = 0;
   514→    let grandPartsCost = 0;
   515→    let grandContractorCost = 0;
   516→
   517→    for (const wo of workOrders) {
   518→      grandTotalCost += wo.totalCost || 0;
   519→      grandLaborCost += wo.laborCost || 0;
   520→      grandPartsCost += wo.partsCost || 0;
   521→      grandContractorCost += wo.contractorCost || 0;
   522→    }
   523→
   524→    // Cost by trade
   525→    const costByTrade: Record<string, { labor: number; parts: number; contractor: number; total: number }> = {};
   526→    for (const wo of workOrders) {
   527→      const t = wo.tradeActivity || 'Unassigned';
   528→      if (!costByTrade[t]) costByTrade[t] = { labor: 0, parts: 0, contractor: 0, total: 0 };
   529→      costByTrade[t].labor += wo.laborCost || 0;
   530→      costByTrade[t].parts += wo.partsCost || 0;
   531→      costByTrade[t].contractor += wo.contractorCost || 0;
   532→      costByTrade[t].total += wo.totalCost || 0;
   533→    }
   534→    const costByTradeArray = Object.entries(costByTrade)
   535→      .map(([trade, data]) => ({
   536→        trade,
   537→        laborCost: Math.round(data.labor * 100) / 100,
   538→        partsCost: Math.round(data.parts * 100) / 100,
   539→        contractorCost: Math.round(data.contractor * 100) / 100,
   540→        totalCost: Math.round(data.total * 100) / 100,
   541→      }))
   542→      .sort((a, b) => b.totalCost - a.totalCost);
   543→
   544→    // Cost by month
   545→    const costByMonth: Record<string, { labor: number; parts: number; contractor: number; total: number }> = {};
   546→    for (const wo of workOrders) {
   547→      const month = wo.createdAt.toISOString().slice(0, 7);
   548→      if (!costByMonth[month]) costByMonth[month] = { labor: 0, parts: 0, contractor: 0, total: 0 };
   549→      costByMonth[month].labor += wo.laborCost || 0;
   550→      costByMonth[month].parts += wo.partsCost || 0;
   551→      costByMonth[month].contractor += wo.contractorCost || 0;
   552→      costByMonth[month].total += wo.totalCost || 0;
   553→    }
   554→    const costByMonthArray = Object.entries(costByMonth)
   555→      .map(([month, data]) => ({
   556→        month,
   557→        laborCost: Math.round(data.labor * 100) / 100,
   558→        partsCost: Math.round(data.parts * 100) / 100,
   559→        contractorCost: Math.round(data.contractor * 100) / 100,
   560→        totalCost: Math.round(data.total * 100) / 100,
   561→      }))
   562→      .sort((a, b) => a.month.localeCompare(b.month));
   563→
   564→    // ========== 10. WO TYPE DISTRIBUTION ==========
   565→    const woByType: Record<string, number> = {};
   566→    const woByStatus: Record<string, number> = {};
   567→    const woByPriority: Record<string, number> = {};
   568→    const woByTrade: Record<string, number> = {};
   569→
   570→    for (const wo of workOrders) {
   571→      woByType[wo.type || 'other'] = (woByType[wo.type || 'other'] || 0) + 1;
   572→      woByStatus[wo.status] = (woByStatus[wo.status] || 0) + 1;
   573→      woByPriority[wo.priority || 'medium'] = (woByPriority[wo.priority || 'medium'] || 0) + 1;
   574→      woByTrade[wo.tradeActivity || 'Unassigned'] = (woByTrade[wo.tradeActivity || 'Unassigned'] || 0) + 1;
   575→    }
   576→
   577→    // ========== BUILD RESPONSE DATA ==========
   578→    const jsonData = {
   579→      success: true,
   580→      data: {
   581→        // 1. Summary
   582→        summary: {
   583→          ...summary,
   584→          avgResponseTime,
   585→          totalManHours: Math.round(grandTotalManHours * 100) / 100,
   586→          totalMaterialCost: Math.round(totalMaterialCost * 100) / 100,
   587→          totalMaterialQty: Math.round(totalMaterialQty * 100) / 100,
   588→          totalDowntimeHours: Math.round((totalDowntimeMinutes / 60) * 100) / 100,
   589→          totalDowntimeEvents,
   590→          totalStoppages,
   591→          totalProductionLoss: Math.round(totalProductionLoss * 100) / 100,
   592→          reworkRate,
   593→          reworkWOs: reworkWOs.length,
   594→        },
   595→
   596→        // 2. Downtime
   597→        downtime: {
   598→          byTrade: downtimeByTradeArray,
   599→          byMonth: downtimeByMonthArray,
   600→          byCategory: downtimeByCategoryArray,
   601→        },
   602→
   603→        // 3. Response Time
   604→        responseTime: {
   605→          overall: { avgHours: avgResponseTime, sampleSize: responseTimes.length },
   606→          byPriority: responseByPriorityArray,
   607→          byTrade: responseByTradeArray,
   608→        },
   609→
   610→        // 4. Breakdowns
   611→        breakdowns: {
   612→          total: breakdownWOs.length,
   613→          byType: breakdownByTypeArray,
   614→          byTrade: breakdownByTradeArray,
   615→          byMonth: breakdownByMonthArray,
   616→          byPriority: breakdownByPriorityArray,
   617→        },
   618→
   619→        // 5. Man Hours
   620→        manHours: {
   621→          grandTotal: Math.round(grandTotalManHours * 100) / 100,
   622→          byTrade: manHoursByTradeArray,
   623→          byTechnician: manHoursByTechnicianArray,
   624→          byMonth: manHoursByMonthArray,
   625→          byActivity: manHoursByActivityArray,
   626→        },
   627→
   628→        // 6. Materials
   629→        materials: {
   630→          totalCost: Math.round(totalMaterialCost * 100) / 100,
   631→          totalQty: Math.round(totalMaterialQty * 100) / 100,
   632→          byType: materialsByCategoryArray,
   633→          topItems: topMaterialsArray,
   634→          costByMonth: materialCostByMonthArray,
   635→        },
   636→
   637→        // 7. Failure Rate
   638→        failureRate: {
   639→          byAsset: failureRateByAsset,
   640→          byType: failureRateByTypeArray,
   641→          mtbf: mtbfByAsset,
   642→          reworkRate,
   643→          reworkWOs: reworkWOs.length,
   644→        },
   645→
   646→        // 8. Stoppages
   647→        stoppages: {
   648→          total: totalStoppages,
   649→          byTrade: stoppagesByTrade,
   650→          byImpact: stoppagesByImpactArray,
   651→          byReason: stoppagesByReasonArray,
   652→        },
   653→
   654→        // 9. Cost Analysis
   655→        cost: {
   656→          grandTotal: Math.round(grandTotalCost * 100) / 100,
   657→          grandLabor: Math.round(grandLaborCost * 100) / 100,
   658→          grandParts: Math.round(grandPartsCost * 100) / 100,
   659→          grandContractor: Math.round(grandContractorCost * 100) / 100,
   660→          byTrade: costByTradeArray,
   661→          byMonth: costByMonthArray,
   662→        },
   663→
   664→        // 10. Distributions
   665→        distribution: {
   666→          byType: Object.entries(woByType).map(([type, count]) => ({ type, count })).sort((a, b) => b.count - a.count),
   667→          byStatus: Object.entries(woByStatus).map(([status, count]) => ({ status, count })).sort((a, b) => b.count - a.count),
   668→          byPriority: Object.entries(woByPriority).map(([priority, count]) => ({ priority, count })).sort((a, b) => b.count - a.count),
   669→          byTrade: Object.entries(woByTrade).map(([trade, count]) => ({ trade, count })).sort((a, b) => b.count - a.count),
   670→        },
   671→      },
   672→    };
   673→
   674→    // ========== PDF FORMAT ==========
   675→    if (format === 'pdf') {
   676→      const pdfBuffer = await generateReportPDF({
   677→        title: 'Work Order Analytics Report',
   678→        subtitle: 'Comprehensive WO Performance Analysis',
   679→        generatedBy: session.fullName || session.userId,
   680→        generatedAt: new Date(),
   681→        filters: { from: from || 'All time', to: to || 'Present', ...(department && { department }), ...(trade && { trade }), ...(priority && { priority }) },
   682→        sections: [
   683→          { title: 'Key Performance Indicators', type: 'summary-cards', data: [
   684→            { label: 'Total WOs', value: jsonData.data.summary.totalWOs },
   685→            { label: 'Completion Rate', value: `${jsonData.data.summary.completionRate}%` },
   686→            { label: 'Breakdown Rate', value: `${jsonData.data.summary.breakdownRate}%` },
   687→            { label: 'Avg Response Time', value: `${jsonData.data.summary.avgResponseTime}h` },
   688→            { label: 'Total Man Hours', value: jsonData.data.summary.totalManHours },
   689→            { label: 'Total Material Cost', value: `$${jsonData.data.summary.totalMaterialCost.toLocaleString()}` },
   690→            { label: 'Total Downtime', value: `${jsonData.data.summary.totalDowntimeHours}h` },
   691→            { label: 'Rework Rate', value: `${jsonData.data.summary.reworkRate}%` },
   692→          ]},
   693→          { title: 'Work Order Distribution by Type', type: 'table', data: {
   694→            headers: ['Type', 'Count'],
   695→            rows: jsonData.data.distribution.byType.map(d => [d.type, String(d.count)]),
   696→          }},
   697→          { title: 'Downtime by Trade', type: 'table', data: {
   698→            headers: ['Trade', 'Hours', 'Events', 'Production Loss'],
   699→            rows: jsonData.data.downtime.byTrade.map(d => [d.trade, String(d.totalHours), String(d.events), String(d.productionLoss)]),
   700→          }},
   701→          { title: 'Response Time by Priority', type: 'table', data: {
   702→            headers: ['Priority', 'Avg Hours', 'Min Hours', 'Max Hours', 'Count'],
   703→            rows: jsonData.data.responseTime.byPriority.map(d => [d.priority, String(d.avgHours), String(d.minHours), String(d.maxHours), String(d.count)]),
   704→          }},
   705→          { title: 'Man Hours by Technician', type: 'table', data: {
   706→            headers: ['Technician', 'Total Hours', 'WO Count', 'Avg Hours/WO'],
   707→            rows: jsonData.data.manHours.byTechnician.slice(0, 15).map(d => [d.name, String(d.totalHours), String(d.woCount), String(d.avgHoursPerWO)]),
   708→          }},
   709→          { title: 'Cost Summary', type: 'summary-cards', data: [
   710→            { label: 'Total Cost', value: `$${jsonData.data.cost.grandTotal.toLocaleString()}` },
   711→            { label: 'Labor Cost', value: `$${jsonData.data.cost.grandLabor.toLocaleString()}` },
   712→            { label: 'Parts Cost', value: `$${jsonData.data.cost.grandParts.toLocaleString()}` },
   713→            { label: 'Contractor Cost', value: `$${jsonData.data.cost.grandContractor.toLocaleString()}` },
   714→          ]},
   715→          { title: 'Top Materials by Cost', type: 'table', data: {
   716→            headers: ['Item', 'Qty', 'Cost', 'WO Count'],
   717→            rows: jsonData.data.materials.topItems.slice(0, 15).map(d => [d.name, String(d.totalQty), `$${d.totalCost.toLocaleString()}`, String(d.woCount)]),
   718→          }},
   719→          { title: 'Failure Rate by Asset', type: 'table', data: {
   720→            headers: ['Asset', 'Total WOs', 'Failures', 'Failure Rate'],
   721→            rows: jsonData.data.failureRate.byAsset.slice(0, 15).map(d => [d.assetName, String(d.totalWOs), String(d.failures), `${d.failureRate}%`]),
   722→          }},
   723→        ],
   724→      });
   725→      return new NextResponse(pdfBuffer, {
   726→        headers: {
   727→          'Content-Type': 'application/pdf',
   728→          'Content-Disposition': 'attachment; filename="work-order-report.pdf"',
   729→        },
   730→      });
   731→    }
   732→
   733→    return NextResponse.json(jsonData);
   734→  } catch (error) {
   735→    console.error('[WO Reports] Error:', error);
   736→    return NextResponse.json({ success: false, error: 'Failed to generate reports' }, { status: 500 });
   737→  }
   738→}
