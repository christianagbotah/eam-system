    42→
    43→export function SafetyIncidentsPage() {
    44→  const [search, setSearch] = useState('');
    45→  const [typeFilter, setTypeFilter] = useState('all');
    46→  const [statusFilter, setStatusFilter] = useState('all');
    47→  const [dialogOpen, setDialogOpen] = useState(false);
    48→  const [incidents, setIncidents] = useState<any[]>([]);
    49→  const [loading, setLoading] = useState(true);
    50→  const [kpis, setKpis] = useState<any>({ total: 0, open: 0, investigating: 0, closed: 0, daysSinceLast: 0 });
    51→  const [form, setForm] = useState({ title: '', description: '', type: 'near_miss', severity: 'medium', location: '', date: '' });
    52→  const [viewItem, setViewItem] = useState<any>(null);
    53→  const [editItem, setEditItem] = useState<any>(null);
    54→  const [editForm, setEditForm] = useState<any>({});
    55→  const [editLoading, setEditLoading] = useState(false);
    56→
    57→  const { hasPermission, hasAnyPermission, isAdmin } = useAuthStore();
    58→
    59→  const fetchData = async () => {
    60→    try {
    61→      const res = await api.get<any>('/api/safety-incidents');
    62→      if (res.success) {
    63→        setIncidents(res.data || []);
    64→        if (res.kpis) setKpis(res.kpis);
    65→      }
    66→    } catch {} finally { setLoading(false); }
    67→  };
    68→
    69→  useEffect(() => { fetchData(); }, []);
    70→
    71→  const severityBadge: Record<string, string> = {
    72→    low: 'bg-sky-50 text-sky-700 border-sky-200',
    73→    medium: 'bg-amber-50 text-amber-700 border-amber-200',
    74→    high: 'bg-orange-50 text-orange-700 border-orange-200',
    75→    critical: 'bg-red-50 text-red-700 border-red-200',
    76→  };
    77→  const statusColors: Record<string, string> = {
    78→    open: 'bg-sky-50 text-sky-700 border-sky-200',
    79→    investigating: 'bg-amber-50 text-amber-700 border-amber-200',
    80→    closed: 'bg-emerald-50 text-emerald-700 border-emerald-200',
    81→  };
    82→  const typeLabel: Record<string, string> = { injury: 'Injury', near_miss: 'Near Miss', property_damage: 'Property Damage', environmental: 'Environmental', fire: 'Fire', chemical_spill: 'Chemical Spill' };
    83→  const severityLabel: Record<string, string> = { low: 'Low', medium: 'Medium', high: 'High', critical: 'Critical' };
    84→
    85→  const filtered = useMemo(() => incidents.filter(i => {
    86→    if (search && !i.title.toLowerCase().includes(search.toLowerCase()) && !(i.incidentNumber || '').toLowerCase().includes(search.toLowerCase())) return false;
    87→    if (typeFilter !== 'all' && i.type !== typeFilter) return false;
    88→    if (statusFilter !== 'all' && i.status !== statusFilter) return false;
    89→    return true;
    90→  }), [incidents, search, typeFilter, statusFilter]);
    91→
    92→  const severityCounts = useMemo(() => {
    93→    const counts: Record<string, number> = {};
    94→    incidents.forEach(i => { counts[i.severity || 'medium'] = (counts[i.severity || 'medium'] || 0) + 1; });
    95→    return Object.entries(counts).map(([name, value]) => ({ name: severityLabel[name] || name, value, fill: name === 'low' ? '#3b82f6' : name === 'medium' ? '#f59e0b' : name === 'high' ? '#f97316' : '#ef4444' }));
    96→  }, [incidents]);
    97→  const severityConfig = useMemo(() => Object.fromEntries(severityCounts.map((s: any) => [s.name, { label: s.name, color: s.fill }])) as any, [severityCounts]);

   100→    { label: 'Total Incidents', value: kpis.total, icon: TriangleAlert, color: 'from-red-500 to-orange-500' },
   101→    { label: 'Open', value: kpis.open, icon: AlertCircle, color: 'from-sky-500 to-blue-500' },
   102→    { label: 'Under Investigation', value: kpis.investigating, icon: Search, color: 'from-amber-500 to-yellow-500' },
   103→    { label: 'Closed', value: kpis.closed, icon: CheckCircle2, color: 'from-emerald-500 to-teal-500' },
   104→    { label: 'Days Since Last Incident', value: kpis.daysSinceLast, icon: ShieldCheck, color: 'from-teal-500 to-emerald-500' },
   105→  ];
   106→
   107→  const handleCreate = async () => {
   108→    if (!form.title) { toast.error('Title is required'); return; }
   109→    if (!form.date) { toast.error('Incident date is required'); return; }
   110→    try {
   111→      const res = await api.post('/api/safety-incidents', {
   112→        title: form.title, description: form.description, type: form.type,
   113→        severity: form.severity, location: form.location || null, incidentDate: form.date,
   114→      });

   116→        toast.success(`Incident "${form.title}" reported successfully`);
   117→        setDialogOpen(false);
   118→        setForm({ title: '', description: '', type: 'near_miss', severity: 'medium', location: '', date: '' });
   119→        fetchData();
   120→      } else { toast.error(res.error || 'Failed to create incident'); }
   121→    } catch { toast.error('Failed to create incident'); }
   122→  };
   123→
   124→  const handleDelete = async (id: string) => {
   125→    try {
   126→      const res = await api.delete(`/api/safety-incidents/${id}`);
   127→      if (res.success) { toast.success('Incident deleted'); fetchData(); }
   128→      else { toast.error(res.error || 'Failed to delete'); }
   129→    } catch { toast.error('Failed to delete'); }
   130→  };
   131→
   132→  const handleSave = async () => {
   133→    if (!editItem) return;
   134→    try {
   135→      setEditLoading(true);
   136→      const res = await api.put(`/api/safety-incidents/${editItem.id}`, editForm);
   137→      if (res.success) {
   138→        toast.success('Incident updated successfully');
   139→        setEditItem(null); setEditForm({}); fetchData();
   140→      } else { toast.error(res.error || 'Failed to update incident'); }
   141→    } catch { toast.error('Failed to update incident'); }
   142→    finally { setEditLoading(false); }
   143→  };
   144→
   145→  return (
   146→    <div className="page-content">
   147→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   148→        <div>
   149→          <h1 className="text-2xl font-bold tracking-tight">Safety Incidents</h1>
   150→          <p className="text-muted-foreground mt-1">Report, investigate, and track safety incidents and near-misses</p>
   151→        </div>
   152→        {(hasAnyPermission(['safety_incidents.create', 'safety_inspections.create', 'safety_equipment.create', 'safety_permits.create']) || isAdmin()) && <>
   153→          <Button className="bg-emerald-600 hover:bg-emerald-700 text-white cursor-pointer" onClick={() => setDialogOpen(true)}><Plus className="h-4 w-4 mr-2" />Report Incident</Button>
   154→          <ResponsiveDialog open={dialogOpen} onOpenChange={setDialogOpen} title="Report New Incident" description="Fill in the details of the safety incident">
   155→            <ScrollArea className="max-h-[70vh]">
   156→              <div className="space-y-4 py-2 pr-3">
   157→                <div className="space-y-2"><Label>Title</Label><Input placeholder="Brief description of incident" value={form.title} onChange={e => setForm(p => ({ ...p, title: e.target.value }))} /></div>
   158→                <div className="space-y-2"><Label>Description</Label><Textarea placeholder="Detailed description..." rows={3} value={form.description} onChange={e => setForm(p => ({ ...p, description: e.target.value }))} /></div>
   159→                <div className="grid grid-cols-2 gap-4">
   160→                  <div className="space-y-2"><Label>Type</Label>
   161→                    <Select value={form.type} onValueChange={v => setForm(p => ({ ...p, type: v }))}>
   162→                      <SelectTrigger><SelectValue /></SelectTrigger>
   163→                      <SelectContent><SelectItem value="near_miss">Near Miss</SelectItem><SelectItem value="injury">Injury</SelectItem><SelectItem value="property_damage">Property Damage</SelectItem><SelectItem value="environmental">Environmental</SelectItem><SelectItem value="fire">Fire</SelectItem><SelectItem value="chemical_spill">Chemical Spill</SelectItem></SelectContent>
   164→                    </Select>
   165→                  </div>
   166→                  <div className="space-y-2"><Label>Severity</Label>
   167→                    <Select value={form.severity} onValueChange={v => setForm(p => ({ ...p, severity: v }))}>
   168→                      <SelectTrigger><SelectValue /></SelectTrigger>
   169→                      <SelectContent><SelectItem value="low">Low</SelectItem><SelectItem value="medium">Medium</SelectItem><SelectItem value="high">High</SelectItem><SelectItem value="critical">Critical</SelectItem></SelectContent>
   170→                    </Select>
   171→                  </div>
   172→                </div>
   173→                <div className="grid grid-cols-2 gap-4">
   174→                  <div className="space-y-2"><Label>Location</Label><Input placeholder="e.g. Warehouse B" value={form.location} onChange={e => setForm(p => ({ ...p, location: e.target.value }))} /></div>
   175→                  <DatePicker label="Date" value={form.date || undefined} onChange={v => setForm(p => ({ ...p, date: v || '' }))} />
   176→                </div>
   177→              </div>
   178→            </ScrollArea>
   179→            
   180→            <div className="flex justify-end gap-2 pt-4">
   181→              <Button variant="outline" onClick={() => setDialogOpen(false)}>Cancel</Button>
   182→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white cursor-pointer" onClick={handleCreate}>Submit Report</Button>
   183→            </div>
   184→          </ResponsiveDialog>
   185→        </>}
   186→      </div>
   187→
   188→      {loading ? <LoadingSkeleton /> : <>
   189→      <div className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-5 gap-4">
   190→        {kpiCards.map(k => { const Icon = k.icon; return (
   191→          <Card key={k.label} className="bg-card text-card-foreground border border-border/60 rounded-xl shadow-sm">
   192→            <CardContent className="p-5">
   193→              <div className="flex items-center justify-between">
   194→                <div><p className="text-xs text-muted-foreground">{k.label}</p><p className="text-2xl font-bold mt-1">{k.value}</p></div>
   195→                <div className={`h-10 w-10 rounded-xl bg-gradient-to-br ${k.color} flex items-center justify-center`}><Icon className="h-5 w-5 text-white" /></div>
   196→              </div>
   197→            </CardContent>
   198→          </Card>
   199→        ); })}
   200→      </div>
   201→
   202→      <div className="grid grid-cols-1 lg:grid-cols-4 gap-6">
   203→        <Card className="bg-card text-card-foreground border border-border/60 rounded-xl shadow-sm lg:col-span-1">

   205→          <CardContent>
   206→            <ChartContainer config={severityConfig} className="h-[240px] w-full">
   207→              <PieChart>
   208→                <ChartTooltip content={<ChartTooltipContent />} />
   209→                <Pie data={severityCounts} cx="50%" cy="50%" innerRadius={50} outerRadius={80} dataKey="value" nameKey="name" strokeWidth={2} stroke="hsl(var(--background))">
   210→                  {severityCounts.map((entry: any, idx: number) => <Cell key={idx} fill={entry.fill} />)}
   211→                </Pie>
   212→              </PieChart>
   213→            </ChartContainer>
   214→            <div className="flex flex-wrap gap-2 mt-2 justify-center">
   215→              {severityCounts.map((s: any) => (
   216→                <div key={s.name} className="flex items-center gap-1.5 text-xs">
   217→                  <div className="h-2.5 w-2.5 rounded-sm" style={{ backgroundColor: s.fill }} />
   218→                  <span className="text-muted-foreground">{s.name} ({s.value})</span>
   219→                </div>
   220→              ))}
   221→            </div>
   222→          </CardContent>
   223→        </Card>
   224→
   225→        <Card className="bg-card text-card-foreground border border-border/60 rounded-xl shadow-sm lg:col-span-3">
   226→          <CardContent className="p-4 sm:p-6">
   227→            <div className="filter-row mb-4">
   228→              <div className="relative flex-1"><Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" /><Input placeholder="Search incidents..." className="pl-9" value={search} onChange={e => setSearch(e.target.value)} /></div>
   229→              <Select value={typeFilter} onValueChange={setTypeFilter}><SelectTrigger className="w-[160px]"><SelectValue placeholder="Type" /></SelectTrigger><SelectContent><SelectItem value="all">All Types</SelectItem><SelectItem value="near_miss">Near Miss</SelectItem><SelectItem value="injury">Injury</SelectItem><SelectItem value="property_damage">Property Damage</SelectItem><SelectItem value="environmental">Environmental</SelectItem><SelectItem value="fire">Fire</SelectItem><SelectItem value="chemical_spill">Chemical Spill</SelectItem></SelectContent></Select>
   230→              <Select value={statusFilter} onValueChange={setStatusFilter}><SelectTrigger className="w-[160px]"><SelectValue placeholder="Status" /></SelectTrigger><SelectContent><SelectItem value="all">All Status</SelectItem><SelectItem value="open">Open</SelectItem><SelectItem value="investigating">Investigating</SelectItem><SelectItem value="closed">Closed</SelectItem></SelectContent></Select>
   231→            </div>
   232→            <div className="overflow-x-auto rounded-lg border max-h-[420px] overflow-y-auto">
   233→              <Table>
   234→                <TableHeader sticky><TableRow className="bg-muted/50"><TableHead className="font-semibold">ID</TableHead><TableHead className="font-semibold">Title</TableHead><TableHead className="font-semibold">Type</TableHead><TableHead className="font-semibold">Severity</TableHead><TableHead className="font-semibold">Location</TableHead><TableHead className="font-semibold">Date</TableHead><TableHead className="font-semibold">Reported By</TableHead><TableHead className="font-semibold">Status</TableHead><TableHead className="font-semibold">Root Cause</TableHead><TableHead className="w-10"></TableHead></TableRow></TableHeader>
   235→                <TableBody>
   236→                  {filtered.length === 0 ? <TableRow><TableCell colSpan={10}><EmptyState icon={TriangleAlert} title="No incidents found" description="Try adjusting your search or filters" /></TableCell></TableRow> : filtered.map(i => (
   237→                    <TableRow key={i.id} className="cursor-pointer hover:bg-muted/30">
   238→                      <TableCell className="font-mono text-xs font-semibold">{i.incidentNumber}</TableCell>
   239→                      <TableCell className="font-medium max-w-[200px] truncate">{i.title}</TableCell>
   240→                      <TableCell><Badge variant="outline" className="bg-slate-50 text-slate-600 border-slate-200">{typeLabel[i.type] || i.type}</Badge></TableCell>
   241→                      <TableCell><Badge variant="outline" className={severityBadge[i.severity] || ''}>{severityLabel[i.severity] || i.severity}</Badge></TableCell>
   242→                      <TableCell className="text-sm text-muted-foreground"><div className="flex items-center gap-1.5"><MapPin className="h-3.5 w-3.5 shrink-0" /><span className="truncate max-w-[120px]">{i.location}</span></div></TableCell>
   243→                      <TableCell className="text-sm text-muted-foreground whitespace-nowrap">{formatDate(i.incidentDate)}</TableCell>
   244→                      <TableCell><div className="flex items-center gap-2"><Avatar className="h-6 w-6"><AvatarFallback className="text-[10px] bg-emerald-100 text-emerald-700">{getInitials(i.reportedBy?.fullName || '')}</AvatarFallback></Avatar><span className="text-sm whitespace-nowrap">{i.reportedBy?.fullName || ''}</span></div></TableCell>
   245→                      <TableCell><Badge variant="outline" className={statusColors[i.status] || ''}>{i.status}</Badge></TableCell>
   246→                      <TableCell className="text-sm text-muted-foreground max-w-[140px] truncate">{i.rootCause || '—'}</TableCell>
   247→                      <TableCell><DropdownMenu><DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="h-8 w-8 cursor-pointer"><MoreHorizontal className="h-4 w-4" /></Button></DropdownMenuTrigger><DropdownMenuContent align="end"><DropdownMenuItem className="cursor-pointer" onClick={() => setViewItem(i)}><Eye className="h-4 w-4 mr-2" />View</DropdownMenuItem>{(hasAnyPermission(['safety_incidents.update', 'safety_inspections.update', 'safety_equipment.update', 'safety_permits.approve']) || isAdmin()) && <DropdownMenuItem className="cursor-pointer" onClick={() => { setEditItem(i); setEditForm({...i}); }}><Pencil className="h-4 w-4 mr-2" />Edit</DropdownMenuItem>}{(hasAnyPermission(['safety_incidents.manage', 'safety_inspections.manage', 'safety_equipment.delete', 'safety_permits.close']) || isAdmin()) && <><DropdownMenuSeparator /><DropdownMenuItem className="cursor-pointer text-red-600" onClick={() => handleDelete(i.id)}><Trash2 className="h-4 w-4 mr-2" />Delete</DropdownMenuItem></>}</DropdownMenuContent></DropdownMenu></TableCell>
   248→                    </TableRow>
   249→                  ))}
   250→                </TableBody>
   251→              </Table>
   252→            </div>
   253→            <div className="flex items-center justify-between mt-4 text-sm text-muted-foreground">
   254→              <span>Showing {filtered.length} of {incidents.length} incidents</span>
   255→            </div>
   256→          </CardContent>
   257→        </Card>
   258→      </div>
   259→      </>}
   260→
   261→      <ResponsiveDialog open={!!viewItem} onOpenChange={(v) => !v && setViewItem(null)} title="Incident Details" large>
   262→          <div className="grid grid-cols-2 gap-4">
   263→            <div><Label className="text-xs text-muted-foreground">Title</Label><p className="text-sm font-medium">{viewItem?.title || '-'}</p></div>
   264→            <div><Label className="text-xs text-muted-foreground">Type</Label><p className="text-sm font-medium">{viewItem?.type?.replace(/_/g, ' ') || '-'}</p></div>
   265→            <div><Label className="text-xs text-muted-foreground">Severity</Label><p className="text-sm font-medium capitalize">{viewItem?.severity || '-'}</p></div>
   266→            <div><Label className="text-xs text-muted-foreground">Status</Label><p className="text-sm font-medium capitalize">{viewItem?.status || '-'}</p></div>
   267→            <div><Label className="text-xs text-muted-foreground">Reported By</Label><p className="text-sm font-medium">{viewItem?.reportedBy?.fullName || '-'}</p></div>
   268→            <div><Label className="text-xs text-muted-foreground">Location</Label><p className="text-sm font-medium">{viewItem?.location || '-'}</p></div>
   269→            <div><Label className="text-xs text-muted-foreground">Occurred At</Label><p className="text-sm font-medium">{viewItem?.incidentDate ? formatDate(viewItem.incidentDate) : viewItem?.occurredAt || '-'}</p></div>
   270→            <div><Label className="text-xs text-muted-foreground">Root Cause</Label><p className="text-sm font-medium">{viewItem?.rootCause || '-'}</p></div>
   271→            <div className="col-span-2"><Label className="text-xs text-muted-foreground">Description</Label><p className="text-sm font-medium">{viewItem?.description || '-'}</p></div>
   272→            <div className="col-span-2"><Label className="text-xs text-muted-foreground">Corrective Action</Label><p className="text-sm font-medium">{viewItem?.correctiveAction || '-'}</p></div>
   273→          </div>
   274→        </ResponsiveDialog>
   275→
   276→      <ResponsiveDialog open={!!editItem} onOpenChange={(v) => !v && setEditItem(null)} title="Edit Incident" large>
   277→          <div className="grid gap-4 py-2">
   278→            <div className="space-y-2"><Label>Title</Label><Input value={editForm.title || ''} onChange={e => setEditForm(p => ({ ...p, title: e.target.value }))} /></div>
   279→            <div className="space-y-2"><Label>Description</Label><Textarea rows={3} value={editForm.description || ''} onChange={e => setEditForm(p => ({ ...p, description: e.target.value }))} /></div>
   280→            <div className="grid grid-cols-2 gap-4">
   281→              <div className="space-y-2"><Label>Type</Label>
   282→                <Select value={editForm.type || 'near_miss'} onValueChange={v => setEditForm(p => ({ ...p, type: v }))}>
   283→                  <SelectTrigger><SelectValue /></SelectTrigger>
   284→                  <SelectContent><SelectItem value="near_miss">Near Miss</SelectItem><SelectItem value="injury">Injury</SelectItem><SelectItem value="property_damage">Property Damage</SelectItem><SelectItem value="environmental">Environmental</SelectItem><SelectItem value="fire">Fire</SelectItem><SelectItem value="chemical_spill">Chemical Spill</SelectItem></SelectContent>
   285→                </Select>
   286→              </div>
   287→              <div className="space-y-2"><Label>Severity</Label>
   288→                <Select value={editForm.severity || 'medium'} onValueChange={v => setEditForm(p => ({ ...p, severity: v }))}>
   289→                  <SelectTrigger><SelectValue /></SelectTrigger>
   290→                  <SelectContent><SelectItem value="low">Low</SelectItem><SelectItem value="medium">Medium</SelectItem><SelectItem value="high">High</SelectItem><SelectItem value="critical">Critical</SelectItem></SelectContent>
   291→                </Select>
   292→              </div>
   293→            </div>
   294→            <div className="grid grid-cols-2 gap-4">
   295→              <div className="space-y-2"><Label>Status</Label>
   296→                <Select value={editForm.status || 'open'} onValueChange={v => setEditForm(p => ({ ...p, status: v }))}>
   297→                  <SelectTrigger><SelectValue /></SelectTrigger>
   298→                  <SelectContent><SelectItem value="open">Open</SelectItem><SelectItem value="investigating">Investigating</SelectItem><SelectItem value="closed">Closed</SelectItem></SelectContent>
   299→                </Select>
   300→              </div>
   301→              <div className="space-y-2"><Label>Location</Label><Input value={editForm.location || ''} onChange={e => setEditForm(p => ({ ...p, location: e.target.value }))} /></div>
   302→            </div>
   303→            <div className="grid grid-cols-2 gap-4">
   304→              <div className="space-y-2"><Label>Root Cause</Label><Input value={editForm.rootCause || ''} onChange={e => setEditForm(p => ({ ...p, rootCause: e.target.value }))} /></div>
   305→              <div className="space-y-2"><Label>Corrective Action</Label><Input value={editForm.correctiveAction || ''} onChange={e => setEditForm(p => ({ ...p, correctiveAction: e.target.value }))} /></div>
