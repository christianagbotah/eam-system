   143→   143→function effColor(v: number): string {
   144→   144→  if (v >= 97) return 'text-emerald-600 font-semibold';
   145→   145→  if (v >= 90) return 'text-amber-600 font-semibold';

   148→   148→
   149→   149→function effBg(v: number): string {
   150→   150→  if (v >= 97) return 'bg-emerald-50';
   151→   151→  if (v >= 90) return 'bg-amber-50';
   152→   152→  return 'bg-red-50';
   153→   153→}
   154→   154→
   155→   155→function dtColor(v: number): string {
   156→   156→  if (v > 5) return 'text-red-600 font-semibold';
   157→   157→  if (v >= 2) return 'text-amber-600 font-semibold';
   158→   158→  return 'text-emerald-600 font-semibold';
   159→   159→}
   160→   160→
   161→   161→function dtBg(v: number): string {
   162→   162→  if (v > 5) return 'bg-red-50';
   163→   163→  if (v >= 2) return 'bg-amber-50';
   164→   164→  return 'bg-emerald-50';
   165→   165→}
   166→   166→
   167→   167→function fmt(n: number, dec = 1): string {
   168→   168→  if (n == null || Number.isNaN(n)) return '—';
   169→   169→  return n.toFixed(dec);
   170→   170→}
   171→   171→
   172→   172→function fmtInt(n: number): string {
   173→   173→  if (n == null || Number.isNaN(n)) return '—';
   174→   174→  return Math.round(n).toLocaleString();
   175→   175→}
   176→   176→
   177→   177→function statusBadge(breakdowns: number): { label: string; variant: 'destructive' | 'outline' | 'secondary' } {
   178→   178→  if (breakdowns > 5) return { label: 'Critical', variant: 'destructive' };
   179→   179→  if (breakdowns >= 2) return { label: 'Warning', variant: 'outline' };
   180→   180→  return { label: 'Normal', variant: 'secondary' };
   181→   181→}
