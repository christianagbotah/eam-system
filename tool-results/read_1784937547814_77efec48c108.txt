    41→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
    42→  const token = localStorage.getItem('eam_token');
    43→  const res = await fetch(url, {
    44→    ...options,
    45→    headers: {
    46→      'Content-Type': 'application/json',
    47→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    48→      ...options?.headers,
    49→    },
    50→  });
    51→  const json = await res.json();
    52→  if (!json.success) {
    53→    throw new Error(json.error || json.message || 'API request failed');
    54→  }
    55→  return json.data as T;
    56→}

   446→function FailureModesTab() {
   447→  const [modes, setModes] = useState<FailureMode[]>([]);
   448→  const [loading, setLoading] = useState(true);
   449→  const [search, setSearch] = useState('');
   450→  const [categoryFilter, setCategoryFilter] = useState('all');
   451→  const [severityFilter, setSeverityFilter] = useState('all');
   452→  const [dialogOpen, setDialogOpen] = useState(false);
   453→  const [selectedMode, setSelectedMode] = useState<FailureMode | null>(null);
   454→  const [detailOpen, setDetailOpen] = useState(false);
   455→  const [form, setForm] = useState({ name: '', code: '', description: '', category: 'mechanical', severity: 'medium', detectionMethod: '' });
   456→
   457→  const loadModes = useCallback(async () => {
   458→    try {
   459→      setLoading(true);
   460→      const params = new URLSearchParams();
   461→      if (search) params.set('search', search);
   462→      if (categoryFilter !== 'all') params.set('category', categoryFilter);
   463→      if (severityFilter !== 'all') params.set('severity', severityFilter);
   464→      const data = await apiFetch<{ items: FailureMode[] }>(`${API_BASE}/failure-modes/route?${params}`);
   465→      setModes(data.items || []);
   466→    } catch (err) {
   467→      console.error(err);
   468→      toast.error('Failed to load failure modes');
   469→    } finally {
   470→      setLoading(false);
   471→    }
   472→  }, [search, categoryFilter, severityFilter]);
   473→
   474→  useEffect(() => { loadModes(); }, [loadModes]);
   475→
   476→  const handleCreate = async () => {
   477→    try {
   478→      await apiFetch(`${API_BASE}/failure-modes/route`, {
   479→        method: 'POST',
   480→        body: JSON.stringify(form),
   481→      });
   482→      toast.success('Failure mode created');
   483→      setDialogOpen(false);
   484→      setForm({ name: '', code: '', description: '', category: 'mechanical', severity: 'medium', detectionMethod: '' });
   485→      loadModes();
   486→    } catch (err) {
   487→      toast.error((err as Error).message);
   488→    }
   489→  };
   490→
   491→  const viewDetail = async (id: string) => {
   492→    try {
   493→      const data = await apiFetch<FailureMode>(`${API_BASE}/failure-modes/${id}/route`);
   494→      setSelectedMode(data);
   495→      setDetailOpen(true);
   496→    } catch (err) {
   497→      toast.error('Failed to load failure mode details');
   498→    }
   499→  };
   500→
   501→  const handleDelete = async (id: string) => {
   502→    try {
   503→      await apiFetch(`${API_BASE}/failure-modes/${id}/route`, { method: 'DELETE' });
   504→      toast.success('Failure mode deleted');
   505→      setDetailOpen(false);
   506→      loadModes();
   507→    } catch (err) {
   508→      toast.error((err as Error).message);
   509→    }
   510→  };
   511→
   512→  return (
   513→    <div className="space-y-4">
   514→      {/* Filters */}
   515→      <Card>
   516→        <CardContent className="p-4">
   517→          <div className="flex flex-col sm:flex-row gap-3">
   518→            <div className="relative flex-1">
   519→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   520→              <Input
