    27→
    28→function DetailRow({ label, value }: { label: string; value: React.ReactNode }) {
    29→  return (
    30→    <div className="flex justify-between text-sm py-1.5">
    31→      <span className="text-muted-foreground">{label}</span>
    32→      <span className="font-medium text-right max-w-[60%] truncate" title={String(value)}>{value || '-'}</span>
    33→    </div>
    34→  );
    35→}
    36→
    37→function EmptyTab({ icon: Icon, title, description, actionLabel, onAction }: { icon: any; title: string; description: string; actionLabel?: string; onAction?: () => void }) {
    38→  return (
    39→    <div className="flex flex-col items-center justify-center py-12 text-center">
    40→      <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-3">
    41→        <Icon className="h-6 w-6 text-muted-foreground" />
    42→      </div>
    43→      <p className="font-medium text-sm">{title}</p>
    44→      <p className="text-xs text-muted-foreground mt-1 max-w-xs">{description}</p>
    45→      {actionLabel && onAction && (
    46→        <Button onClick={onAction} size="sm" className="mt-4"><Plus className="h-3.5 w-3.5 mr-1.5" />{actionLabel}</Button>
    47→      )}
    48→    </div>
    49→  );
    50→}
    51→
    52→export function AssetDetailPage({ id }: { id: string }) {
    53→  const [activeTab, setActiveTab] = useState('overview');
    54→  const [asset, setAsset] = useState<any>(null);
    55→  const [loading, setLoading] = useState(true);
    56→  const [bomItems, setBomItems] = useState<any[]>([]);
    57→  const [bomAsChild, setBomAsChild] = useState<any[]>([]);
    58→  const [components, setComponents] = useState<any[]>([]);
    59→  const [twin, setTwin] = useState<any>(null);
    60→  const [diagrams, setDiagrams] = useState<any[]>([]);
    61→  const [tabDataLoading, setTabDataLoading] = useState(false);
    62→  const loadedTabsRef = useRef<Set<string>>(new Set(['overview']));
    63→  const tabDataLoadingRef = useRef(false);
    64→
    65→  // Create dialogs
    66→  const [showComponentForm, setShowComponentForm] = useState(false);
    67→  const [showTwinForm, setShowTwinForm] = useState(false);
    68→  const [showDiagramForm, setShowDiagramForm] = useState(false);
    69→  const [saving, setSaving] = useState(false);
    70→
    71→  // Component form
    72→  const [compForm, setCompForm] = useState({ componentCode: '', name: '', componentType: 'component', criticality: 'medium', manufacturer: '', modelNumber: '', serialNumber: '', description: '', expectedLifeHours: '', operatingHours: '' });
    73→  // Twin form
    74→  const [twinForm, setTwinForm] = useState({ name: '', type: 'pump', syncInterval: '5min' });
    75→  // Diagram form
    76→  const [diagForm, setDiagForm] = useState({ name: '', type: 'process', description: '' });
    77→
    78→  // Load main asset data
    79→  useEffect(() => {
    80→    api.get<any>(`/api/assets/${id}`).then(res => {
    81→      if (res.success && res.data) {
    82→        setAsset(res.data);
    83→        if (res.data.digitalTwin) setTwin(res.data.digitalTwin);
    84→      }
    85→      setLoading(false);
    86→    });
    87→  }, [id]);
    88→
    89→  // Reload components
    90→  const reloadComponents = useCallback(() => {
    91→    api.get(`/api/component-registry?assetId=${id}&limit=100`).then(res => {
    92→      if (res.success && res.data) setComponents(Array.isArray(res.data) ? res.data : []);
    93→    }).catch(() => {});
    94→  }, [id]);
    95→
    96→  // Reload digital twin
    97→  const reloadTwin = useCallback(() => {
    98→    api.get(`/api/digital-twins?assetId=${id}&limit=1`).then(res => {
    99→      if (res.success && res.data && Array.isArray(res.data) && res.data.length > 0) {
   100→        setTwin(res.data[0]);
   101→      }
   102→    }).catch(() => {});
   103→  }, [id]);
   104→
   105→  // Reload diagrams
   106→  const reloadDiagrams = useCallback(() => {
   107→    api.get(`/api/system-diagrams?limit=100`).then(res => {
   108→      if (res.success && res.data) setDiagrams(Array.isArray(res.data) ? res.data : []);
   109→    }).catch(() => {});
   110→  }, []);
   111→
   112→  // Lazy load tab data when tab changes
   113→  // NOTE: loadedTabsRef is a ref (not state) to avoid infinite re-render loops.
   114→  // Using a Set in useEffect deps (via useState) creates a new object reference
   115→  // on every mutation, which triggers the effect again → Error #185.
   116→  useEffect(() => {
   117→    if (!asset || loadedTabsRef.current.has(activeTab) || tabDataLoadingRef.current) return;
   118→    tabDataLoadingRef.current = true;
   119→    setTabDataLoading(true);
   120→    const promises: Promise<void>[] = [];
   121→
   122→    if (activeTab === 'hierarchy' || activeTab === 'overview') {
   123→      loadedTabsRef.current.add('hierarchy');
   124→    }
   125→    if (activeTab === 'bom') {
   126→      promises.push(
   127→        api.get(`/api/bill-of-materials?parentId=${id}&limit=100`).then(res => {
   128→          if (res.success && res.data) setBomItems(Array.isArray(res.data) ? res.data : []);
   129→        }).catch(() => {}),
   130→        api.get(`/api/bill-of-materials?search=${encodeURIComponent(asset.name)}&limit=100`).then(res => {
   131→          if (res.success && res.data) {
   132→            const items = Array.isArray(res.data) ? res.data : [];
   133→            setBomAsChild(items.filter((b: any) => b.childAssetId === id));
   134→          }
   135→        }).catch(() => {}),
   136→      );
   137→      loadedTabsRef.current.add('bom');
   138→    }
   139→    if (activeTab === 'components') {
   140→      promises.push(reloadComponents());
   141→      loadedTabsRef.current.add('components');
   142→    }
   143→    if (activeTab === 'digital-twin' && !twin) {
   144→      promises.push(reloadTwin());
   145→      loadedTabsRef.current.add('digital-twin');
   146→    }
   147→    if (activeTab === 'diagrams') {
   148→      promises.push(reloadDiagrams());
   149→      loadedTabsRef.current.add('diagrams');
   150→    }
   151→
   152→    Promise.all(promises).finally(() => {
   153→      tabDataLoadingRef.current = false;
   154→      setTabDataLoading(false);
   155→    });
   156→  }, [activeTab, asset, id, twin, reloadComponents, reloadTwin, reloadDiagrams]);
   157→
   158→  // Handlers
   159→  const handleCreateComponent = async () => {
   160→    if (!compForm.componentCode.trim() || !compForm.name.trim()) {
   161→      toast.error('Component code and name are required');
   162→      return;
   163→    }
   164→    setSaving(true);
   165→    try {
   166→      const res = await api.post('/api/component-registry', {
   167→        ...compForm,
   168→        assetId: id,
   169→        healthScore: 100,
   170→        lifecycleStatus: 'operational',
   171→        expectedLifeHours: compForm.expectedLifeHours ? parseFloat(compForm.expectedLifeHours) : null,
   172→        operatingHours: compForm.operatingHours ? parseFloat(compForm.operatingHours) : 0,
   173→      });
   174→      if (res.success) {
   175→        toast.success('Component registered successfully');
   176→        setShowComponentForm(false);
   177→        setCompForm({ componentCode: '', name: '', componentType: 'component', criticality: 'medium', manufacturer: '', modelNumber: '', serialNumber: '', description: '', expectedLifeHours: '', operatingHours: '' });
   178→        reloadComponents();
   179→      } else {
   180→        toast.error(res.error || 'Failed to create component');
   181→      }
   182→    } catch {
   183→      toast.error('Failed to create component');
   184→    }
   185→    setSaving(false);
   186→  };
   187→
   188→  const handleCreateTwin = async () => {
   189→    if (!twinForm.name.trim()) {
   190→      toast.error('Twin name is required');
   191→      return;
   192→    }
   193→    setSaving(true);
   194→    try {
   195→      const res = await api.post('/api/digital-twins', {
   196→        name: twinForm.name,
   197→        assetId: id,
   198→        type: twinForm.type,
   199→        syncInterval: twinForm.syncInterval,
   200→      });
   201→      if (res.success) {
   202→        toast.success('Digital twin created successfully');
   203→        setShowTwinForm(false);
   204→        setTwinForm({ name: '', type: 'pump', syncInterval: '5min' });
   205→        reloadTwin();
   206→      } else {
   207→        toast.error(res.error || 'Failed to create digital twin');
   208→      }
   209→    } catch {
   210→      toast.error('Failed to create digital twin');
   211→    }
   212→    setSaving(false);
   213→  };
   214→
   215→  const handleCreateDiagram = async () => {
   216→    if (!diagForm.name.trim()) {
   217→      toast.error('Diagram name is required');
   218→      return;
   219→    }
   220→    setSaving(true);
   221→    try {
   222→      const res = await api.post('/api/system-diagrams', {
   223→        name: diagForm.name,
   224→        type: diagForm.type,
   225→        description: diagForm.description || null,
   226→        nodes: [],
   227→        edges: [],
   228→        isTemplate: false,
   229→      });
   230→      if (res.success) {
   231→        toast.success('System diagram created successfully');
   232→        setShowDiagramForm(false);
   233→        setDiagForm({ name: '', type: 'process', description: '' });
   234→        reloadDiagrams();
   235→      } else {
   236→        toast.error(res.error || 'Failed to create diagram');
   237→      }
   238→    } catch {
   239→      toast.error('Failed to create diagram');
   240→    }
   241→    setSaving(false);
   242→  };
   243→
   244→  if (loading) return <LoadingSkeleton />;
   245→  if (!asset) return <div className="p-6">Asset not found</div>;
   246→
   247→  let specs: Record<string, string> = {};
   248→  try { if (asset.specification) specs = JSON.parse(asset.specification); } catch (_e) { specs = {}; }
   249→  const statusColors: Record<string, string> = { operational: 'bg-emerald-100 text-emerald-700 border-emerald-200', under_maintenance: 'bg-amber-100 text-amber-700 border-amber-200', standby: 'bg-sky-100 text-sky-700 border-sky-200', decommissioned: 'bg-slate-100 text-slate-600 border-slate-200', disposed: 'bg-red-100 text-red-600 border-red-200' };
   250→  const criticalityColors: Record<string, string> = { low: 'bg-slate-100 text-slate-600 border-slate-200', medium: 'bg-sky-100 text-sky-700 border-sky-200', high: 'bg-amber-100 text-amber-700 border-amber-200', critical: 'bg-red-100 text-red-700 border-red-200' };
   251→  const condColors: Record<string, string> = { new: 'bg-emerald-100 text-emerald-700 border-emerald-200', good: 'bg-emerald-100 text-emerald-700 border-emerald-200', fair: 'bg-amber-100 text-amber-700 border-amber-200', poor: 'bg-red-100 text-red-700 border-red-200', out_of_service: 'bg-red-100 text-red-800 border-red-300' };
   252→  const hasHierarchy = asset.parent || (asset.children && asset.children.length > 0);
   253→  const hasIoT = asset.iotDevices && asset.iotDevices.length > 0;
   254→  const hasMR = asset.maintenanceRequests && asset.maintenanceRequests.length > 0;
   255→  const hasWO = asset.workOrders && asset.workOrders.length > 0;
   256→
   257→  const tabs = [
   258→    { id: 'overview', label: 'Overview', icon: Building2 },
   259→    { id: 'hierarchy', label: 'Hierarchy', icon: GitBranch, badge: hasHierarchy ? asset.children.length + 1 : 0 },
   260→    { id: 'bom', label: 'BOM', icon: Layers },
   261→    { id: 'components', label: 'Components', icon: Cpu, badge: components.length || 0 },
   262→    { id: 'condition', label: 'Monitoring', icon: Activity, badge: hasIoT ? asset.iotDevices.length : 0 },
   263→    { id: 'digital-twin', label: 'Digital Twin', icon: Monitor },
   264→    { id: 'diagrams', label: 'Diagrams', icon: MapPin },
   265→  ];
   266→
   267→  return (
   268→    <>
   269→      {/* Header */}
   270→      <SheetHeader>
   271→        <SheetTitle>{asset.name}</SheetTitle>
   272→        <SheetDescription>{asset.assetTag} · {asset.category?.name || 'Asset'}</SheetDescription>
   273→      </SheetHeader>
   274→      <div className="flex items-center gap-2 flex-wrap mt-3">
   275→        <Badge variant="outline" className={`capitalize ${statusColors[asset.status] || ''}`}>{(asset.status || '').replace(/_/g, ' ')}</Badge>
   276→        <Badge variant="outline" className={`capitalize ${condColors[asset.condition] || ''}`}>{asset.condition || '-'}</Badge>
   277→        <Badge variant="outline" className={`uppercase ${criticalityColors[asset.criticality] || ''}`}>{asset.criticality || '-'}</Badge>
   278→      </div>
   279→
   280→      {/* Tabs */}
   281→      <Tabs value={activeTab} onValueChange={setActiveTab} className="mt-4">
   282→        <TabsList className="w-full flex overflow-x-auto p-0 h-auto gap-0 bg-transparent border-b rounded-none">
   283→          {tabs.map((tab) => {
   284→            const TabIcon = tab.icon;
   285→            return (
   286→            <TabsTrigger
   287→              key={tab.id}
   288→              value={tab.id}
   289→              className="flex items-center gap-1.5 px-3 py-2 rounded-none border-b-2 border-transparent data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:shadow-none text-xs whitespace-nowrap flex-shrink-0"
   290→            >
   291→              <TabIcon className="h-3.5 w-3.5" />
   292→              {tab.label}
   293→              {tab.badge ? <span className="ml-0.5 h-4 min-w-4 px-1 rounded-full bg-primary/10 text-primary text-[9px] font-bold flex items-center justify-center">{tab.badge}</span> : null}
   294→            </TabsTrigger>
   295→            );
   296→          })}
   297→        </TabsList>
   298→
   299→        <ScrollArea className="max-h-[calc(100vh-14rem)]">
   300→          <div className="pb-6">
   301→            {/* ==================== OVERVIEW TAB ==================== */}
   302→            <TabsContent value="overview" className="mt-4 space-y-4">
   303→              {asset.description && (
   304→                <Card className="border-0 shadow-sm">
   305→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Description</CardTitle></CardHeader>
   306→                  <CardContent><p className="text-sm text-muted-foreground whitespace-pre-wrap">{asset.description}</p></CardContent>
   307→                </Card>
   308→              )}
   309→              {Object.keys(specs).length > 0 && (
   310→                <Card className="border-0 shadow-sm">
   311→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Specifications</CardTitle></CardHeader>
   312→                  <CardContent>
   313→                    <div className="grid grid-cols-2 gap-x-6">
   314→                      {Object.entries(specs).map(([key, value]) => (
   315→                        <DetailRow key={key} label={key.replace(/_/g, ' ')} value={String(value)} />
   316→                      ))}
   317→                    </div>
   318→                  </CardContent>
   319→                </Card>
   320→              )}
   321→              {asset.pmSchedules && asset.pmSchedules.length > 0 && (
   322→                <Card className="border-0 shadow-sm">
   323→                  <CardHeader className="pb-2"><CardTitle className="text-sm">PM Schedules ({asset.pmSchedules.length})</CardTitle></CardHeader>
   324→                  <CardContent>
   325→                    <div className="space-y-2">
   326→                      {asset.pmSchedules.map((pm: any) => (
   327→                        <div key={pm.id} className="flex items-center justify-between text-sm py-2 border-b last:border-0">
   328→                          <div>
   329→                            <p className="font-medium">{pm.title}</p>
   330→                            <p className="text-xs text-muted-foreground">{pm.frequencyType} · {(pm.priority || '').toUpperCase()}</p>
   331→                          </div>
   332→                          <div className="text-right">
   333→                            <p className="text-xs text-muted-foreground">Next Due</p>
   334→                            <p className="text-sm font-medium">{pm.nextDueDate ? formatDate(pm.nextDueDate) : 'N/A'}</p>
   335→                          </div>
   336→                        </div>
   337→                      ))}
   338→                    </div>
   339→                  </CardContent>
   340→                </Card>
   341→              )}
   342→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   343→                <Card className="border-0 shadow-sm">
   344→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Details</CardTitle></CardHeader>
   345→                  <CardContent className="space-y-0">
   346→                    <DetailRow label="Category" value={asset.category?.name} />
   347→                    <Separator className="my-1" />
   348→                    <DetailRow label="Serial Number" value={asset.serialNumber} />
   349→                    <Separator className="my-1" />
   350→                    <DetailRow label="Manufacturer" value={asset.manufacturer} />
   351→                    <Separator className="my-1" />
   352→                    <DetailRow label="Model" value={asset.model} />
   353→                    <Separator className="my-1" />
   354→                    <DetailRow label="Location" value={asset.location} />
   355→                    <Separator className="my-1" />
   356→                    <DetailRow label="Plant" value={asset.plant?.name} />
   357→                  </CardContent>
   358→                </Card>
   359→                <Card className="border-0 shadow-sm">
   360→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Financial</CardTitle></CardHeader>
   361→                  <CardContent className="space-y-0">
   362→                    <DetailRow label="Purchase Cost" value={asset.purchaseCost ? formatCurrency(asset.purchaseCost) : undefined} />
   363→                    <Separator className="my-1" />
   364→                    <DetailRow label="Current Value" value={asset.currentValue ? formatCurrency(asset.currentValue) : undefined} />
   365→                    <Separator className="my-1" />
   366→                    <DetailRow label="Purchase Date" value={formatDate(asset.purchaseDate)} />
   367→                    <Separator className="my-1" />
   368→                    <DetailRow label="Warranty Expiry" value={formatDate(asset.warrantyExpiry)} />
   369→                    <Separator className="my-1" />
   370→                    <DetailRow label="Expected Life" value={asset.expectedLifeYears ? `${asset.expectedLifeYears} years` : undefined} />
   371→                  </CardContent>
   372→                </Card>
   373→              </div>
   374→              {asset.assignedTo && (
   375→                <Card className="border-0 shadow-sm">
   376→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Assigned To</CardTitle></CardHeader>
   377→                  <CardContent>
   378→                    <div className="flex items-center gap-2 text-sm">
   379→                      <Avatar className="h-7 w-7"><AvatarFallback className="text-[10px]">{getInitials(asset.assignedTo.fullName || 'U')}</AvatarFallback></Avatar>
   380→                      <span className="font-medium">{asset.assignedTo.fullName}</span>
   381→                      <span className="text-muted-foreground">@{asset.assignedTo.username}</span>
   382→                    </div>
   383→                  </CardContent>
   384→                </Card>
   385→              )}
   386→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   387→                {hasHierarchy && (
   388→                  <button onClick={() => setActiveTab('hierarchy')} className="flex items-center gap-2 p-3 rounded-lg border hover:bg-muted/50 transition-colors text-left">
   389→                    <GitBranch className="h-4 w-4 text-muted-foreground" />
   390→                    <div><p className="text-xs font-medium">Hierarchy</p><p className="text-[10px] text-muted-foreground">{asset.children.length} child(ren)</p></div>
   391→                  </button>
   392→                )}
   393→                <button onClick={() => setActiveTab('bom')} className="flex items-center gap-2 p-3 rounded-lg border hover:bg-muted/50 transition-colors text-left">
   394→                  <Layers className="h-4 w-4 text-muted-foreground" />
   395→                  <div><p className="text-xs font-medium">BOM</p><p className="text-[10px] text-muted-foreground">View components</p></div>
   396→                </button>
   397→                {hasIoT && (
   398→                  <button onClick={() => setActiveTab('condition')} className="flex items-center gap-2 p-3 rounded-lg border hover:bg-muted/50 transition-colors text-left">
   399→                    <Activity className="h-4 w-4 text-muted-foreground" />
   400→                    <div><p className="text-xs font-medium">Monitoring</p><p className="text-[10px] text-muted-foreground">{asset.iotDevices.length} device(s)</p></div>
   401→                  </button>
   402→                )}
   403→                <button onClick={() => setActiveTab('digital-twin')} className="flex items-center gap-2 p-3 rounded-lg border hover:bg-muted/50 transition-colors text-left">
   404→                  <Monitor className="h-4 w-4 text-muted-foreground" />
   405→                  <div><p className="text-xs font-medium">Digital Twin</p><p className="text-[10px] text-muted-foreground">{twin ? 'Active' : 'Not created'}</p></div>
   406→                </button>
   407→              </div>
   408→              {hasMR && (
   409→                <Card className="border-0 shadow-sm">
   410→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Recent Maintenance Requests</CardTitle></CardHeader>
   411→                  <CardContent>
   412→                    <div className="space-y-2">
   413→                      {asset.maintenanceRequests.map((mr: any) => (
   414→                        <div key={mr.id} className="flex items-center justify-between text-sm py-1.5 border-b last:border-0">
   415→                          <div><p className="font-medium">{mr.title || mr.requestNumber}</p><p className="text-xs text-muted-foreground">{mr.requestNumber}</p></div>
   416→                          <Badge variant="outline" className="text-[10px] capitalize">{(mr.status || '').replace(/_/g, ' ')}</Badge>
   417→                        </div>
   418→                      ))}
   419→                    </div>
   420→                  </CardContent>
   421→                </Card>
   422→              )}
   423→              {hasWO && (
   424→                <Card className="border-0 shadow-sm">
   425→                  <CardHeader className="pb-2"><CardTitle className="text-sm">Recent Work Orders</CardTitle></CardHeader>
   426→                  <CardContent>
   427→                    <div className="space-y-2">
