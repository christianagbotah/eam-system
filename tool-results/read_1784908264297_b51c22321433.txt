    55→// Safe trade name extraction — API may return Trade object {id, name, code} or string
    56→function tradeName(trade: any): string {
    57→  if (!trade) return 'Unassigned';
    58→  if (typeof trade === 'string') return trade;
    59→  if (typeof trade === 'object' && trade.name) return trade.name;
    60→  return String(trade);

    94→const exportCSV = (filename: string, headers: string[], rows: string[][]) => {
    95→  const csv = [headers.join(','), ...rows.map(r => r.map(c => `"${(c ?? '').toString().replace(/"/g, '""')}"`).join(','))].join('\n');
    96→  const blob = new Blob([csv], { type: 'text/csv' });
    97→  const url = URL.createObjectURL(blob);
    98→  const a = document.createElement('a');
    99→  a.href = url; a.download = `${filename}.csv`; a.click();
   100→  URL.revokeObjectURL(url);
   101→  toast.success(`Exported ${filename}.csv`);
   102→};

   305→export function WOReportsPage() {
   306→  const { startDate, setStartDate, endDate, setEndDate } = useDateRange();
   307→  const [reportData, setReportData] = useState<ReportData | null>(null);
   308→  const [loading, setLoading] = useState(false);
   309→  const [activeTab, setActiveTab] = useState('overview');
   310→  const [pdfLoading, setPdfLoading] = useState(false);
   311→  const [generatedAt, setGeneratedAt] = useState<string | null>(null);
   312→
   313→  // Module visibility
   314→  const woEnabled = useModuleEnabled(MODULE_CODES.WORK_ORDERS);
   315→  const repairsEnabled = useModuleEnabled(MODULE_CODES.REPAIRS);
   316→  const pmEnabled = useModuleEnabled(MODULE_CODES.PM_SCHEDULES);
   317→  const invEnabled = useModuleEnabled(MODULE_CODES.INVENTORY);
   318→  const dtEnabled = useModuleEnabled(MODULE_CODES.DOWNTIME);
   319→
   320→  // Filters
   321→  const [tradeFilter, setTradeFilter] = useState('');
   322→  const [priorityFilter, setPriorityFilter] = useState('');
   323→  const [deptFilter, setDeptFilter] = useState('');
   324→  const [moduleFilter, setModuleFilter] = useState<'all' | 'repairs' | 'pm'>('all');
   325→
   326→  // Fetch report data
   327→  const fetchReport = useCallback(() => {
   328→    // Don't fetch if the selected module filter's module is disabled
   329→    if (moduleFilter === 'repairs' && !repairsEnabled) return;
   330→    if (moduleFilter === 'pm' && !pmEnabled) return;
   331→    setLoading(true);
   332→    api
   333→      .get<ReportData>(
   334→        `/api/work-orders/reports?from=${startDate}&to=${endDate}&trade=${tradeFilter || ''}&priority=${priorityFilter || ''}&department=${deptFilter || ''}&moduleFilter=${moduleFilter}`,
   335→        { timeout: 60_000 }
   336→      )
   337→      .then((res) => {
   338→        if (res.success && res.data) {
   339→          setReportData(res.data);
   340→          setGeneratedAt(formatTimestamp());
   341→        } else {
   342→          setReportData(null);
   343→        }
   344→        setLoading(false);
   345→      })
   346→      .catch(() => {
   347→        setReportData(null);
   348→        setLoading(false);
   349→        toast.error('Failed to load report data');
   350→      });
   351→  }, [startDate, endDate, tradeFilter, priorityFilter, deptFilter, moduleFilter, repairsEnabled, pmEnabled]);
   352→
   353→  // Auto-fetch on mount
   354→  React.useEffect(() => {
   355→    fetchReport();
   356→  }, [fetchReport]);
   357→
   358→  const s = reportData?.summary;
   359→
   360→  // ============================================================================
   361→  // CSV EXPORT (Enhanced per tab)
   362→  // ============================================================================
   363→
   364→  const handleExportCSV = useCallback(() => {
   365→    if (!reportData) return;
   366→    const tab = activeTab;
   367→    let filename = `wo-report-${startDate}-to-${endDate}`;
   368→    let headers: string[] = [];
   369→    let rows: string[][] = [];
   370→
   371→    switch (tab) {
   372→      case 'overview': {
   373→        filename += '-overview';
   374→        headers = ['Category', 'Sub-Category', 'Count', 'Percentage'];
   375→        const total = s?.totalWOs || 1;
   376→        (reportData.distribution?.byType || []).forEach((d: any) => rows.push(['Type', d.type, String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   377→        (reportData.distribution?.byStatus || []).forEach((d: any) => rows.push(['Status', d.status, String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   378→        (reportData.distribution?.byTrade || []).forEach((d: any) => rows.push(['Trade', tradeName(d.trade), String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   379→        (reportData.distribution?.byPriority || []).forEach((d: any) => rows.push(['Priority', d.priority, String(d.count), `${((d.count/total)*100).toFixed(1)}%`]));
   380→        break;
   381→      }
   382→      case 'downtime': {
   383→        filename += '-downtime';
   384→        const dtData = reportData.downtime?.byTrade || [];
   385→        const totalHrs = dtData.reduce((a: number, b: any) => a + b.totalHours, 0);
   386→        const totalEvts = dtData.reduce((a: number, b: any) => a + b.events, 0);
   387→        headers = ['Trade', 'Events', 'Hours', 'Avg Hours/Event', '% of Total', 'Production Loss'];
   388→        rows = dtData.map((d: any) => [
   389→          tradeName(d.trade), String(d.events), String(d.totalHours),
   390→          d.events > 0 ? (d.totalHours / d.events).toFixed(1) : '0',
   391→          totalHrs > 0 ? `${((d.totalHours / totalHrs) * 100).toFixed(1)}%` : '0%',
   392→          String(d.productionLoss || 0),
   393→        ]);
   394→        rows.push(['TOTAL', String(totalEvts), String(totalHrs), '-', '100%', String(dtData.reduce((a: number, b: any) => a + (b.productionLoss || 0), 0))]);
   395→        break;
   396→      }
   397→      case 'response-time': {
   398→        filename += '-response-time';
   399→        headers = ['Group', 'Type', 'Count', 'Avg Hours', 'Min Hours', 'Max Hours', 'SLA Target (h)', 'SLA Compliance'];
   400→        (reportData.responseTime?.byPriority || []).forEach((d: any) => {
   401→          const target = SLA_TARGETS[d.priority] || 8;
   402→          const compliance = d.avgHours <= target ? 100 : Math.round((target / d.avgHours) * 100);
   403→          rows.push([d.priority, 'Priority', String(d.count), String(d.avgHours), String(d.minHours), String(d.maxHours), String(target), `${compliance}%`]);
   404→        });
   405→        (reportData.responseTime?.byTrade || []).forEach((d: any) =>
   406→          rows.push([tradeName(d.trade), 'Trade', String(d.count), String(d.avgHours), '-', '-', '-', '-'])
   407→        );
   408→        break;
   409→      }
   410→      case 'breakdowns': {
   411→        filename += '-breakdowns';
   412→        const bdTotal = reportData.breakdowns?.total || 1;
   413→        headers = ['Group', 'Type', 'Count', '% of Total'];
   414→        (reportData.breakdowns?.byTrade || []).forEach((d: any) => rows.push([tradeName(d.trade), 'Trade', String(d.count), `${((d.count/bdTotal)*100).toFixed(1)}%`]));
   415→        (reportData.breakdowns?.byType || []).forEach((d: any) => rows.push([d.type, 'Type', String(d.count), `${((d.count/bdTotal)*100).toFixed(1)}%`]));
   416→        (reportData.breakdowns?.byPriority || []).forEach((d: any) => rows.push([d.priority, 'Priority', String(d.count), `${((d.count/bdTotal)*100).toFixed(1)}%`]));
   417→        break;
   418→      }
   419→      case 'man-hours': {
   420→        filename += '-man-hours';
   421→        const mhTotal = reportData.manHours?.grandTotal || 1;
   422→        const techs = reportData.manHours?.byTechnician || [];
   423→        headers = ['Technician', 'Total Hours', 'WO Count', 'Avg Hrs/WO', '% of Grand Total'];
   424→        rows = techs.map((d: any) => [
   425→          d.name, String(d.totalHours), String(d.woCount),
   426→          String(d.avgHoursPerWO), `${((d.totalHours / mhTotal) * 100).toFixed(1)}%`,
   427→        ]);
   428→        rows.push(['TOTAL', String(reportData.manHours?.grandTotal || 0), '-', '-', '100%']);
   429→        break;
   430→      }
   431→      case 'materials': {
   432→        filename += '-materials';
   433→        const matTotal = reportData.materials?.totalCost || 1;
   434→        const items = reportData.materials?.topItems || [];
   435→        headers = ['Item Name', 'Part #', 'Supplier', 'Unit', 'Total Qty', 'Unit Cost', 'Total Cost', 'WOs Used', '% of Total Cost'];
   436→        rows = items.map((d: any) => [
   437→          d.name, d.inventoryItem?.itemCode || '', d.inventoryItem?.supplier || '', d.inventoryItem?.unitOfMeasure || '',
   438→          String(d.totalQty), String(d.unitCost || ''), String(d.totalCost),
   439→          String(d.woCount), `${((d.totalCost / matTotal) * 100).toFixed(1)}%`,
   440→        ]);
   441→        rows.push(['TOTAL', '', '', '', String(reportData.materials?.totalQty || 0), '', String(reportData.materials?.totalCost || 0), '-', '100%']);
   442→        break;
   443→      }
   444→      case 'failure-rate': {
   445→        filename += '-failure-rate';
   446→        headers = ['Equipment', 'Tag', 'Manufacturer', 'Model', 'Category', 'Criticality', 'Location', 'Total WOs', 'Failures', 'Failure Rate', 'MTBF (Days)', 'Risk Level'];
   447→        const assets = reportData.failureRate?.byAsset || [];
   448→        rows = assets.map((d: any) => {
   449→          const mtbf = reportData.failureRate?.mtbf?.find((m: any) => m.assetId === d.assetId);
   450→          const a = d.asset || {};
   451→          const risk = d.failureRate >= 40 ? 'High' : d.failureRate >= 20 ? 'Medium' : 'Low';
   452→          return [d.assetName || d.assetId, a.assetTag || '', a.manufacturer || '', a.model || '', typeof a.category === 'string' ? a.category : a.category?.name || '', a.criticality || '',
   453→            [a.building, a.floor, a.area].filter(Boolean).join('/') || a.location || '',
   454→            String(d.totalWOs), String(d.failures), `${d.failureRate}%`, String(mtbf?.mtbfDays || '-'), risk];
   455→        });
   456→        break;
   457→      }
   458→      case 'stoppages': {
   459→        filename += '-stoppages';
   460→        const spData = reportData.stoppages?.byTrade || [];
   461→        const totalSp = spData.reduce((a: number, b: any) => a + b.count, 0);
   462→        const totalSpHrs = spData.reduce((a: number, b: any) => a + b.totalHours, 0);
   463→        headers = ['Trade', 'Stoppage Count', 'Total Hours', 'Avg Hours', '% of Total'];
   464→        rows = spData.map((d: any) => [
