    15→// Usage:
    16→//   bun run prisma/seed-permissions-only.ts

    43→    port: parseInt(_url.port || '3306', 10),

    46→    database: _url.pathname.slice(1),

   620→function formatModuleName(mod: string): string {
   621→  return mod
   622→    .split('_')
   623→    .map(w => w.charAt(0).toUpperCase() + w.slice(1))
   624→    .join(' ');
   625→}

   627→function formatActionName(action: string): string {
   628→  return action
   629→    .split('_')
   630→    .map(w => w.charAt(0).toUpperCase() + w.slice(1))
   631→    .join(' ');
   632→}

   638→async function seedPermissionsOnly() {
   639→  console.log('\n═══════════════════════════════════════════════════════════════');
   640→  console.log('  iAssetsPro EAM — Permission-Only Seed (NON-DESTRUCTIVE)');
   641→  console.log('═══════════════════════════════════════════════════════════════\n');
   642→  console.log('⚠️  This script ONLY modifies: permissions, roles, role_permissions');
   643→  console.log('   All other data (users, assets, work orders, etc.) is UNTOUCHED.\n');
   644→
   645→  // ── Test connection ──
   646→  try {
   647→    await db.$queryRawUnsafe('SELECT 1 as ok');
   648→    console.log('✅ Database connection successful\n');
   649→  } catch (connErr) {
   650→    console.error('❌ FATAL: Cannot connect to database!');
   651→    console.error('   Error:', (connErr as Error).message);
   652→    process.exit(1);
   653→  }
   654→
   655→  // ══════════════════════════════════════════════════════════════════════════
   656→  // STEP 1: UPSERT PERMISSIONS (non-destructive — only creates new, updates existing names)
   657→  // ══════════════════════════════════════════════════════════════════════════
   658→  console.log('📋 Upserting permissions...');
   659→  const permissionMap: Record<string, string> = {};
   660→  let createdCount = 0;
   661→  let updatedCount = 0;
   662→
   663→  for (const [moduleName, actions] of Object.entries(modulePermissions)) {
   664→    for (const action of actions) {
   665→      const slug = `${moduleName}.${action}`;
   666→      const name = `${formatModuleName(moduleName)} - ${formatActionName(action)}`;
   667→      const description = `${formatActionName(action)} access for ${formatModuleName(moduleName)} module`;
   668→
   669→      const permission = await db.permission.upsert({
   670→        where: { slug },
   671→        update: { name, module: moduleName, action, description },
   672→        create: { slug, name, module: moduleName, action, description },
   673→      });
   674→
   675→      permissionMap[slug] = permission.id;
   676→      createdCount++;
   677→    }
   678→  }
   679→  console.log(`  ✅ Processed ${createdCount} permission slugs across ${Object.keys(modulePermissions).length} modules\n`);
   680→
   681→  // ══════════════════════════════════════════════════════════════════════════
   682→  // STEP 2: UPSERT ROLES
   683→  // ══════════════════════════════════════════════════════════════════════════
   684→  console.log('🔑 Upserting roles...');
   685→  const createdRoles: Record<string, string> = {};
   686→
   687→  for (const roleDef of roleDefinitions) {
   688→    const role = await db.role.upsert({
   689→      where: { slug: roleDef.slug },
   690→      update: {
   691→        name: roleDef.name,
   692→        description: roleDef.description,
   693→        level: roleDef.level,
   694→        isSystem: roleDef.isSystem,
   695→      },
   696→      create: {
   697→        name: roleDef.name,
   698→        slug: roleDef.slug,
   699→        description: roleDef.description,
   700→        level: roleDef.level,
   701→        isSystem: roleDef.isSystem,
   702→      },
   703→    });
   704→    createdRoles[roleDef.slug] = role.id;
   705→  }
   706→  console.log(`  ✅ Processed ${roleDefinitions.length} roles\n`);
   707→
   708→  // ══════════════════════════════════════════════════════════════════════════
   709→  // STEP 3: CLEAR AND RE-SYNC ROLE-PERMISSION MAPPINGS
   710→  // ══════════════════════════════════════════════════════════════════════════
   711→  console.log('🔗 Syncing role-permission mappings...');
   712→
   713→  // Clear ALL existing mappings
   714→  const deleted = await db.rolePermission.deleteMany();
   715→  console.log(`  🗑️  Cleared ${deleted.count} existing mappings`);
   716→
   717→  // Admin gets ALL permissions
   718→  const allPermIds = Object.values(permissionMap);
   719→  await db.rolePermission.createMany({
   720→    data: allPermIds.map((pid) => ({ roleId: createdRoles['admin'], permissionId: pid })),
   721→    skipDuplicates: true,
   722→  });
   723→  console.log(`  ✅ admin: ${allPermIds.length} permissions (ALL)`);
   724→
   725→  // Other roles get their bundles
   726→  let totalMappings = 0;
   727→  for (const [roleSlug, permSlugs] of Object.entries(rolePermissionBundles)) {
   728→    const roleId = createdRoles[roleSlug];
   729→    if (!roleId) {
   730→      console.log(`  ⚠️  Role "${roleSlug}" not found in roleDefinitions — skipping`);
   731→      continue;
   732→    }
   733→
   734→    const validPermIds: string[] = [];
   735→    for (const slug of permSlugs) {
   736→      const pid = permissionMap[slug];
   737→      if (pid) validPermIds.push(pid);
   738→    }
   739→
   740→    if (validPermIds.length > 0) {
   741→      await db.rolePermission.createMany({
   742→        data: validPermIds.map((pid) => ({ roleId, permissionId: pid })),
   743→        skipDuplicates: true,
   744→      });
   745→    }
   746→    console.log(`  ✅ ${roleSlug}: ${validPermIds.length} permissions`);
   747→    totalMappings += validPermIds.length;
   748→  }
   749→
   750→  console.log(`\n  📊 Total non-admin role-permission mappings: ${totalMappings}`);
   751→  console.log(`  📊 Total including admin: ${totalMappings + allPermIds.length}\n`);
   752→
   753→  // ══════════════════════════════════════════════════════════════════════════
   754→  // VERIFICATION
   755→  // ══════════════════════════════════════════════════════════════════════════
   756→  console.log('🔍 Verification...');
   757→  const totalPerms = await db.permission.count();
   758→  const totalRoles = await db.role.count();
   759→  const totalMappingsDb = await db.rolePermission.count();
   760→
   761→  console.log(`  Permissions in DB: ${totalPerms}`);
   762→  console.log(`  Roles in DB: ${totalRoles}`);
   763→  console.log(`  Role-permission mappings in DB: ${totalMappingsDb}`);
   764→
   765→  if (totalPerms === createdCount && totalRoles === roleDefinitions.length) {
   766→    console.log('\n✅ ═════════════════════════════════════════════════════════════');
   767→    console.log('   PERMISSION SEED COMPLETE — All users should re-login');
   768→    console.log('══════════════════════════════════════════════════════════════ ✅\n');
   769→  } else {
   770→    console.log('\n⚠️  Permission seed completed with warnings — check counts above\n');
   771→  }
   772→
   773→  await db.$disconnect();
   774→}
   775→
   776→seedPermissionsOnly().catch((err) => {
   777→  console.error('❌ Seed failed:', err);
   778→  process.exit(1);
   779→});
