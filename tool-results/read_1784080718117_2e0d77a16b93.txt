     1→     1→import { NextRequest, NextResponse } from 'next/server';
     2→     2→import { db } from '@/lib/db';
     3→     3→import { getSession, hasPermission, isAdmin, hasAnyPermission } from '@/lib/auth';
     4→     4→import { getPlantScope, getPlantFilterWhere } from '@/lib/plant-scope';
     5→     5→import { generateReportPDF } from '@/lib/generate-report-pdf';
     6→     6→
     7→     7→// GET /api/work-orders/reports
     8→     8→// Comprehensive WO reports: downtime, response time, breakdowns, man hours, materials, failure rate, stoppages
     9→     9→export async function GET(request: NextRequest) {
    10→    10→  try {
    11→    11→    const session = getSession(request);
    12→    12→    if (!session) {
    13→    13→      return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
    14→    14→    }
    15→    15→    // Permission gate: require reports.view, work_orders.dashboard, or admin
    16→    16→    if (!hasAnyPermission(session, ['reports.view', 'reports.export', 'work_orders.dashboard', 'analytics.view']) && !isAdmin(session)) {
    17→    17→      return NextResponse.json({ success: false, error: 'Insufficient permissions: reports.view required' }, { status: 403 });
    18→    18→    }
    19→    19→
    20→    20→    const { searchParams } = new URL(request.url);
    21→    21→    const from = searchParams.get('from');
    22→    22→    const to = searchParams.get('to');
    23→    23→    const department = searchParams.get('department');
    24→    24→    const trade = searchParams.get('trade');
    25→    25→    const priority = searchParams.get('priority');
    26→    26→    const format = searchParams.get('format');
    27→    27→
    28→    28→    const plantScope = await getPlantScope(request, session);
    29→    29→    const plantFilter = getPlantFilterWhere(plantScope);
    30→    30→
    31→    31→    // Date filter
    32→    32→    const dateFilter: Record<string, unknown> = {};
    33→    33→    if (from) dateFilter.gte = new Date(from + 'T00:00:00');
    34→    34→    if (to) dateFilter.lte = new Date(to + 'T23:59:59');
    35→    35→
    36→    36→    const woWhere: Record<string, unknown> = {
    37→    37→      ...plantFilter,
    38→    38→      status: { notIn: ['draft', 'cancelled'] },
    39→    39→    };
    40→    40→    if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
    41→    41→    if (department) woWhere.departmentId = department;
    42→    42→    if (trade) woWhere.tradeActivity = trade;
    43→    43→    if (priority) woWhere.priority = priority;
    44→    44→
    45→    45→    // Fetch all relevant WOs with relations
    46→    46→    const workOrders = await db.workOrder.findMany({
    47→    47→      where: Object.keys(woWhere).length > 0 ? woWhere : undefined,
    48→    48→      include: {
    49→    49→        assignee: { select: { id: true, fullName: true, department: true } },
    50→    50→        teamLeader: { select: { id: true, fullName: true } },
    51→    51→        materials: true,
    52→    52→        timeLogs: true,
    53→    53→        workOrderDowntimes: true,
    54→    54→        repairCompletion: true,
    55→    55→        statusHistory: { orderBy: { createdAt: 'asc' } },
    56→    56→        taskExecutions: true,
    57→    57→        failureRecords: true,
    58→    58→      },
    59→    59→      orderBy: { createdAt: 'desc' },
    60→    60→    });
    61→    61→
    62→    62→    const total = workOrders.length;
    63→    63→    const closedWOs = workOrders.filter(wo => wo.status === 'closed');
    64→    64→    const completedWOs = workOrders.filter(wo => ['completed', 'verified', 'closed'].includes(wo.status));
    65→    65→    const breakdownWOs = workOrders.filter(wo => ['emergency', 'corrective'].includes(wo.type));
    66→    66→
    67→    67→    // ========== 1. SUMMARY KPIs ==========
    68→    68→    const summary = {
    69→    69→      totalWOs: total,
    70→    70→      completedWOs: completedWOs.length,
    71→    71→      closedWOs: closedWOs.length,
    72→    72→      completionRate: total > 0 ? Math.round((completedWOs.length / total) * 10000) / 100 : 0,
    73→    73→      breakdownCount: breakdownWOs.length,
    74→    74→      breakdownRate: total > 0 ? Math.round((breakdownWOs.length / total) * 10000) / 100 : 0,
    75→    75→    };
    76→    76→
    77→    77→    // ========== 2. DOWNTIME (per trade) ==========
    78→    78→    const downtimeByTrade: Record<string, { totalMinutes: number; events: number; productionLoss: number }> = {};
    79→    79→    for (const wo of workOrders) {
    80→    80→      const dtEvents = wo.workOrderDowntimes || [];
    81→    81→      const trade = wo.tradeActivity || 'Unassigned';
    82→    82→      for (const dt of dtEvents) {
    83→    83→        if (!downtimeByTrade[trade]) downtimeByTrade[trade] = { totalMinutes: 0, events: 0, productionLoss: 0 };
    84→    84→        downtimeByTrade[trade].totalMinutes += dt.durationMinutes;
    85→    85→        downtimeByTrade[trade].events++;
    86→    86→        downtimeByTrade[trade].productionLoss += dt.productionLoss || 0;
    87→    87→      }
    88→    88→    }
    89→    89→    const totalDowntimeMinutes = Object.values(downtimeByTrade).reduce((s, d) => s + d.totalMinutes, 0);
    90→    90→    const totalDowntimeEvents = Object.values(downtimeByTrade).reduce((s, d) => s + d.events, 0);
    91→    91→    const totalProductionLoss = Object.values(downtimeByTrade).reduce((s, d) => s + d.productionLoss, 0);
    92→    92→
    93→    93→    const downtimeByTradeArray = Object.entries(downtimeByTrade)
    94→    94→      .map(([tradeName, data]) => ({
    95→    95→        trade: tradeName,
    96→    96→        totalHours: Math.round((data.totalMinutes / 60) * 100) / 100,
    97→    97→        totalMinutes: data.totalMinutes,
    98→    98→        events: data.events,
    99→    99→        productionLoss: data.productionLoss,
   100→   100→      }))
   101→   101→      .sort((a, b) => b.totalHours - a.totalHours);
   102→   102→
   103→   103→    // Downtime by month
   104→   104→    const downtimeByMonth: Record<string, { minutes: number; events: number }> = {};
   105→   105→    for (const wo of workOrders) {
   106→   106→      for (const dt of (wo.workOrderDowntimes || [])) {
   107→   107→        const month = dt.downtimeStart?.toISOString().slice(0, 7) || 'unknown';
   108→   108→        if (!downtimeByMonth[month]) downtimeByMonth[month] = { minutes: 0, events: 0 };
   109→   109→        downtimeByMonth[month].minutes += dt.durationMinutes;
   110→   110→        downtimeByMonth[month].events++;
   111→   111→      }
   112→   112→    }
   113→   113→    const downtimeByMonthArray = Object.entries(downtimeByMonth)
   114→   114→      .map(([month, data]) => ({
   115→   115→        month,
   116→   116→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   117→   117→        events: data.events,
   118→   118→      }))
   119→   119→      .sort((a, b) => a.month.localeCompare(b.month));
   120→   120→
   121→   121→    // Downtime by category
   122→   122→    const downtimeByCategory: Record<string, { minutes: number; events: number }> = {};
   123→   123→    for (const wo of workOrders) {
   124→   124→      for (const dt of (wo.workOrderDowntimes || [])) {
   125→   125→        const cat = dt.category || 'unplanned';
   126→   126→        if (!downtimeByCategory[cat]) downtimeByCategory[cat] = { minutes: 0, events: 0 };
   127→   127→        downtimeByCategory[cat].minutes += dt.durationMinutes;
   128→   128→        downtimeByCategory[cat].events++;
   129→   129→      }
   130→   130→    }
   131→   131→    const downtimeByCategoryArray = Object.entries(downtimeByCategory)
   132→   132→      .map(([category, data]) => ({
   133→   133→        category,
   134→   134→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   135→   135→        events: data.events,
   136→   136→      }))
   137→   137→      .sort((a, b) => b.totalHours - a.totalHours);
   138→   138→
   139→   139→    // ========== 3. RESPONSE TIME ==========
   140→   140→    // Response time = time from WO creation to first status that indicates work started (in_progress)
   141→   141→    const responseTimes: { hours: number; priority: string; type: string; trade: string; woId: string }[] = [];
   142→   142→    for (const wo of workOrders) {
   143→   143→      const history = wo.statusHistory || [];
   144→   144→      if (history.length === 0) continue;
   145→   145→
   146→   146→      // Find the transition to 'in_progress' (or 'assigned' as fallback)
   147→   147→      const startEntry = history.find(h => h.toStatus === 'in_progress')
   148→   148→        || history.find(h => h.toStatus === 'assigned');
   149→   149→
   150→   150→      if (startEntry && wo.createdAt) {
   151→   151→        const hours = (startEntry.createdAt.getTime() - wo.createdAt.getTime()) / (1000 * 60 * 60);
   152→   152→        if (hours >= 0 && hours < 10000) { // sanity check
   153→   153→          responseTimes.push({
   154→   154→            hours: Math.round(hours * 100) / 100,
   155→   155→            priority: wo.priority,
   156→   156→            type: wo.type,
   157→   157→            trade: wo.tradeActivity || 'Unassigned',
   158→   158→            woId: wo.id,
   159→   159→          });
   160→   160→        }
   161→   161→      }
   162→   162→    }
   163→   163→
   164→   164→    const avgResponseTime = responseTimes.length > 0
   165→   165→      ? Math.round(responseTimes.reduce((s, r) => s + r.hours, 0) / responseTimes.length * 100) / 100
   166→   166→      : 0;
   167→   167→
   168→   168→    // Response time by priority
   169→   169→    const responseByPriority: Record<string, number[]> = {};
   170→   170→    for (const r of responseTimes) {
   171→   171→      if (!responseByPriority[r.priority]) responseByPriority[r.priority] = [];
   172→   172→      responseByPriority[r.priority].push(r.hours);
   173→   173→    }
   174→   174→    const responseByPriorityArray = Object.entries(responseByPriority)
   175→   175→      .map(([p, hours]) => ({
   176→   176→        priority: p,
   177→   177→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   178→   178→        minHours: Math.round(Math.min(...hours) * 100) / 100,
   179→   179→        maxHours: Math.round(Math.max(...hours) * 100) / 100,
   180→   180→        count: hours.length,
   181→   181→      }))
   182→   182→      .sort((a, b) => a.avgHours - b.avgHours);
   183→   183→
   184→   184→    // Response time by trade
   185→   185→    const responseByTrade: Record<string, number[]> = {};
   186→   186→    for (const r of responseTimes) {
   187→   187→      if (!responseByTrade[r.trade]) responseByTrade[r.trade] = [];
   188→   188→      responseByTrade[r.trade].push(r.hours);
   189→   189→    }
   190→   190→    const responseByTradeArray = Object.entries(responseByTrade)
   191→   191→      .map(([t, hours]) => ({
   192→   192→        trade: t,
   193→   193→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   194→   194→        count: hours.length,
   195→   195→      }))
   196→   196→      .sort((a, b) => a.avgHours - b.avgHours);
   197→   197→
   198→   198→    // ========== 4. BREAKDOWNS ==========
   199→   199→    // Breakdown by type
   200→   200→    const breakdownByType: Record<string, number> = {};
   201→   201→    for (const wo of breakdownWOs) {
   202→   202→      const t = wo.type || 'unknown';
   203→   203→      breakdownByType[t] = (breakdownByType[t] || 0) + 1;
   204→   204→    }
   205→   205→    const breakdownByTypeArray = Object.entries(breakdownByType)
   206→   206→      .map(([type, count]) => ({ type, count }))
   207→   207→      .sort((a, b) => b.count - a.count);
   208→   208→
   209→   209→    // Breakdown by trade
   210→   210→    const breakdownByTrade: Record<string, number> = {};
   211→   211→    for (const wo of breakdownWOs) {
   212→   212→      const t = wo.tradeActivity || 'Unassigned';
   213→   213→      breakdownByTrade[t] = (breakdownByTrade[t] || 0) + 1;
   214→   214→    }
   215→   215→    const breakdownByTradeArray = Object.entries(breakdownByTrade)
   216→   216→      .map(([trade, count]) => ({ trade, count }))
   217→   217→      .sort((a, b) => b.count - a.count);
   218→   218→
   219→   219→    // Breakdown by month
   220→   220→    const breakdownByMonth: Record<string, number> = {};
   221→   221→    for (const wo of breakdownWOs) {
   222→   222→      const month = wo.createdAt.toISOString().slice(0, 7);
   223→   223→      breakdownByMonth[month] = (breakdownByMonth[month] || 0) + 1;
   224→   224→    }
   225→   225→    const breakdownByMonthArray = Object.entries(breakdownByMonth)
   226→   226→      .map(([month, count]) => ({ month, count }))
   227→   227→      .sort((a, b) => a.month.localeCompare(b.month));
   228→   228→
   229→   229→    // Breakdown by priority
   230→   230→    const breakdownByPriority: Record<string, number> = {};
   231→   231→    for (const wo of breakdownWOs) {
   232→   232→      const p = wo.priority || 'medium';
   233→   233→      breakdownByPriority[p] = (breakdownByPriority[p] || 0) + 1;
   234→   234→    }
   235→   235→    const breakdownByPriorityArray = Object.entries(breakdownByPriority)
   236→   236→      .map(([priority, count]) => ({ priority, count }))
   237→   237→      .sort((a, b) => b.count - a.count);
   238→   238→
   239→   239→    // ========== 5. MAN HOURS ==========
   240→   240→    // From timeLogs and repairCompletion
   241→   241→    const manHoursByTrade: Record<string, { totalHours: number; woCount: number }> = {};
   242→   242→    const manHoursByTechnician: Record<string, { name: string; totalHours: number; woCount: number }> = {};
   243→   243→    let grandTotalManHours = 0;
   244→   244→
   245→   245→    for (const wo of workOrders) {
   246→   246→      const trade = wo.tradeActivity || 'Unassigned';
   247→   247→
   248→   248→      // Get hours from timeLogs (primary), actualHours (fallback), repairCompletion (override)
   249→   249→      let woHours = 0;
   250→   250→      for (const tl of (wo.timeLogs || [])) {
   251→   251→        if (tl.duration && tl.duration > 0) {
   252→   252→          woHours += tl.duration;
   253→   253→        }
   254→   254→      }
   255→   255→      // Fallback to WO's actualHours when no time logs exist
   256→   256→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   257→   257→        woHours = wo.actualHours;
   258→   258→      }
   259→   259→      // Override with repairCompletion if higher (supervisor-verified hours)
   260→   260→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   261→   261→        woHours = wo.repairCompletion.totalLaborHours;
   262→   262→      }
   263→   263→
   264→   264→      grandTotalManHours += woHours;
   265→   265→
   266→   266→      // By trade
   267→   267→      if (!manHoursByTrade[trade]) manHoursByTrade[trade] = { totalHours: 0, woCount: 0 };
   268→   268→      manHoursByTrade[trade].totalHours += woHours;
   269→   269→      manHoursByTrade[trade].woCount++;
   270→   270→
   271→   271→      // By technician
   272→   272→      const techId = wo.assignedTo || 'unassigned';
   273→   273→      const techName = wo.assignee?.fullName || 'Unassigned';
   274→   274→      if (!manHoursByTechnician[techId]) {
   275→   275→        manHoursByTechnician[techId] = { name: techName, totalHours: 0, woCount: 0 };
   276→   276→      }
   277→   277→      manHoursByTechnician[techId].totalHours += woHours;
   278→   278→      manHoursByTechnician[techId].woCount++;
   279→   279→    }
   280→   280→
   281→   281→    const manHoursByTradeArray = Object.entries(manHoursByTrade)
   282→   282→      .map(([trade, data]) => ({
   283→   283→        trade,
   284→   284→        totalHours: Math.round(data.totalHours * 100) / 100,
   285→   285→        woCount: data.woCount,
   286→   286→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   287→   287→      }))
   288→   288→      .sort((a, b) => b.totalHours - a.totalHours);
   289→   289→
   290→   290→    const manHoursByTechnicianArray = Object.values(manHoursByTechnician)
   291→   291→      .map(data => ({
   292→   292→        ...data,
   293→   293→        totalHours: Math.round(data.totalHours * 100) / 100,
   294→   294→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   295→   295→      }))
   296→   296→      .sort((a, b) => b.totalHours - a.totalHours);
   297→   297→
   298→   298→    // Man hours by month
   299→   299→    const manHoursByMonth: Record<string, number> = {};
   300→   300→    for (const wo of workOrders) {
   301→   301→      const month = wo.createdAt.toISOString().slice(0, 7);
   302→   302→      let woHours = 0;
   303→   303→      for (const tl of (wo.timeLogs || [])) {
   304→   304→        if (tl.duration && tl.duration > 0) woHours += tl.duration;
   305→   305→      }
   306→   306→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   307→   307→        woHours = wo.actualHours;
   308→   308→      }
   309→   309→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   310→   310→        woHours = wo.repairCompletion.totalLaborHours;
   311→   311→      }
   312→   312→      manHoursByMonth[month] = (manHoursByMonth[month] || 0) + woHours;
   313→   313→    }
   314→   314→    const manHoursByMonthArray = Object.entries(manHoursByMonth)
   315→   315→      .map(([month, totalHours]) => ({
   316→   316→        month,
   317→   317→        totalHours: Math.round(totalHours * 100) / 100,
   318→   318→      }))
   319→   319→      .sort((a, b) => a.month.localeCompare(b.month));
   320→   320→
   321→   321→    // Man hours by activity type
   322→   322→    const manHoursByActivity: Record<string, number> = {};
   323→   323→    for (const wo of workOrders) {
   324→   324→      for (const tl of (wo.timeLogs || [])) {
   325→   325→        const act = tl.activityType || 'maintenance';
   326→   326→        manHoursByActivity[act] = (manHoursByActivity[act] || 0) + (tl.duration || 0);
   327→   327→      }
   328→   328→    }
   329→   329→    const manHoursByActivityArray = Object.entries(manHoursByActivity)
   330→   330→      .map(([activity, totalHours]) => ({
   331→   331→        activity,
   332→   332→        totalHours: Math.round(totalHours * 100) / 100,
   333→   333→      }))
   334→   334→      .sort((a, b) => b.totalHours - a.totalHours);
   335→   335→
   336→   336→    // ========== 6. MATERIAL QUANTITY AND COST ==========
   337→   337→    let totalMaterialCost = 0;
   338→   338→    let totalMaterialQty = 0;
   339→   339→    const materialsByCategory: Record<string, { qty: number; cost: number; woCount: number }> = {};
   340→   340→    const topMaterials: Record<string, { name: string; qty: number; cost: number; woCount: Set<string> }> = {};
   341→   341→
   342→   342→    for (const wo of workOrders) {
   343→   343→      for (const mat of (wo.materials || [])) {
   344→   344→        const qty = mat.quantity || 0;
   345→   345→        const cost = mat.totalCost || (mat.unitCost && mat.quantity ? mat.unitCost * mat.quantity : 0);
   346→   346→        totalMaterialQty += qty;
   347→   347→        totalMaterialCost += cost;
   348→   348→
   349→   349→        // By WO type
   350→   350→        const woType = wo.type || 'other';
   351→   351→        if (!materialsByCategory[woType]) materialsByCategory[woType] = { qty: 0, cost: 0, woCount: 0 };
   352→   352→        materialsByCategory[woType].qty += qty;
   353→   353→        materialsByCategory[woType].cost += cost;
   354→   354→        materialsByCategory[woType].woCount++;
   355→   355→
   356→   356→        // Top items
   357→   357→        const itemName = mat.itemName || mat.itemId || 'Unknown Item';
   358→   358→        if (!topMaterials[itemName]) topMaterials[itemName] = { name: itemName, qty: 0, cost: 0, woCount: new Set() };
   359→   359→        topMaterials[itemName].qty += qty;
   360→   360→        topMaterials[itemName].cost += cost;
   361→   361→        topMaterials[itemName].woCount.add(wo.id);
   362→   362→      }
   363→   363→    }
   364→   364→
   365→   365→    const materialsByCategoryArray = Object.entries(materialsByCategory)
   366→   366→      .map(([type, data]) => ({
   367→   367→        type,
   368→   368→        totalQty: Math.round(data.qty * 100) / 100,
   369→   369→        totalCost: Math.round(data.cost * 100) / 100,
   370→   370→        woCount: data.woCount,
   371→   371→      }))
   372→   372→      .sort((a, b) => b.totalCost - a.totalCost);
   373→   373→
   374→   374→    const topMaterialsArray = Object.values(topMaterials)
   375→   375→      .map(data => ({
   376→   376→        name: data.name,
   377→   377→        totalQty: Math.round(data.qty * 100) / 100,
   378→   378→        totalCost: Math.round(data.cost * 100) / 100,
   379→   379→        woCount: data.woCount.size,
   380→   380→      }))
   381→   381→      .sort((a, b) => b.totalCost - a.totalCost)
   382→   382→      .slice(0, 20);
   383→   383→
   384→   384→    // Material cost by month
   385→   385→    const materialCostByMonth: Record<string, number> = {};
   386→   386→    for (const wo of workOrders) {
   387→   387→      const month = wo.createdAt.toISOString().slice(0, 7);
   388→   388→      let woMatCost = 0;
   389→   389→      for (const mat of (wo.materials || [])) {
   390→   390→        woMatCost += mat.totalCost || (mat.unitCost && mat.quantity ? mat.unitCost * mat.quantity : 0);
   391→   391→      }
   392→   392→      materialCostByMonth[month] = (materialCostByMonth[month] || 0) + woMatCost;
   393→   393→    }
   394→   394→    const materialCostByMonthArray = Object.entries(materialCostByMonth)
   395→   395→      .map(([month, totalCost]) => ({
   396→   396→        month,
   397→   397→        totalCost: Math.round(totalCost * 100) / 100,
   398→   398→      }))
   399→   399→      .sort((a, b) => a.month.localeCompare(b.month));
   400→   400→
   401→   401→    // ========== 7. FAILURE RATE ==========
   402→   402→    // Failure rate per asset: (failure WOs / total WOs for that asset) * 100
   403→   403→    const assetWOs: Record<string, { name: string; total: number; failures: number }> = {};
   404→   404→    for (const wo of workOrders) {
   405→   405→      const assetId = wo.assetId || 'unassigned';
   406→   406→      const assetName = wo.assetName || 'Unknown Asset';
   407→   407→      if (!assetWOs[assetId]) assetWOs[assetId] = { name: assetName, total: 0, failures: 0 };
   408→   408→      assetWOs[assetId].total++;
   409→   409→      if (['emergency', 'corrective'].includes(wo.type)) {
   410→   410→        assetWOs[assetId].failures++;
   411→   411→      }
   412→   412→    }
   413→   413→
   414→   414→    const failureRateByAsset = Object.entries(assetWOs)
   415→   415→      .filter(([, data]) => data.total >= 2) // Only assets with 2+ WOs
   416→   416→      .map(([assetId, data]) => ({
   417→   417→        assetId,
   418→   418→        assetName: data.name,
   419→   419→        totalWOs: data.total,
   420→   420→        failures: data.failures,
   421→   421→        failureRate: Math.round((data.failures / data.total) * 10000) / 100,
   422→   422→      }))
   423→   423→      .sort((a, b) => b.failureRate - a.failureRate)
   424→   424→      .slice(0, 20);
   425→   425→
   426→   426→    // Failure rate by type
   427→   427→    const failureRateByType: Record<string, { total: number; failures: number }> = {};
   428→   428→    for (const wo of workOrders) {
   429→   429→      const t = wo.type || 'other';
   430→   430→      if (!failureRateByType[t]) failureRateByType[t] = { total: 0, failures: 0 };
   431→   431→      failureRateByType[t].total++;
   432→   432→      if (['emergency', 'corrective'].includes(wo.type)) {
   433→   433→        failureRateByType[t].failures++;
   434→   434→      }
   435→   435→    }
   436→   436→    const failureRateByTypeArray = Object.entries(failureRateByType)
   437→   437→      .map(([type, data]) => ({
   438→   438→        type,
   439→   439→        total: data.total,
   440→   440→        failures: data.failures,
   441→   441→        failureRate: data.total > 0 ? Math.round((data.failures / data.total) * 10000) / 100 : 0,
   442→   442→      }))
   443→   443→      .sort((a, b) => b.failureRate - a.failureRate);
   444→   444→
   445→   445→    // MTBF (Mean Time Between Failures) per asset
   446→   446→    const mtbfByAsset = Object.entries(assetWOs)
   447→   447→      .filter(([, data]) => data.failures >= 2)
   448→   448→      .map(([assetId, data]) => {
   449→   449→        // Calculate operating period from first to last WO
   450→   450→        const assetWOList = workOrders.filter(wo => wo.assetId === assetId);
   451→   451→        if (assetWOList.length < 2) return null;
   452→   452→        const firstDate = assetWOList[assetWOList.length - 1].createdAt;
   453→   453→        const lastDate = assetWOList[0].createdAt;
   454→   454→        const operatingDays = (lastDate.getTime() - firstDate.getTime()) / (1000 * 60 * 60 * 24);
   455→   455→        const mtbfDays = data.failures > 0 ? operatingDays / data.failures : 0;
   456→   456→        return {
   457→   457→          assetId,
   458→   458→          assetName: data.name,
   459→   459→          mtbfDays: Math.round(mtbfDays * 10) / 10,
   460→   460→          mtbfHours: Math.round(mtbfDays * 24 * 10) / 10,
   461→   461→          failureCount: data.failures,
   462→   462→          operatingDays: Math.round(operatingDays),
   463→   463→        };
   464→   464→      })
   465→   465→      .filter(Boolean)
   466→   466→      .sort((a, b) => (a as any).mtbfDays - (b as any).mtbfDays) as any[];
   467→   467→
   468→   468→    // Rework rate (repeat failures)
   469→   469→    const reworkWOs = workOrders.filter(wo => wo.repairCompletion && wo.repairCompletion.reworkCount > 0);
   470→   470→    const reworkRate = total > 0 ? Math.round((reworkWOs.length / total) * 10000) / 100 : 0;
   471→   471→
   472→   472→    // ========== 8. STOPPAGES ==========
   473→   473→    const totalStoppages = totalDowntimeEvents;
   474→   474→    const stoppagesByTrade = downtimeByTradeArray.map(d => ({
   475→   475→      trade: d.trade,
   476→   476→      count: d.events,
   477→   477→      totalHours: d.totalHours,
   478→   478→    }));
   479→   479→
   480→   480→    // Stoppages by impact level
   481→   481→    const stoppagesByImpact: Record<string, { count: number; minutes: number }> = {};
   482→   482→    for (const wo of workOrders) {
   483→   483→      for (const dt of (wo.workOrderDowntimes || [])) {
   484→   484→        const impact = dt.impactLevel || 'medium';
   485→   485→        if (!stoppagesByImpact[impact]) stoppagesByImpact[impact] = { count: 0, minutes: 0 };
   486→   486→        stoppagesByImpact[impact].count++;
   487→   487→        stoppagesByImpact[impact].minutes += dt.durationMinutes;
   488→   488→      }
   489→   489→    }
   490→   490→    const stoppagesByImpactArray = Object.entries(stoppagesByImpact)
   491→   491→      .map(([impact, data]) => ({
   492→   492→        impact,
   493→   493→        count: data.count,
   494→   494→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   495→   495→      }))
   496→   496→      .sort((a, b) => b.count - a.count);
   497→   497→
   498→   498→    // Stoppages by reason
   499→   499→    const stoppagesByReason: Record<string, number> = {};
   500→   500→    for (const wo of workOrders) {
   501→   501→      for (const dt of (wo.workOrderDowntimes || [])) {
   502→   502→        const reason = dt.reason || 'Unassigned';
   503→   503→        stoppagesByReason[reason] = (stoppagesByReason[reason] || 0) + 1;
   504→   504→      }
   505→   505→    }
   506→   506→    const stoppagesByReasonArray = Object.entries(stoppagesByReason)
   507→   507→      .map(([reason, count]) => ({ reason, count }))
   508→   508→      .sort((a, b) => b.count - a.count)
   509→   509→      .slice(0, 10);
   510→   510→
   511→   511→    // ========== 9. COST ANALYSIS ==========
   512→   512→    let grandTotalCost = 0;
   513→   513→    let grandLaborCost = 0;
   514→   514→    let grandPartsCost = 0;
   515→   515→    let grandContractorCost = 0;
   516→   516→
   517→   517→    for (const wo of workOrders) {
   518→   518→      grandTotalCost += wo.totalCost || 0;
   519→   519→      grandLaborCost += wo.laborCost || 0;
   520→   520→      grandPartsCost += wo.partsCost || 0;
   521→   521→      grandContractorCost += wo.contractorCost || 0;
   522→   522→    }
   523→   523→
   524→   524→    // Cost by trade
   525→   525→    const costByTrade: Record<string, { labor: number; parts: number; contractor: number; total: number }> = {};
   526→   526→    for (const wo of workOrders) {
   527→   527→      const t = wo.tradeActivity || 'Unassigned';
   528→   528→      if (!costByTrade[t]) costByTrade[t] = { labor: 0, parts: 0, contractor: 0, total: 0 };
   529→   529→      costByTrade[t].labor += wo.laborCost || 0;
   530→   530→      costByTrade[t].parts += wo.partsCost || 0;
   531→   531→      costByTrade[t].contractor += wo.contractorCost || 0;
   532→   532→      costByTrade[t].total += wo.totalCost || 0;
   533→   533→    }
   534→   534→    const costByTradeArray = Object.entries(costByTrade)
   535→   535→      .map(([trade, data]) => ({
   536→   536→        trade,
   537→   537→        laborCost: Math.round(data.labor * 100) / 100,
   538→   538→        partsCost: Math.round(data.parts * 100) / 100,
   539→   539→        contractorCost: Math.round(data.contractor * 100) / 100,
   540→   540→        totalCost: Math.round(data.total * 100) / 100,
   541→   541→      }))
   542→   542→      .sort((a, b) => b.totalCost - a.totalCost);
   543→   543→
   544→   544→    // Cost by month
   545→   545→    const costByMonth: Record<string, { labor: number; parts: number; contractor: number; total: number }> = {};
   546→   546→    for (const wo of workOrders) {
   547→   547→      const month = wo.createdAt.toISOString().slice(0, 7);
   548→   548→      if (!costByMonth[month]) costByMonth[month] = { labor: 0, parts: 0, contractor: 0, total: 0 };
   549→   549→      costByMonth[month].labor += wo.laborCost || 0;
   550→   550→      costByMonth[month].parts += wo.partsCost || 0;
   551→   551→      costByMonth[month].contractor += wo.contractorCost || 0;
   552→   552→      costByMonth[month].total += wo.totalCost || 0;
   553→   553→    }
   554→   554→    const costByMonthArray = Object.entries(costByMonth)
   555→   555→      .map(([month, data]) => ({
   556→   556→        month,
   557→   557→        laborCost: Math.round(data.labor * 100) / 100,
   558→   558→        partsCost: Math.round(data.parts * 100) / 100,
   559→   559→        contractorCost: Math.round(data.contractor * 100) / 100,
   560→   560→        totalCost: Math.round(data.total * 100) / 100,
   561→   561→      }))
   562→   562→      .sort((a, b) => a.month.localeCompare(b.month));
   563→   563→
   564→   564→    // ========== 10. WO TYPE DISTRIBUTION ==========
   565→   565→    const woByType: Record<string, number> = {};
   566→   566→    const woByStatus: Record<string, number> = {};
   567→   567→    const woByPriority: Record<string, number> = {};
   568→   568→    const woByTrade: Record<string, number> = {};
   569→   569→
   570→   570→    for (const wo of workOrders) {
   571→   571→      woByType[wo.type || 'other'] = (woByType[wo.type || 'other'] || 0) + 1;
   572→   572→      woByStatus[wo.status] = (woByStatus[wo.status] || 0) + 1;
   573→   573→      woByPriority[wo.priority || 'medium'] = (woByPriority[wo.priority || 'medium'] || 0) + 1;
   574→   574→      woByTrade[wo.tradeActivity || 'Unassigned'] = (woByTrade[wo.tradeActivity || 'Unassigned'] || 0) + 1;
   575→   575→    }
   576→   576→
   577→   577→    // ========== BUILD RESPONSE DATA ==========
   578→   578→    const jsonData = {
   579→   579→      success: true,
   580→   580→      data: {
   581→   581→        // 1. Summary
   582→   582→        summary: {
   583→   583→          ...summary,
   584→   584→          avgResponseTime,
   585→   585→          totalManHours: Math.round(grandTotalManHours * 100) / 100,
   586→   586→          totalMaterialCost: Math.round(totalMaterialCost * 100) / 100,
   587→   587→          totalMaterialQty: Math.round(totalMaterialQty * 100) / 100,
   588→   588→          totalDowntimeHours: Math.round((totalDowntimeMinutes / 60) * 100) / 100,
   589→   589→          totalDowntimeEvents,
   590→   590→          totalStoppages,
   591→   591→          totalProductionLoss: Math.round(totalProductionLoss * 100) / 100,
   592→   592→          reworkRate,
   593→   593→          reworkWOs: reworkWOs.length,
   594→   594→        },
   595→   595→
   596→   596→        // 2. Downtime
   597→   597→        downtime: {
   598→   598→          byTrade: downtimeByTradeArray,
   599→   599→          byMonth: downtimeByMonthArray,
   600→   600→          byCategory: downtimeByCategoryArray,
   601→   601→        },
   602→   602→
   603→   603→        // 3. Response Time
   604→   604→        responseTime: {
   605→   605→          overall: { avgHours: avgResponseTime, sampleSize: responseTimes.length },
   606→   606→          byPriority: responseByPriorityArray,
   607→   607→          byTrade: responseByTradeArray,
   608→   608→        },
   609→   609→
   610→   610→        // 4. Breakdowns
   611→   611→        breakdowns: {
   612→   612→          total: breakdownWOs.length,
   613→   613→          byType: breakdownByTypeArray,
   614→   614→          byTrade: breakdownByTradeArray,
   615→   615→          byMonth: breakdownByMonthArray,
   616→   616→          byPriority: breakdownByPriorityArray,
   617→   617→        },
   618→   618→

   620→   620→        manHours: {
   621→   621→          grandTotal: Math.round(grandTotalManHours * 100) / 100,
   622→   622→          byTrade: manHoursByTradeArray,

   633→   633→          topItems: topMaterialsArray,

   635→   635→        },
   636→   636→

   638→   638→        failureRate: {
   639→   639→          byAsset: failureRateByAsset,
   640→   640→          byType: failureRateByTypeArray,
   641→   641→          mtbf: mtbfByAsset,
   642→   642→          reworkRate,
   643→   643→          reworkWOs: reworkWOs.length,
   644→   644→        },
   645→   645→
   646→   646→        // 8. Stoppages
   647→   647→        stoppages: {
   648→   648→          total: totalStoppages,
   649→   649→          byTrade: stoppagesByTrade,
   650→   650→          byImpact: stoppagesByImpactArray,
   651→   651→          byReason: stoppagesByReasonArray,
   652→   652→        },
   653→   653→
   654→   654→        // 9. Cost Analysis
   655→   655→        cost: {
   656→   656→          grandTotal: Math.round(grandTotalCost * 100) / 100,
   657→   657→          grandLabor: Math.round(grandLaborCost * 100) / 100,
   658→   658→          grandParts: Math.round(grandPartsCost * 100) / 100,
   659→   659→          grandContractor: Math.round(grandContractorCost * 100) / 100,

   661→   661→          byMonth: costByMonthArray,

   663→   663→
   664→   664→        // 10. Distributions

   667→   667→          byStatus: Object.entries(woByStatus).map(([status, count]) => ({ status, count })).sort((a, b) => b.count - a.count),
   668→   668→          byPriority: Object.entries(woByPriority).map(([priority, count]) => ({ priority, count })).sort((a, b) => b.count - a.count),
   669→   669→          byTrade: Object.entries(woByTrade).map(([trade, count]) => ({ trade, count })).sort((a, b) => b.count - a.count),
   670→   670→        },
   671→   671→      },
   672→   672→    };

   675→   675→    if (format === 'pdf') {

   681→   681→        filters: { from: from || 'All time', to: to || 'Present', ...(department && { department }), ...(trade && { trade }), ...(priority && { priority }) },
   682→   682→        sections: [

   733→   733→    return NextResponse.json(jsonData);
   734→   734→  } catch (error) {
   735→   735→    console.error('[WO Reports] Error:', error);
   736→   736→    return NextResponse.json({ success: false, error: 'Failed to generate reports' }, { status: 500 });
   737→   737→  }
