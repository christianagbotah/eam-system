   733→   733→// GLOBAL ERROR BOUNDARY — catches ALL errors including .map() crashes
   734→   734→// ============================================================================
   735→   735→
   736→   736→interface ErrorBoundaryState {
   737→   737→  hasError: boolean;
   738→   738→  error: Error | null;
   739→   739→  componentStack: string | null;
   740→   740→  isChunkError: boolean;
   741→   741→}
   742→   742→
   743→   743→  // Error message pattern for React #185
   744→   744→  const ERROR_185_PATTERN = /Cannot update a component (?:from within the render function|while rendering a different component)/;
   745→   745→
   746→   746→class GlobalErrorBoundary extends React.Component<
   747→   747→  { children: React.ReactNode },
   748→   748→  ErrorBoundaryState & { retryTimer: ReturnType<typeof setTimeout> | null; retryCount: number }
   749→   749→> {
   750→   750→  private _retryTimer: ReturnType<typeof setTimeout> | null = null;
   751→   751→  private _retryCount: number = 0;
   752→   752→
   753→   753→  constructor(props: { children: React.ReactNode }) {
   754→   754→    super(props);
   755→   755→    this.state = { hasError: false, error: null, componentStack: null, isChunkError: false };
   756→   756→  }
   757→   757→
   758→   758→  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
   759→   759→    const msg = error?.message || '';
   760→   760→    const isChunkError =
   761→   761→      msg.includes('ChunkLoadError') ||
   762→   762→      msg.includes('Failed to load chunk') ||
   763→   763→      msg.includes('Loading chunk') ||
   764→   764→      msg.includes('Loading CSS chunk');
   765→   765→
   766→   766→    // CRITICAL: For Error #185, do NOT show the error screen.
   767→   767→    // This error is caused by external store updates (Zustand set()) firing
   768→   768→    // during React's concurrent render phase — it's a timing issue, not a
   769→   769→    // real bug. Returning hasError: false lets React retry the render
   770→   770→    // automatically, which almost always succeeds.
   771→   771→    if (ERROR_185_PATTERN.test(msg)) {
   772→   772→      return { hasError: false, error, componentStack: null, isChunkError: false };
   773→   773→    }
   774→   774→
   775→   775→    // Catch ALL other errors
   776→   776→    return { hasError: true, error, componentStack: null, isChunkError };
   777→   777→  }
   778→   778→
   779→   779→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   780→   780→    // Log full error details to console and state
   781→   781→    console.error('=== EAM GLOBAL ERROR BOUNDARY ===');
   782→   782→    console.error('Error:', error.message);
   783→   783→    console.error('Stack:', error.stack);
   784→   784→    console.error('Component Stack:', info.componentStack);
   785→   785→    console.error('=== END ERROR ===');
   786→   786→    this.setState({ componentStack: info.componentStack });
   787→   787→
   788→   788→    // For Error #185, schedule a silent retry after a short delay.
   789→   789→    // This handles the case where the automatic retry (hasError: false from
   790→   790→    // getDerivedStateFromError) still fails because the Zustand store update
   791→   791→    // hasn't completed yet. We force a re-render after 50ms which almost
   792→   792→    // always succeeds.
   793→   793→    if (ERROR_185_PATTERN.test(error.message)) {
   794→   794→      this._retryCount++;
   795→   795→      console.warn(`[ErrorBoundary] React #185 caught, scheduling silent retry #${this._retryCount}`);
   796→   796→
   797→   797→      this._retryTimer = setTimeout(() => {
   798→   798→        this._retryTimer = null;
   799→   799→        // Force a re-render by toggling state
   800→   800→        this.setState((prev) => {
   801→   801→          if (prev.hasError) {
   802→   802→            return { hasError: false, error: null, componentStack: null, isChunkError: false };
   803→   803→          }
   804→   804→          return prev;
   805→   805→        });
   806→   806→      }, 50);
   807→   807→    }
   808→   808→  }
   809→   809→
   810→   810→  componentWillUnmount() {
   811→   811→    if (this._retryTimer) {
   812→   812→      clearTimeout(this._retryTimer);
   813→   813→      this._retryTimer = null;
   814→   814→    }
   815→   815→  }
   816→   816→
   817→   817→  handleRetry = () => {
   818→   818→    if (this._retryTimer) {
   819→   819→      clearTimeout(this._retryTimer);
   820→   820→      this._retryTimer = null;
   821→   821→    }
   822→   822→    // Always reload for chunk errors — the JS bundle is stale
   823→   823→    if (this.state.isChunkError) {
   824→   824→      window.location.reload();
   825→   825→      return;
   826→   826→    }
   827→   827→    // For runtime errors, try re-rendering
   828→   828→    this.setState({ hasError: false, error: null, componentStack: null, isChunkError: false });
   829→   829→  };
   830→   830→
   831→   831→  handleHardRefresh = () => {
   832→   832→    window.location.href = window.location.href.split('?')[0] + '?_t=' + Date.now();
   833→   833→  };
   834→   834→
   835→   835→  render() {
   836→   836→    if (this.state.hasError && this.state.error) {
   837→   837→      const isChunk = this.state.isChunkError;
   838→   838→      return (
   839→   839→        <div className="flex items-center justify-center min-h-[400px]">
   840→   840→          <div className="text-center space-y-4 p-6 max-w-lg">
   841→   841→            <div className={`h-12 w-12 mx-auto rounded-full flex items-center justify-center ${isChunk ? 'bg-amber-100 dark:bg-amber-950/30' : 'bg-red-100 dark:bg-red-950/30'}`}>
   842→   842→              <svg className={`h-6 w-6 ${isChunk ? 'text-amber-600' : 'text-red-600'}`} fill="none" viewBox="0 0 24 24" stroke="currentColor">
   843→   843→                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
   844→   844→              </svg>
   845→   845→            </div>
   846→   846→            <div>
   847→   847→              <h3 className="text-sm font-semibold text-foreground">{isChunk ? 'Module Loading Error' : 'Runtime Error'}</h3>
   848→   848→              <p className="text-xs text-red-600 font-mono mt-2 p-2 bg-red-50 dark:bg-red-950/20 rounded break-all text-left">
   849→   849→                {this.state.error.message}
   850→   850→              </p>
   851→   851→              {this.state.componentStack && (
   852→   852→                <pre className="text-[10px] text-muted-foreground mt-2 p-2 bg-muted rounded overflow-auto max-h-32 text-left whitespace-pre-wrap">
   853→   853→                  {this.state.componentStack}
   854→   854→                </pre>
   855→   855→              )}
   856→   856→            </div>
   857→   857→            <div className="flex gap-2 justify-center">
   858→   858→              <button
   859→   859→                onClick={this.handleRetry}
   860→   860→                className="inline-flex items-center gap-2 px-4 py-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium rounded-lg transition-colors"
   861→   861→              >
   862→   862→                <svg className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor">
   863→   863→                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15" />
   864→   864→                </svg>
   865→   865→                Retry
   866→   866→              </button>
   867→   867→              <button
   868→   868→                onClick={this.handleHardRefresh}
   869→   869→                className="inline-flex items-center gap-2 px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white text-sm font-medium rounded-lg transition-colors"
   870→   870→              >
   871→   871→                Hard Refresh
   872→   872→              </button>
   873→   873→            </div>
   874→   874→          </div>
   875→   875→        </div>
   876→   876→      );
   877→   877→    }
   878→   878→    return this.props.children;
   879→   879→  }
   880→   880→}
   881→   881→

   887→   887→  return (
   888→   888→    <GlobalErrorBoundary>
   889→   889→      <AppShell />
   890→   890→    </GlobalErrorBoundary>
   891→   891→  );
