   311→  // Compute the label of the group that should auto-expand (based on active child page).
   312→  // This is pure derived state — NO setState, NO useEffect, NO queueMicrotask.
   313→  // Fixes React Error #185 caused by queueMicrotask calling setOpenMenus during render.
   314→  const autoExpandLabel = useMemo<string | null>(() => {
   315→    for (const group of menuGroups) {
   316→      if (!group.children) continue;
   317→      const childMatch = group.children.some(c => {
   318→        if (c.page === currentPage) return true;

   320→        if (c.page === 'maintenance-work-orders' && currentPage === 'wo-detail') return true;
   321→        return false;
   322→      });

   324→    }
   325→    return null;
   326→  }, [currentPage, menuGroups]);
   327→

   336→      });

   340→  // Final open menus = manually opened ∪ auto-expanded (but NOT manually closed)

   344→    return Array.from(set);

   351→      (autoExpandLabel === label && !closedMenus.has(label));
   352→
   353→    setManualOpenMenus(prev => {
   354→      const next = new Set(prev);
   355→      if (isActuallyOpen) next.delete(label);
   356→      else next.add(label);
   357→      return next;
   358→    });
   359→    setClosedMenus(prev => {
   360→      const next = new Set(prev);
   361→      if (isActuallyOpen) next.add(label);
   362→      else next.delete(label);
   363→      return next;
   364→    });
   365→  }, [manualOpenMenus, autoExpandLabel, closedMenus]);
   366→

   368→  const isGroupActive = useCallback((group: NavGroup) => {

   372→      if (c.page === currentPage) return true;
   373→      if (c.page === 'maintenance-requests' && (currentPage === 'mr-detail' || currentPage === 'create-mr')) return true;
   374→      if (c.page === 'maintenance-work-orders' && currentPage === 'wo-detail') return true;
   375→      return false;
   376→    });
   377→  }, [currentPage]);
   378→
   379→  // Filter visible groups based on permissions and module activation

   384→      if (g.adminOnly && !isAdm) return false;
   385→
   386→      // Permission check (supports permOr for multiple alternative permissions)
   387→      if (!isAdm && permissions && permissions.length > 0) {

   389→          ? g.permOr.some(p => hasPermission(p))
   390→          : hasPermission(g.perm);
   391→        if (!permOk) return false;
   392→      }
   393→
   394→      // Module activation check — only apply once modules have loaded
   395→      if (enabledModules.size > 0) {
   396→        // Single module code check
   397→        if (g.moduleCode && g.moduleCode !== 'core' && !enabledModules.has(g.moduleCode.toLowerCase())) return false;

   399→        if (g.moduleCodes && g.moduleCodes.length > 0) {
   400→          if (!g.moduleCodes.some(code => enabledModules.has(code.toLowerCase()))) return false;
   401→        }
   402→      }
   403→
   404→      return true;
   405→    });
   406→  }, [menuGroups, hasPermission, permissions, isAdmin, enabledModules]);
   407→
   408→  // Get tooltip text for collapsed sidebar
   409→  const getGroupTooltip = (group: NavGroup) => {
   410→    if (!group.children) return group.label;
   411→    const activeChild = group.children.find(c => {
   412→      if (c.page === currentPage) return true;
   413→      if (c.page === 'maintenance-requests' && (currentPage === 'mr-detail' || currentPage === 'create-mr')) return true;
   414→      if (c.page === 'maintenance-work-orders' && currentPage === 'wo-detail') return true;
   415→      return false;
   416→    });
   417→    return activeChild ? `${group.label} → ${activeChild.label}` : group.label;
   418→  };

   425→          <Factory className="h-5 w-5 text-white" />

   430→            <p className="text-[10px] text-sidebar-foreground/40 uppercase tracking-widest font-medium">Enterprise EAM</p>
   431→          </div>
   432→        )}

   435→      {/* Navigation */}
   436→      <nav className="flex-1 overflow-y-auto overflow-x-hidden px-3 py-4">
   437→        <div className="space-y-0.5">
   438→          {visibleGroups.map(group => {

   442→            // Auto-collapse manually-opened menus whose children are no longer active
   443→            // (handled implicitly since manualOpenMenus only changes on user click)
   444→            const hasChildren = !!group.children;
   445→
   446→            // Standalone item (no children)
   447→            if (!hasChildren) {
   448→              return (
   449→                <TooltipProvider key={group.label} delayDuration={0}>
   450→                  <Tooltip>
   451→                    <TooltipTrigger asChild>
   452→                      <button
   453→                        onClick={() => navigate(group.page!)}
   454→                        className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm transition-all relative ${
   455→                          active
   456→                            ? 'bg-sidebar-accent text-sidebar-primary font-medium shadow-sm shadow-black/10'
   457→                            : 'text-sidebar-foreground/60 hover:bg-sidebar-accent/40 hover:text-sidebar-foreground/90'
   458→                        }`}
   459→                      >
   460→                        {active && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full bg-sidebar-primary" />}
   461→                        <Icon className={`h-[18px] w-[18px] shrink-0 ${active ? 'text-sidebar-primary' : ''}`} />
   462→                        {expanded && <span className="flex-1 text-left">{group.label}</span>}
   463→                      </button>
   464→                    </TooltipTrigger>
   465→                    {!expanded && <TooltipContent side="right">{group.label}</TooltipContent>}
   466→                  </Tooltip>
   467→                </TooltipProvider>
   468→              );
   469→            }
   470→
   471→            // Collapsed sidebar — show only icon with tooltip
   472→            if (!expanded) {
   473→              return (
   474→                <TooltipProvider key={group.label} delayDuration={0}>
   475→                  <Tooltip>
   476→                    <TooltipTrigger asChild>
   477→                      <button
   478→                        onClick={() => {
   479→                          const firstChild = group.children![0];
   480→                          navigate(firstChild.page);
   481→                        }}
   482→                        className={`w-full flex items-center justify-center px-3 py-2.5 rounded-lg text-sm transition-all relative ${
   483→                          active
   484→                            ? 'bg-sidebar-accent text-sidebar-primary font-medium shadow-sm shadow-black/10'
   485→                            : 'text-sidebar-foreground/60 hover:bg-sidebar-accent/40 hover:text-sidebar-foreground/90'
   486→                        }`}
   487→                      >
   488→                        {active && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full bg-sidebar-primary" />}
   489→                        <Icon className={`h-[18px] w-[18px] shrink-0 ${active ? 'text-sidebar-primary' : ''}`} />
   490→                      </button>
   491→                    </TooltipTrigger>
   492→                    <TooltipContent side="right">{getGroupTooltip(group)}</TooltipContent>
   493→                  </Tooltip>
   494→                </TooltipProvider>
   495→              );
   496→            }
   497→
   498→            // Expanded sidebar — show icon + label + chevron + children
   499→            return (
   500→              <div key={group.label}>
   501→                <button
   502→                  onClick={() => toggleMenu(group.label)}
   503→                  className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm transition-all relative ${
   504→                    active
   505→                      ? 'bg-sidebar-accent text-sidebar-primary font-medium shadow-sm shadow-black/10'
   506→                      : 'text-sidebar-foreground/60 hover:bg-sidebar-accent/40 hover:text-sidebar-foreground/90'
   507→                  }`}
   508→                >
   509→                  {active && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full bg-sidebar-primary" />}
   510→                  <Icon className={`h-[18px] w-[18px] shrink-0 ${active ? 'text-sidebar-primary' : ''}`} />
   511→                  <span className="flex-1 text-left">{group.label}</span>
   512→                  <ChevronRight className={`h-3.5 w-3.5 shrink-0 transition-transform duration-200 ${isOpen ? 'rotate-90' : ''}`} />
   513→                </button>
   514→                {isOpen && (
   515→                  <div className="ml-4 mt-0.5 space-y-0.5 border-l border-sidebar-border pl-3">
   516→                    {group.children!.map(child => {
   517→                      const childActive = child.page === currentPage ||
   518→                        (child.page === 'maintenance-requests' && (currentPage === 'mr-detail' || currentPage === 'create-mr')) ||
   519→                        (child.page === 'maintenance-work-orders' && currentPage === 'wo-detail');
   520→                      const ChildIcon = child.icon;
   521→                      return (
   522→                        <button
   523→                          key={child.page}
   524→                          onClick={() => navigate(child.page)}
   525→                          className={`w-full flex items-center gap-2.5 px-2.5 py-2 rounded-md text-[13px] transition-all relative ${
   526→                            childActive
   527→                              ? 'bg-sidebar-accent text-sidebar-primary font-medium'
   528→                              : 'text-sidebar-foreground/50 hover:bg-sidebar-accent/30 hover:text-sidebar-foreground/80'
   529→                          }`}
   530→                        >
   531→                          {childActive && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[2px] h-4 rounded-r-full bg-sidebar-primary" />}
   532→                          {ChildIcon && <ChildIcon className={`h-[15px] w-[15px] shrink-0 ${childActive ? 'text-sidebar-primary' : ''}`} />}
   533→                          <span className="flex-1 text-left">{child.label}</span>
   534→                        </button>
   535→                      );
   536→                    })}
   537→                  </div>
   538→                )}
   539→              </div>
   540→            );
   541→          })}
   542→        </div>
   543→      </nav>
   544→
   545→      {/* User — Clean profile section */}
   546→      <div className="border-t border-sidebar-border p-3">
   547→        <div className="flex items-center gap-3 px-2">
   548→          <div className="h-8 w-8 rounded-full bg-gradient-to-br from-emerald-500/80 to-emerald-700/80 flex items-center justify-center shrink-0 ring-1 ring-sidebar-border">
   549→            <span className="text-xs font-bold text-white">{user ? getInitials(user.fullName) : '?'}</span>
   550→          </div>
   551→          {expanded && (
   552→            <>
   553→              <div className="flex-1 min-w-0">
   554→                <div className="flex items-center gap-1.5 min-w-0">
   555→                  <p className="text-sm font-medium text-sidebar-foreground truncate">{user?.fullName}</p>
   556→                  {user?.roles?.[0] && (
   557→                    <span className="inline-flex items-center shrink-0 rounded-full bg-emerald-500/15 px-1.5 py-0.5 text-[10px] font-medium text-emerald-600 ring-1 ring-emerald-500/20">
   558→                      {user.roles[0].name}
   559→                    </span>
   560→                  )}
   561→                </div>
   562→                <p className="text-[10px] text-sidebar-foreground/40 truncate">{user?.email}</p>
   563→              </div>
   564→              <TooltipProvider delayDuration={0}>
   565→                <Tooltip>
   566→                  <TooltipTrigger asChild>
   567→                    <button
   568→                      className="h-8 w-8 flex items-center justify-center rounded-lg text-sidebar-foreground/40 hover:text-red-400 hover:bg-red-500/10 transition-all"
   569→                      onClick={logout}
   570→                    >
   571→                      <LogOut className="h-4 w-4" />
   572→                    </button>
   573→                  </TooltipTrigger>
   574→                  <TooltipContent side="top">Sign Out</TooltipContent>
   575→                </Tooltip>
   576→              </TooltipProvider>
   577→            </>
   578→          )}
   579→        </div>
   580→      </div>
   581→    </div>
   582→  );
   583→}
