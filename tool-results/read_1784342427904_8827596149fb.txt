   139→// ---------------------------------------------------------------------------
   140→// Helpers
   141→// ---------------------------------------------------------------------------
   142→
   143→function effColor(v: number): string {
   144→  if (v >= 97) return 'text-emerald-600 font-semibold';
   145→  if (v >= 90) return 'text-amber-600 font-semibold';
   146→  return 'text-red-600 font-semibold';
   147→}
   148→
   149→function effBg(v: number): string {
   150→  if (v >= 97) return 'bg-emerald-50';
   151→  if (v >= 90) return 'bg-amber-50';
   152→  return 'bg-red-50';
   153→}
   154→
   155→function dtColor(v: number): string {
   156→  if (v > 5) return 'text-red-600 font-semibold';
   157→  if (v >= 2) return 'text-amber-600 font-semibold';
   158→  return 'text-emerald-600 font-semibold';
   159→}
   160→
   161→function dtBg(v: number): string {
   162→  if (v > 5) return 'bg-red-50';
   163→  if (v >= 2) return 'bg-amber-50';
   164→  return 'bg-emerald-50';
   165→}
   166→
   167→function fmt(n: number, dec = 1): string {
   168→  if (n == null || Number.isNaN(n)) return '—';
   169→  return n.toFixed(dec);
   170→}
   171→
   172→function fmtInt(n: number): string {
   173→  if (n == null || Number.isNaN(n)) return '—';
   174→  return Math.round(n).toLocaleString();
   175→}
   176→
   177→function statusBadge(breakdowns: number): { label: string; variant: 'destructive' | 'outline' | 'secondary' } {
   178→  if (breakdowns > 5) return { label: 'Critical', variant: 'destructive' };
   179→  if (breakdowns >= 2) return { label: 'Warning', variant: 'outline' };
   180→  return { label: 'Normal', variant: 'secondary' };
   181→}

'text-emerald-600' : selectedKPI.weightedAvgEfficiency >= 90 ? 'text-amber-600' : 'text-red-600', targetTab: 'efficiency' as const },
   372→                repairsEnabled ? { label: 'Total Breakdowns', value: String(selectedKPI.totalBreakdowns), icon: AlertTriangle, color: selectedKPI.totalBreakdowns <= (data.targets.breakdownsWeekly ?? 2) ? 'text-emerald-600' : 'text-red-600', targetTab: 'breakdown' as const } : null,
   373→              ].filter(Boolean).map((item) => (
