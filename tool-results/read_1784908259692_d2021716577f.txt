    68→// ── Helpers ──────────────────────────────────────────────────────────────────

    70→function formatUptime(seconds: number): string {
    71→  const d = Math.floor(seconds / 86400);
    72→  const h = Math.floor((seconds % 86400) / 3600);
    73→  const m = Math.floor((seconds % 3600) / 60);
    74→  if (d > 0) return `${d}d ${h}h ${m}m`;
    75→  if (h > 0) return `${h}h ${m}m`;
    76→  return `${m}m`;
    77→}
    78→
    79→function formatBytes(mb: number): string {
    80→  if (mb >= 1024) return `${(mb / 1024).toFixed(1)} GB`;
    81→  return `${mb} MB`;
    82→}

   138→// ── Throughput Chart (canvas-based sparkline) ────────────────────────────────
   139→
   140→function ThroughputChart({ data }: { data: number[] }) {
   141→  const canvasRef = useRef<HTMLCanvasElement>(null);
   142→
   143→  useEffect(() => {
   144→    const canvas = canvasRef.current;
   145→    if (!canvas || data.length === 0) return;
   146→
   147→    const ctx = canvas.getContext('2d');
   148→    if (!ctx) return;
   149→
   150→    const dpr = window.devicePixelRatio || 1;
   151→    const rect = canvas.getBoundingClientRect();
   152→    canvas.width = rect.width * dpr;
   153→    canvas.height = rect.height * dpr;
   154→    ctx.scale(dpr, dpr);
   155→
   156→    const w = rect.width;
   157→    const h = rect.height;
   158→    const maxVal = Math.max(...data, 1);
   159→    const padding = 4;
   160→    const chartW = w - padding * 2;
   161→    const chartH = h - padding * 2;
   162→
   163→    ctx.clearRect(0, 0, w, h);
   164→
   165→    // Fill area
   166→    ctx.beginPath();
   167→    ctx.moveTo(padding, h - padding);
   168→    for (let i = 0; i < data.length; i++) {
   169→      const x = padding + (i / (data.length - 1)) * chartW;
   170→      const y = h - padding - (data[i] / maxVal) * chartH;
   171→      if (i === 0) ctx.lineTo(x, y);
   172→      else ctx.lineTo(x, y);
   173→    }
   174→    ctx.lineTo(padding + chartW, h - padding);
   175→    ctx.closePath();
   176→    const gradient = ctx.createLinearGradient(0, 0, 0, h);
   177→    gradient.addColorStop(0, 'rgba(16, 185, 129, 0.2)');
   178→    gradient.addColorStop(1, 'rgba(16, 185, 129, 0.0)');
   179→    ctx.fillStyle = gradient;
   180→    ctx.fill();
   181→
   182→    // Line
   183→    ctx.beginPath();
   184→    for (let i = 0; i < data.length; i++) {
   185→      const x = padding + (i / (data.length - 1)) * chartW;
   186→      const y = h - padding - (data[i] / maxVal) * chartH;
   187→      if (i === 0) ctx.moveTo(x, y);
   188→      else ctx.lineTo(x, y);
   189→    }
   190→    ctx.strokeStyle = '#10b981';
   191→    ctx.lineWidth = 2;
   192→    ctx.stroke();
   193→  }, [data]);
   194→
   195→  return (
   196→    <canvas
   197→      ref={canvasRef}
   198→      className="w-full h-full"
   199→      style={{ width: '100%', height: '100%' }}
   200→    />
   201→  );
   202→}

   204→// ── Error Breakdown Donut ───────────────────────────────────────────────────
   205→
   206→function ErrorDonut({ data }: { data: { label: string; value: number; color: string }[] }) {
   207→  const total = data.reduce((s, d) => s + d.value, 0);
   208→
   209→  if (total === 0) {
   210→    return (
   211→      <div className="flex items-center justify-center h-full text-sm text-muted-foreground">
   212→        No errors recorded
   213→      </div>
   214→    );
   215→  }
   216→
   217→  // Simple CSS conic-gradient donut
   218→  let gradientParts: string[] = [];
   219→  let currentAngle = 0;
   220→
   221→  for (const item of data) {
   222→    const percentage = (item.value / total) * 100;
   223→    gradientParts.push(`${item.color} ${currentAngle}deg ${currentAngle + percentage}deg`);
   224→    currentAngle += percentage;
   225→  }
   226→
   227→  return (
   228→    <div className="flex items-center gap-6 w-full">
   229→      <div
   230→        className="w-32 h-32 rounded-full shrink-0"
   231→        style={{
   232→          background: `conic-gradient(${gradientParts.join(', ')})`,
   233→          maskImage: 'radial-gradient(circle at center, transparent 60%, black 61%)',
   234→          WebkitMaskImage: 'radial-gradient(circle at center, transparent 60%, black 61%)',
   235→        }}
   236→      />
   237→      <div className="space-y-2 flex-1 min-w-0">
   238→        {data.filter(d => d.value > 0).map((item, i) => (
   239→          <div key={i} className="flex items-center gap-2 text-sm">
   240→            <div className="w-3 h-3 rounded-sm shrink-0" style={{ backgroundColor: item.color }} />
   241→            <span className="text-muted-foreground truncate">{item.label}</span>
   242→            <span className="font-semibold tabular-nums ml-auto">{item.value}</span>
   243→          </div>
   244→        ))}
   245→      </div>
   246→    </div>
   247→  );
   248→}

   258→  const [logSearch, setLogSearch] = useState('');

   260→  const refreshTimerRef = useRef<NodeJS.Timeout | null>(null);
   261→

   263→    try {
   264→      const [healthRes, logsRes, tracesRes] = await Promise.all([
   265→        api.get('/api/observability/health'),
   266→        api.get('/api/observability/logs?limit=100&level=' + (logFilter === 'all' ? 'info' : logFilter === 'warn' ? 'warn' : logFilter === 'error' ? 'error' : 'info')),
   267→        api.get('/api/observability/traces?limit=20'),
   268→      ]);
   269→
   270→      if (healthRes.success) setHealthData(healthRes.data);
   271→      if (logsRes.success) setLogs(logsRes.data?.entries || []);
   272→      if (tracesRes.success) setTraces(tracesRes.data?.traces || []);
   273→    } catch {
   274→      // Silently handle fetch errors
   275→    } finally {
   276→      setLoading(false);
   277→    }
   278→  }, [logFilter]);
   279→
   280→  // Initial fetch + polling every 10s
   281→  useEffect(() => {
   282→    fetchData();
   283→    refreshTimerRef.current = setInterval(fetchData, 10_000);
   284→    return () => {
   285→      if (refreshTimerRef.current) clearInterval(refreshTimerRef.current);
   286→    };
   287→  }, [fetchData]);
   288→
   289→  // Auto-scroll log stream
   290→  useEffect(() => {
   291→    if (logStreamRef.current) {
   292→      logStreamRef.current.scrollTop = logStreamRef.current.scrollHeight;
   293→    }
   294→  }, [logs]);
   295→
   296→  // ── Derived data ─────────────────────────────────────────────────────────
   297→
   298→  const metricCards: DashboardMetricCard[] = useMemo(() => {
   299→    if (!healthData) return [];

   301→    return [
   302→      {
   303→        label: 'Response Time',
   304→        value: `${healthData.responseTimeMs}ms`,
   305→        subtext: healthData.responseTimeMs < 200 ? 'Normal' : 'Slow',
   306→        icon: <Zap className="h-5 w-5" />,
   307→        color: healthData.responseTimeMs < 200 ? 'text-emerald-500' : 'text-amber-500',
   308→      },
   309→      {
   310→        label: 'Uptime',
   311→        value: formatUptime(sys.uptime),
   312→        subtext: `${Math.round(sys.uptime)}s total`,
   313→        icon: <Clock className="h-5 w-5" />,
   314→        color: 'text-emerald-500',
   315→      },
   316→      {
   317→        label: 'Heap Memory',
   318→        value: `${sys.memory.heapPercent}%`,
   319→        subtext: `${sys.memory.heapUsedMB} / ${sys.memory.heapTotalMB} MB`,
   320→        icon: <MemoryStick className="h-5 w-5" />,
   321→        color: sys.memory.heapPercent < 70 ? 'text-emerald-500' : sys.memory.heapPercent < 90 ? 'text-amber-500' : 'text-red-500',
   322→      },
   323→      {
   324→        label: 'RSS Memory',
   325→        value: formatBytes(sys.memory.rssMB),
   326→        icon: <Server className="h-5 w-5" />,
   327→        color: 'text-muted-foreground',
   328→      },
   329→    ];
   330→  }, [healthData]);
   331→
   332→  // Simulated throughput data (would come from metrics service in production)
   333→  const throughputData = useMemo(() => {
   334→    if (!healthData) return Array(60).fill(0);
   335→    // Generate pseudo-random based on current minute to look realistic
   336→    return Array.from({ length: 60 }, (_, i) => {
   337→      const base = 15 + Math.sin((i + Date.now() / 60_000) * 0.3) * 8;
   338→      return Math.max(0, Math.round(base + Math.random() * 6));
   339→    });
   340→  }, [healthData]);
   341→
   342→  const errorBreakdown = useMemo(() => {
   343→    const filtered = logFilter === 'all' ? logs : logs.filter(l => l.level === logFilter);
   344→    const errors = filtered.filter(l => l.level === 'error' || l.level === 'fatal');
   345→    const bySource: Record<string, number> = {};
   346→    for (const e of errors) {
   347→      const src = e.service || 'unknown';
   348→      bySource[src] = (bySource[src] || 0) + 1;
   349→    }
   350→    const colors = ['#ef4444', '#f97316', '#eab308', '#a855f7', '#06b6d4', '#10b981'];
   351→    return Object.entries(bySource)
   352→      .sort((a, b) => b[1] - a[1])
   353→      .slice(0, 6)
   354→      .map(([label, value], i) => ({ label, value, color: colors[i % colors.length] }));
   355→  }, [logs, logFilter]);
   356→
   357→  const filteredLogs = useMemo(() => {
   358→    let result = logFilter === 'all' ? logs : logs.filter(l => l.level === logFilter);
   359→    if (logSearch) {
   360→      const q = logSearch.toLowerCase();
   361→      result = result.filter(l => l.message.toLowerCase().includes(q) || (l.service && l.service.toLowerCase().includes(q)));
   362→    }
   363→    return result;
   364→  }, [logs, logFilter, logSearch]);

   369→      name,
   370→      status: check.status,
   371→      details: check.details,

   382→          {Array.from({ length: 4 }).map((_, i) => (
   383→            <Skeleton key={i} className="h-28 rounded-xl" />
   384→          ))}
   385→        </div>
   386→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   387→          <Skeleton className="h-64 rounded-xl" />
   388→          <Skeleton className="h-64 rounded-xl" />
   389→        </div>
   390→      </div>
   391→    );
   392→  }
   393→
   394→  // ── Render ────────────────────────────────────────────────────────────────

   409→        >
   410→          <RefreshCw className="h-3.5 w-3.5" />
   411→          Refresh
   412→        </Button>
   413→      </div>
   414→
   415→      {/* ── 1. System Health Banner ───────────────────────────────────────── */}

   417→        <div className="flex flex-col sm:flex-row sm:items-center gap-3">
   418→          <div className="flex items-center gap-3">

   518→                <div
   519→                  key={svc.name}
   520→                  className="flex items-start gap-2.5 p-3 rounded-lg border bg-muted/30"
   521→                >
   522→                  {statusIcon(svc.status)}
   523→                  <div className="min-w-0 flex-1">
   524→                    <p className="text-sm font-medium capitalize">{svc.name.replace(/_/g, ' ')}</p>
   525→                    <p className="text-xs text-muted-foreground mt-0.5 truncate">{svc.details}</p>
   526→                  </div>
   527→                  <Badge
   528→                    variant="outline"
   529→                    className={`text-[10px] shrink-0 ${statusColor(svc.status)}`}
   530→                  >
   531→                    {svc.status}
   532→                  </Badge>
   533→                </div>
   534→              ))}
   535→            </div>
   536→          </CardContent>
   537→        </Card>
   538→
   539→        {/* Resource Usage */}
   540→        <Card className="gap-4 p-4">
   541→          <CardHeader className="p-0 gap-1">
   542→            <div className="flex items-center justify-between">
   543→              <div>
   544→                <CardTitle className="text-sm">Resource Usage</CardTitle>
   545→                <CardDescription>Memory and CPU utilization</CardDescription>
   546→              </div>
   547→              <Cpu className="h-4 w-4 text-muted-foreground" />
   548→            </div>
   549→          </CardHeader>
   550→          <CardContent className="p-0 space-y-5">
   551→            {/* Heap Memory */}
   552→            <div className="space-y-2">
   553→              <div className="flex items-center justify-between text-sm">
   554→                <span className="font-medium flex items-center gap-1.5">
   555→                  <MemoryStick className="h-3.5 w-3.5 text-muted-foreground" />
   556→                  Heap Memory
   557→                </span>
   558→                <span className="tabular-nums text-muted-foreground">
   559→                  {healthData?.system.memory.heapUsedMB || 0} / {healthData?.system.memory.heapTotalMB || 0} MB
   560→                </span>
   561→              </div>
   562→              <Progress
   563→                value={healthData?.system.memory.heapPercent || 0}
   564→                className={`h-2.5 ${((healthData?.system.memory.heapPercent || 0) > 85) ? '[&>div]:bg-red-500' : ((healthData?.system.memory.heapPercent || 0) > 70) ? '[&>div]:bg-amber-500' : '[&>div]:bg-emerald-500'}`}
   565→              />
   566→              <p className={`text-xs font-medium ${statusColor(healthData?.status || 'healthy')}`}>
   567→                {healthData?.system.memory.heapPercent || 0}% utilized
   568→              </p>
   569→            </div>
   570→
   571→            {/* RSS Memory */}
   572→            <div className="space-y-2">
   573→              <div className="flex items-center justify-between text-sm">
   574→                <span className="font-medium flex items-center gap-1.5">
   575→                  <Server className="h-3.5 w-3.5 text-muted-foreground" />
   576→                  RSS Memory
   577→                </span>
   578→                <span className="tabular-nums text-muted-foreground">
   579→                  {formatBytes(healthData?.system.memory.rssMB || 0)}
   580→                </span>
   581→              </div>
   582→              <Progress
   583→                value={Math.min(((healthData?.system.memory.rssMB || 0) / 512) * 100, 100)}
   584→                className="h-2.5 [&>div]:bg-blue-500"
   585→              />
   586→              <p className="text-xs text-muted-foreground">of 512 MB typical limit</p>
   587→            </div>
   588→
   589→            {/* External Memory */}
   590→            <div className="space-y-2">
   591→              <div className="flex items-center justify-between text-sm">
   592→                <span className="font-medium flex items-center gap-1.5">
   593→                  <Database className="h-3.5 w-3.5 text-muted-foreground" />
   594→                  External (Buffers)
   595→                </span>
   596→                <span className="tabular-nums text-muted-foreground">
   597→                  {formatBytes(healthData?.system.memory.externalMB || 0)}
   598→                </span>
   599→              </div>
   600→              <Progress
   601→                value={Math.min(((healthData?.system.memory.externalMB || 0) / 128) * 100, 100)}
   602→                className="h-2.5 [&>div]:bg-purple-500"
   603→              />
   604→            </div>
   605→
   606→            {/* Connections */}
   607→            <div className="flex items-center justify-between p-3 rounded-lg border bg-muted/30">
   608→              <div className="flex items-center gap-2">
   609→                <Radio className="h-4 w-4 text-muted-foreground" />
   610→                <span className="text-sm font-medium">Active Connections</span>
   611→              </div>
   612→              <span className="text-sm font-bold tabular-nums">
   613→                {healthData?.checks.connections?.details?.match(/Active:\s*(\d+)/)?.[1] || 'N/A'}
   614→              </span>
   615→            </div>
   616→          </CardContent>
   617→        </Card>
   618→      </div>
   619→
   620→      {/* ── 7 & 8. Traces + Logs ────────────────────────────────────────── */}
   621→      <Tabs defaultValue="traces" className="w-full">
   622→        <TabsList className="grid w-full grid-cols-2">
   623→          <TabsTrigger value="traces" className="gap-1.5">
   624→            <Activity className="h-3.5 w-3.5" />
   625→            Recent Traces
   626→          </TabsTrigger>
   627→          <TabsTrigger value="logs" className="gap-1.5">
   628→            <FileText className="h-3.5 w-3.5" />
   629→            Log Stream
   630→          </TabsTrigger>
   631→        </TabsList>
   632→
   633→        <TabsContent value="traces">
   634→          <Card className="gap-0 overflow-hidden">
   635→            <CardContent className="p-0">
   636→              {traces.length === 0 ? (
   637→                <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   638→                  <Activity className="h-8 w-8 mb-2 opacity-40" />
   639→                  <p className="text-sm">No traces recorded yet</p>
   640→                  <p className="text-xs mt-1">Traces will appear as API requests are made</p>
   641→                </div>
   642→              ) : (
   643→                <div className="overflow-x-auto">
   644→                  <table className="w-full text-sm">
   645→                    <thead>
   646→                      <tr className="border-b bg-muted/50">
   647→                        <th className="text-left p-3 font-medium text-muted-foreground">Trace ID</th>
   648→                        <th className="text-left p-3 font-medium text-muted-foreground">Name</th>
   649→                        <th className="text-left p-3 font-medium text-muted-foreground">Service</th>
   650→                        <th className="text-left p-3 font-medium text-muted-foreground">Status</th>
   651→                        <th className="text-right p-3 font-medium text-muted-foreground">Duration</th>
   652→                        <th className="text-left p-3 font-medium text-muted-foreground">Time</th>
   653→                      </tr>
   654→                    </thead>
   655→                    <tbody>
   656→                      {traces.map(span => (
   657→                        <tr key={span.spanId} className="border-b last:border-0 hover:bg-muted/30 transition-colors">
   658→                          <td className="p-3 font-mono text-xs text-muted-foreground max-w-[100px] truncate" title={span.traceId}>
   659→                            {span.traceId.slice(0, 12)}…
   660→                          </td>
   661→                          <td className="p-3 font-medium max-w-[200px] truncate">{span.name}</td>
   662→                          <td className="p-3 text-muted-foreground">{span.serviceName}</td>
   663→                          <td className="p-3">
   664→                            <Badge
   665→                              variant="outline"
   666→                              className={`text-[10px] ${span.status === 'ok' ? 'border-emerald-200 text-emerald-700 dark:border-emerald-800 dark:text-emerald-400' : 'border-red-200 text-red-700 dark:border-red-800 dark:text-red-400'}`}
   667→                            >
   668→                              {span.status}
   669→                            </Badge>
   670→                          </td>
   671→                          <td className="p-3 text-right tabular-nums">
   672→                            {span.durationMs !== undefined ? `${span.durationMs.toFixed(1)}ms` : '—'}
   673→                          </td>
   674→                          <td className="p-3 text-xs text-muted-foreground">
   675→                            {new Date(span.startTime).toLocaleTimeString()}
   676→                          </td>
   677→                        </tr>
   678→                      ))}
   679→                    </tbody>
   680→                  </table>
   681→                </div>
   682→              )}
   683→            </CardContent>
   684→          </Card>
   685→        </TabsContent>
   686→
   687→        <TabsContent value="logs">
   688→          <Card className="gap-0 overflow-hidden">
   689→            {/* Log Controls */}
   690→            <div className="flex flex-col sm:flex-row items-start sm:items-center gap-2 p-3 border-b bg-muted/30">
   691→              <div className="flex gap-1.5">
   692→                {(['all', 'error', 'warn', 'info'] as const).map(level => (
   693→                  <Button
   694→                    key={level}
   695→                    variant={logFilter === level ? 'default' : 'outline'}
   696→                    size="sm"
   697→                    className="text-xs h-7 px-2.5 capitalize"
   698→                    onClick={() => setLogFilter(level)}
   699→                  >
   700→                    {level}
   701→                  </Button>
   702→                ))}
   703→              </div>
   704→              <div className="relative flex-1 w-full sm:w-auto sm:ml-auto">
   705→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   706→                <input
   707→                  type="text"
   708→                  placeholder="Search logs..."
   709→                  value={logSearch}
   710→                  onChange={e => setLogSearch(e.target.value)}
   711→                  className="w-full h-7 pl-8 pr-3 text-xs rounded-md border bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   712→                />
   713→              </div>
   714→            </div>
   715→
   716→            {/* Log Stream */}
   717→            <div
   718→              ref={logStreamRef}
   719→              className="max-h-96 overflow-y-auto font-mono text-xs"
   720→            >
   721→              {filteredLogs.length === 0 ? (
   722→                <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   723→                  <FileText className="h-8 w-8 mb-2 opacity-40" />
   724→                  <p className="text-sm">No log entries found</p>
   725→                  <p className="text-xs mt-1">Adjust filters or wait for new entries</p>
   726→                </div>
   727→              ) : (
   728→                filteredLogs.map(entry => (
   729→                  <div
   730→                    key={entry.id}
   731→                    className="flex items-start gap-2 px-3 py-2 border-b last:border-0 hover:bg-muted/30 transition-colors"
   732→                  >
   733→                    <span className={`inline-flex items-center px-1.5 py-0.5 rounded text-[10px] font-bold uppercase shrink-0 mt-0.5 ${levelColor(entry.level)}`}>
   734→                      {entry.level}
   735→                    </span>
   736→                    <span className="text-muted-foreground shrink-0 tabular-nums">
   737→                      {new Date(entry.timestamp).toLocaleTimeString()}
   738→                    </span>
   739→                    <span className="text-muted-foreground shrink-0">
   740→                      [{entry.service}]
   741→                    </span>
   742→                    <span className="text-foreground break-all min-w-0">
   743→                      {entry.message}
   744→                    </span>
   745→                    {entry.error && (
   746→                      <span className="text-red-500 shrink-0 truncate max-w-[120px]" title={entry.error.message}>
   747→                        {entry.error.message}
   748→                      </span>
   749→                    )}
   750→                  </div>
   751→                ))
   752→              )}
   753→            </div>
   754→          </Card>
   755→        </TabsContent>
   756→      </Tabs>
   757→    </div>
   758→  );
   759→}
