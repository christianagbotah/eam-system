   740→// ============================================================================
   741→// GLOBAL ERROR BOUNDARY — catches ALL errors including .map() crashes
   742→// ============================================================================
   743→
   744→interface ErrorBoundaryState {
   745→  hasError: boolean;
   746→  error: Error | null;
   747→  componentStack: string | null;
   748→  isChunkError: boolean;
   749→}
   750→
   751→  // Error message pattern for React #185
   752→  const ERROR_185_PATTERN = /Cannot update a component (?:from within the render function|while rendering a different component)/;
   753→
   754→class GlobalErrorBoundary extends React.Component<
   755→  { children: React.ReactNode },
   756→  ErrorBoundaryState & { retryTimer: ReturnType<typeof setTimeout> | null; retryCount: number }
   757→> {
   758→  private _retryTimer: ReturnType<typeof setTimeout> | null = null;
   759→  private _retryCount: number = 0;
   760→
   761→  constructor(props: { children: React.ReactNode }) {
   762→    super(props);
   763→    this.state = { hasError: false, error: null, componentStack: null, isChunkError: false };
   764→  }
   765→
   766→  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
   767→    const msg = error?.message || '';
   768→    const isChunkError =
   769→      msg.includes('ChunkLoadError') ||
   770→      msg.includes('Failed to load chunk') ||
   771→      msg.includes('Loading chunk') ||
   772→      msg.includes('Loading CSS chunk');
   773→
   774→    // CRITICAL: For Error #185, do NOT show the error screen.
   775→    // This error is caused by external store updates (Zustand set()) firing
   776→    // during React's concurrent render phase — it's a timing issue, not a
   777→    // real bug. Returning hasError: false lets React retry the render
   778→    // automatically, which almost always succeeds.
   779→    if (ERROR_185_PATTERN.test(msg)) {
   780→      return { hasError: false, error, componentStack: null, isChunkError: false };
   781→    }
   782→
   783→    // Catch ALL other errors
   784→    return { hasError: true, error, componentStack: null, isChunkError };
   785→  }
   786→
   787→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   788→    // Log full error details to console and state
   789→    console.error('=== EAM GLOBAL ERROR BOUNDARY ===');
   790→    console.error('Error:', error.message);
   791→    console.error('Stack:', error.stack);
   792→    console.error('Component Stack:', info.componentStack);
   793→    console.error('=== END ERROR ===');
   794→    this.setState({ componentStack: info.componentStack });
   795→
   796→    // For Error #185, schedule a silent retry after a short delay.
   797→    // This handles the case where the automatic retry (hasError: false from
   798→    // getDerivedStateFromError) still fails because the Zustand store update
   799→    // hasn't completed yet. We force a re-render after 50ms which almost
   800→    // always succeeds.
   801→    if (ERROR_185_PATTERN.test(error.message)) {
   802→      this._retryCount++;
   803→      console.warn(`[ErrorBoundary] React #185 caught, scheduling silent retry #${this._retryCount}`);
   804→
   805→      this._retryTimer = setTimeout(() => {
   806→        this._retryTimer = null;
   807→        // Force a re-render by toggling state
   808→        this.setState((prev) => {
   809→          if (prev.hasError) {
   810→            return { hasError: false, error: null, componentStack: null, isChunkError: false };
   811→          }
   812→          return prev;
   813→        });
   814→      }, 50);
   815→    }
   816→  }
   817→
   818→  componentWillUnmount() {
   819→    if (this._retryTimer) {
   820→      clearTimeout(this._retryTimer);
   821→      this._retryTimer = null;
   822→    }
   823→  }
   824→
   825→  handleRetry = () => {
   826→    if (this._retryTimer) {
   827→      clearTimeout(this._retryTimer);
   828→      this._retryTimer = null;
   829→    }
   830→    // Always reload for chunk errors — the JS bundle is stale
   831→    if (this.state.isChunkError) {
   832→      window.location.reload();
   833→      return;
   834→    }
   835→    // For runtime errors, try re-rendering
   836→    this.setState({ hasError: false, error: null, componentStack: null, isChunkError: false });
   837→  };
   838→
   839→  handleHardRefresh = () => {
   840→    window.location.href = window.location.href.split('?')[0] + '?_t=' + Date.now();
   841→  };
   842→
   843→  render() {
   844→    if (this.state.hasError && this.state.error) {
   845→      const isChunk = this.state.isChunkError;
   846→      return (
   847→        <div className="flex items-center justify-center min-h-[400px]">
   848→          <div className="text-center space-y-4 p-6 max-w-lg">
   849→            <div className={`h-12 w-12 mx-auto rounded-full flex items-center justify-center ${isChunk ? 'bg-amber-100 dark:bg-amber-950/30' : 'bg-red-100 dark:bg-red-950/30'}`}>
   850→              <svg className={`h-6 w-6 ${isChunk ? 'text-amber-600' : 'text-red-600'}`} fill="none" viewBox="0 0 24 24" stroke="currentColor">
   851→                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
   852→              </svg>
   853→            </div>
   854→            <div>
   855→              <h3 className="text-sm font-semibold text-foreground">{isChunk ? 'Module Loading Error' : 'Runtime Error'}</h3>
   856→              <p className="text-xs text-red-600 font-mono mt-2 p-2 bg-red-50 dark:bg-red-950/20 rounded break-all text-left">
   857→                {this.state.error.message}
   858→              </p>
   859→              {this.state.componentStack && (
   860→                <pre className="text-[10px] text-muted-foreground mt-2 p-2 bg-muted rounded overflow-auto max-h-32 text-left whitespace-pre-wrap">
   861→                  {this.state.componentStack}
   862→                </pre>
   863→              )}
   864→            </div>
   865→            <div className="flex gap-2 justify-center">
   866→              <button
   867→                onClick={this.handleRetry}
   868→                className="inline-flex items-center gap-2 px-4 py-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium rounded-lg transition-colors"
   869→              >
   870→                <svg className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor">
   871→                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15" />
   872→                </svg>
   873→                Retry
   874→              </button>
   875→              <button
   876→                onClick={this.handleHardRefresh}
   877→                className="inline-flex items-center gap-2 px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white text-sm font-medium rounded-lg transition-colors"
   878→              >
   879→                Hard Refresh
   880→              </button>
   881→            </div>
   882→          </div>
   883→        </div>
   884→      );
   885→    }
   886→    return this.props.children;
   887→  }
   888→}
