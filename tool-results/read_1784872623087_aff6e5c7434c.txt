     1→import { NextRequest, NextResponse } from 'next/server';
     2→import { db } from '@/lib/db';
     3→import { getSession, hasPermission, isAdmin, hasAnyPermission } from '@/lib/auth';
     4→import { getPlantScope, getPlantFilterWhere } from '@/lib/plant-scope';
     5→import { generateReportPDF } from '@/lib/generate-report-pdf';
     6→
     7→// GET /api/work-orders/reports
     8→// Comprehensive WO reports: downtime, response time, breakdowns, man hours, materials, failure rate, stoppages
     9→export async function GET(request: NextRequest) {
    10→  try {
    11→    const session = getSession(request);
    12→    if (!session) {
    13→      return NextResponse.json({ success: false, error: 'Not authenticated' }, { status: 401 });
    14→    }
    15→    // Permission gate: require reports.view, work_orders.dashboard, or admin
    16→    if (!hasAnyPermission(session, ['reports.view', 'reports.export', 'work_orders.dashboard', 'analytics.view']) && !isAdmin(session)) {
    17→      return NextResponse.json({ success: false, error: 'Insufficient permissions: reports.view required' }, { status: 403 });
    18→    }
    19→
    20→    const { searchParams } = new URL(request.url);
    21→    const from = searchParams.get('from');
    22→    const to = searchParams.get('to');
    23→    const department = searchParams.get('department');
    24→    const trade = searchParams.get('trade');
    25→    const priority = searchParams.get('priority');
    26→    const format = searchParams.get('format');
    27→    const moduleFilter = searchParams.get('moduleFilter') || 'all';
    28→
    29→    const plantScope = await getPlantScope(request, session);
    30→    const plantFilter = getPlantFilterWhere(plantScope);
    31→
    32→    // Date filter
    33→    const dateFilter: Record<string, unknown> = {};
    34→    if (from) dateFilter.gte = new Date(from + 'T00:00:00');
    35→    if (to) dateFilter.lte = new Date(to + 'T23:59:59');
    36→
    37→    const woWhere: Record<string, unknown> = {
    38→      ...plantFilter,
    39→      status: { notIn: ['draft', 'cancelled'] },
    40→    };
    41→    if (Object.keys(dateFilter).length > 0) woWhere.createdAt = dateFilter;
    42→    if (department) woWhere.departmentId = department;
    43→    if (trade) woWhere.tradeActivity = trade;
    44→    if (priority) woWhere.priority = priority;
    45→    if (moduleFilter === 'repairs') woWhere.type = { in: ['corrective', 'emergency'] };
    46→    if (moduleFilter === 'pm') woWhere.type = { in: ['preventive'] };
    47→
    48→    // Fetch all relevant WOs with relations
    49→    const workOrders = await db.workOrder.findMany({
    50→      where: Object.keys(woWhere).length > 0 ? woWhere : undefined,
    51→      include: {
    52→        assignee: { select: { id: true, fullName: true, department: true } },
    53→        teamLeader: { select: { id: true, fullName: true } },
    54→        materials: true,
    55→        timeLogs: true,
    56→        workOrderDowntimes: true,
    57→        repairCompletion: true,
    58→        statusHistory: { orderBy: { createdAt: 'asc' } },
    59→        taskExecutions: true,
    60→        failureRecords: true,
    61→      },
    62→      orderBy: { createdAt: 'desc' },
    63→    });
    64→
    65→    // Fetch enriched asset data for all referenced assets
    66→    const assetIds = [...new Set(workOrders.map(wo => wo.assetId).filter((id): id is string => !!id))];
    67→    const assets = assetIds.length > 0 ? await db.asset.findMany({
    68→      where: { id: { in: assetIds } },
    69→      include: { category: { select: { name: true } } },
    70→    }) : [];
    71→    const assetMap = new Map(assets.map(a => [a.id, a]));
    72→
    73→    // Fetch enriched inventory item data for all referenced materials
    74→    const itemIds = [...new Set(workOrders.flatMap(wo =>
    75→      (wo.materials || []).map(m => m.itemId).filter((id): id is string => !!id)
    76→    ))];
    77→    const inventoryItems = itemIds.length > 0 ? await db.inventoryItem.findMany({
    78→      where: { id: { in: itemIds } },
    79→      select: { id: true, itemCode: true, name: true, unitOfMeasure: true, supplier: true, supplierPartNumber: true, binLocation: true, shelfLocation: true, specification: true, currentStock: true },
    80→    }) : [];
    81→    const itemMap = new Map(inventoryItems.map(i => [i.id, i]));
    82→
    83→    // Helper: get enriched asset details from a work order
    84→    function getAssetDetails(wo: { assetId?: string | null; assetName?: string | null }) {
    85→      const asset = wo.assetId ? assetMap.get(wo.assetId) : null;
    86→      return {
    87→        assetId: wo.assetId || null,
    88→        assetName: wo.assetName || asset?.name || 'Unassigned',
    89→        assetTag: asset?.assetTag || null,
    90→        manufacturer: asset?.manufacturer || null,
    91→        model: asset?.model || null,
    92→        serialNumber: asset?.serialNumber || null,
    93→        category: asset?.category?.name || null,
    94→        criticality: asset?.criticality || null,
    95→        condition: asset?.condition || null,
    96→        location: asset?.location || null,
    97→        building: asset?.building || null,
    98→        floor: asset?.floor || null,
    99→        area: asset?.area || null,
   100→        purchaseCost: asset?.purchaseCost || null,
   101→        currentValue: asset?.currentValue || null,
   102→      };
   103→    }
   104→
   105→    // Build itemName -> first itemId mapping for material enrichment
   106→    const itemNameToItemId: Record<string, string> = {};
   107→    for (const wo of workOrders) {
   108→      for (const mat of (wo.materials || [])) {
   109→        if (mat.itemId && mat.itemName && !itemNameToItemId[mat.itemName]) {
   110→          itemNameToItemId[mat.itemName] = mat.itemId;
   111→        }
   112→      }
   113→    }
   114→
   115→    // Helper: get enriched inventory item details
   116→    function getItemDetails(itemName: string) {
   117→      const itemId = itemNameToItemId[itemName];
   118→      const item = itemId ? itemMap.get(itemId) : null;
   119→      return {
   120→        itemCode: item?.itemCode || null,
   121→        unitOfMeasure: item?.unitOfMeasure || null,
   122→        supplier: item?.supplier || null,
   123→        supplierPartNumber: item?.supplierPartNumber || null,
   124→        binLocation: item?.binLocation || null,
   125→        shelfLocation: item?.shelfLocation || null,
   126→        specification: item?.specification || null,
   127→        currentStock: item?.currentStock || null,
   128→      };
   129→    }
   130→
   131→    const total = workOrders.length;
   132→    const closedWOs = workOrders.filter(wo => wo.status === 'closed');
   133→    const completedWOs = workOrders.filter(wo => ['completed', 'verified', 'closed'].includes(wo.status));
   134→    const breakdownWOs = workOrders.filter(wo => ['emergency', 'corrective'].includes(wo.type));
   135→
   136→    // ========== 1. SUMMARY KPIs ==========
   137→    const summary = {
   138→      totalWOs: total,
   139→      completedWOs: completedWOs.length,
   140→      closedWOs: closedWOs.length,
   141→      completionRate: total > 0 ? Math.round((completedWOs.length / total) * 10000) / 100 : 0,
   142→      breakdownCount: breakdownWOs.length,
   143→      breakdownRate: total > 0 ? Math.round((breakdownWOs.length / total) * 10000) / 100 : 0,
   144→    };
   145→
   146→    // ========== 2. DOWNTIME (per trade) ==========
   147→    const downtimeByTrade: Record<string, { totalMinutes: number; events: number; productionLoss: number }> = {};
   148→    for (const wo of workOrders) {
   149→      const dtEvents = wo.workOrderDowntimes || [];
   150→      const trade = wo.tradeActivity || 'Unassigned';
   151→      for (const dt of dtEvents) {
   152→        if (!downtimeByTrade[trade]) downtimeByTrade[trade] = { totalMinutes: 0, events: 0, productionLoss: 0 };
   153→        downtimeByTrade[trade].totalMinutes += dt.durationMinutes;
   154→        downtimeByTrade[trade].events++;
   155→        downtimeByTrade[trade].productionLoss += dt.productionLoss || 0;
   156→      }
   157→    }
   158→    const totalDowntimeMinutes = Object.values(downtimeByTrade).reduce((s, d) => s + d.totalMinutes, 0);
   159→    const totalDowntimeEvents = Object.values(downtimeByTrade).reduce((s, d) => s + d.events, 0);
   160→    const totalProductionLoss = Object.values(downtimeByTrade).reduce((s, d) => s + d.productionLoss, 0);
   161→
   162→    const downtimeByTradeArray = Object.entries(downtimeByTrade)
   163→      .map(([tradeName, data]) => ({
   164→        trade: tradeName,
   165→        totalHours: Math.round((data.totalMinutes / 60) * 100) / 100,
   166→        totalMinutes: data.totalMinutes,
   167→        events: data.events,
   168→        productionLoss: data.productionLoss,
   169→      }))
   170→      .sort((a, b) => b.totalHours - a.totalHours);
   171→
   172→    // Downtime by month
   173→    const downtimeByMonth: Record<string, { minutes: number; events: number }> = {};
   174→    for (const wo of workOrders) {
   175→      for (const dt of (wo.workOrderDowntimes || [])) {
   176→        const month = dt.downtimeStart?.toISOString().slice(0, 7) || 'unknown';
   177→        if (!downtimeByMonth[month]) downtimeByMonth[month] = { minutes: 0, events: 0 };
   178→        downtimeByMonth[month].minutes += dt.durationMinutes;
   179→        downtimeByMonth[month].events++;
   180→      }
   181→    }
   182→    const downtimeByMonthArray = Object.entries(downtimeByMonth)
   183→      .map(([month, data]) => ({
   184→        month,
   185→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   186→        events: data.events,
   187→      }))
   188→      .sort((a, b) => a.month.localeCompare(b.month));
   189→
   190→    // Downtime by category
   191→    const downtimeByCategory: Record<string, { minutes: number; events: number }> = {};
   192→    for (const wo of workOrders) {
   193→      for (const dt of (wo.workOrderDowntimes || [])) {
   194→        const cat = dt.category || 'unplanned';
   195→        if (!downtimeByCategory[cat]) downtimeByCategory[cat] = { minutes: 0, events: 0 };
   196→        downtimeByCategory[cat].minutes += dt.durationMinutes;
   197→        downtimeByCategory[cat].events++;
   198→      }
   199→    }
   200→    const downtimeByCategoryArray = Object.entries(downtimeByCategory)
   201→      .map(([category, data]) => ({
   202→        category,
   203→        totalHours: Math.round((data.minutes / 60) * 100) / 100,
   204→        events: data.events,
   205→      }))
   206→      .sort((a, b) => b.totalHours - a.totalHours);
   207→
   208→    // ========== 3. RESPONSE TIME ==========
   209→    // Response time = time from WO creation to first status that indicates work started (in_progress)
   210→    const responseTimes: { hours: number; priority: string; type: string; trade: string; woId: string }[] = [];
   211→    for (const wo of workOrders) {
   212→      const history = wo.statusHistory || [];
   213→      if (history.length === 0) continue;
   214→
   215→      // Find the transition to 'in_progress' (or 'assigned' as fallback)
   216→      const startEntry = history.find(h => h.toStatus === 'in_progress')
   217→        || history.find(h => h.toStatus === 'assigned');
   218→
   219→      if (startEntry && wo.createdAt) {
   220→        const hours = (startEntry.createdAt.getTime() - wo.createdAt.getTime()) / (1000 * 60 * 60);
   221→        if (hours >= 0 && hours < 10000) { // sanity check
   222→          responseTimes.push({
   223→            hours: Math.round(hours * 100) / 100,
   224→            priority: wo.priority,
   225→            type: wo.type,
   226→            trade: wo.tradeActivity || 'Unassigned',
   227→            woId: wo.id,
   228→          });
   229→        }
   230→      }
   231→    }
   232→
   233→    const avgResponseTime = responseTimes.length > 0
   234→      ? Math.round(responseTimes.reduce((s, r) => s + r.hours, 0) / responseTimes.length * 100) / 100
   235→      : 0;
   236→
   237→    // Response time by priority
   238→    const responseByPriority: Record<string, number[]> = {};
   239→    for (const r of responseTimes) {
   240→      if (!responseByPriority[r.priority]) responseByPriority[r.priority] = [];
   241→      responseByPriority[r.priority].push(r.hours);
   242→    }
   243→    const responseByPriorityArray = Object.entries(responseByPriority)
   244→      .map(([p, hours]) => ({
   245→        priority: p,
   246→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   247→        minHours: Math.round(Math.min(...hours) * 100) / 100,
   248→        maxHours: Math.round(Math.max(...hours) * 100) / 100,
   249→        count: hours.length,
   250→      }))
   251→      .sort((a, b) => a.avgHours - b.avgHours);
   252→
   253→    // Response time by trade
   254→    const responseByTrade: Record<string, number[]> = {};
   255→    for (const r of responseTimes) {
   256→      if (!responseByTrade[r.trade]) responseByTrade[r.trade] = [];
   257→      responseByTrade[r.trade].push(r.hours);
   258→    }
   259→    const responseByTradeArray = Object.entries(responseByTrade)
   260→      .map(([t, hours]) => ({
   261→        trade: t,
   262→        avgHours: Math.round(hours.reduce((s, h) => s + h, 0) / hours.length * 100) / 100,
   263→        count: hours.length,
   264→      }))
   265→      .sort((a, b) => a.avgHours - b.avgHours);
   266→
   267→    // ========== 4. BREAKDOWNS ==========
   268→    // Breakdown by type
   269→    const breakdownByType: Record<string, number> = {};
   270→    for (const wo of breakdownWOs) {
   271→      const t = wo.type || 'unknown';
   272→      breakdownByType[t] = (breakdownByType[t] || 0) + 1;
   273→    }
   274→    const breakdownByTypeArray = Object.entries(breakdownByType)
   275→      .map(([type, count]) => ({ type, count }))
   276→      .sort((a, b) => b.count - a.count);
   277→
   278→    // Breakdown by trade
   279→    const breakdownByTrade: Record<string, number> = {};
   280→    for (const wo of breakdownWOs) {
   281→      const t = wo.tradeActivity || 'Unassigned';
   282→      breakdownByTrade[t] = (breakdownByTrade[t] || 0) + 1;
   283→    }
   284→    const breakdownByTradeArray = Object.entries(breakdownByTrade)
   285→      .map(([trade, count]) => ({ trade, count }))
   286→      .sort((a, b) => b.count - a.count);
   287→
   288→    // Breakdown by month
   289→    const breakdownByMonth: Record<string, number> = {};
   290→    for (const wo of breakdownWOs) {
   291→      const month = wo.createdAt.toISOString().slice(0, 7);
   292→      breakdownByMonth[month] = (breakdownByMonth[month] || 0) + 1;
   293→    }
   294→    const breakdownByMonthArray = Object.entries(breakdownByMonth)
   295→      .map(([month, count]) => ({ month, count }))
   296→      .sort((a, b) => a.month.localeCompare(b.month));
   297→
   298→    // Breakdown by priority
   299→    const breakdownByPriority: Record<string, number> = {};
   300→    for (const wo of breakdownWOs) {
   301→      const p = wo.priority || 'medium';
   302→      breakdownByPriority[p] = (breakdownByPriority[p] || 0) + 1;
   303→    }
   304→    const breakdownByPriorityArray = Object.entries(breakdownByPriority)
   305→      .map(([priority, count]) => ({ priority, count }))
   306→      .sort((a, b) => b.count - a.count);
   307→
   308→    // ========== 5. MAN HOURS ==========
   309→    // From timeLogs and repairCompletion
   310→    const manHoursByTrade: Record<string, { totalHours: number; woCount: number }> = {};
   311→    const manHoursByTechnician: Record<string, { name: string; totalHours: number; woCount: number }> = {};
   312→    let grandTotalManHours = 0;
   313→
   314→    for (const wo of workOrders) {
   315→      const trade = wo.tradeActivity || 'Unassigned';
   316→
   317→      // Get hours from timeLogs (primary), actualHours (fallback), repairCompletion (override)
   318→      let woHours = 0;
   319→      for (const tl of (wo.timeLogs || [])) {
   320→        if (tl.duration && tl.duration > 0) {
   321→          woHours += tl.duration;
   322→        }
   323→      }
   324→      // Fallback to WO's actualHours when no time logs exist
   325→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   326→        woHours = wo.actualHours;
   327→      }
   328→      // Override with repairCompletion if higher (supervisor-verified hours)
   329→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   330→        woHours = wo.repairCompletion.totalLaborHours;
   331→      }
   332→
   333→      grandTotalManHours += woHours;
   334→
   335→      // By trade
   336→      if (!manHoursByTrade[trade]) manHoursByTrade[trade] = { totalHours: 0, woCount: 0 };
   337→      manHoursByTrade[trade].totalHours += woHours;
   338→      manHoursByTrade[trade].woCount++;
   339→
   340→      // By technician
   341→      const techId = wo.assignedTo || 'unassigned';
   342→      const techName = wo.assignee?.fullName || 'Unassigned';
   343→      if (!manHoursByTechnician[techId]) {
   344→        manHoursByTechnician[techId] = { name: techName, totalHours: 0, woCount: 0 };
   345→      }
   346→      manHoursByTechnician[techId].totalHours += woHours;
   347→      manHoursByTechnician[techId].woCount++;
   348→    }
   349→
   350→    const manHoursByTradeArray = Object.entries(manHoursByTrade)
   351→      .map(([trade, data]) => ({
   352→        trade,
   353→        totalHours: Math.round(data.totalHours * 100) / 100,
   354→        woCount: data.woCount,
   355→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   356→      }))
   357→      .sort((a, b) => b.totalHours - a.totalHours);
   358→
   359→    const manHoursByTechnicianArray = Object.values(manHoursByTechnician)
   360→      .map(data => ({
   361→        ...data,
   362→        totalHours: Math.round(data.totalHours * 100) / 100,
   363→        avgHoursPerWO: data.woCount > 0 ? Math.round((data.totalHours / data.woCount) * 100) / 100 : 0,
   364→      }))
   365→      .sort((a, b) => b.totalHours - a.totalHours);
   366→
   367→    // Man hours by month
   368→    const manHoursByMonth: Record<string, number> = {};
   369→    for (const wo of workOrders) {
   370→      const month = wo.createdAt.toISOString().slice(0, 7);
   371→      let woHours = 0;
   372→      for (const tl of (wo.timeLogs || [])) {
   373→        if (tl.duration && tl.duration > 0) woHours += tl.duration;
   374→      }
   375→      if (woHours === 0 && wo.actualHours && wo.actualHours > 0) {
   376→        woHours = wo.actualHours;
   377→      }
   378→      if (wo.repairCompletion && wo.repairCompletion.totalLaborHours > woHours) {
   379→        woHours = wo.repairCompletion.totalLaborHours;
   380→      }
   381→      manHoursByMonth[month] = (manHoursByMonth[month] || 0) + woHours;
   382→    }
   383→    const manHoursByMonthArray = Object.entries(manHoursByMonth)
   384→      .map(([month, totalHours]) => ({
   385→        month,
   386→        totalHours: Math.round(totalHours * 100) / 100,
   387→      }))
   388→      .sort((a, b) => a.month.localeCompare(b.month));
   389→
   390→    // Man hours by activity type
   391→    const manHoursByActivity: Record<string, number> = {};
   392→    for (const wo of workOrders) {
   393→      for (const tl of (wo.timeLogs || [])) {
   394→        const act = tl.activityType || 'maintenance';
   395→        manHoursByActivity[act] = (manHoursByActivity[act] || 0) + (tl.duration || 0);
   396→      }
   397→    }
   398→    const manHoursByActivityArray = Object.entries(manHoursByActivity)
   399→      .map(([activity, totalHours]) => ({
   400→        activity,
   401→        totalHours: Math.round(totalHours * 100) / 100,
   402→      }))
   403→      .sort((a, b) => b.totalHours - a.totalHours);
   404→
   405→    // ========== 6. MATERIAL QUANTITY AND COST ==========
   406→    let totalMaterialCost = 0;
   407→    let totalMaterialQty = 0;
   408→    const materialsByCategory: Record<string, { qty: number; cost: number; woCount: number }> = {};
   409→    const topMaterials: Record<string, { name: string; qty: number; cost: number; woCount: Set<string> }> = {};
   410→
   411→    for (const wo of workOrders) {
   412→      for (const mat of (wo.materials || [])) {
   413→        const qty = mat.quantity || 0;
   414→        const cost = mat.totalCost || (mat.unitCost && mat.quantity ? mat.unitCost * mat.quantity : 0);
   415→        totalMaterialQty += qty;
   416→        totalMaterialCost += cost;
   417→
   418→        // By WO type
   419→        const woType = wo.type || 'other';
   420→        if (!materialsByCategory[woType]) materialsByCategory[woType] = { qty: 0, cost: 0, woCount: 0 };
   421→        materialsByCategory[woType].qty += qty;
   422→        materialsByCategory[woType].cost += cost;
   423→        materialsByCategory[woType].woCount++;
   424→
   425→        // Top items
   426→        const itemName = mat.itemName || mat.itemId || 'Unknown Item';
   427→        if (!topMaterials[itemName]) topMaterials[itemName] = { name: itemName, qty: 0, cost: 0, woCount: new Set() };
   428→        topMaterials[itemName].qty += qty;
   429→        topMaterials[itemName].cost += cost;
   430→        topMaterials[itemName].woCount.add(wo.id);
   431→      }
   432→    }
   433→
   434→    const materialsByCategoryArray = Object.entries(materialsByCategory)
   435→      .map(([type, data]) => ({
   436→        type,
   437→        totalQty: Math.round(data.qty * 100) / 100,
   438→        totalCost: Math.round(data.cost * 100) / 100,
   439→        woCount: data.woCount,
   440→      }))
   441→      .sort((a, b) => b.totalCost - a.totalCost);
   442→
   443→    const topMaterialsArray = Object.values(topMaterials)
   444→      .map(data => ({
   445→        name: data.name,
   446→        ...getItemDetails(data.name),
   447→        totalQty: Math.round(data.qty * 100) / 100,
   448→        totalCost: Math.round(data.cost * 100) / 100,
   449→        woCount: data.woCount.size,
   450→      }))
   451→      .sort((a, b) => b.totalCost - a.totalCost)
   452→      .slice(0, 20);
   453→
   454→    // Material cost by month
   455→    const materialCostByMonth: Record<string, number> = {};
   456→    for (const wo of workOrders) {
   457→      const month = wo.createdAt.toISOString().slice(0, 7);
   458→      let woMatCost = 0;
   459→      for (const mat of (wo.materials || [])) {
   460→        woMatCost += mat.totalCost || (mat.unitCost && mat.quantity ? mat.unitCost * mat.quantity : 0);
   461→      }
   462→      materialCostByMonth[month] = (materialCostByMonth[month] || 0) + woMatCost;
   463→    }
   464→    const materialCostByMonthArray = Object.entries(materialCostByMonth)
   465→      .map(([month, totalCost]) => ({
   466→        month,
   467→        totalCost: Math.round(totalCost * 100) / 100,
   468→      }))
   469→      .sort((a, b) => a.month.localeCompare(b.month));
   470→
   471→    // ========== 7. FAILURE RATE ==========
   472→    // Failure rate per asset: (failure WOs / total WOs for that asset) * 100
   473→    const assetWOs: Record<string, { name: string; total: number; failures: number }> = {};
