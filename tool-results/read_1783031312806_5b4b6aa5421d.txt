    46→  const [startDate, setStartDate] = useState(() => {
    47→    const d = new Date();
    48→    d.setMonth(d.getMonth() - 1);

    57→// Client-side date range filter helper
    58→function filterByDateRange<T extends { createdAt?: string | null }>(items: T[], startDate: string, endDate: string): T[] {
    59→  const start = new Date(startDate + 'T00:00:00').getTime();
    60→  const end = new Date(endDate + 'T23:59:59').getTime();
    61→  return items.filter(item => {
    62→    if (!item.createdAt) return false;
    63→    const t = new Date(item.createdAt).getTime();
    64→    return t >= start && t <= end;
    65→  });
    66→}

    68→// Reusable CSV export helper
    69→const exportCSV = (filename: string, headers: string[], rows: string[][]) => {
    70→  const csv = [headers.join(','), ...rows.map(r => r.map(c => `"${(c ?? '').toString().replace(/"/g, '""')}"`).join(','))].join('\n');
    71→  const blob = new Blob([csv], { type: 'text/csv' });
    72→  const url = URL.createObjectURL(blob);
    73→  const a = document.createElement('a');
    74→  a.href = url; a.download = `${filename}.csv`; a.click();
    75→  URL.revokeObjectURL(url);
    76→  toast.success(`Exported ${filename}.csv`);
    77→};
