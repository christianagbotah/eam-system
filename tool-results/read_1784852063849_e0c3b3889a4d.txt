     1→'use client';
     2→
     3→import React, { useEffect, useState, useCallback, useMemo } from 'react';
     4→import { toast } from 'sonner';
     5→import { useAuthStore } from '@/stores/authStore';
     6→import { api } from '@/lib/api';
     7→import type { InventoryItem } from '@/types';
     8→
     9→import { Button } from '@/components/ui/button';
    10→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';
    11→import { Input } from '@/components/ui/input';

    14→import { Textarea } from '@/components/ui/textarea';
    15→import { Badge } from '@/components/ui/badge';
    16→import { Separator } from '@/components/ui/separator';
    17→import { ScrollArea } from '@/components/ui/scroll-area';
    18→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    19→import { Switch } from '@/components/ui/switch';
    20→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    21→import {
    22→  Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle, DialogTrigger,
    23→} from '@/components/ui/dialog'
    24→import { ResponsiveDialog } from '@/components/shared/ResponsiveDialog';;
    25→import {
    26→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    27→} from '@/components/ui/table';
    28→import {
    29→  DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuSeparator, DropdownMenuTrigger,
    30→} from '@/components/ui/dropdown-menu';
    31→import { Accordion, AccordionItem, AccordionTrigger, AccordionContent } from '@/components/ui/accordion';
    32→import {
    33→  Package, Plus, Search, Pencil, Trash2, MoreHorizontal, AlertTriangle,
    34→  TrendingUp, ArrowUpDown, History, X, ShoppingCart, Building, FolderOpen,
    35→  ChevronRight,
    36→  MapPin, ArrowRightLeft, FileText, Truck, Download, RefreshCw, Eye,
    37→  CheckCircle2, Check, ClipboardCheck, ClipboardList, Clock, Filter, DollarSign, Box, Star,
    38→  XCircle, Loader2,
    39→} from 'lucide-react';
    40→import { EmptyState, StatusBadge, PriorityBadge, getInitials, formatDate, formatDateTime, timeAgo, LoadingSkeleton, formatCurrency } from '@/components/shared/helpers';
    41→import { AsyncSearchableSelect, SearchableSelect } from '@/components/ui/searchable-select';
    42→
    43→export function InventoryPage() {
    44→  const { hasPermission, isAdmin } = useAuthStore();
    45→  const [items, setItems] = useState<any[]>([]);
    46→  const [loading, setLoading] = useState(true);
    47→  const [dialogOpen, setDialogOpen] = useState(false);
    48→  const [saving, setSaving] = useState(false);
    49→  const [editId, setEditId] = useState<string | null>(null);
    50→  const [searchText, setSearchText] = useState('');
    51→  const [categoryFilter, setCategoryFilter] = useState('all');
    52→  const [lowStockOnly, setLowStockOnly] = useState(false);
    53→  // Delete
    54→  const [deleteId, setDeleteId] = useState<string | null>(null);
    55→  // Stock movement
    56→  const [movementOpen, setMovementOpen] = useState(false);
    57→  const [movType, setMovType] = useState('in');
    58→  const [movItemId, setMovItemId] = useState('');
    59→  const [movQty, setMovQty] = useState('');
    60→  const [movReason, setMovReason] = useState('');
    61→  const [movLoading, setMovLoading] = useState(false);
    62→  // Stock movements list
    63→  const [selectedMovItemId, setSelectedMovItemId] = useState<string | null>(null);
    64→  const [stockMovements, setStockMovements] = useState<any[]>([]);
    65→  // KPI data from API
    66→  const [kpi, setKpi] = useState<{
    67→    total: number;
    68→    byCategory: Record<string, number>;
    69→    lowStock: number;
    70→    totalValue: number;
    71→    avgUnitCost: number;
    72→    pendingRequests: number;
    73→    pendingAdjustments: number;
    74→    movements: { thisMonth: number; lastMonth: number; trendPercent: number };
    75→    stockSummary: { totalQuantity: number; totalMinLevels: number; totalMaxCapacity: number };
    76→  } | null>(null);
    77→
    78→  const emptyForm = { itemCode: '', name: '', description: '', category: 'spare_part', unitOfMeasure: 'each', currentStock: '', minStockLevel: '', maxStockLevel: '', reorderQuantity: '', unitCost: '', supplier: '', location: '', binLocation: '', plantId: '' };
    79→  const [form, setForm] = useState(emptyForm);
    80→
    81→  const loadData = useCallback(() => {
    82→    api.get<any[]>('/api/inventory').then((iRes) => {
    83→      if (iRes.success && iRes.data) setItems(Array.isArray(iRes.data) ? iRes.data : []);
    84→      setLoading(false);
    85→    });
    86→  }, []);
    87→
    88→  useEffect(() => { loadData(); }, [loadData]);
    89→
    90→  useEffect(() => {
    91→    api.get('/api/inventory/kpi').then((res: any) => {
    92→      if (res.success && res.data) setKpi(res.data);
    93→    });
    94→  }, []);
    95→
    96→  const filteredItems = useMemo(() => {
    97→    return items.filter(i => {
    98→      if (searchText) { const q = searchText.toLowerCase(); if (!i.name.toLowerCase().includes(q) && !i.itemCode.toLowerCase().includes(q)) return false; }
    99→      if (categoryFilter !== 'all' && i.category !== categoryFilter) return false;
   100→      if (lowStockOnly && i.currentStock > i.minStockLevel) return false;
   101→      return true;
   102→    });
   103→  }, [items, searchText, categoryFilter, lowStockOnly]);
   104→
   105→  const stats = useMemo(() => {
   106→    const lowStock = items.filter(i => i.currentStock <= i.minStockLevel).length;
   107→    const totalValue = items.reduce((sum, i) => sum + (i.currentStock * (i.unitCost || 0)), 0);
   108→    return { total: items.length, lowStock, totalValue };
   109→  }, [items]);
   110→
   111→  const openCreate = () => { setEditId(null); setForm(emptyForm); setDialogOpen(true); };
   112→  const openEdit = (i: any) => { setEditId(i.id); setForm({ itemCode: i.itemCode, name: i.name, description: i.description || '', category: i.category || 'spare_part', unitOfMeasure: i.unitOfMeasure || 'each', currentStock: i.currentStock?.toString() || '', minStockLevel: i.minStockLevel?.toString() || '', maxStockLevel: i.maxStockLevel?.toString() || '', reorderQuantity: i.reorderQuantity?.toString() || '', unitCost: i.unitCost?.toString() || '', supplier: i.supplier || '', location: i.location || '', binLocation: i.binLocation || '', plantId: i.plantId || '' }); setDialogOpen(true); };
   113→
   114→  const handleSave = async () => {
   115→    if (!form.name || !form.itemCode) { toast.error('Name and item code required'); return; }
   116→    setSaving(true);
   117→    const payload: any = { ...form, currentStock: form.currentStock ? parseFloat(form.currentStock) : 0, minStockLevel: form.minStockLevel ? parseFloat(form.minStockLevel) : 0, maxStockLevel: form.maxStockLevel ? parseFloat(form.maxStockLevel) : undefined, reorderQuantity: form.reorderQuantity ? parseFloat(form.reorderQuantity) : undefined, unitCost: form.unitCost ? parseFloat(form.unitCost) : undefined };
   118→    const res = editId ? await api.put(`/api/inventory/${editId}`, payload) : await api.post('/api/inventory', payload);
   119→    if (res.success) { toast.success(editId ? 'Item updated' : 'Item created'); setDialogOpen(false); loadData(); } else { toast.error(res.error || 'Failed'); }
   120→    setSaving(false);
   121→  };
   122→
   123→  const handleDelete = async () => {
   124→    if (!deleteId) return;
   125→    const res = await api.delete(`/api/inventory/${deleteId}`);
   126→    if (res.success) { toast.success('Item deleted'); setDeleteId(null); loadData(); }
   127→    else { toast.error(res.error || 'Failed to delete'); }
   128→  };
   129→
   130→  const handleStockMovement = async () => {
   131→    if (!movItemId || !movQty) { toast.error('Item and quantity required'); return; }
   132→    setMovLoading(true);
   133→    const res = await api.post(`/api/inventory/${movItemId}/stock-movements`, { type: movType, quantity: parseFloat(movQty), reason: movReason || undefined });
   134→    if (res.success) { toast.success('Stock movement recorded'); setMovementOpen(false); setMovQty(''); setMovReason(''); loadData(); }
   135→    else { toast.error(res.error || 'Failed'); }
   136→    setMovLoading(false);
   137→  };
   138→
   139→  const loadMovements = useCallback((itemId: string) => {
   140→    setSelectedMovItemId(itemId);
   141→    api.get<any[]>(`/api/inventory/${itemId}/stock-movements`).then(res => {
   142→      if (res.success && res.data) setStockMovements(Array.isArray(res.data) ? res.data : []);
   143→    });
   144→  }, []);
   145→
   146→  if (loading) return <LoadingSkeleton />;
   147→
   148→  return (
   149→    <div className="page-content">
   150→      <div className="flex items-center justify-between flex-wrap gap-3">
   151→        <div>
   152→          <h1 className="text-2xl font-bold tracking-tight">Inventory</h1>
   153→          <p className="text-muted-foreground text-sm mt-1">Manage spare parts, consumables, and supplies</p>
   154→        </div>
   155→        <div className="flex gap-2">
   156→          {(hasPermission('inventory.manage') || isAdmin()) && <Button variant="outline" onClick={() => { setMovItemId(''); setMovQty(''); setMovReason(''); setMovType('in'); setMovementOpen(true); }} className="gap-1.5"><ArrowUpDown className="h-4 w-4" />Stock Movement</Button>}
   157→          {(hasPermission('inventory.create') || isAdmin()) && <Button onClick={openCreate} className="bg-emerald-600 hover:bg-emerald-700 text-white"><Plus className="h-4 w-4 mr-1.5" />Add Item</Button>}
   158→        </div>
   159→      </div>
   160→
   161→      {/* KPI Cards */}
   162→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-6 gap-4">
   163→        <Card className="border-0 shadow-sm">
   164→          <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   165→            <CardTitle className="text-sm font-medium text-muted-foreground">Total Items</CardTitle>
   166→            <Package className="h-4 w-4 text-muted-foreground" />
   167→          </CardHeader>
   168→          <CardContent>
   169→            <div className="text-2xl font-bold">{kpi?.total ?? stats.total}</div>
   170→            <p className="text-xs text-muted-foreground mt-1">
   171→              {kpi?.byCategory ? `${Object.keys(kpi.byCategory).length} categories` : 'across all categories'}
   172→            </p>
   173→          </CardContent>
   174→        </Card>
   175→        <Card className="border-0 shadow-sm">
   176→          <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   177→            <CardTitle className="text-sm font-medium text-muted-foreground">Stock Value</CardTitle>
   178→            <DollarSign className="h-4 w-4 text-muted-foreground" />
   179→          </CardHeader>
   180→          <CardContent>
   181→            <div className="text-2xl font-bold">
   182→              {formatCurrency(kpi?.totalValue ?? stats.totalValue)}
   183→            </div>
   184→            <p className="text-xs text-muted-foreground mt-1">
   185→              {kpi?.avgUnitCost != null ? `Avg ${formatCurrency(kpi.avgUnitCost)}/unit` : 'total inventory value'}
   186→            </p>
   187→          </CardContent>
   188→        </Card>
   189→        <Card className={"border-0 shadow-sm" + ((kpi?.lowStock ?? stats.lowStock) > 0 ? ' border-l-4 border-l-amber-400' : '')}>
   190→          <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   191→            <CardTitle className="text-sm font-medium text-muted-foreground">Low Stock</CardTitle>
   192→            <AlertTriangle className={`h-4 w-4 ${(kpi?.lowStock ?? stats.lowStock) > 0 ? 'text-amber-500' : 'text-muted-foreground'}`} />
   193→          </CardHeader>
   194→          <CardContent>
   195→            <div className={`text-2xl font-bold ${(kpi?.lowStock ?? stats.lowStock) > 0 ? 'text-amber-600' : ''}`}>{kpi?.lowStock ?? stats.lowStock}</div>
   196→            <p className="text-xs text-muted-foreground mt-1">
   197→              {(kpi?.lowStock ?? stats.lowStock) > 0 ? 'items below min level' : 'all items in stock'}
   198→            </p>
   199→          </CardContent>
   200→        </Card>
   201→        <Card className="border-0 shadow-sm">
   202→          <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   203→            <CardTitle className="text-sm font-medium text-muted-foreground">Pending Requests</CardTitle>
   204→            <ClipboardList className="h-4 w-4 text-muted-foreground" />
   205→          </CardHeader>
   206→          <CardContent>
   207→            <div className="text-2xl font-bold">{kpi?.pendingRequests ?? 0}</div>
   208→            <p className="text-xs text-muted-foreground mt-1">
   209→              {kpi?.pendingRequests != null ? 'awaiting approval' : 'pending requests'}
   210→            </p>
   211→          </CardContent>
   212→        </Card>
   213→        <Card className="border-0 shadow-sm">
   214→          <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   215→            <CardTitle className="text-sm font-medium text-muted-foreground">Movements This Month</CardTitle>
   216→            <TrendingUp className="h-4 w-4 text-muted-foreground" />
   217→          </CardHeader>
   218→          <CardContent>
   219→            <div className="text-2xl font-bold">{kpi?.movements?.thisMonth ?? 0}</div>
   220→            <p className="text-xs text-muted-foreground mt-1">
   221→              {kpi?.movements?.trendPercent != null
   222→                ? (kpi.movements.trendPercent >= 0 ? '+' : '') + kpi.movements.trendPercent.toFixed(1) + '% vs last month'
   223→                : 'stock movements'}
   224→            </p>
   225→          </CardContent>
   226→        </Card>
   227→        <Card className="border-0 shadow-sm">
   228→          <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   229→            <CardTitle className="text-sm font-medium text-muted-foreground">Pending Adjustments</CardTitle>
   230→            <ClipboardCheck className="h-4 w-4 text-muted-foreground" />
   231→          </CardHeader>
   232→          <CardContent>
   233→            <div className="text-2xl font-bold">{kpi?.pendingAdjustments ?? 0}</div>
   234→            <p className="text-xs text-muted-foreground mt-1">
   235→              {kpi?.pendingAdjustments != null ? 'awaiting review' : 'pending adjustments'}
   236→            </p>
   237→          </CardContent>
   238→        </Card>
   239→      </div>
   240→
   241→      <div className="filter-row flex flex-wrap gap-3 items-center">
   242→        <div className="relative min-w-[200px] flex-1 max-w-sm">
   243→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   244→          <Input placeholder="Search inventory..." value={searchText} onChange={e => setSearchText(e.target.value)} className="pl-9" />
   245→        </div>
   246→        <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   247→          <SelectTrigger className="w-[170px]"><SelectValue placeholder="Category" /></SelectTrigger>
   248→          <SelectContent><SelectItem value="all">All Categories</SelectItem><SelectItem value="spare_part">Spare Part</SelectItem><SelectItem value="consumable">Consumable</SelectItem><SelectItem value="tool">Tool</SelectItem><SelectItem value="material">Material</SelectItem><SelectItem value="other">Other</SelectItem></SelectContent>
   249→        </Select>
   250→        <label className="filter-row-checkbox flex items-center gap-2 text-sm cursor-pointer">
   251→          <input type="checkbox" checked={lowStockOnly} onChange={e => setLowStockOnly(e.target.checked)} className="rounded" />
   252→          <span>Low Stock Only</span>
   253→        </label>
   254→      </div>
   255→
   256→      <Card className="border-0 shadow-sm dark:bg-card overflow-hidden">
   257→        <div className="overflow-x-auto">
   258→          <Table>
   259→            <TableHeader>
   260→              <TableRow>
   261→                <TableHead>Item Code</TableHead>
   262→                <TableHead>Name</TableHead>
   263→                <TableHead className="hidden md:table-cell">Category</TableHead>
   264→                <TableHead>Stock Level</TableHead>
   265→                <TableHead className="hidden md:table-cell">Unit Cost</TableHead>
   266→                <TableHead className="hidden lg:table-cell">Location</TableHead>
   267→                <TableHead className="w-10">Actions</TableHead>
   268→              </TableRow>
   269→            </TableHeader>
   270→            <TableBody>
   271→              {filteredItems.length === 0 ? (
   272→                <TableRow><TableCell colSpan={7} className="h-48"><EmptyState icon={Package} title="No items found" description="Try adjusting your filters." /></TableCell></TableRow>
   273→              ) : filteredItems.map(i => {
   274→                const isLow = i.currentStock <= i.minStockLevel;
   275→                const max = i.maxStockLevel || Math.max(i.minStockLevel * 2, i.currentStock, 100);
   276→                const pct = Math.min(100, (i.currentStock / max) * 100);
   277→                return (
   278→                  <TableRow key={i.id} className={isLow ? 'bg-amber-50/50 dark:bg-amber-950/10' : ''}>
   279→                    <TableCell className="font-mono text-xs">{i.itemCode}</TableCell>
   280→                    <TableCell className="font-medium text-sm">{i.name}</TableCell>
   281→                    <TableCell className="text-sm hidden md:table-cell capitalize">{i.category?.replace(/_/g, ' ')}</TableCell>
   282→                    <TableCell>
   283→                      <div className="w-24">
   284→                        <div className="flex justify-between text-[10px] text-muted-foreground mb-0.5"><span>{i.currentStock}</span><span>{i.maxStockLevel || '—'}</span></div>
   285→                        <div className="h-1.5 bg-muted rounded-full overflow-hidden">
   286→                          <div className={`h-full rounded-full ${isLow ? 'bg-amber-500' : 'bg-emerald-500'}`} style={{ width: `${pct}%` }} />
   287→                        </div>
   288→                        <p className="text-[10px] text-muted-foreground mt-0.5">Min: {i.minStockLevel}</p>
   289→                      </div>
   290→                    </TableCell>
   291→                    <TableCell className="text-sm hidden md:table-cell">{i.unitCost ? formatCurrency(i.unitCost) : '-'}</TableCell>
   292→                    <TableCell className="text-sm hidden lg:table-cell">{i.location || '-'}</TableCell>
   293→                    <TableCell>
   294→                      <DropdownMenu>
   295→                        <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="h-4 w-4" /></Button></DropdownMenuTrigger>
   296→                        <DropdownMenuContent align="end">
   297→                          {(hasPermission('inventory.update') || isAdmin()) && <DropdownMenuItem onClick={() => openEdit(i)}><Pencil className="h-3.5 w-3.5 mr-2" />Edit</DropdownMenuItem>}
   298→                          <DropdownMenuItem onClick={() => loadMovements(i.id)}><History className="h-3.5 w-3.5 mr-2" />Movements</DropdownMenuItem>
   299→                          <DropdownMenuSeparator />
   300→                          {(hasPermission('inventory.delete') || isAdmin()) && <DropdownMenuItem className="text-red-600" onClick={() => setDeleteId(i.id)}><Trash2 className="h-3.5 w-3.5 mr-2" />Delete</DropdownMenuItem>}
   301→                        </DropdownMenuContent>
   302→                      </DropdownMenu>
   303→                    </TableCell>
   304→                  </TableRow>
   305→                );
   306→              })}
   307→            </TableBody>
   308→          </Table>
   309→        </div>
   310→      </Card>
   311→
   312→      <ResponsiveDialog open={dialogOpen} onOpenChange={setDialogOpen}>
   313→        
   314→          <div className="space-y-1.5 mb-4"><h2 className="text-lg font-semibold leading-none tracking-tight">{editId ? 'Edit Item' : 'Add Inventory Item'}</h2><p className="text-sm text-muted-foreground">{editId ? 'Update inventory item.' : 'Add a new item to inventory.'}</p></div>
   315→          <div className="grid gap-4 py-2">
   316→            <div className="grid grid-cols-2 gap-3">
   317→              <div className="space-y-1.5"><Label>Item Code *</Label><Input value={form.itemCode} onChange={e => setForm(f => ({ ...f, itemCode: e.target.value }))} /></div>
   318→              <div className="space-y-1.5"><Label>Name *</Label><Input value={form.name} onChange={e => setForm(f => ({ ...f, name: e.target.value }))} /></div>
   319→            </div>
   320→            <div className="space-y-1.5"><Label>Description</Label><Textarea value={form.description} onChange={e => setForm(f => ({ ...f, description: e.target.value }))} rows={2} /></div>
   321→            <div className="grid grid-cols-3 gap-3">
   322→              <div className="space-y-1.5"><Label>Category</Label>
   323→                <Select value={form.category} onValueChange={v => setForm(f => ({ ...f, category: v }))}>
   324→                  <SelectTrigger><SelectValue /></SelectTrigger>
   325→                  <SelectContent><SelectItem value="spare_part">Spare Part</SelectItem><SelectItem value="consumable">Consumable</SelectItem><SelectItem value="tool">Tool</SelectItem><SelectItem value="material">Material</SelectItem><SelectItem value="other">Other</SelectItem></SelectContent>
   326→                </Select>
   327→              </div>
   328→              <div className="space-y-1.5"><Label>Unit of Measure</Label>
   329→                <Select value={form.unitOfMeasure} onValueChange={v => setForm(f => ({ ...f, unitOfMeasure: v }))}>
   330→                  <SelectTrigger><SelectValue /></SelectTrigger>
   331→                  <SelectContent><SelectItem value="each">Each</SelectItem><SelectItem value="kg">Kg</SelectItem><SelectItem value="liter">Liter</SelectItem><SelectItem value="meter">Meter</SelectItem><SelectItem value="set">Set</SelectItem><SelectItem value="box">Box</SelectItem><SelectItem value="roll">Roll</SelectItem></SelectContent>
   332→                </Select>
   333→              </div>
   334→              <div className="space-y-1.5"><Label>Plant</Label>
   335→                <AsyncSearchableSelect
   336→                  value={form.plantId}
   337→                  onValueChange={(val) => setForm(f => ({ ...f, plantId: val }))}
   338→                  fetchOptions={async () => {
   339→                    const res = await api.get('/api/plants');
   340→                    if (res.success && res.data) {
   341→                      return (Array.isArray(res.data) ? res.data : []).map((p: any) => ({
   342→                        value: p.id,
   343→                        label: p.name,
   344→                        group: p.city || p.location || undefined,
   345→                      }));
   346→                    }
   347→                    return [];
   348→                  }}
   349→                  placeholder="Select plant..."
   350→                />
   351→              </div>
   352→            </div>
   353→            <div className="grid grid-cols-4 gap-3">
   354→              <div className="space-y-1.5"><Label>Current Stock</Label><Input type="number" value={form.currentStock} onChange={e => setForm(f => ({ ...f, currentStock: e.target.value }))} /></div>
   355→              <div className="space-y-1.5"><Label>Min Stock Level</Label><Input type="number" value={form.minStockLevel} onChange={e => setForm(f => ({ ...f, minStockLevel: e.target.value }))} /></div>
   356→              <div className="space-y-1.5"><Label>Max Stock Level</Label><Input type="number" value={form.maxStockLevel} onChange={e => setForm(f => ({ ...f, maxStockLevel: e.target.value }))} /></div>
   357→              <div className="space-y-1.5"><Label>Reorder Qty</Label><Input type="number" value={form.reorderQuantity} onChange={e => setForm(f => ({ ...f, reorderQuantity: e.target.value }))} /></div>
   358→            </div>
   359→            <div className="grid grid-cols-2 gap-3">
   360→              <div className="space-y-1.5"><Label>Unit Cost (₵)</Label><Input type="number" value={form.unitCost} onChange={e => setForm(f => ({ ...f, unitCost: e.target.value }))} /></div>
   361→              <div className="space-y-1.5"><Label>Supplier</Label>
   362→                <AsyncSearchableSelect
   363→                  value={form.supplier}
   364→                  onValueChange={(val) => setForm(f => ({ ...f, supplier: val }))}
   365→                  fetchOptions={async () => {
