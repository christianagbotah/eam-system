   313→  // Compute the label of the group that should auto-expand (based on active child page).
   314→  // This is pure derived state — NO setState, NO useEffect, NO queueMicrotask.
   315→  // Fixes React Error #185 caused by queueMicrotask calling setOpenMenus during render.
   316→  const autoExpandLabel = useMemo<string | null>(() => {
   317→    for (const group of menuGroups) {
   318→      if (!group.children) continue;
   319→      const childMatch = group.children.some(c => {
   320→        if (c.page === currentPage) return true;
   321→        if (c.page === 'maintenance-requests' && (currentPage === 'mr-detail' || currentPage === 'create-mr')) return true;
   322→        if (c.page === 'maintenance-work-orders' && currentPage === 'wo-detail') return true;
   323→        return false;
   324→      });
   325→      if (childMatch) return group.label;
   326→    }
   327→    return null;
   328→  }, [currentPage, menuGroups]);
   329→

   337→        return next;
   338→      });

   341→
   342→  // Final open menus = manually opened ∪ auto-expanded (but NOT manually closed)

   344→    const set = new Set(manualOpenMenus);
   345→    if (autoExpandLabel && !closedMenus.has(autoExpandLabel)) set.add(autoExpandLabel);
   346→    return Array.from(set);

   349→  const toggleMenu = useCallback((label: string) => {

   351→    const isActuallyOpen =
   352→      manualOpenMenus.has(label) ||
   353→      (autoExpandLabel === label && !closedMenus.has(label));
   354→
   355→    setManualOpenMenus(prev => {
   356→      const next = new Set(prev);
   357→      if (isActuallyOpen) next.delete(label);
   358→      else next.add(label);
   359→      return next;
   360→    });
   361→    setClosedMenus(prev => {
   362→      const next = new Set(prev);
   363→      if (isActuallyOpen) next.add(label);
   364→      else next.delete(label);
   365→      return next;
   366→    });
   367→  }, [manualOpenMenus, autoExpandLabel, closedMenus]);

   375→      if (c.page === 'maintenance-requests' && (currentPage === 'mr-detail' || currentPage === 'create-mr')) return true;
   376→      if (c.page === 'maintenance-work-orders' && currentPage === 'wo-detail') return true;
   377→      return false;
   378→    });

   389→      if (!isAdm && permissions && permissions.length > 0) {

   392→          : hasPermission(g.perm);
   393→        if (!permOk) return false;

   397→      if (enabledModules.size > 0) {

   402→          if (!g.moduleCodes.some(code => enabledModules.has(code.toLowerCase()))) return false;
   403→        }
   404→      }
   405→
   406→      return true;
   407→    });

   410→  // Get tooltip text for collapsed sidebar

   416→      if (c.page === 'maintenance-work-orders' && currentPage === 'wo-detail') return true;
   417→      return false;
   418→    });
   419→    return activeChild ? `${group.label} → ${activeChild.label}` : group.label;
   420→  };

   426→        <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-400 to-emerald-600 shrink-0 shadow-lg shadow-emerald-900/30">
   427→          <Factory className="h-5 w-5 text-white" />

   434→        )}

   436→
   437→      {/* Navigation */}

   439→        <div className="space-y-0.5">
   440→          {visibleGroups.map(group => {

   445→            // (handled implicitly since manualOpenMenus only changes on user click)
   446→            const hasChildren = !!group.children;
   447→
   448→            // Standalone item (no children)
   449→            if (!hasChildren) {
   450→              return (
   451→                <TooltipProvider key={group.label} delayDuration={0}>
   452→                  <Tooltip>
   453→                    <TooltipTrigger asChild>
   454→                      <button
   455→                        onClick={() => navigate(group.page!)}
   456→                        className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm transition-all relative ${
   457→                          active
   458→                            ? 'bg-sidebar-accent text-sidebar-primary font-medium shadow-sm shadow-black/10'
   459→                            : 'text-sidebar-foreground/60 hover:bg-sidebar-accent/40 hover:text-sidebar-foreground/90'
   460→                        }`}
   461→                      >
   462→                        {active && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full bg-sidebar-primary" />}
   463→                        <Icon className={`h-[18px] w-[18px] shrink-0 ${active ? 'text-sidebar-primary' : ''}`} />
   464→                        {expanded && <span className="flex-1 text-left">{group.label}</span>}
   465→                      </button>
   466→                    </TooltipTrigger>
   467→                    {!expanded && <TooltipContent side="right">{group.label}</TooltipContent>}
   468→                  </Tooltip>
   469→                </TooltipProvider>
   470→              );
   471→            }
   472→
   473→            // Collapsed sidebar — show only icon with tooltip
   474→            if (!expanded) {
   475→              return (
   476→                <TooltipProvider key={group.label} delayDuration={0}>
   477→                  <Tooltip>
   478→                    <TooltipTrigger asChild>
   479→                      <button
   480→                        onClick={() => {
   481→                          const firstChild = group.children![0];
   482→                          navigate(firstChild.page);
   483→                        }}
   484→                        className={`w-full flex items-center justify-center px-3 py-2.5 rounded-lg text-sm transition-all relative ${
   485→                          active
   486→                            ? 'bg-sidebar-accent text-sidebar-primary font-medium shadow-sm shadow-black/10'
   487→                            : 'text-sidebar-foreground/60 hover:bg-sidebar-accent/40 hover:text-sidebar-foreground/90'
   488→                        }`}
   489→                      >
   490→                        {active && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full bg-sidebar-primary" />}
   491→                        <Icon className={`h-[18px] w-[18px] shrink-0 ${active ? 'text-sidebar-primary' : ''}`} />
   492→                      </button>
   493→                    </TooltipTrigger>
   494→                    <TooltipContent side="right">{getGroupTooltip(group)}</TooltipContent>
   495→                  </Tooltip>
   496→                </TooltipProvider>
   497→              );
   498→            }
   499→
   500→            // Expanded sidebar — show icon + label + chevron + children
   501→            return (
   502→              <div key={group.label}>
   503→                <button
   504→                  onClick={() => toggleMenu(group.label)}
   505→                  className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm transition-all relative ${
   506→                    active
   507→                      ? 'bg-sidebar-accent text-sidebar-primary font-medium shadow-sm shadow-black/10'
   508→                      : 'text-sidebar-foreground/60 hover:bg-sidebar-accent/40 hover:text-sidebar-foreground/90'
   509→                  }`}
   510→                >
   511→                  {active && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full bg-sidebar-primary" />}
   512→                  <Icon className={`h-[18px] w-[18px] shrink-0 ${active ? 'text-sidebar-primary' : ''}`} />
   513→                  <span className="flex-1 text-left">{group.label}</span>
   514→                  <ChevronRight className={`h-3.5 w-3.5 shrink-0 transition-transform duration-200 ${isOpen ? 'rotate-90' : ''}`} />
   515→                </button>
   516→                {isOpen && (
   517→                  <div className="ml-4 mt-0.5 space-y-0.5 border-l border-sidebar-border pl-3">
   518→                    {group.children!.filter(child => {
   519→                      if (child.moduleCode && enabledModules.size > 0 && child.moduleCode !== 'core') {
   520→                        return enabledModules.has(child.moduleCode.toLowerCase());
   521→                      }
   522→                      return true;
   523→                    }).map(child => {
   524→                      const childActive = child.page === currentPage ||
   525→                        (child.page === 'maintenance-requests' && (currentPage === 'mr-detail' || currentPage === 'create-mr')) ||
   526→                        (child.page === 'maintenance-work-orders' && currentPage === 'wo-detail');
   527→                      const ChildIcon = child.icon;
   528→                      return (
   529→                        <button
   530→                          key={child.page}
   531→                          onClick={() => navigate(child.page)}
   532→                          className={`w-full flex items-center gap-2.5 px-2.5 py-2 rounded-md text-[13px] transition-all relative ${
   533→                            childActive
   534→                              ? 'bg-sidebar-accent text-sidebar-primary font-medium'
   535→                              : 'text-sidebar-foreground/50 hover:bg-sidebar-accent/30 hover:text-sidebar-foreground/80'
   536→                          }`}
   537→                        >
   538→                          {childActive && <div className="absolute left-0 top-1/2 -translate-y-1/2 w-[2px] h-4 rounded-r-full bg-sidebar-primary" />}
   539→                          {ChildIcon && <ChildIcon className={`h-[15px] w-[15px] shrink-0 ${childActive ? 'text-sidebar-primary' : ''}`} />}
   540→                          <span className="flex-1 text-left">{child.label}</span>
   541→                        </button>
   542→                      );
   543→                    })}
   544→                  </div>
   545→                )}
   546→              </div>
   547→            );
   548→          })}
   549→        </div>
   550→      </nav>
   551→
   552→      {/* User — Clean profile section */}
   553→      <div className="border-t border-sidebar-border p-3">
   554→        <div className="flex items-center gap-3 px-2">
   555→          <div className="h-8 w-8 rounded-full bg-gradient-to-br from-emerald-500/80 to-emerald-700/80 flex items-center justify-center shrink-0 ring-1 ring-sidebar-border">
   556→            <span className="text-xs font-bold text-white">{user ? getInitials(user.fullName) : '?'}</span>
   557→          </div>
   558→          {expanded && (
   559→            <>
   560→              <div className="flex-1 min-w-0">
   561→                <div className="flex items-center gap-1.5 min-w-0">
   562→                  <p className="text-sm font-medium text-sidebar-foreground truncate">{user?.fullName}</p>
   563→                  {user?.roles?.[0] && (
   564→                    <span className="inline-flex items-center shrink-0 rounded-full bg-emerald-500/15 px-1.5 py-0.5 text-[10px] font-medium text-emerald-600 ring-1 ring-emerald-500/20">
   565→                      {user.roles[0].name}
   566→                    </span>
   567→                  )}
   568→                </div>
   569→                <p className="text-[10px] text-sidebar-foreground/40 truncate">{user?.email}</p>
   570→              </div>
   571→              <TooltipProvider delayDuration={0}>
   572→                <Tooltip>
   573→                  <TooltipTrigger asChild>
   574→                    <button
   575→                      className="h-8 w-8 flex items-center justify-center rounded-lg text-sidebar-foreground/40 hover:text-red-400 hover:bg-red-500/10 transition-all"
   576→                      onClick={logout}
   577→                    >
   578→                      <LogOut className="h-4 w-4" />
   579→                    </button>
   580→                  </TooltipTrigger>
   581→                  <TooltipContent side="top">Sign Out</TooltipContent>
   582→                </Tooltip>
   583→              </TooltipProvider>
   584→            </>
   585→          )}
   586→        </div>
   587→      </div>
   588→    </div>
   589→  );
   590→}
