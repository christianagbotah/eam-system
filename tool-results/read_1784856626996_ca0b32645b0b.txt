    54→const useDateRange = () => {
    55→  const [startDate, setStartDate] = useState(() => {
    56→    const d = new Date(); d.setMonth(d.getMonth() - 1); return d.toISOString().slice(0, 10);
    57→  });
    58→  const [endDate, setEndDate] = useState(() => new Date().toISOString().slice(0, 10));
    59→  return { startDate, setStartDate, endDate, setEndDate };
    60→};

    62→const exportCSV = (filename: string, headers: string[], rows: string[][]) => {
    63→  const csv = [headers.join(','), ...rows.map(r => r.map(c => `"${(c ?? '').toString().replace(/"/g, '""')}"`).join(','))].join('\n');
    64→  const blob = new Blob([csv], { type: 'text/csv' });
    65→  const url = URL.createObjectURL(blob);
    66→  const a = document.createElement('a');
    67→  a.href = url; a.download = `${filename}.csv`; a.click();
    68→  URL.revokeObjectURL(url);
    69→  toast.success(`Exported ${filename}.csv`);
    70→};
