    53→    53→const useDateRange = () => {
    54→    54→  const [startDate, setStartDate] = useState(() => {
    55→    55→    const d = new Date(); d.setMonth(d.getMonth() - 1); return d.toISOString().slice(0, 10);
    56→    56→  });
    57→    57→  const [endDate, setEndDate] = useState(() => new Date().toISOString().slice(0, 10));
    58→    58→  return { startDate, setStartDate, endDate, setEndDate };

    61→    61→const exportCSV = (filename: string, headers: string[], rows: string[][]) => {
    62→    62→  const csv = [headers.join(','), ...rows.map(r => r.map(c => `"${(c ?? '').toString().replace(/"/g, '""')}"`).join(','))].join('\n');
    63→    63→  const blob = new Blob([csv], { type: 'text/csv' });
    64→    64→  const url = URL.createObjectURL(blob);
    65→    65→  const a = document.createElement('a');
    66→    66→  a.href = url; a.download = `${filename}.csv`; a.click();
    67→    67→  URL.revokeObjectURL(url);
    68→    68→  toast.success(`Exported ${filename}.csv`);
    69→    69→};
