   377→          pageCache.set(page, C);
   378→          setComponent(() => C);
   379→        }
   380→      })
   381→      .catch(err => {
   382→        if (!cancelled) {
   383→          console.error(`[PageSwitcher] Failed to load page "${page}":`, err);
   384→          setError(err instanceof Error ? err.message : 'Failed to load page');
   385→        }
   386→      });
   387→
   388→    return () => { cancelled = true; };
   389→  }, [page]);
   390→
   391→  if (error) {
   392→    return (
   393→      <div className="flex items-center justify-center min-h-[300px]">
   394→        <div className="text-center space-y-3 p-6 max-w-md">
   395→          <div className="h-10 w-10 mx-auto rounded-full bg-red-100 dark:bg-red-950/30 flex items-center justify-center">
   396→            <svg className="h-5 w-5 text-red-600" fill="none" viewBox="0 0 24 24" stroke="currentColor">
   397→              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
   398→            </svg>
   399→          </div>
   400→          <p className="text-sm font-medium">Page failed to load</p>
   401→          <p className="text-xs text-muted-foreground">This usually happens after an update. Try reloading.</p>
   402→          <button
   403→            onClick={() => window.location.reload()}
   404→            className="px-4 py-1.5 bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-medium rounded-lg transition-colors"
   405→          >
   406→            Reload Page
   407→          </button>
   408→        </div>
   409→      </div>
   410→    );
   411→  }
   412→
   413→  if (!Component) return <LoadingSkeleton />;
   414→  return <Component />;
   415→}

   735→// ============================================================================
   736→// GLOBAL ERROR BOUNDARY — catches ALL errors including .map() crashes
   737→// ============================================================================
   738→
   739→interface ErrorBoundaryState {
   740→  hasError: boolean;
   741→  error: Error | null;
   742→  componentStack: string | null;
   743→  isChunkError: boolean;
   744→}
   745→
   746→  // Error message pattern for React #185
   747→  const ERROR_185_PATTERN = /Cannot update a component (?:from within the render function|while rendering a different component)/;
   748→
   749→class GlobalErrorBoundary extends React.Component<
   750→  { children: React.ReactNode },
   751→  ErrorBoundaryState & { retryTimer: ReturnType<typeof setTimeout> | null; retryCount: number }
   752→> {
   753→  private _retryTimer: ReturnType<typeof setTimeout> | null = null;
   754→  private _retryCount: number = 0;
   755→
   756→  constructor(props: { children: React.ReactNode }) {
   757→    super(props);
   758→    this.state = { hasError: false, error: null, componentStack: null, isChunkError: false };
   759→  }
   760→
   761→  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
   762→    const msg = error?.message || '';
   763→    const isChunkError =
   764→      msg.includes('ChunkLoadError') ||
   765→      msg.includes('Failed to load chunk') ||
   766→      msg.includes('Loading chunk') ||
   767→      msg.includes('Loading CSS chunk');
   768→
   769→    // CRITICAL: For Error #185, do NOT show the error screen.
   770→    // This error is caused by external store updates (Zustand set()) firing
   771→    // during React's concurrent render phase — it's a timing issue, not a
   772→    // real bug. Returning hasError: false lets React retry the render
   773→    // automatically, which almost always succeeds.
   774→    if (ERROR_185_PATTERN.test(msg)) {
   775→      return { hasError: false, error, componentStack: null, isChunkError: false };
   776→    }
   777→
   778→    // Catch ALL other errors
   779→    return { hasError: true, error, componentStack: null, isChunkError };
   780→  }
   781→
   782→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   783→    // Log full error details to console and state
   784→    console.error('=== EAM GLOBAL ERROR BOUNDARY ===');
   785→    console.error('Error:', error.message);
   786→    console.error('Stack:', error.stack);
   787→    console.error('Component Stack:', info.componentStack);
   788→    console.error('=== END ERROR ===');
   789→    this.setState({ componentStack: info.componentStack });
   790→
   791→    // For Error #185, schedule a silent retry after a short delay.
   792→    // This handles the case where the automatic retry (hasError: false from
   793→    // getDerivedStateFromError) still fails because the Zustand store update
   794→    // hasn't completed yet. We force a re-render after 50ms which almost
   795→    // always succeeds.
   796→    if (ERROR_185_PATTERN.test(error.message)) {
   797→      this._retryCount++;
   798→      console.warn(`[ErrorBoundary] React #185 caught, scheduling silent retry #${this._retryCount}`);
   799→
   800→      this._retryTimer = setTimeout(() => {
   801→        this._retryTimer = null;
   802→        // Force a re-render by toggling state
   803→        this.setState((prev) => {
   804→          if (prev.hasError) {
   805→            return { hasError: false, error: null, componentStack: null, isChunkError: false };
   806→          }
   807→          return prev;
   808→        });
   809→      }, 50);
   810→    }
   811→  }
   812→
   813→  componentWillUnmount() {
   814→    if (this._retryTimer) {
   815→      clearTimeout(this._retryTimer);
   816→      this._retryTimer = null;
   817→    }
   818→  }
   819→
   820→  handleRetry = () => {
   821→    if (this._retryTimer) {
   822→      clearTimeout(this._retryTimer);
   823→      this._retryTimer = null;
   824→    }
   825→    // Always reload for chunk errors — the JS bundle is stale
   826→    if (this.state.isChunkError) {
   827→      window.location.reload();
   828→      return;
   829→    }
   830→    // For runtime errors, try re-rendering
   831→    this.setState({ hasError: false, error: null, componentStack: null, isChunkError: false });
   832→  };
   833→
   834→  handleHardRefresh = () => {
   835→    window.location.href = window.location.href.split('?')[0] + '?_t=' + Date.now();
   836→  };
   837→
   838→  render() {
   839→    if (this.state.hasError && this.state.error) {
   840→      const isChunk = this.state.isChunkError;
   841→      return (
   842→        <div className="flex items-center justify-center min-h-[400px]">
   843→          <div className="text-center space-y-4 p-6 max-w-lg">
   844→            <div className={`h-12 w-12 mx-auto rounded-full flex items-center justify-center ${isChunk ? 'bg-amber-100 dark:bg-amber-950/30' : 'bg-red-100 dark:bg-red-950/30'}`}>
   845→              <svg className={`h-6 w-6 ${isChunk ? 'text-amber-600' : 'text-red-600'}`} fill="none" viewBox="0 0 24 24" stroke="currentColor">
   846→                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
   847→              </svg>
   848→            </div>
   849→            <div>
   850→              <h3 className="text-sm font-semibold text-foreground">{isChunk ? 'Module Loading Error' : 'Runtime Error'}</h3>
   851→              <p className="text-xs text-red-600 font-mono mt-2 p-2 bg-red-50 dark:bg-red-950/20 rounded break-all text-left">
   852→                {this.state.error.message}
   853→              </p>
   854→              {this.state.componentStack && (
   855→                <pre className="text-[10px] text-muted-foreground mt-2 p-2 bg-muted rounded overflow-auto max-h-32 text-left whitespace-pre-wrap">
   856→                  {this.state.componentStack}
   857→                </pre>
   858→              )}
   859→            </div>
   860→            <div className="flex gap-2 justify-center">
   861→              <button
   862→                onClick={this.handleRetry}
   863→                className="inline-flex items-center gap-2 px-4 py-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium rounded-lg transition-colors"
