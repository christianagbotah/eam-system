    53→const useDateRange = () => {
    54→  const [startDate, setStartDate] = useState(() => {
    55→    const d = new Date(); d.setMonth(d.getMonth() - 1); return d.toISOString().slice(0, 10);
    56→  });
    57→  const [endDate, setEndDate] = useState(() => new Date().toISOString().slice(0, 10));
    58→  return { startDate, setStartDate, endDate, setEndDate };
    59→};

    61→const exportCSV = (filename: string, headers: string[], rows: string[][]) => {
    62→  const csv = [headers.join(','), ...rows.map(r => r.map(c => `"${(c ?? '').toString().replace(/"/g, '""')}"`).join(','))].join('\n');
    63→  const blob = new Blob([csv], { type: 'text/csv' });
    64→  const url = URL.createObjectURL(blob);
    65→  const a = document.createElement('a');
    66→  a.href = url; a.download = `${filename}.csv`; a.click();
    67→  URL.revokeObjectURL(url);
    68→  toast.success(`Exported ${filename}.csv`);
    69→};

   330→    return { currentPeriod: cur, previousPeriod: prev, hasYoY, comparisonKPIs, comparisonBarData, comparisonRows, yoyData };
   331→  }, [enterpriseData?.periodComparison]);
   332→
   333→  if (loading && !reportData) return <div className="page-content"><LoadingSkeleton /></div>;

   341→            <BarChart3 className="h-6 w-6 text-emerald-600" />
   342→            Enterprise Reporting
   343→          </h1>

   347→          <Button size="sm" onClick={fetchReport} disabled={loading}>
   348→            {loading ? <Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> : <RefreshCw className="h-4 w-4 mr-1.5" />}
   349→            Refresh
   350→          </Button>

   352→            <FileDown className="h-4 w-4 mr-1.5" />PDF
   353→          </Button>
   354→          <Button variant="outline" size="sm" onClick={handleCsvExport} disabled={!reportData}>
   355→            <Download className="h-4 w-4 mr-1.5" />CSV
   356→          </Button>
   357→        </div>
   358→      </div>
   359→
   360→      {/* Date Range */}
   361→      <Card className="border border-border/60 shadow-sm">
   362→        <CardContent className="p-4">
   363→          <div className="flex items-center gap-3 flex-wrap">
   364→            <DateRangePicker label="Date Range" from={startDate || undefined} to={endDate || undefined} onChange={(f, t) => { setStartDate(f || ''); setEndDate(t || ''); }} />
